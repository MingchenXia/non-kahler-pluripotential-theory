\section{Relative boundedness for \texorpdfstring{$L^p$}{Lp} data}
\label{sec:big-model-lp}

If the datum has finite moments of sufficiently high order against normalized $\theta$-plurisubharmonic functions, every normalized $u\in\mathcal E_\varphi$ whose measure is dominated by the datum differs from $\varphi$ by a bounded function. We work directly with $u\in\mathcal E_\varphi$: truncating $u$ against $\varphi-j$ would introduce a contact measure that need not be controlled by the datum. The concave-obstacle envelopes of \cref{lem:big-model-concave-envelope} keep this domination and give a differential inequality for the distribution of $\varphi-u$.

\Needspace{14\baselineskip}
\begin{proposition}\label{prop:big-model-moment-bound}\index{Monge--Ampere equation@Monge--Amp\`ere equation!relative boundedness}
Let $\mu$ be a finite non-pluripolar measure. Suppose that, for some real $m>2n+1$,
\[
 A_m:=\sup_{\substack{v\in\PSH(X,\theta)\\\sup_Xv=0}}
             \int_X(-v)^m\,\mathrm{d}\mu<\infty.
\]
There is a finite constant $C$, depending only on
$n,\ell,M,m,\mu(X)$ and $A_m$, such that every $u\in\mathcal E_\varphi$ with $\sup_Xu=0$ and $\mu_u\leq\mu$ satisfies
\[
 \varphi-C\leq u\leq\varphi.
\]
\end{proposition}
\begin{proof}
As in \cref{prop:big-minimal-higher-moment}, we build a concave weight from the distribution of $\varphi-u$ and apply \cref{lem:big-model-concave-envelope}. The moment bound on $\mu$ now yields a differential inequality for this weight, which bounds the depth of $\varphi-u$ under $\mu_u$.

The envelope identity gives $u\leq\varphi\leq0$. Put $x=\varphi-u$, defined off the poles, and $N=2n+1$. Since $x\leq-u$ and $\mu_u\leq\mu$, H\"older's inequality gives the uniform first-moment bound
\[
 B:=\int_Xx\,\mu_u
 \leq\int_X(-u)\,\mathrm{d}\mu
 \leq\mu(X)^{1-1/m}A_m^{1/m}=:A_1.
\]
Let $H(t)=\mu_u(\{x>t\})$ and
$T=\sup\{t\geq0:H(t)>0\}$, with $T=0$ if this set is empty; at this point $T$ may be infinite. If $T<\infty$, zero domination applied to $u$ and $\varphi-t$ for $t>T$ gives $u\geq\varphi-T$. It is therefore enough to bound $T$. We may assume $T>1$, and we choose any $S$ with $1<S<T$.

Define $g$, $F$, $W$ and $\chi$ as $g_s$, $F_s$, $W_s$ and $\chi_s$ in the proof of \cref{prop:big-minimal-higher-moment}, with $\varphi$, $H$ and $S$ in place of $V$, $\mu_u(\{x>\cdot\})$ and $T-s$. Since $H(S)>0$, $F$ is bounded, although its bound may depend on $S$ and $u$; thus $\chi$ satisfies the hypotheses of \cref{lem:big-model-concave-envelope}. Also $W(t)\geq t$ and $W(t)\leq tF(t)^{1/N}$. Let $p=P_\theta(\varphi-W(x))\in\mathcal E_\varphi$ be the envelope of that lemma. The estimate \eqref{eq:big-weight-derivative-moment}, now with $M$ in place of $M_V$ and the integral in $t$ over $[0,\infty)$, and the Cauchy--Schwarz estimate in that proof give bounds independent of $S$ and $u$:
\[
 \int_X F(x)\,\mu_u\leq3\max(1,M)=:K,
 \qquad
 \int_X(\varphi-p)^{1/2}\,\mu_p\leq\sqrt{KB}.
\]
The identity $P_\theta[p]=\varphi$ implies $p-\sup_Xp\leq\varphi$. Since $p\leq\varphi\leq0$ and $\mu_p(X)\geq\ell$, the last integral is at least $\ell\sqrt{-\sup_Xp}$. Consequently
\[
 0\leq-\sup_Xp\leq KA_1/\ell^2=:D.
\]
Using $W(x)\leq\varphi-p\leq-p$, and applying the moment hypothesis to $p-\sup_Xp$, we obtain
\begin{equation}\label{eq:big-model-weight-moment}
 \int_XW(x)^m\,\mathrm{d}\mu
 \leq2^{m-1}\bigl(A_m+D^m\mu(X)\bigr)<C_0,
\end{equation}
where we choose $C_0=1+2^{m-1}(A_m+D^m\mu(X))$. This constant is independent of $S$ and $u$.

For $0<t\leq S$, Chebyshev's inequality and $\mu_u\leq\mu$ give
\[
 H(t)\leq C_0W(t)^{-m},
 \qquad
 F'(t)\geq\frac{W(t)^m}{C_0(1+t)^{3/2}}
 \quad\text{for almost every }t.
\]
Since $W'=F^{1/N}$, the derivative of $(W')^{N+1}$ is at least
\[
 \frac{N+1}{NC_0}\,
 \frac{W^mW'}{(1+t)^{3/2}}.
\]
All functions being differentiated are absolutely continuous on compact intervals. Integrating from $0$ to $t$ and using $W(0)=0$ gives
\[
 (W'(t))^{N+1}
 \geq\frac{N+1}{NC_0(m+1)}
        \frac{W(t)^{m+1}}{(1+t)^{3/2}}.
\]
In particular
\[
 W'(t)\geq c_0(1+t)^{-\gamma}W(t)^\kappa,
 \quad
 \gamma=\frac{3}{2(N+1)}<1,\quad
 \kappa=\frac{m+1}{N+1}>1,
\]
with $c_0>0$ independent of $S$ and $u$. Integrating the derivative of $W^{1-\kappa}$ over $[1,S]$ yields
\[
 (\kappa-1)c_0\int_1^S(1+t)^{-\gamma}\,\mathrm{d}t
 \leq W(1)^{1-\kappa}-W(S)^{1-\kappa}\leq1,
\]
because $W(1)\geq1$. Since the integral on the left diverges as $S\to\infty$, every $S<T$ is bounded by a constant independent of $S$ and $u$, and hence so is $T$. The zero-domination argument at the beginning of the proof now gives $u\geq\varphi-C$. The case $T\leq1$, which includes $B=0$, was covered there as well.
\end{proof}

\enlargethispage{8\baselineskip}
\begin{corollary}\label{cor:big-model-lp-type}
Let $0\leq f\in L^p(X,\mathrm{d}V)$, with $p>1$ and $\int_Xf\,\mathrm{d}V>0$.
For $\lambda>0$, the solution in \cref{thm:big-model-exponential} of
\[
 \mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}f\,\mathrm{d}V
\]
satisfies $u\sim\varphi$. Every normalized solution in $\mathcal E_\varphi$ of
\[
 \mathrm{MA}_\theta(u)=cf\,\mathrm{d}V,\qquad c>0,
\]
also satisfies $u\sim\varphi$. In particular, both existence statements hold in the ordinary singularity type of $\varphi$.
\end{corollary}
\begin{proof}
\Cref{lem:uniform-exponential}, applied in a Hermitian background dominating $\theta$, gives uniform moments of every finite order against $\mathrm{d}V$ for normalized $\theta$-plurisubharmonic functions. H\"older's inequality therefore implies, for every finite $m$,
\[
 \sup_{\substack{v\in\PSH(X,\theta)\\\sup_Xv=0}}
       \int_X(-v)^m f\,\mathrm{d}V<\infty.
\]
Choose $m>2n+1$. For the solution of the exponential equation, set $S=\sup_Xu$ and $v=u-S$. Then $v\in\mathcal E_\varphi$, $\sup_Xv=0$, and
\[
 \mu_v=\mathrm{e}^{\lambda u}f\,\mathrm{d}V
 \leq \mathrm{e}^{\lambda S}f\,\mathrm{d}V.
\]
The preceding proposition, applied to this fixed dominating measure, shows that $v$, and hence $u$, has the ordinary type of $\varphi$. For the normalized equation, apply the proposition to the measure $cf\,\mathrm{d}V$.
\end{proof}
