\section{Comparison and invariant mass}
\label{uni-sec:bounded}

The preceding argument controls the extremal level sets of $u-v$ through connectedness of the set where the density is positive. When the background satisfies the Guan--Li condition below, invariance of the total mass gives the ordinary comparison principle instead. An auxiliary solution whose density is larger on the set where $u>v$ yields small perturbations $q$ of $u$ for which $\{v<q\}$ has zero mass for the common Monge--Amp\`ere measure, and the domination principle then proves uniqueness.

\Needspace{10\baselineskip}
Set $V=\int_X\omega^n$. The \emph{Guan--Li condition}
\begin{equation}
   \ddc\omega=0,\qquad \ddc(\omega^2)=0
   \label{uni-eq:GL}
\end{equation}
is equivalent to invariance of the total mass $\int_X\mathrm{MA}_\omega(u)$ by \cref{lem:power-torsion,thm:mass-invariance-characterization}. In dimension one it holds automatically, for degree reasons.

\Needspace{17\baselineskip}
\begin{theorem}\label{uni-thm:GL-intro}\index{Uniqueness!invariant mass}
Assume \eqref{uni-eq:GL}, let $p>1$, and let $f\geq0$ belong to
$L^p(X,\omega^n)$ with positive integral. Then there is exactly one
pair $(c,u)$, with $c>0$, $u\in\PSH(X,\omega)\cap L^\infty(X)$ and
$\sup_Xu=0$, satisfying
\[
  (\omega+\ddc u)^n=c f\,\omega^n.
\]
Moreover
\[
   c=\frac{\int_X\omega^n}{\int_Xf\,\omega^n},
\]
and $u$ is continuous.  Thus two bounded solutions with the same
$L^p$ density differ by a constant, without any positive lower bound on $f$.
\end{theorem}

We first show that the ordinary comparison principle is equivalent to invariance of the mass.

\begin{theorem}\label{uni-thm:GL-equivalence}
Condition \eqref{uni-eq:GL} holds if and only if every pair of bounded $\omega$-plurisubharmonic functions $a,b$ satisfies
\begin{equation}\label{uni-eq:ordinaryCP}
 \int_{\{a<b\}}\mathrm{MA}_\omega(b)
 \leq\int_{\{a<b\}}\mathrm{MA}_\omega(a).
\end{equation}
For the converse implication, it suffices to require this inequality for $a,b\in\HH_\omega$.
\end{theorem}

\begin{proof}
Under \eqref{uni-eq:GL}, \cref{thm:mass-invariance-characterization} gives $\int_X\mathrm{MA}_\omega(a)=V$ for every bounded potential $a$; for $n=1$ this is Stokes' theorem.

Set $q=\max(a,b)$. On $\{a<b\}$, plurifine locality gives $\mathrm{MA}_\omega(q)=\mathrm{MA}_\omega(b)$. On $\{a\geq b\}$, one has $a=q$, and $a\leq q$ everywhere, so bounded ordered contact (\cref{lem:bounded-ordered-contact}) gives
\[
 \mathbf1_{\{a\geq b\}}\mathrm{MA}_\omega(a)
 \leq \mathbf1_{\{a\geq b\}}\mathrm{MA}_\omega(q).
\]
The total masses of $a$, $b$ and $q$ all equal $V$. Hence
\[
 \int_{\{a<b\}}\mathrm{MA}_\omega(b)=V-\int_{\{a\geq b\}}\mathrm{MA}_\omega(q)
 \leq V-\int_{\{a\geq b\}}\mathrm{MA}_\omega(a)=\int_{\{a<b\}}\mathrm{MA}_\omega(a),
\]
which is \eqref{uni-eq:ordinaryCP}.

Conversely, if \eqref{uni-eq:ordinaryCP} holds for all $a,b\in\HH_\omega$, then \cref{prop:comparison-forces-mass} makes the mass constant on $\HH_\omega$. By \cref{thm:mass-invariance-characterization,lem:power-torsion}, this implies \eqref{uni-eq:GL} when $n\geq2$; for $n=1$ that condition holds for degree reasons.
\end{proof}

For the uniqueness argument we need, in addition to comparison, auxiliary solutions and the mixed inequality \eqref{eq:interior-minkowski-background}. The auxiliary solutions come from continuous existence: if $G\geq0$ lies in $L^p(X,\omega^n)$ and $\int_XG\omega^n=V$, \cref{thm:kn-continuous} gives a continuous solution of $\mathrm{MA}_\omega(\rho)=cG\omega^n$ with a constant $c>0$, and since all bounded potentials have mass $V$, this constant equals one. Thus
\begin{equation}\label{uni-eq:aux-solvability}
 \mathrm{MA}_\omega(\rho)=G\omega^n
\end{equation}
has a continuous solution $\rho$ for every such $G$, in particular for the density with an indicator factor used below.

\begin{proof}[of \cref{uni-thm:GL-intro}]
We use the domination principle in the form of the case $c=0$ of \cref{thm:bounded-strict-domination}: if $\mathrm{MA}_\omega(a)(\{a<b\})=0$ for bounded $\omega$-plurisubharmonic functions $a,b$, then $a\geq b$.

We first prove that two bounded solutions $u,v$ of $\mathrm{MA}_\omega(u)=\mathrm{MA}_\omega(v)=\mu=F\omega^n$, with
$F\in L^p$, $p>1$, differ by a constant.  The mass identity gives $\mu(X)=V$.
If $u-v$ is nonconstant, add a constant to $u$ so that $u-v$
has supremum $s>0$ and infimum $-s$.  Since $\{v<u\}$ and $\{u<v\}$ are disjoint, after interchanging $u$ and $v$ if
necessary, the set $S=\{v<u\}$ satisfies $\mu(S)\leq V/2$.
Define
\[
 b_0=\frac{V-2\mu(S)}V\geq0,
 \qquad G=2F\mathbf1_S+b_0.
\]
This nonnegative $L^p$ function has total integral $V$, so we can choose a
continuous solution $\rho$ of \eqref{uni-eq:aux-solvability}.
Take $C_0$ with $\rho-C_0\leq u$, and set
\[
 q_\delta=(1-\delta)u+\delta(\rho-C_0),\qquad
 E_\delta=\{v<q_\delta\}\subset S,
 \qquad 0<\delta<1.
\]
Since $G\geq2F$ on $S$, the mixed inequality \eqref{eq:interior-minkowski-background} gives, on $S$,
\begin{equation}\label{uni-eq:mixed-bounded}
 \mathrm{MA}_\omega(q_\delta)\geq
 \bigl((1-\delta)+\delta2^{1/n}\bigr)^n\mu=:A_\delta\mu,
 \qquad A_\delta>1.
\end{equation}
Ordinary comparison \eqref{uni-eq:ordinaryCP}, applied to the pair $v,q_\delta$, yields
\[
 A_\delta\mu(E_\delta)
 \leq\int_{E_\delta}\mathrm{MA}_\omega(q_\delta)
 \leq\int_{E_\delta}\mathrm{MA}_\omega(v)=\mu(E_\delta).
\]
Thus $\mu(E_\delta)=0$, that is, $\mathrm{MA}_\omega(v)(\{v<q_\delta\})=0$.  Bounded domination gives
$v\geq q_\delta$ everywhere, and letting $\delta\downarrow0$ gives
$v\geq u$, which contradicts $\sup_X(u-v)=s>0$.  Therefore $u-v$
is constant.  The argument never divides by $F$, so $F$ may vanish on an open set.

Finally, for the density $f$ in the theorem, put
$I=\int_Xf\omega^n>0$ and $c=V/I$.  By \eqref{uni-eq:aux-solvability} with
$G=cf$, there is a continuous solution, which we normalize by subtracting its supremum.  Any other bounded
solution, with constant $c'$, satisfies $V=c'I$ by mass invariance, hence $c'=c$;
the uniqueness just proved and the common normalization then identify the
potentials.  In particular, every bounded solution with this $L^p$
density is continuous, although continuity was not assumed.
\end{proof}

\begin{remark}
Uniqueness does not require the Guan--Li condition: the other criteria in this chapter apply without it. The condition must hold for $\omega$ itself: on the torus of \cref{ex:torus-mass}, the flat metric satisfies \eqref{uni-eq:GL}, but \eqref{uni-eq:ordinaryCP} fails for $\mathrm{MA}_\omega$ by \cref{cor:comparison-fails}.
\end{remark}
