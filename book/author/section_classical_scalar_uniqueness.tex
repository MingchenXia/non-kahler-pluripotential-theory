\begin{proposition}\label{prop:classical-scalar-uniqueness}\index{Normalization scalar!classical uniqueness}
Let $u,v\in C^2(X,\mathbb R)$ and set $A=\omega+\ddc u$, $B=\omega+\ddc v$.
If $A,B\geq0$ and $A^n=cB^n$ for a constant $c>0$, then $c=1$.
\end{proposition}

\begin{proof}
Fix $\varepsilon>0$. At a maximum of $u-(1+\varepsilon)v$,
\[
 A+\varepsilon\omega\leq(1+\varepsilon)B.
\]
At this maximum, $B>0$ because $A+\varepsilon\omega>0$. Determinant monotonicity and $A^n=cB^n$ give $c\leq(1+\varepsilon)^n$. Let $\varepsilon\downarrow0$ and interchange the two potentials to obtain $c\leq1$ and $c^{-1}\leq1$.
\end{proof}
