\section{The modified comparison principle}

For a Hermitian background, the total Monge--Amp\`ere mass varies with the potential, and the ordinary comparison principle fails in general (see \cref{uni-thm:GL-equivalence}). A weaker form survives. Replacing the comparison potential $v$ by $(1-\varepsilon)v$ leaves the strictly positive form $\varepsilon\omega$ in reserve, and this reserve absorbs the torsion terms. On a sublevel set of height $s$, integration by parts makes the error proportional to $s$, and a finite induction controls the intermediate mixed products. The resulting multiplicative correction tends to one as $s$ tends to zero with $\varepsilon$ fixed.

Since $X$ is compact and $\omega$ is positive, we can choose $B_0\geq1$ such that
\[
 -B_0\omega^2\leq\ddc\omega\leq B_0\omega^2,
 \qquad
 -B_0\omega^3\leq \mathrm{d}\omega\wedge\dc\omega\leq B_0\omega^3.
\]
These inequalities are understood after wedging with positive forms of complementary degree; forms of degree larger than $2n$ vanish, and the corresponding terms below are omitted.

\Needspace{14\baselineskip}
\begin{theorem}\label{thm:modified-comparison}\index{Comparison principle!modified}
Let $u,v\in\PSH(X,\omega)\cap L^\infty(X)$ and $0<\varepsilon<1$. For $s>0$, put
\[
 m_\varepsilon=\inf_X\bigl(u-(1-\varepsilon)v\bigr),\qquad
 U_{\varepsilon,s}=\{u<(1-\varepsilon)v+m_\varepsilon+s\}.
\]
If $0<s<\varepsilon^3/(32B_0n^2)$, then, with
$a=B_0n^2s\varepsilon^{-3}$,
\begin{equation}\label{eq:modified-comparison}
 (1-4a)^n\int_{U_{\varepsilon,s}}\omega_{(1-\varepsilon)v}^{\,n}
 \leq\int_{U_{\varepsilon,s}}\omega_u^n.
\end{equation}
\end{theorem}
\begin{proof}
Write $v_\varepsilon=(1-\varepsilon)v$, $U=U_{\varepsilon,s}$ and
\[
 \phi=\max(u,v_\varepsilon+m_\varepsilon+s),\qquad
 T_k=\omega_u^k\wedge\omega_\phi^{n-k},\qquad I_k=\int_U T_k\quad(0\leq k\leq n).
\]
Set $I_{-1}=I_{-2}=0$. Then $0\leq\phi-u\leq s$, this difference vanishes outside $U$, and on $U$ plurifine locality gives
$\omega_\phi=\omega_{v_\varepsilon}\geq\varepsilon\omega$ in all bounded products.

By \cref{lem:bounded-ordered-contact}, for every fixed positive mixed product $W$ of $n-1$ bounded-potential factors,
\[
 \mathbf1_{\{u=\phi\}}\omega_u\wedge W
 \leq\mathbf1_{\{u=\phi\}}\omega_\phi\wedge W.
\]

For $0\leq k\leq n-1$, apply this with $W=\omega_u^k\wedge\omega_\phi^{n-k-1}$. Since $u=\phi$ on $X\setminus U$, the difference $T_k-T_{k+1}=(\omega_\phi-\omega_u)\wedge W$ is nonnegative there. Consequently
\begin{equation}\label{eq:comparison-stokes}
 I_k-I_{k+1}
 \leq\int_X\ddc(\phi-u)\wedge W
 =\int_X(\phi-u)\ddc W.
\end{equation}
This integration by parts is valid for bounded potentials. To see this, approximate $u$ and $\phi$ by decreasing sequences of smooth, uniformly bounded $\omega$-plurisubharmonic functions, using \cref{thm:bounded-exact-regularization}. For smooth functions the identity is ordinary integration by parts. The form $\ddc W$ expands as below into bounded Hessian products and smooth forms, and bounded Bedford--Taylor convergence, also against bounded quasi-continuous coefficients converging in capacity, passes each term to the limit.

\Needspace{11\baselineskip}
Set $r=n-k-1$. Since $\mathrm{d}\omega_u=\mathrm{d}\omega_\phi=\mathrm{d}\omega$, expansion gives
\[
 \begin{split}
 \ddc W={}&k\ddc\omega\wedge\omega_u^{k-1}\wedge\omega_\phi^r
       +r\ddc\omega\wedge\omega_u^k\wedge\omega_\phi^{r-1}\\
 &+k(k-1)\mathrm{d}\omega\wedge\dc\omega\wedge\omega_u^{k-2}\wedge\omega_\phi^r\\
 &+2kr\,\mathrm{d}\omega\wedge\dc\omega\wedge\omega_u^{k-1}\wedge\omega_\phi^{r-1}\\
 &+r(r-1)\mathrm{d}\omega\wedge\dc\omega\wedge\omega_u^k\wedge\omega_\phi^{r-2}.
 \end{split}
\]
Terms containing a negative power have coefficient zero and are omitted. On $U$ we may replace each $\omega^j$ in an upper bound by $\varepsilon^{-j}\omega_\phi^j$, since $\omega_\phi\geq\varepsilon\omega$ there. Using $\phi-u\leq s$ and $\varepsilon^{-2}\leq\varepsilon^{-3}$ in \eqref{eq:comparison-stokes}, we obtain
\[
 I_k-I_{k+1}\leq B_0s\varepsilon^{-3}
 \bigl(k(k-1)I_{k-2}+k(1+2r)I_{k-1}+r^2I_k\bigr).
\]
Each coefficient is at most $n^2$, and $a=B_0n^2s\varepsilon^{-3}$. Hence
\begin{equation}\label{eq:comparison-recurrence}
 I_k-I_{k+1}\leq a(I_{k-2}+I_{k-1}+I_k).
\end{equation}
Put $q=1-4a$; the assumption on $s$ gives $a<1/32$, so $q>7/8$. Suppose inductively that $qI_j\leq I_{j+1}$ for $j<k$. Then
$I_{k-1}\leq q^{-1}I_k$ and $I_{k-2}\leq q^{-2}I_k$ whenever the indices exist. Since
\[
 1+q^{-1}+q^{-2}\leq1+8/7+64/49<4,
\]
\eqref{eq:comparison-recurrence} gives $qI_k\leq I_{k+1}$. The case $k=0$ follows from the same formula with the two absent terms set to zero. By induction, $q^n I_0\leq I_n$. Since $T_0=\omega_{v_\varepsilon}^n$ on $U$ and $T_n=\omega_u^n$, this is the theorem. In dimension one all torsion terms vanish and the same argument applies.
\end{proof}

The reverse factor $(1-4a)^{-n}$ in \eqref{eq:modified-comparison} tends to one as $s\downarrow0$ for fixed $\varepsilon$. A strict multiplicative inequality between the Monge--Amp\`ere measures on $\{u<v\}$ therefore absorbs the error in modified comparison. One first chooses the positivity reserve $\varepsilon$, and then a sublevel set thin enough to preserve the strict inequality.

\begin{theorem}\label{thm:bounded-strict-domination}\index{Domination principle!strict bounded comparison}
Let $u,v\in\PSH(X,\omega)\cap L^\infty(X)$, and let $0\leq c<1$. If
\[
 \mathbf1_{\{u<v\}}\omega_u^n
 \leq c\,\mathbf1_{\{u<v\}}\omega_v^n,
\]
then $u\geq v$. In particular, $\omega_u^n(\{u<v\})=0$ implies $u\geq v$.
\end{theorem}
\begin{proof}
Suppose $m=\inf_X(u-v)<0$. Choose $\varepsilon>0$ so small that
\[
 (1-\varepsilon)^n>c,
 \qquad 2\varepsilon\|v\|_\infty<-m/4.
\]
Let $m_\varepsilon$ and $U_{\varepsilon,s}$ be defined as in \cref{thm:modified-comparison} for this pair $u,v$. Since
$|m_\varepsilon-m|\leq\varepsilon\|v\|_\infty$, for all sufficiently small $s>0$ the set $U=U_{\varepsilon,s}$ is contained in $\{u<v\}$: there
\[
 u-v<m_\varepsilon+s-\varepsilon v
       \leq m+2\varepsilon\|v\|_\infty+s<0.
\]
The set $U$ is nonempty by the definition of the infimum and has positive volume by \cref{lem:strict-set-positive-volume}.

\Needspace{8\baselineskip}
Decrease $s$ further so that \cref{thm:modified-comparison} applies and
$(1-4B_0n^2s\varepsilon^{-3})^n(1-\varepsilon)^n>c$. Write the first factor as $q^n$. Positivity of every mixed term gives
\[
 \omega_{(1-\varepsilon)v}^n
 \geq(1-\varepsilon)^n\omega_v^n+\varepsilon^n\omega^n.
\]
Modified comparison and the assumption would therefore imply
\[
 q^n(1-\varepsilon)^n\int_U\omega_v^n
       +q^n\varepsilon^n\int_U\omega^n
 \leq\int_U\omega_u^n\leq c\int_U\omega_v^n.
\]
All integrals are finite, so the choice $q^n(1-\varepsilon)^n>c$ and the positive volume of $U$ make this impossible. Hence $m\geq0$.
\end{proof}

\Needspace{8\baselineskip}
\begin{corollary}\label{cor:bounded-mass-positive}
Every bounded $\omega$-plurisubharmonic function has strictly positive Monge--Amp\`ere mass.
\end{corollary}
\begin{proof}
Let $u$ be a bounded $\omega$-plurisubharmonic function. If $\omega_u^n=0$, then \cref{thm:bounded-strict-domination}, applied with $c=0$ and the constant potential $v=\sup_Xu+1$, would give $u\geq v$, a contradiction.
\end{proof}

The positive lower bound in \cref{cor:bounded-mass-positive} depends on the potential. A bound that is uniform over all bounded potentials is the condition $v_-(\omega)>0$ of \cref{def:bmp}.

\Needspace{8\baselineskip}
\begin{corollary}\label{cor:bounded-exponential-comparison}\index{Comparison principle!exponential}
If $u,v\in\PSH(X,\omega)\cap L^\infty(X)$ and $\lambda>0$, then
\[
 \mathrm{e}^{-\lambda v}\omega_v^n\geq \mathrm{e}^{-\lambda u}\omega_u^n
 \quad\Longrightarrow\quad v\leq u.
\]
\end{corollary}
\begin{proof}
For every $t>0$, the assumption gives
$\omega_u^n\leq \mathrm{e}^{-\lambda t}\omega_v^n$ on $\{u<v-t\}$. Since $\omega_{v-t}=\omega_v$, \cref{thm:bounded-strict-domination}, applied to $u$ and $v-t$, gives $u\geq v-t$. Letting $t\downarrow0$ proves $v\leq u$.
\end{proof}
