\subsection{A family with nonanalytic isolated poles}
\label{sec:curved-nonanalytic-references}
\index{Singularity type!nonanalytic isolated pole}

The hypothesis of a common simplex allows ordinary singularity types that contain no potential with analytic singularities. Let $o=[1:0:\cdots:0]$ be the origin of the affine chart $Z_0\ne0$, with affine coordinates $z_i=Z_i/Z_0$. For $0<a\leq1$, define
\[
 P_a=\{p\in\Sigma:\textstyle\sum_i\sqrt{p_i}\geq\sqrt a\},
 \qquad
 \kappa_n=\frac{2^n n!}{(2n)!}=\frac1{(2n-1)!!}.
\]

\begin{proposition}\label{prop:curved-nonanalytic-reference}
The function $\Phi_{P_a}$ is locally bounded outside the single point $o$, where it has a pole, and it does not have the ordinary singularity type of any function with analytic singularities. Its generic Lelong number along every prime divisor of $\mathbb P^n$ is zero, and
\begin{equation}\label{eq:curved-nonanalytic-mass}
 \int_{\mathbb P^n}\mathrm{MA}_{\omega_{\mathrm{FS}}}(\Phi_{P_a})
 =1-\kappa_n a^n>0.
\end{equation}
The same assertions about the pole, nonanalyticity and generic Lelong numbers hold for $\pi^*\Phi_{P_a}$ on the blowup $\operatorname{Bl}_q\mathbb P^n$ at a fixed point $q\ne o$.
\end{proposition}
\begin{proof}
The function $p\mapsto\sum_i\sqrt{p_i}$ is concave, so $P_a$ is convex. Set
\[
 b=\frac{n+1}{2n},\qquad p_{*,i}=\frac bn,
 \qquad\delta=\frac{n-1}{4n}.
\]
Then $b+\delta<1$ and $\sum_i\sqrt{p_{*,i}}=\sqrt{(n+1)/2}>1$. Adding a point of $\delta\Sigma$ preserves the last inequality and keeps the sum of the coordinates below one, so $p_*+\delta\Sigma$ lies in every $P_a$.

All the standard basis vectors $e_i$ belong to $P_a$, and therefore
\[
 \psi_{P_a}\geq\log\frac{\max_{1\leq i\leq n}|Z_i|^2}
                                  {\sum_{i=0}^n|Z_i|^2}.
\]
Hence the reference is locally bounded away from $o$. The Cauchy--Schwarz inequality gives $\sum_ip_i\geq a/n$ on $P_a$. When all $x_i<0$, this yields $h_{P_a}(x)\leq(a/n)\max_i x_i$, which tends to $-\infty$ as $z\to o$, and \cref{lem:convex-body-model} gives the pole at $o$. Local boundedness outside one point implies that all generic Lelong numbers along prime divisors vanish, and pulling back to the blowup at a different point introduces no pole.

To test analyticity, fix positive $\lambda_1,\ldots,\lambda_n$ and approach $o$ along $|z_i|^2=\mathrm{e}^{-R\lambda_i}$, $R\to\infty$. Writing $y_i=\sqrt{p_i}$, the Cauchy--Schwarz inequality gives
\begin{equation}\label{eq:curved-directional-coefficient}
 \min_{p\in P_a}\sum_i\lambda_i p_i
 =\frac a{\sum_i\lambda_i^{-1}}=:d_a(\lambda),\qquad
 y_i=\frac{\sqrt a}{\lambda_i\sum_k\lambda_k^{-1}},
\end{equation}
where the minimizer satisfies $\sum_ip_i\leq a\leq1$, so that it lies in $P_a$. Since $L(-R\lambda)\to0$ and $\Phi_{P_a}-\psi_{P_a}$ is bounded, $-\Phi_{P_a}/R\to d_a(\lambda)$ along these curves.

Suppose that the type were that of $c\log\sum_k|f_k|^2$ near $o$, with $c>0$ and holomorphic germs $f_k=\sum_\alpha c_{k,\alpha}z^\alpha$. Exponentiating the two inequalities expressing this equivalence and averaging over the arguments of the $z_i$ gives the positive series
\[
 \sum_{k,\alpha}|c_{k,\alpha}|^2\mathrm{e}^{-R\alpha\cdot\lambda},
\]
whose exponential decay rate in $R$ must be $d_a(\lambda)/c$. That rate is the minimum of $\alpha\cdot\lambda$ over the monomials with nonzero coefficients: a minimizing term gives one bound, and convergence at a fixed $R_0$ bounds the series above by a constant times $\mathrm{e}^{-(R-R_0)\min\alpha\cdot\lambda}$. Near any $\lambda$ with positive coordinates, only finitely many monomials can attain the minimum, because these coordinates have a common positive lower bound and any one nonzero coefficient bounds the degree of the minimizing monomials. The rate is therefore locally piecewise linear in $\lambda$. But
\[
 d_a(s,1,\ldots,1)=\frac{as}{1+(n-1)s},\qquad
 \frac{\mathrm{d}^2}{\mathrm{d}s^2}d_a(s,1,\ldots,1)
 =-\frac{2a(n-1)}{(1+(n-1)s)^3}<0.
\]
This contradiction proves nonanalyticity. The assertion persists on a blowup at a point different from $o$, which does not change a neighborhood of the pole.

The removed corner $C_a=\{p\in\mathbb R^n:p_i\geq0,\ \sum_i\sqrt{p_i}<\sqrt a\}=aC_1$ lies in $\Sigma$. The substitution $p_i=y_i^2$ gives
\[
 \operatorname{Vol}_n(C_1)
 =2^n\int_{y_i\geq0,\,\sum y_i<1}\prod_i y_i\,\mathrm{d}y
 =\frac{2^n}{(2n)!}.
\]
The last equality follows by iterated integration of monomials over a simplex. Removing the corner from $\Sigma$, whose volume is $1/n!$, and applying \eqref{eq:convex-body-mass} gives \eqref{eq:curved-nonanalytic-mass}. Since $n\geq2$, one has $\kappa_n<1$.
\end{proof}

\begin{corollary}\label{cor:curved-reference-stability}
Put $a_j=1-1/(j+2)$ and $r_j=1+1/(j+2)$ for $j\geq1$. On $\mathbb P^n$, or on its blowup at a fixed point $q\ne o$, the references $r_j\pi^*\Phi_{P_{a_j}}$ satisfy the hypotheses of \cref{thm:convex-body-reference-stability}, with limit $\pi^*\Phi_{P_1}$. Their ordinary types and masses vary with $j$, and
\[
 \sup_G|B_j(G)-B(G)|
 \leq n|\log r_j|+\frac{n\kappa_n}{1-\kappa_n}|1-a_j|\longrightarrow0.
\]
The equation is solvable in the limiting ordinary type, whose mass is $1-\kappa_n$; for $n=2$ this mass is $2/3$.
\end{corollary}
\begin{proof}
If $p\in P_{a_j}\setminus P_1$, then $u=(\sum_i\sqrt{p_i})^2\in[a_j,1)$ and $p/u\in P_1$. Since $|p|\leq\sum_ip_i\leq u$, one has $|p/u-p|\leq1-u\leq1-a_j$. Thus $d_H(P_{a_j},P_1)\leq1-a_j$, and the hypotheses of the theorem hold with the simplex found in the proof of \cref{prop:curved-nonanalytic-reference}. The mass formula \eqref{eq:curved-nonanalytic-mass} and the derivative bound $|\mathrm{d}\log(1-\kappa_n a^n)/\mathrm{d}a|\leq n\kappa_n/(1-\kappa_n)$ give the estimate for the functionals. By \eqref{eq:curved-directional-coefficient}, the directional coefficients of the $j$th reference are multiplied by $r_ja_j=1-(j+2)^{-2}$, which strictly increases with $j$, so the ordinary types are distinct, while the masses $r_j^n(1-\kappa_n a_j^n)$ strictly decrease to $1-\kappa_n>0$.
\end{proof}
