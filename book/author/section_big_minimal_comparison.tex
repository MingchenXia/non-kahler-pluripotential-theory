\section{Comparison above a strict singular barrier}
\label{sec:big-minimal-comparison}

From now on, $\theta$ is an arbitrary smooth big real $(1,1)$-form. Then $V_\theta$ may be unbounded, and the smaller-background construction used above is no longer available. Its role is taken by a strict analytic barrier. When two potentials bounded below by the barrier are compared, adding a small multiple of the barrier forces a negative infimum of their difference to be attained away from its pole set, and there a local comparison principle rules it out.

Since $\theta$ is big, \cref{thm:resolvable-regularization-input} provides $\rho\in\PSH(X,\theta)$ with analytic singularities and a smooth remainder after resolution such that $\theta_\rho\geq\kappa\omega$ for some $\kappa>0$. Replacing $\omega$ by a small positive multiple and subtracting a constant from $\rho$, we fix such a $\rho$ with $\rho\leq0$ and $\theta_\rho\geq\omega$. Put $Z=\{\rho=-\infty\}$ and $\Omega=X\setminus Z$. From now on until the end of \cref{sec:nonclosed-interior-smoothness}, these data remain fixed, and $\Omega$ always denotes this open set rather than an auxiliary Hermitian form. A potential bounded below by $\rho-C$ is locally bounded on $\Omega$, and there its extended Monge--Amp\`ere measure agrees with the local bounded product.

\begin{lemma}\label{lem:big-local-comparison}
In this lemma, $\theta$ may be any smooth real $(1,1)$-form on $X$. Let $u$ and $v$ be bounded $\theta$-plurisubharmonic functions on a coordinate ball $B\subset X$, and assume
\[
 \liminf_{x\to\partial B}(u-v)(x)\geq0,
 \qquad \mathbf1_{\{u<v\}}\mathrm{MA}_\theta(u)\leq
             \mathbf1_{\{u<v\}}\mathrm{MA}_\theta(v).
\]
Then $u\geq v$ on $B$.
\end{lemma}
\begin{proof}
Fix $\varepsilon>0$ and set $v_\varepsilon=\max(u,v-\varepsilon)$, which equals $u$ near the boundary. We claim that
\[
 \mathrm{MA}_\theta(u)\leq\mathrm{MA}_\theta(v_\varepsilon)
\]
throughout $B$. On $\{u>v-\varepsilon\}$ this follows from locality, since $v_\varepsilon=u$ there. On $\{u\leq v-\varepsilon\}$ it follows from ordered contact and from the assumed inequality, which applies because this set is contained in $\{u<v\}$.

Let $q$ be the plurisubharmonic envelope on $B$ of the obstacle $u-v_\varepsilon$, in the sense of \eqref{eq:local-envelope-contact}. This obstacle is bounded, quasi-continuous and zero near the boundary. Hence it lies between $Ah$ and zero for a sufficiently large $A$, where $h<0$ is a smooth strictly plurisubharmonic defining function for the ball. Thus $q$ tends to zero at the boundary, and \eqref{eq:local-envelope-contact} shows that $(\ddc q)^n$ is carried by $D=\{q=u-v_\varepsilon\}$.

Since $q\leq u-v_\varepsilon$ quasi-everywhere, the canonical representatives satisfy $q+v_\varepsilon\leq u$. Ordered contact, followed by the measure inequality above, yields
\[
 \mathbf1_D\mathrm{MA}_\theta(q+v_\varepsilon)
 \leq\mathbf1_D\mathrm{MA}_\theta(u)
 \leq\mathbf1_D\mathrm{MA}_\theta(v_\varepsilon).
\]
The positive expansion of $(\theta_{v_\varepsilon}+\ddc q)^n$ contains both $\mathrm{MA}_\theta(v_\varepsilon)$ and $(\ddc q)^n$. Thus $\mathbf1_D(\ddc q)^n=0$, and since $(\ddc q)^n$ is carried by $D$, we get $(\ddc q)^n=0$. The classical Bedford--Taylor comparison principle for bounded plurisubharmonic functions with zero boundary values implies $q=0$. Consequently $u\geq v_\varepsilon$, first quasi-everywhere and then everywhere. Finally let $\varepsilon\downarrow0$.
\end{proof}

\begin{theorem}\label{thm:big-barrier-domination}\index{Domination principle!strict analytic barrier}
Suppose $u,v\in\PSH(X,\theta)$ satisfy, for constants $C_0,C_1\geq0$,
\[
 \min(u,v)\geq\rho-C_0,\qquad v\leq u+C_1.
\]
If, for some $0\leq c<1$,
\[
 \mathbf1_{\{u<v\}}\mathrm{MA}_\theta(u)\leq
 c\,\mathbf1_{\{u<v\}}\mathrm{MA}_\theta(v),
\]
then $u\geq v$.
\end{theorem}
\begin{proof}
We first modify the barrier. Let $r=a\rho-A$, where $a>1$ is sufficiently close to one and $A$ is sufficiently large. Since
\[
 \theta+\ddc(a\rho)=a\theta_\rho+(1-a)\theta,
\]
we can arrange $\theta_r\geq\omega/2$ and $r-v\leq-1$, and moreover $u-r\to+\infty$ at $Z$, because $u-r\geq(a-1)(-\rho)+A-C_0$.

Choose $0<\delta<1$ with $(1-\delta)^n>c$, and put $v_\delta=(1-\delta)v+\delta r$. Then
\[
 u-v_\delta\geq\delta(u-r)-(1-\delta)C_1.
\]
Suppose that the infimum $m$ of $u-v_\delta$ on $\Omega$ were negative. Since $u-v_\delta\to+\infty$ at $Z$, a sequence $x_j$ along which $u-v_\delta$ tends to $m$ has a subsequence converging to a point $x_0\in\Omega$. Choose a coordinate ball $B\Subset\Omega$ centered at $x_0$, identify it with $\{|z|<1\}$, and take
\[
 \eta(z)=b(1-|z|^2),\qquad 0<b<1,
 \qquad \omega/2+\ddc\eta\geq0.
\]
The function $w=v_\delta+\delta\eta+m$ is $\theta$-plurisubharmonic on $B$, and $\liminf_{\partial B}(u-w)\geq0$ because $\eta$ vanishes on $\partial B$. Also
\[
 w-v=\delta(r-v+\eta)+m<0.
\]
Thus $\{u<w\}\subset\{u<v\}$, and positivity of mixed products gives on this set
\[
 \mathrm{MA}_\theta(u)\leq c\mathrm{MA}_\theta(v)\leq(1-\delta)^n\mathrm{MA}_\theta(v)\leq\mathrm{MA}_\theta(w).
\]
By \cref{lem:big-local-comparison}, $u\geq w$ on $B$. Evaluating at $x_j$ gives $u(x_j)-v_\delta(x_j)\geq m+\delta\eta(x_j)$, which contradicts $\eta(x_0)>0$ in the limit. Hence $u\geq v_\delta$ on $\Omega$. Letting $\delta\downarrow0$ along values with $(1-\delta)^n>c$, we obtain $u\geq v$ on $\Omega$, and \cref{lem:qpsh-almost-everywhere-order} extends this inequality to $X$.
\end{proof}

\Needspace{20\baselineskip}
\begin{corollary}\label{cor:big-barrier-orders}
Let $u,v\in\PSH(X,\theta)$ satisfy $u,v\geq\rho-C$ for some $C\geq0$ and $u\sim v$.
\begin{enumerate}
\item If $\mathrm{MA}_\theta(u)\leq c\mathrm{MA}_\theta(v)$ for some $c\geq0$, then $c\geq1$.
\item If $\mathrm{MA}_\theta(u)\leq \mathrm{e}^{\lambda(u-v)}\mathrm{MA}_\theta(v)$ for some $\lambda>0$, then $u\geq v$.
\end{enumerate}
In particular, let $\mu$ be a positive measure. Among potentials bounded below by $\rho$ minus a constant, each ordinary singularity type contains at most one solution of $\mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}\mu$, and the constant $c>0$ in $\mathrm{MA}_\theta(u)=c\mu$ is uniquely determined within each type.
\end{corollary}
\begin{proof}
If $c<1$, apply the theorem to $u$ and $v+A$ for every constant $A>0$; the constants in its hypotheses may depend on $A$. The resulting inequality $u\geq v+A$ for all $A$ is impossible. For the second assertion, fix $a>0$ and restrict the measure inequality to $\{u<v-a\}$. There the factor is at most $\mathrm{e}^{-\lambda a}<1$, so the theorem gives $u\geq v-a$; now let $a\downarrow0$. For two solutions of the exponential equation, apply the second assertion in both orders; for two solutions of the normalized equation, apply the first.
\end{proof}

\section{Subsolutions and the Gauduchon trace}
\label{sec:big-minimal-estimates}

The strict barrier also yields subsolutions within a bounded distance of any reference between $\rho$ and zero. The curvature estimate comes from composing a convex function with the difference between the reference and a barrier-based subsolution, which produces a convex combination of positive currents. Pairing the equation with a Gauduchon metric then bounds the normalizing constants through a linear trace.

\begin{lemma}\label{lem:big-reference-subsolution}
Fix $p>1$ and $L>0$. There is $m>0$ such that, whenever
\[
 \rho\leq\eta\leq0,\quad\eta\in\PSH(X,\theta),
 \qquad g\geq0,\quad\|g\|_{L^p(\mathrm{d}V)}\leq L,
\]
there exists $v\in\PSH(X,\theta)$ with
\[
 \eta\leq v\leq\eta+1,\qquad
 \mathrm{MA}_\theta(v)\geq m g\,\mathrm{d}V.
\]
The constant $m$ depends only on the fixed geometry and on $\rho,p,L$.
\end{lemma}
\begin{proof}
Choose $1<q<p$. By H\"older's inequality and the finiteness of all polynomial moments of the fixed quasi-plurisubharmonic function $\rho$, the functions $h=g(2-\rho)^{2n}$ have uniformly bounded $L^q$ norms. \Cref{lem:big-lp-subsolution}, applied in the Hermitian background $\omega$ with the constant strict barrier $-1$, gives $s\in\PSH(X,\omega)$ with $-1\leq s\leq0$ and
\[
 \omega_s^n\geq m h\,\mathrm{d}V
\]
for a fixed $m>0$. Put $u_1=\rho+s$. Then $\theta_{u_1}\geq\omega_s$ and $u_1\leq\rho\leq\eta$. On $\Omega$ define
\[
 t=1+\eta-u_1,\qquad v=\eta+t^{-1},
\]
so that $1\leq t\leq2-\rho$. \Cref{lem:bounded-convex-pair}, applied on relatively compact subsets of $\Omega$ with $a=\eta$, $h=u_1$ and $\chi(x)=(1-x)^{-1}$, gives
\[
 \theta_v=(1-t^{-2})\theta_\eta+t^{-2}\theta_{u_1}
       +2t^{-3}\,\mathrm{d}(\eta-u_1)\wedge\dc(\eta-u_1)
 \geq t^{-2}\theta_{u_1}.
\]
Since $\eta\leq v\leq\eta+1$, \cref{lem:qpsh-removable-analytic} extends $v$ across $Z$, and the bounds extend by \cref{lem:qpsh-almost-everywhere-order}. On $\Omega$, positivity of the local products gives
\[
 \mathrm{MA}_\theta(v)\geq t^{-2n}\theta_{u_1}^n
       \geq(2-\rho)^{-2n}\omega_s^n
       \geq m g\,\mathrm{d}V.
\]
Neither side charges $Z$, so the inequality holds on $X$.
\end{proof}

\begin{proposition}\label{prop:big-singular-gauduchon-trace}\index{Gauduchon trace!singular potentials}
Let $\gamma$ be a Gauduchon metric on $X$, write $\gamma^n=G\,\mathrm{d}V$, and put
\[
 T_\gamma=\int_X\theta\wedge\gamma^{n-1}.
\]
Then $T_\gamma>0$. If $u\in\PSH(X,\theta)$ satisfies $u\geq\rho-C$ and $\mathrm{MA}_\theta(u)\geq F\,\mathrm{d}V$ with $F\geq0$, then
\begin{equation}\label{eq:big-singular-gauduchon-trace}
 \int_X F^{1/n}G^{1-1/n}\,\mathrm{d}V\leq T_\gamma.
\end{equation}
Consequently, let $0\leq f\in L^p(\mathrm{d}V)$, $p>1$, be nonzero, and let $\lambda>0$. Every solution of $\mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}f\,\mathrm{d}V$ that is bounded below by $\rho$ minus a constant satisfies $\sup_Xu\leq C'$, where $C'$ depends only on $\theta$, $\gamma$, $f$ and $\lambda$. Moreover, if $v_j\in\PSH(X,\theta)$, $j\geq1$, and constants $C_j\geq0$ and $c_j>0$ satisfy
\[
 \sup_Xv_j=0,\qquad
 \mathrm{MA}_\theta(v_j)=c_j \mathrm{e}^{v_j/j}f\,\mathrm{d}V,
 \qquad v_j\geq\rho-C_j,
\]
then the constants $c_j$ are uniformly bounded above.
\end{proposition}
\begin{proof}
The current $\theta_u\wedge\gamma^{n-1}$ is linear in $\theta_u$, so it is a globally defined positive current, even at $Z$. Since $\ddc\gamma^{n-1}=0$, Stokes' theorem in the sense of currents gives
\[
 \int_X\theta_u\wedge\gamma^{n-1}
 =\int_X\theta\wedge\gamma^{n-1}.
\]
Applied to $\rho$, this identity proves $T_\gamma\geq\int_X\omega\wedge\gamma^{n-1}>0$.

On $\Omega$, \cref{lem:interior-mixed-background}, applied to $\theta_u$ and $\gamma$, gives
\[
 \theta_u\wedge\gamma^{n-1}\geq
 F^{1/n}G^{1-1/n}\,\mathrm{d}V.
\]
Exhausting $\Omega$ and using positivity of the global trace measure, we obtain \eqref{eq:big-singular-gauduchon-trace}, since the complement $Z$ has zero volume.

\Needspace{5\baselineskip}
For the exponential equation, write $a=\sup_Xu$ and $w=u-a$. The estimate becomes
\[
 \mathrm{e}^{\lambda a/n}\int_X\mathrm{e}^{\lambda w/n}f^{1/n}G^{1-1/n}\,\mathrm{d}V
 \leq T_\gamma.
\]
The integral has a positive lower bound over all normalized $w\in\PSH(X,\theta)$. Otherwise, normalized compactness would give a sequence of such $w$ with integrals tending to zero and converging almost everywhere to a $\theta$-plurisubharmonic function; since $w\leq0$ and $f^{1/n}G^{1-1/n}$ is integrable, dominated convergence would give a strictly positive limit integral, a contradiction. This bounds $a$ from above.

For the last assertion, the same estimate contains the factors $c_j^{1/n}$ and $\mathrm{e}^{v_j/(jn)}$. If the integrals $\int_X\mathrm{e}^{v_j/(jn)}f^{1/n}G^{1-1/n}\,\mathrm{d}V$ tended to zero along a subsequence with $j\to\infty$, normalized compactness and almost-everywhere convergence would give $v_j/j\to0$ almost everywhere along a further subsequence. Dominated convergence would then give the positive limit $\int_Xf^{1/n}G^{1-1/n}\,\mathrm{d}V$. This contradiction bounds $c_j$ for all large $j$, and each of the finitely many remaining constants is finite. Neither argument uses the total Monge--Amp\`ere mass.
\end{proof}
