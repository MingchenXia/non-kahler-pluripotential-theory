\begin{proof}
We compute the Ricci tensor without assuming that $\lambda$ is co-closed, and compare the integrated Bianchi and Bochner identities; the Gauduchon condition enters only in Step~3. All norms, traces and contractions use $g$. Put
\[
 B=JA,\qquad S=\nabla\lambda,\qquad
 S^+(U,V)=\tfrac12\bigl(S(U,V)+S(JU,JV)\bigr),\qquad
 u=|\lambda|_g^2.
\]
Closedness of $\lambda$ makes $S$ symmetric, and $\operatorname{tr}_gS=-q$. The tensors $S^+$ and $S^-=S-S^+$ are pointwise orthogonal. Write $J\lambda=-\lambda\circ J$, so that $\lambda^{\mathrm c}=J\lambda/(4\pi)$.

\emph{Step 1: the Ricci tensor.} We first show that
\begin{equation}\label{eq:lck-ricci-real}
 \operatorname{Ric}_g
 =nS^+-(n-1)S+\tfrac{q-u}{2}g
       +\tfrac12\lambda\otimes\lambda
       +\tfrac n2 J\lambda\otimes J\lambda.
\end{equation}
Locally write $\lambda=\mathrm df$ and put $\widetilde g=\mathrm e^{-f}g$. The formula for the Ricci tensor under a conformal change in real dimension $2n$ gives
\[
 \operatorname{Ric}_{\widetilde g}
 =\operatorname{Ric}_g+(n-1)S
       +\tfrac{n-1}{2}\lambda\otimes\lambda
       -\tfrac12\bigl(q+(n-1)u\bigr)g.
\]
The local metric $\widetilde g$ is K\"ahler. Since the Chern--Ricci form of $g$ vanishes, the Ricci form of $\widetilde g$, normalized as $\operatorname{Ric}$, is $n\mathrm i\partial\bar\partial f=(n/2)\mathrm d(J\lambda)$.
Applying \eqref{eq:lck-complex-connection} to
$(\nabla_UJ\lambda)(V)=-S(U,JV)-\lambda((\nabla_UJ)V)$ and antisymmetrizing gives
\[
 \mathrm d(J\lambda)(U,JV)
 =2S^+(U,V)-u g(U,V)
             +\lambda(U)\lambda(V)+J\lambda(U)J\lambda(V).
\]
Evaluating the K\"ahler Ricci form on $(U,JV)$ and comparing with the conformal formula proves \eqref{eq:lck-ricci-real}. Taking its trace gives the scalar curvature
\[
 s=(n-1)q-\tfrac{n-1}{2}u.
\]

\emph{Step 2: the integral identity.} Write $\|\cdot\|$ for the $L^2(g)$ norm and set
\[
 E_A=\int_X S(A,A)\,\mathrm d\operatorname{vol}_g,
 \qquad E_B=\int_X S(B,B)\,\mathrm d\operatorname{vol}_g.
\]
Since $S(A,A)=A(u)/2$ and $\operatorname{div}A=-q$, integration by parts gives
\begin{equation}\label{eq:lck-aa-divergence}
 \int_Xu q\,\mathrm d\operatorname{vol}_g=2E_A.
\end{equation}
Pair the contracted Bianchi identity $\delta\operatorname{Ric}_g+\mathrm ds/2=0$ with $\lambda$ and integrate. Here $\delta$, minus the divergence, is the formal adjoint of the symmetrized covariant derivative. Thus
\[
 0=\int_X\langle\operatorname{Ric}_g,S\rangle\,\mathrm d\operatorname{vol}_g
       +\tfrac12\int_Xs q\,\mathrm d\operatorname{vol}_g.
\]
Substituting \eqref{eq:lck-ricci-real} and using $\langle S^+,S\rangle=|S^+|^2$, $\operatorname{tr}_gS=-q$, $(J\lambda)^\sharp=B$ and \eqref{eq:lck-aa-divergence}, we obtain
\begin{equation}\label{eq:lck-bianchi-integral}
 0=n\|S^+\|^2-(n-1)\|S\|^2+\tfrac{n-2}{2}\|q\|^2
       +\tfrac{4-n}{2}E_A+\tfrac n2E_B.
\end{equation}
For the closed one-form $\lambda$, the integrated Bochner formula reads
\[
 \|q\|^2=\|S\|^2
          +\int_X\operatorname{Ric}_g(A,A)\,\mathrm d\operatorname{vol}_g.
\]
The quartic terms in \eqref{eq:lck-ricci-real} cancel on $(A,A)$, and $J\lambda(A)=0$, so
\[
 \operatorname{Ric}_g(A,A)
 =(1-\tfrac n2)S(A,A)+\tfrac n2 S(B,B)+\tfrac12q u.
\]
Using \eqref{eq:lck-aa-divergence}, we obtain
\begin{equation}\label{eq:lck-bochner-integral}
 \|q\|^2=\|S\|^2+\tfrac{4-n}{2}E_A+\tfrac n2E_B.
\end{equation}
Substituting the value of $\tfrac{4-n}{2}E_A+\tfrac n2E_B$ given by \eqref{eq:lck-bochner-integral} into \eqref{eq:lck-bianchi-integral} gives
$n(\|S^+\|^2-\|S\|^2)+n\|q\|^2/2=0$. Since $\|S\|^2=\|S^+\|^2+\|S^-\|^2$ by orthogonality, this is the asserted identity for $S^-$.

\emph{Step 3: rigidity.} By \cref{prop:lck-gauduchon}, the Gauduchon condition is equivalent to $q=0$. If $q=0$, the integral identity forces $S=S^+$, hence $S(B,B)=S(JA,JA)=S(A,A)$. Equation \eqref{eq:lck-aa-divergence} gives $E_A=E_B=0$, and \eqref{eq:lck-bochner-integral} implies $S=0$. Since $\omega$ is strictly LCK, it is then Vaisman.

Equation \eqref{eq:lck-complex-connection} gives $\nabla_AJ=0$. Let $T$ be the symmetric endomorphism defined by $g(TU,V)=S(U,V)$, so that $\nabla_UA=TU$. Since $\nabla$ is torsion free,
\[
 (\mathcal L_AJ)V=JTV-TJV.
\]
Consequently $A$ is real holomorphic exactly when $T$ commutes with $J$, that is, when $S=S^+$. In that case the integral identity gives $q=0$, hence $S=0$ by the previous paragraph. Conversely, if $\omega$ is Vaisman, then $S=0$, so $q=-\operatorname{tr}_gS=0$ and $\mathcal L_AJ=0$.
\end{proof}
