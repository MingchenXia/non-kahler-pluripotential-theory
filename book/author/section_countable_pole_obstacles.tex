\subsection{Countable poles and all obstacles}
\label{subsec:countable-pole-obstacles}\index{Pole set!countable dense poles}\index{Envelope!countable poles}

If a potential $w$ is at least as singular as every multiple $bu$, $b\geq1$, of a potential $u$, then convex combinations with $w$ supply ordinary competitors for $u$ below every continuous obstacle; this is the mechanism of the single-pole example in the proof of \cref{prop:all-obstacle-uniformity}. We construct such a pair with poles on any prescribed countable set. The potentials have trivial multiplier ideals even when their poles are dense, so both envelopes lose this ordinary singularity information.

We keep the Hermitian form $\omega$ and the volume form $\mathrm dV$. Envelopes below a continuous obstacle $F$ are defined by the competitor families of \cref{def:singularity-envelope,def:hermitian-i-envelope} with the bound $v\leq F$.

\Needspace{14\baselineskip}
\begin{theorem}\label{thm:countable-pole-stretching}\index{Relative envelope class!countable poles}
For every nonempty finite or countably infinite subset $\{x_j\}\subset X$, there are nonpositive $u,w\in\PSH(X,\omega)$ such that
\[
 \omega+\ddc u,\ \omega+\ddc w\geq\tfrac34\omega,
 \qquad u(x_j)=w(x_j)=-\infty.
\]
The same $w$ satisfies $tw\preceq u$ for every $0<t\leq1$, and
\[
 u\in\mathcal E_{\mathrm{env}}(X,\omega,0),\qquad
 \mathrm e^{-su},\ \mathrm e^{-sw}\in L^1(X,\mathrm dV)
 \quad(s>0).
\]
Thus $\mathcal I(su)=\mathcal I(sw)=\mathcal O_X$ for every $s>0$. If the prescribed set is dense, both potentials are unbounded on every nonempty open set.
\end{theorem}
\begin{proof}
Fix $0<\beta<\beta'<1$, and index the prescribed points by integers $j\geq1$. Choose a finite atlas of relatively compact charts with larger coordinate neighborhoods, on which the coefficients of $\mathrm dV$ are uniformly bounded. For each $j$, let $z_j$ be the coordinate of $x_j$ in a chart of the atlas containing it, and choose a coordinate ball of radius $r_j\leq1$ centered at $z_j$ and contained in the corresponding larger neighborhood. Let $\chi$ be a smooth cutoff supported inside the unit ball, with $0\leq\chi\leq1$ and $\chi=1$ near zero. On this ball set
\[
 \chi_j(z)=\chi((z-z_j)/r_j),\qquad
 L_j=1-\log(|z-z_j|^2/r_j^2).
\]
The functions $-\chi_jL_j^\sigma$, for $\sigma=\beta,\beta'$, extend by zero outside the ball. Near the center, where $\chi_j=1$, they are plurisubharmonic off $z_j$: for $s=\log(|z-z_j|^2/r_j^2)$ the profile $q_\sigma(s)=-(1-s)^\sigma$ is convex and increasing, since
\[
 q_\sigma'=\sigma(1-s)^{\sigma-1}>0,
 \qquad q_\sigma''=\sigma(1-\sigma)(1-s)^{\sigma-2}>0.
\]
Being nonpositive, they extend across $z_j$ with the value $-\infty$ by \cref{lem:qpsh-removable-analytic}. On the annulus where the cutoff varies they are smooth, so their complex Hessians have a finite lower bound.

Put $e_j=2^{-j-2}$. Scale both profiles by the same $0<\delta_j\leq2^{-j}$ so that the nonpositive functions
\begin{equation}\label{eq:countable-pole-terms}
 a_j=-\delta_j\chi_jL_j^\beta,
 \qquad c_j=-\delta_j\chi_jL_j^{\beta'},
 \qquad \ddc a_j,\ \ddc c_j\geq-e_j\omega
\end{equation}
have these lower bounds globally; only the cutoff annulus contributes a negative Hessian. Coordinate rescaling and the finite atlas give
\[
 \|a_j\|_{L^1}+\|c_j\|_{L^1}\leq C\delta_j.
\]
Consequently the nonpositive partial sums decrease with uniformly bounded $L^1$ norms. Since $\sum_{j\geq1}e_j=\frac14$, the partial sums $\sum_{j\leq N}a_j$ and $\sum_{j\leq N}c_j$ have complex Hessians at least $-\frac14\omega$. By \cref{cor:decreasing-limit}, their pointwise limits
\[
 u=\sum_j a_j,\qquad w=\sum_j c_j
\]
belong to $\PSH(X,\omega)$, and the lower bound $\frac34\omega$ for their currents passes to the limits. Each prescribed point is a pole of its own summand, and the other summands are nonpositive, so $u(x_j)=w(x_j)=-\infty$. This also proves the assertion when the prescribed set is dense.

For $b\geq1$ let
\[
 S=\sum_j\delta_j>0,\qquad
 M_b=\max\{0,\sup_{L\geq1}(bL^\beta-L^{\beta'})\}<\infty.
\]
Multiplying the inequality $-L^{\beta'}\leq-bL^\beta+M_b$, valid for $L\geq1$, at $L=L_j$ by $\delta_j\chi_j\geq0$ and summing over $j$ gives
\begin{equation}\label{eq:countable-fixed-minorant}
 w\leq bu+M_bS.
\end{equation}
The inequality first holds for finite sums and then for their decreasing limits, including common poles. Taking $b=1/t$ gives $tw\preceq u$; here $tw=tw+(1-t)\cdot0$ is $\omega$-plurisubharmonic by convexity, since $\omega>0$. Also, $b^{-1}(w-M_bS)\leq u$, so $w-M_bS$ belongs to the family $\mathcal A_b(u,0)$ of stretching competitors for every $b\geq1$, and \cref{def:relative-envelope-class} gives $u\in\mathcal E_{\mathrm{env}}(X,\omega,0)$.

It remains to prove integrability at every positive exponential scale. Fix $s>0$ and use Jensen's inequality with weights $\delta_j/S$, assigning the unused weight in each finite sum to zero:
\[
 \mathrm e^{-s\sum_{j\leq N}a_j}
 \leq1-\sum_{j\leq N}\frac{\delta_j}{S}
     +\sum_{j\leq N}\frac{\delta_j}{S}
          \mathrm e^{sS\chi_jL_j^\beta}.
\]
The integrals on the right are bounded uniformly in $j$, since coordinate rescaling reduces the contribution inside the ball to a constant times
\[
 \int_0^1\exp\bigl(sS(1-\log R^2)^\beta\bigr)R^{2n-1}\,\mathrm dR.
\]
With $x=-\log R^2$, its exponent is $-nx+sS(1+x)^\beta$, which is at most $-nx/2$ for large $x$. Integrating the Jensen inequality and using monotone convergence proves $\mathrm e^{-su}\in L^1$. The same calculation with $\beta'$ proves the assertion for $w$. Every holomorphic germ is bounded on a smaller coordinate neighborhood, so \cref{def:multiplier-ideal} gives the stated trivial ideals.
\end{proof}

\Needspace{10\baselineskip}
\begin{proposition}\label{prop:countable-pole-all-obstacles}\index{Envelope!all continuous obstacles}
For the potentials of \cref{thm:countable-pole-stretching} and every $F\in C(X,\mathbb R)$,
\[
 P_{\omega,F}[u]=P_{\omega,F}[u]_{\mathcal I}=P_\omega(F).
\]
In particular, $P_\omega[u]=P_\omega[u]_{\mathcal I}=0$.
\end{proposition}
\begin{proof}
The proof of the nonseparation assertion of \cref{prop:all-obstacle-uniformity} uses only that $w\leq0$ is $\omega$-plurisubharmonic with $\delta w\preceq u$ for $0<\delta\leq1$, that $u$ has trivial multiplier ideals, and that $F$ is continuous. By \cref{thm:countable-pole-stretching}, these conditions hold here, so that proof gives $P_{\omega,F}[u]=P_{\omega,F}[0]=P_\omega(F)$ and $P_{\omega,F}[u]_{\mathcal I}=P_{\omega,F}[0]_{\mathcal I}$. The latter equals $P_\omega(F)$ because $P_{\omega,F}[0]\leq P_{\omega,F}[0]_{\mathcal I}\leq P_\omega(F)$. For $F=0$, zero is a competitor, so $P_\omega(0)=0$.
\end{proof}

For a dense prescribed set, the ordinary type of $u$ differs from the bounded type, although their envelopes agree below every continuous obstacle. The construction applies to every Hermitian form $\omega$, including one with unbounded upper mass or zero lower mass.
