\chapter[Relative equations under positive lower mass]{Relative equations under positive lower mass}
\label{chap:relative-finite-measures}

Throughout this chapter $X$ is a connected compact complex manifold of dimension $n$, $\theta$ is a smooth semipositive big real $(1,1)$-form on $X$, and $\varphi\in\PSH(X,\theta)$ satisfies \eqref{eq:als-model-condition}. We assume
\begin{equation}\label{eq:relative-positive-lower-volume}
 v_-(\theta)>0.
\end{equation}
Here $v_-(\theta)$ is the infimum of $\int_X(\theta+\ddc a)^n$ over bounded $a\in\PSH(X,\theta)$. No upper mass bound is imposed. Fix $0<\varepsilon<1$ with $P_{\varepsilon\theta}(\varphi)$ genuine, and write
\[
 \mathcal E_\varphi=\mathcal E_{\mathrm{env}}(X,\theta,\varphi),
 \qquad \mathcal E_0=\mathcal E_{\mathrm{env}}(X,\theta,0),
 \qquad \mu_u=\mathrm{MA}^{\mathrm{ext}}_\theta(u).
\]
The envelope lemmas in \cref{sec:relative-envelope-room} supply rooftops and room below the reference. Here the uniform lower bound prevents collapse of auxiliary envelopes and gives equations for arbitrary finite non-pluripolar measures. Members of $\mathcal E_\varphi$ can still have infinite mass; solutions acquire finite mass from their equation.

\section{The general-measure equations}
\label{sec:relative-general-setting}

Retain the connected compact complex $n$-fold $X$, semipositive big form $\theta$, model reference $\varphi$, parameter $\varepsilon$ and positive lower mass specified above. Thus $\mathcal E_\varphi=\mathcal E_{\mathrm{env}}(X,\theta,\varphi)$, $\mathcal E_0=\mathcal E_{\mathrm{env}}(X,\theta,0)$ and $\mu_a=\mathrm{MA}^{\mathrm{ext}}_\theta(a)$ for $a\in\PSH(X,\theta)$. Fix a Hermitian form $\omega$ on $X$ for capacity and a smooth positive volume form for $L^1$ convergence.

\Needspace{15\baselineskip}
\begin{theorem}
\label{thm:als-general-measure}
Let $\theta$ be semipositive and big, let $\varphi$ satisfy \eqref{eq:als-model-condition}, and assume $v_-(\theta)>0$. If $\mu$ is a nonzero positive Radon measure vanishing on pluripolar sets, then:
\begin{enumerate}
\item For every $\lambda>0$ there is a unique
$u\in\mathcal E_{\mathrm{env}}(X,\theta,\varphi)$ with
$\mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}\mu$.
\item There are $u\in\mathcal E_{\mathrm{env}}(X,\theta,\varphi)$ and a unique $c>0$ with
$\mathrm{MA}_\theta(u)=c\mu$.
\end{enumerate}
No bounded mass assumption is made.
\end{theorem}

These are \cite[Theorem 5.8 and Corollary 5.9]{ALS26}. The minimal-reference case is proved in \cref{thm:general-measure-minimal}, and the full prescribed-reference argument is completed in \cref{sec:general-measure-relative}. The solutions have finite mass: in the first equation $u$ is bounded above on compact $X$, so the right-hand side is finite; in the second this is immediate. This does not assert finite mass for every $\theta$-plurisubharmonic function, or bounded difference $u-\varphi$ for these general data.

\section{Domination with one finite measure}
\label{sec:relative-finite-domination}

Use the data and measure notation of \cref{sec:relative-general-setting}. For potentials $a,w$ and a real number $b\geq1$, write $Q_b(a,w)=P_\theta(ba-(b-1)w)$ whenever this envelope is genuine; differences are interpreted off the pole sets. The relation $a\preceq w$ means $a\leq w+C$ for a real constant $C$.

Set $\kappa=(1-\varepsilon)^n v_-(\theta)>0$. This condition concerns a lower bound. It is compatible with $v_+(\theta)=+\infty$, as the Hopf example shows.

\begin{theorem}\label{thm:relative-finite-domination}
Let $u\in\mathcal E_\varphi$ have finite $\mu_u$, and let $v\in\PSH(X,\theta)$ satisfy $v\preceq\varphi$. If, for some $0\leq c<1$,
\[
 \mu_u\leq c\mu_v\quad\hbox{on }\{u<v\},
\]
then $u\geq v$.
\end{theorem}
\begin{proof}
Replace $v$ by $w=\max(u,v)$. Upward stability gives $w\in\mathcal E_\varphi$ and locality gives $\mu_u\leq c\mu_w$ on $\{u<w\}$. For $b>1$ put $q_b=Q_b(u,w)\in\mathcal E_\varphi$. It satisfies $q_b\leq u$, and its measure is carried by
\[
 D_b=\{b^{-1}q_b+(1-b^{-1})w=u\}.
\]
On $D_b$, positivity and ordered contact give
\[
 b^{-n}\mu_{q_b}+(1-b^{-1})^n\mu_w\leq\mu_u.
\]
Choose $b$ so large that $(1-b^{-1})^n>c$. On $D_b\cap\{u<w\}$ the preceding inequality forces both $\mu_w$ and $\mu_{q_b}$ to vanish. To justify this even when $\mu_w(X)=+\infty$, restrict first to common finite truncation levels of the potentials. On each such set the measures are finite and subtraction is legitimate. Exhausting the complements of the pole sets proves the assertion for the extended measures.

It follows that $\mu_{q_b}$ is carried by $D_b\cap\{u=w\}\subseteq\{q_b=u\}$. Since $q_b\leq u$, another application of ordered contact gives the bound with coefficient one
\begin{equation}\label{eq:relative-contact-fixed-measure}
 \mu_{q_b}\leq\mathbf1_{\{q_b=u\}}\mu_u.
\end{equation}
This coefficient is independent of $b$.

Along integers $b\to\infty$, the potentials $q_b$ decrease. If they collapsed identically to $-\infty$, compactness and Hartogs' lemma would give $M_b:=\sup_Xq_b\to-\infty$. But \eqref{eq:relative-contact-fixed-measure} would then imply
\[
 \kappa\leq\int_X\mu_{q_b}
 \leq\mu_u(\{u\leq M_b\})\longrightarrow0.
\]
Here finiteness of $\mu_u$ and its vanishing on the pole set are exactly what gives the limit. This is impossible. Hence the decreasing limit $q$ is genuine. On $\{u<w-a\}$, for any $a>0$, the defining inequality gives $q_b\leq u-(b-1)a$. Outside the poles its limit is $-\infty$ there. Such a set has smooth volume zero because $q$ is genuine. Thus $u\geq w$ almost everywhere, and then everywhere for canonical representatives. This proves the theorem.
\end{proof}

The theorem reconstructs the finite-$\mu_u$ case of \cite[Theorem 3.14]{ALS26}. Its proof displays the fixed finite measure that prevents collapse. It does not assume that every member of $\mathcal E_\varphi$ has finite mass.

\Needspace{13\baselineskip}
\begin{corollary}\label{cor:relative-finite-exponential-order}
Under the hypotheses on $u,v$ in \cref{thm:relative-finite-domination}, let $\lambda>0$. Then
\[
 \mathrm{e}^{-\lambda u}\mu_u\leq \mathrm{e}^{-\lambda v}\mu_v
 \quad\Longrightarrow\quad u\geq v.
\]
Also $\mu_u\leq c\mu_v$ on $X$ with a finite constant $c\geq0$ implies $c\geq1$.
\end{corollary}
\begin{proof}
For each $a>0$, the exponential inequality gives $\mu_u\leq \mathrm{e}^{-\lambda a}\mu_v$ on $\{u<v-a\}$. Apply the theorem to $v-a$ and then let $a\downarrow0$. The weights are finite and positive off the pole sets, so their cancellation is done there. If the second assertion failed, apply the theorem to $v+A$ for every real constant $A$, since its Monge--Amp\`ere measure is unchanged. The resulting inequalities $u\geq v+A$ contradict finiteness at a common point.
\end{proof}

In particular, two solutions in $\mathcal E_\varphi$ of an exponential equation with a fixed finite non-pluripolar right-hand measure must agree. Their Monge--Amp\`ere measures are finite because their potentials are bounded above. For the equation $\mu_u=c\nu$, the positive scalar is unique whenever two such finite-measure solutions exist. No assertion of uniqueness modulo constants for the zero-exponent potentials follows from this argument.
