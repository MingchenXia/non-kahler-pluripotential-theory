\section{Higher-dimensional Hopf manifolds}
\label{sec:hopf-all-dimensions}

The threefold mass formula locates the source of unbounded mass, but it does not by itself give a construction in higher dimension. The same heat potentials do work there: the radial--vertical block is unchanged, and each extra angular direction contributes a factor whose lower bound is uniform where the threefold mass diverges.

For $n\geq2$, let
\[
 H_n=(\mathbb C^n\setminus\{0\})/
             \langle z\mapsto\mathrm e^{-1}z\rangle,
 \qquad p_n\colon H_n\longrightarrow\mathbb P^{n-1},
\]
with coordinates $z=(z_0,\dots,z_{n-1})$ on $\mathbb C^n$.
Put $s=\log|z|^2$, $\alpha_n=\ddc s=p_n^*\omega_{\FS}$,
$\beta_n=\mathrm ds\wedge\dc s$ and $\omega_n=\alpha_n+\beta_n$, where
$\omega_{\FS}$ is the Fubini--Study form on $\mathbb P^{n-1}$, so that
$\int_{\mathbb P^{n-1}}\omega_{\FS}^{n-1}=1$.
The forms descend and $\omega_n$ is positive for the same reasons as in dimension three. The fiber is the same elliptic curve $E$, with $\int_E\beta_n=2$. The form $\alpha_n$ is closed of rank $n-1$, $\beta_n^2=0$, and $\ddc\beta_n=-\alpha_n^2$ by the computation following \eqref{eq:hopf-torsion}. Hence $\omega_n^n=n\alpha_n^{n-1}\wedge\beta_n$ and
\[
 \int_{H_n}\omega_n^n=2n,\qquad
 \ddc(\omega_n^{n-1})=-(n-1)\alpha_n^n=0.
\]
Thus $\omega_n$ is Gauduchon.

Potentials pulled back from the projective base reduce to the closed manifold $\mathbb P^{n-1}$. For the invariant backgrounds $a\alpha_n+b\beta_n$, positivity of such a potential is decided on the base, also in families over a parameter manifold. Since $\beta_n^2=0$, the vertical form contributes exactly one factor of the Monge--Amp\`ere measure. Later chapters use this reduction for subgeodesics over strips, trace envelopes, descended Monge--Amp\`ere equations and envelope systems.

\Needspace{18\baselineskip}
\begin{lemma}\label{lem:hopf-basic-potentials}\label{thm:hopf-descended-hermitian-ma}\index{Hopf manifold!basic potentials}\index{Hopf manifold!descended equation}
Let $n\geq2$, let $a,b\geq0$ be real numbers, and let $S$ be a complex manifold, possibly a point. Put $P=p_n\times\mathrm{id}_S\colon H_n\times S\to\mathbb P^{n-1}\times S$, and write $\operatorname{pr}$ for the projections of $H_n\times S$ to $H_n$ and of $\mathbb P^{n-1}\times S$ to $\mathbb P^{n-1}$.
\begin{enumerate}
\item For every quasi-plurisubharmonic function $V$ on $\mathbb P^{n-1}\times S$,
\[
 a\operatorname{pr}^*\alpha_n+b\operatorname{pr}^*\beta_n+\ddc P^*V\geq0
 \quad\Longleftrightarrow\quad
 a\operatorname{pr}^*\omega_{\FS}+\ddc V\geq0.
\]
\item A quasi-plurisubharmonic function on $H_n\times S$ that is constant on the fibers of $P$, equivalently invariant under scalar multiplication on the first factor, is $P^*V$ for a unique quasi-plurisubharmonic function $V$ on $\mathbb P^{n-1}\times S$.
\item For a positive Borel measure $\nu$ on $\mathbb P^{n-1}$, let $\beta_n\wedge p_n^*\nu$ denote the measure $F\mapsto\int_{\mathbb P^{n-1}}\bigl(\int_{p_n^{-1}(y)}F\beta_n\bigr)\mathrm d\nu(y)$ on $H_n$. For every $v\in\PSH(\mathbb P^{n-1},a\omega_{\FS})$,
\[
 \mathrm{MA}^{\mathrm{ext}}_{a\alpha_n+b\beta_n}(p_n^*v)
 =nb\,\beta_n\wedge p_n^*\mathrm{MA}^{\mathrm{ext}}_{a\omega_{\FS}}(v).
\]
Consequently, for every nonnegative Borel function $F$ on $\mathbb P^{n-1}$,
\[
 \int_{H_n}p_n^*F\,\mathrm{MA}^{\mathrm{ext}}_{a\alpha_n+b\beta_n}(p_n^*v)
 =2nb\int_{\mathbb P^{n-1}}F\,\mathrm{MA}^{\mathrm{ext}}_{a\omega_{\FS}}(v).
\]
The total masses of $\mathrm{MA}^{\mathrm{ext}}_{a\omega_{\FS}}(v)$ and $\mathrm{MA}^{\mathrm{ext}}_{a\alpha_n+b\beta_n}(p_n^*v)$ are at most $a^{n-1}$ and $2nba^{n-1}$, with equality for bounded $v$. In particular, $\mathrm{MA}^{\mathrm{ext}}_{a\alpha_n}(p_n^*v)=0$.
\end{enumerate}
\end{lemma}
\begin{proof}
(1) In local product coordinates for the submersion $P$, the current $\ddc P^*V$ is the pullback of $\ddc V$. Thus the current on the left is $T=P^*\Theta+b\operatorname{pr}^*\beta_n$, where $\Theta=a\operatorname{pr}^*\omega_{\FS}+\ddc V$. If $\Theta\geq0$, then $P^*\Theta\geq0$, because holomorphic pullback preserves positivity (\cref{lem:smooth-disk-criterion}), and $T\geq0$ because $\beta_n\geq0$. Conversely, suppose that $T\geq0$. Its wedge product with the smooth semipositive form $\operatorname{pr}^*\beta_n$ is a positive current, and $\beta_n^2=0$ gives $T\wedge\operatorname{pr}^*\beta_n=P^*\Theta\wedge\operatorname{pr}^*\beta_n$. Pairing a pulled-back current with a test form amounts to pairing the original current with the fiber integral of the form, as in the proof of \cref{lem:bounded-products-submersion}. The fibers of $P$ are those of $p_n$, on which $\beta_n$ has integral two, so
\[
 P_*\bigl(T\wedge\operatorname{pr}^*\beta_n\bigr)=2\Theta.
\]
The map $P$ is proper, so this push-forward is positive, and $\Theta\geq0$.

(2) Each fiber of $P$ is an orbit of scalar multiplication on the first factor, so the two conditions agree. Since $P$ is surjective, a function $U$ satisfying them is $P^*V$ for a unique function $V$. Near each point of $\mathbb P^{n-1}\times S$, write $V=U\circ\gamma$ for a local holomorphic section $\gamma$ of $P$; as in the second paragraph of the proof of \cref{prop:psh-descent-submersion}, $V$ is quasi-plurisubharmonic, since otherwise $U$ would equal $-\infty$ on a nonempty open set.

(3) By (1), $p_n^*v$ belongs to $\PSH(H_n,a\alpha_n+b\beta_n)$, so its extended measure is defined. Adding a constant to $v$ changes neither side, so we may assume $v\leq0$. The functions $v_j=\max(v,-j)$ are bounded and $a\omega_{\FS}$-plurisubharmonic, and $p_n^*v_j=\max(p_n^*v,-j)$. Put $\nu_j=(a\omega_{\FS}+\ddc v_j)^{n-1}$. By \cref{lem:bounded-products-submersion}, applied with $n-1$ and with $n$ factors,
\[
 (a\alpha_n+\ddc p_n^*v_j)^{n-1}=p_n^*\nu_j,
 \qquad
 (a\alpha_n+\ddc p_n^*v_j)^n=0.
\]
Expanding the $n$th power of $(a\alpha_n+\ddc p_n^*v_j)+b\beta_n$ by multilinearity of bounded products (\cref{prop:bounded-products}), and using $\beta_n^2=0$, we obtain
\[
 (a\alpha_n+b\beta_n+\ddc p_n^*v_j)^n
 =nb\,\beta_n\wedge p_n^*\nu_j.
\]
On the right, the wedge product is taken in local product coordinates. Wedging with the pullback of a top-degree measure on the base keeps only the restriction of $\beta_n$ to the fibers, so this is the measure defined in the statement. Restrict both sides to the set $\{p_n^*v>-j\}=p_n^{-1}\{v>-j\}$. As $j\to\infty$, the measures $\mathbf 1_{\{v>-j\}}\nu_j$ increase to $\mathrm{MA}^{\mathrm{ext}}_{a\omega_{\FS}}(v)$. Monotone convergence in the outer integral defining $\beta_n\wedge p_n^*\nu$ and the definition \eqref{eq:extended-ma} give the measure identity. Since $\beta_n$ has integral two on each fiber, the integral identity follows. Finally, $\nu_j$ is a positive measure of mass $\int_{\mathbb P^{n-1}}(a\omega_{\FS})^{n-1}=a^{n-1}$, by Stokes' theorem for the closed form $\omega_{\FS}$, and $\mathbf 1_{\{v>-j\}}\nu_j\leq\nu_j$. For bounded $v$ and large $j$, nothing is removed.
\end{proof}

For the rest of the section, let $n\geq3$. On the affine chart $\{z_0\neq0\}$ write $\xi=(z_1/z_0,\dots,z_{n-1}/z_0)\in\mathbb C^{n-1}$,
$x=\log|\xi|^2$, $w=\log z_0\pmod\Lambda$ and
$\mu=\mathrm e^x/(1+\mathrm e^x)$ as before. For a smooth real function $V(x,w)$ with the endpoint properties required before \cref{lem:hopf-positivity}, let $J_n(V;x,w)$ be the density obtained by integrating $(\omega_n+\ddc V)^n$ over the angular variables $\xi/|\xi|$ against their rotation-invariant probability measure. Thus
\[
 \int_{H_n}(\omega_n+\ddc V)^n
   =\int_{\mathbb R\times E}J_n(V;x,w)\,\mathrm dx\,\mathrm dA_E.
\]
The endpoints and the complement of the affine chart have zero smooth volume.

\Needspace{12\baselineskip}
\begin{lemma}\label{lem:hopf-dimensional-density}\index{Hopf manifold!dimensional density comparison}
The positivity criterion \eqref{eq:hopf-positivity} holds in every dimension $n\geq3$, and
\begin{equation}\label{eq:hopf-dimensional-density}
 J_n(V;x,w)=\frac{n(n-1)}6
                  (\mu+V_x)^{n-3}J_3(V;x,w).
\end{equation}
Here $J_3$ is evaluated for the same function $V$ on the threefold.
\end{lemma}
\begin{proof}
As in the proof of \cref{lem:hopf-positivity}, choose horizontal vectors forming an orthonormal frame for $2\pi\alpha_n$: $n-2$ angular vectors orthogonal to the radial direction, and the radial vector $e_r$. Adjoin the vertical vector $\partial_w$. In this frame, each angular diagonal entry of the matrix of $2\pi(\omega_n+\ddc V)$ is $A=(\mu+V_x)/\mu$, the off-diagonal entries in the angular rows and columns vanish, and the radial--vertical block is
\[
 K_V=\begin{pmatrix}
 (\mu+V_{xx})/[\mu(1-\mu)]&
       (\mu+V_{x\bar w})/\sqrt{\mu(1-\mu)}\\
 (\mu+V_{wx})/\sqrt{\mu(1-\mu)}&1+V_{w\bar w}
 \end{pmatrix}.
\]
This follows from the formula for $2\pi\alpha$ in the proof of \cref{lem:hopf-positivity}: the form $\mathrm i\partial\bar\partial x$ vanishes in the radial direction and equals $1/\mu$ on each angular unit vector, where $\partial x$ vanishes, so that $\mathrm i\partial\bar\partial V$ contributes $V_x/\mu$ there. The Schur complement of the last entry gives the stated positivity criterion. At $\xi=0$ the same endpoint argument as in dimension three applies.

Set $d=n-1$. After averaging over the angular variables, $\omega_{\FS}^{d}$ becomes the probability measure
$d\,\mu^{d-1}\,\mathrm d\mu=d\,\mu^d(1-\mu)\,\mathrm dx$;
this follows from the polar-coordinate determinant of
$\ddc\log(1+|\xi|^2)$. The background block $K_0$ has determinant one. Since $\omega_n^n=n\alpha_n^{n-1}\wedge\beta_n$ and the fiber mass of $\beta_n$ is two, therefore,
\[
 J_n(V)=2n\,d\,\mu^d(1-\mu)A^{n-2}\det K_V.
\]
For $n=3$ this is $12\mu^2(1-\mu)A\det K_V$.
Comparing the two expressions proves \eqref{eq:hopf-dimensional-density}.
\end{proof}

\Needspace{12\baselineskip}
\begin{theorem}\label{thm:hopf-all-dimensional-unbounded}\index{Hopf manifold!unbounded mass in all dimensions}
For every $n\geq3$, the Gauduchon metric $\omega_n$ admits smooth potentials $V_\eta$ with $\omega_n+\ddc V_\eta>0$ whose total Monge--Amp\`ere masses tend to infinity as $\eta\downarrow0$.
\end{theorem}
\begin{proof}
Use the functions $V_\eta=W+g_\eta$ of the proof of \cref{thm:hopf-unbounded}, with $\varepsilon=\eta^{1/\eta^2}$. Their endpoint properties give smooth potentials on every $H_n$, and \cref{lem:hopf-dimensional-density} gives strict positivity. Since $\mu\geq1/2$ on $x\geq0$, the slope bound $\max\{-W_x,0\}=o(1)$, the inequalities $-m_\eta\leq g_\eta'\leq0$ of \cref{lem:hopf-base-correction} and $m_\eta=o(1)$ imply
\[
 \mu+(V_\eta)_x\geq\tfrac14
 \qquad(x\geq0)
\]
for all sufficiently small $\eta$.

We next show that the threefold mass of $V_\eta$ on $\{x<0\}$ stays bounded, so that it diverges on $\{x\geq0\}$. On $x<0$, the heat potential is $W=F_\eta(\varepsilon,w)$ and has no $x$ derivatives. The determinant formula in the preceding proof and $\int_E(F_\eta)_{w\bar w}(\varepsilon,\cdot)\,\mathrm dA_E=0$ give
\[
 \int_E J_3(V_\eta;x,w)\,\mathrm dA_E
 =12(\mu+g_\eta')
                \bigl(g_\eta''+\mu(1-\mu)\bigr).
\]
Here $0<\mu+g_\eta'\leq\mu$, and
$g_\eta''=r_\eta-m_\eta\psi$ with
$\int r_\eta=m_\eta$ and $\int\psi=1$. Hence
\[
 \int_{x<0}\!J_3(V_\eta)\,\mathrm dx\,\mathrm dA_E
 \leq12\int_{-\infty}^0\mu^2(1-\mu)\,\mathrm dx
       +12\int_{\mathbb R}|g_\eta''|\,\mathrm dx
 \leq\tfrac32+24m_\eta.
\]
No bound on the amplitude of the fiber Hessian is required; its mean has been integrated exactly. Since the total threefold mass tends to infinity, its mass on $x\geq0$ tends to infinity. Since $J_n\geq0$ by positivity, the dimensional comparison on $x\geq0$ now gives
\[
 \int_{H_n}(\omega_n+\ddc V_\eta)^n
 \geq\frac{n(n-1)}6\,4^{-(n-3)}
       \int_{x\geq0}J_3(V_\eta)\,\mathrm dx\,\mathrm dA_E
 \longrightarrow+\infty.
\]
\end{proof}

\begin{corollary}\label{cor:hopf-all-metrics-unbounded}\index{Bounded mass property!failure on scalar Hopf manifolds}
For $n\geq3$, every Hermitian form $h$ on $H_n$ satisfies $v_+(h)=+\infty$. In particular, $H_n$ does not have the bounded mass property.
\end{corollary}
\begin{proof}
For a Hermitian form $h$, compactness gives $h\geq c\omega_n$ for some $c>0$. The smooth potentials $cV_\eta$ satisfy
$h+\ddc(cV_\eta)\geq c(\omega_n+\ddc V_\eta)>0$,
and positivity of determinants bounds their masses below by $c^n$ times the divergent masses in \cref{thm:hopf-all-dimensional-unbounded}. Hence $v_+(h)=+\infty$ (\cref{def:bmp}).
\end{proof}
