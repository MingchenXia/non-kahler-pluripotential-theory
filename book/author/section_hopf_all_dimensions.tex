\section{Higher-dimensional Hopf manifolds}
\label{sec:hopf-all-dimensions}

The threefold mass formula locates the source of unbounded mass, but it does not by itself give a construction in higher dimension. The same heat potentials do work there: the radial--vertical block is unchanged, and each extra angular direction contributes a factor whose lower bound is uniform where the threefold mass diverges.

For $n\geq3$, let
\[
 H_n=(\mathbb C^n\setminus\{0\})/
             \langle z\mapsto\mathrm e^{-1}z\rangle,
 \qquad p_n:H_n\longrightarrow\mathbb P^{n-1}.
\]
Put $s=\log|z|^2$, $\alpha_n=\ddc s=p_n^*\omega_{\FS}$,
$\beta_n=\mathrm ds\wedge\dc s$ and $\omega_n=\alpha_n+\beta_n$, where
$\int_{\mathbb P^{n-1}}\omega_{\FS}^{n-1}=1$.
The forms descend and $\omega_n$ is positive for the same reasons as in dimension three. The fiber is the same elliptic curve $E$, with $\int_E\beta_n=2$. Since $\alpha_n$ has rank $n-1$ and $\beta_n^2=0$,
\[
 \int_{H_n}\omega_n^n=2n,\qquad
 \ddc(\omega_n^{n-1})=-(n-1)\alpha_n^n=0.
\]
Thus $\omega_n$ is Gauduchon.

On the affine chart write $\xi\in\mathbb C^{n-1}$,
$x=\log|\xi|^2$, $w=\log z_0\pmod\Lambda$ and
$\mu=\mathrm e^x/(1+\mathrm e^x)$ as before. For a smooth real function $V(x,w)$ with the endpoint properties of \cref{lem:hopf-positivity}, let $J_n(V;x,w)$ be the density obtained by integrating $(\omega_n+\ddc V)^n$ over the angular variables, with angular probability measure. Thus
\[
 \int_{H_n}(\omega_n+\ddc V)^n
   =\int_{\mathbb R\times E}J_n(V;x,w)\,\mathrm dx\,\mathrm dA_E.
\]
The endpoints and the complement of the affine chart have zero smooth volume.

\Needspace{12\baselineskip}
\begin{lemma}\label{lem:hopf-dimensional-density}\index{Hopf manifold!dimensional density comparison}
The positivity criterion \eqref{eq:hopf-positivity} holds in every dimension $n\geq3$, and
\begin{equation}\label{eq:hopf-dimensional-density}
 J_n(V;x,w)=\frac{n(n-1)}6
                  (\mu+V_x)^{n-3}J_3(V;x,w).
\end{equation}
Here $J_3$ is evaluated for the same function $V$ on the threefold.
\end{lemma}
\begin{proof}
Choose $n-2$ horizontal angular vectors, one horizontal radial vector and the vertical vector $\partial_w$. In the normalization of the proof of \cref{lem:hopf-positivity}, each angular entry is $A=(\mu+V_x)/\mu$, and the radial--vertical block is
\[
 K_V=\begin{pmatrix}
 (\mu+V_{xx})/[\mu(1-\mu)]&
       (\mu+V_{x\bar w})/\sqrt{\mu(1-\mu)}\\
 (\mu+V_{wx})/\sqrt{\mu(1-\mu)}&1+V_{w\bar w}
 \end{pmatrix}.
\]
This follows by differentiating $s=w+\bar w+\log(1+\mathrm e^x)$: the radial part of $\partial\bar\partial x$ is zero, whereas its angular part contributes $V_x/\mu$. The Schur complement of the last entry gives the stated positivity criterion. At $\xi=0$ the same endpoint argument as in dimension three applies.

Set $d=n-1$. The radial Fubini--Study probability density is
$d\mu^{d-1}\mathrm d\mu=d\mu^d(1-\mu)\mathrm dx$;
it follows from the polar-coordinate determinant of
$\ddc\log(1+|\xi|^2)$ and its integral is one. The background block $K_0$ has determinant one. Using the fiber mass two, therefore,
\[
 J_n(V)=2nd\mu^d(1-\mu)A^{n-2}\det K_V.
\]
For $n=3$ this is $12\mu^2(1-\mu)A\det K_V$.
Comparing the two expressions proves \eqref{eq:hopf-dimensional-density}.
\end{proof}

\Needspace{12\baselineskip}
\begin{theorem}\label{thm:hopf-all-dimensional-unbounded}\index{Hopf manifold!unbounded mass in all dimensions}
For every $n\geq3$, the Gauduchon metric $\omega_n$ admits smooth potentials $V_\eta$ with $\omega_n+\ddc V_\eta>0$ whose total Monge--Amp\`ere masses tend to infinity as $\eta\downarrow0$.
\end{theorem}
\begin{proof}
Use exactly the functions $V_\eta=W+g_\eta$ constructed in the proof of \cref{thm:hopf-unbounded}, including the heat scale $\varepsilon=\eta^{1/\eta^2}$. Their endpoint properties give smooth potentials on every $H_n$, and \cref{lem:hopf-dimensional-density} gives strict positivity. The slope bounds in \cref{lem:hopf-small-defect} and $-m_\eta\leq g_\eta'\leq0$ imply
\[
 \mu+(V_\eta)_x\geq\tfrac14
 \qquad(x\geq0)
\]
for all sufficiently small $\eta$.

We must first localize the divergent threefold mass to this region. On $x<0$, the heat potential is $W=F_\eta(\varepsilon,w)$ and has no $x$ derivatives. The determinant formula in the preceding proof and $\int_E F_{w\bar w}\,\mathrm dA_E=0$ give
\[
 \int_E J_3(V_\eta;x,w)\,\mathrm dA_E
 =12(\mu+g_\eta')
                \bigl(g_\eta''+\mu(1-\mu)\bigr).
\]
Here $0<\mu+g_\eta'\leq\mu$, and
$g_\eta''=r_\eta-m_\eta\psi$ with
$\int r_\eta=m_\eta$ and $\int\psi=1$. Hence
\[
 \int_{x<0}\!J_3(V_\eta)\,\mathrm dx\,\mathrm dA_E
 \leq12\int_{-\infty}^0\mu^2(1-\mu)\,\mathrm dx
       +12\int_{\mathbb R}|g_\eta''|\,\mathrm dx
 \leq\tfrac32+24m_\eta.
\]
No bound on the amplitude of the fiber Hessian is required; its mean has been integrated exactly. Since the total threefold mass tends to infinity, its mass on $x\geq0$ tends to infinity. The dimensional comparison now gives
\[
 \int_{H_n}(\omega_n+\ddc V_\eta)^n
 \geq\frac{n(n-1)}6\,4^{-(n-3)}
       \int_{x\geq0}J_3(V_\eta)\,\mathrm dx\,\mathrm dA_E
 \longrightarrow+\infty.
\]
\end{proof}

\begin{corollary}\label{cor:hopf-all-metrics-unbounded}\index{Bounded mass property!failure on scalar Hopf manifolds}
Every Hermitian metric on $H_n$, $n\geq3$, fails the bounded mass property.
\end{corollary}
\begin{proof}
For a Hermitian form $h$, compactness gives $h\geq c\omega_n$ for some $c>0$. The potentials $cV_\eta$ satisfy
$h+\ddc(cV_\eta)\geq c(\omega_n+\ddc V_\eta)>0$,
and positivity of determinants bounds their masses below by $c^n$ times the divergent masses in \cref{thm:hopf-all-dimensional-unbounded}.
\end{proof}
