\section{Moving bounded carrier measures}

When the bounded carrier measure varies, exponential domination supplies a common lower envelope. A smaller-background potential and a strict big potential give comparison functions with constants uniform over the family. If every approximant has the ordinary type of $\varphi$, smaller-background comparison gives the same construction even when $v_-(\theta)=0$. Convergence of the carrier potentials in capacity will then identify the limiting equation.

For a bounded $\omega$-plurisubharmonic potential $\psi$, write $\omega_\psi=\omega+\ddc\psi$.

\Needspace{19\baselineskip}
\begin{proposition}\label{prop:exponential-common-rooftop}
Fix $A,\lambda>0$, $C\in\mathbb R$ and $L\geq0$. Suppose $u_j\in\mathcal E_\varphi$, $u_j\leq\varphi+C$, and
\begin{equation}\label{eq:exponential-bounded-reference-domination}
 \mu_{u_j}\leq A \mathrm{e}^{\lambda u_j}\omega_{\psi_j}^n,
 \qquad \psi_j\in\PSH(X,\omega),\qquad
 \|\psi_j\|_\infty\leq L.
\end{equation}
The index $j$ ranges over the positive integers. Assume either $v_-(\theta)>0$ or $u_j\sim\varphi$ for every $j$. Then
\[
 h=P_\theta\left(\inf_j u_j\right)\in\mathcal E_\varphi.
\]
No convergence of $\psi_j$ or $u_j$ is required. In the ordinary-type alternative, the constants in $u_j\sim\varphi$ need not be uniform.
\end{proposition}
\begin{proof}
Translate all $u_j$ by one constant so that $u_j\leq\varphi\leq0$; this changes $A$ by a fixed factor. Translate each $\psi_j$ to lie in $[-2L,0]$, without changing its measure. Set $q=P_{\varepsilon\theta}(\varphi)$ and choose $\rho\in\PSH(X,\theta)$ and $a>0$ such that
\[
 q\leq\varphi,
 \qquad \rho\leq0,\qquad \theta+\ddc\rho\geq a\omega.
\]
The first potential is genuine by the model hypothesis, and the second by bigness. Fix $0<\delta<(1-\varepsilon)a$.

\Needspace{8\baselineskip}
For a fixed real $b\geq1$, put $r_{j,b}=Q_b(u_j,\varphi)$. The stretching lemma puts it in $\mathcal E_\varphi$, and $r_{j,b}\leq u_j\leq0$. Contact support and positivity imply
\[
 \mu_{r_{j,b}}
 \leq b^n\mathbf1_{D_{j,b}}\mu_{u_j},
 \qquad
 D_{j,b}=\{b^{-1}r_{j,b}+(1-b^{-1})\varphi=u_j\}.
\]
On this contact set, $u_j\leq r_{j,b}/b$. Thus
\begin{equation}\label{eq:stretched-exponential-domination}
 \mathrm{e}^{-\lambda r_{j,b}/b}\mu_{r_{j,b}}
 \leq b^n A\omega_{\psi_j}^n.
\end{equation}
The weights are taken off the pluripolar pole sets.

Choose $C_b\geq0$ so large that
\[
 \delta^n\exp(\lambda C_b/b)\geq b^n A,
\]
and define
\[
 s_{j,b}=q+(1-\varepsilon)\rho+\delta\psi_j-C_b.
\]
This is a genuine $\theta$-plurisubharmonic function, since
\[
 \begin{split}
 \theta+\ddc s_{j,b}
 &= (\varepsilon\theta+\ddc q)
      +(1-\varepsilon)(\theta+\ddc\rho)
      +\delta\ddc\psi_j\\
 &\geq \delta\omega_{\psi_j}.
 \end{split}
\]
Consequently $\mu_{s_{j,b}}\geq\delta^n\omega_{\psi_j}^n$. This inequality follows first away from the poles by the local positive product inequality, and then globally because the bounded reference measure does not charge pluripolar sets. Also $s_{j,b}\leq-C_b$ and $s_{j,b}\preceq\varphi$. We conclude
\[
 \mathrm{e}^{-\lambda s_{j,b}/b}\mu_{s_{j,b}}
 \geq b^n A\omega_{\psi_j}^n
 \geq \mathrm{e}^{-\lambda r_{j,b}/b}\mu_{r_{j,b}}.
\]

If $v_-(\theta)>0$, apply \cref{cor:relative-extended-exponential-order} with first potential $r_{j,b}$ and second potential $s_{j,b}$. It gives $r_{j,b}\geq s_{j,b}$.

In the ordinary-type alternative, $r_{j,b}\sim\varphi$: if $u_j\geq\varphi-K_j$, the potential $\varphi-bK_j$ is a competitor for $r_{j,b}$. The smaller-background envelope of $r_{j,b}$ is therefore genuine, and $s_{j,b}\preceq r_{j,b}$. The same conclusion follows instead from \cref{cor:singular-exponential-order}, without a lower-volume hypothesis.

The bound $\psi_j\geq-2L$ now supplies one genuine lower bound, independent of $j$:
\[
 H_b=q+(1-\varepsilon)\rho-2\delta L-C_b
 \leq r_{j,b}.
\]
The potential $H_b$ is $\theta$-plurisubharmonic. Therefore
\[
 b^{-1}H_b+(1-b^{-1})\varphi\leq u_j
 \quad\hbox{for every }j.
\]
For $b=1$ this proves that the infimum envelope $h$ is genuine. For general $b$, the left side is a competitor for $h$, so it also lies below $h$. This is a competitor for every defining stretched-envelope condition of $h$. Since $h\leq\varphi$, we obtain $h\in\mathcal E_\varphi$. Restore the common translation to finish.
\end{proof}

\Needspace{8\baselineskip}
\begin{corollary}\label{cor:exponential-relative-class-closure}
Under the hypotheses of \cref{prop:exponential-common-rooftop}, if $u_j\to u\in\PSH(X,\theta)$ in $L^1(X)$, then $u\in\mathcal E_\varphi$.
\end{corollary}
\begin{proof}
The common envelope gives $h\leq u_j\leq\varphi+C$. Pass to an almost-everywhere convergent subsequence and use \cref{lem:qpsh-almost-everywhere-order} to get $h\leq u\leq\varphi+C$. Upward stability gives the conclusion.
\end{proof}
