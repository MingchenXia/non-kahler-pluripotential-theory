\subsubsection{Local regularity and strict weights}
\label{sec:affine-local-regularity}

Whether the regularized envelope touches the obstacle at a point $a\in K$ depends only on the parts of $K$ near $a$. Without weight, this local regularity is detected by the relative extremal function $h_{E,B}$ of \cref{def:relative-extremal-function}, for compact pieces $E$ of $K$ near $a$ and small balls $B$ centered at $a$. If the weight extends to a global function whose complex Hessian has a strict lower bound near $a$, a failure of this regularity produces a quantitative defect of the weighted envelope.

For a non-pluripolar compact set $E\subset\mathbb C^n$, write $V_E$ and $V_E^*$ for the raw and regularized envelopes of growth one with zero obstacle on $E$; thus $V_K=V_{1,0}$ as before. For other weights and growth allowances, write $F_{\delta,E,q}$ for the regularized envelope with growth allowance $\delta$ and obstacle $q$ imposed at every point of $E$.

\begin{lemma}\label{lem:relative-affine-zero-value}
Let $a\in\mathbb C^n$, $0<r<R$ and $B=B(a,R)$, and let $E\subseteq\overline B(a,r)$ be compact. If $E$ is pluripolar, then $h_{E,B}^*=0$ on $B$. If $E$ is non-pluripolar, then
\[
 h_{E,B}^*(a)=-1\quad\Longleftrightarrow\quad V_E^*(a)=0.
\]
\end{lemma}
\begin{proof}
In the pluripolar case, choose $b\in\mathcal L_1$ with $b=-\infty$ on $E$ by \cref{lem:affine-logarithmic-pole-barrier}, and a finite upper bound $C$ for $b$ on $B$. The functions
\[
 u_\epsilon=\max\{-1,\epsilon(b-C)\}
\]
are competitors for $h_{E,B}$, and they tend to zero almost everywhere as $\epsilon\downarrow0$. Hence $h_{E,B}^*\geq0$ almost everywhere, and then everywhere by \cref{lem:qpsh-almost-everywhere-order}. Since $h_{E,B}^*\leq0$, it vanishes.

Now suppose that $E$ is non-pluripolar. \Cref{prop:affine-projective-envelope} gives a finite upper bound $M\geq1$ for $V_E^*$ on $B$. For each competitor $v$ for $V_E$, the function $\max(-1,v/M-1)$ is a competitor for $h_{E,B}$. Taking suprema and regularizing gives
\[
 h_{E,B}^*(a)\geq\max\{-1,V_E^*(a)/M-1\}.
\]
As $V_E^*\geq0$, the equality $h_{E,B}^*(a)=-1$ forces $V_E^*(a)=0$.

For the converse, choose $r<R_0<R$ and $k>0$ with $k\log(R_0/r)>1$. The plurisubharmonic function
\[
 L(z)=k\log\bigl(\lVert z-a\rVert/r\bigr)
\]
belongs to $\mathcal L_k$ and equals $-\infty$ at $a$. For a competitor $u$ for $h_{E,B}$, define $W=\max(u+1,L)$ on $B$ and $W=L$ outside $B$. On the collar $R_0<\lVert z-a\rVert<R$ we have $L>1\geq u+1$, so the two definitions agree there, and $W$ is plurisubharmonic by \cref{lem:qpsh-gluing}. On $E$ one has $u=-1$ and $L\leq0$, so $W=0$; on the ball $B(a,r)$ one has $L<0\leq u+1$, so $W=u+1$. Hence $W/k$ is a competitor for $V_E$, with $k$ independent of $u$. Taking the supremum over $u$ and regularizing at $a$ gives
\[
 V_E^*(a)\geq(1+h_{E,B}^*(a))/k.
\]
Thus $V_E^*(a)=0$ forces $h_{E,B}^*(a)=-1$.
\end{proof}

Let $K\subset\mathbb C^n$ again be a non-pluripolar compact set. For $a\in K$ and $r>0$, put $K_r=K\cap\overline B(a,r)$.

\begin{definition}\label{def:affine-local-pluriregularity}\index{Pluriregular compact!local}
The compact set $K$ is \emph{locally pluriregular} at $a\in K$ if every $K_r$, $r>0$, is non-pluripolar and $V_{K_r}^*(a)=0$. For $\delta>0$ and a bounded function $q$ on $K$, it is \emph{locally $(\delta,q)$-regular} at $a$ if $F_{\delta,K_r,q}(a)=q(a)$ for every $r>0$.
\end{definition}

For local pluriregularity it suffices to check arbitrarily small radii, since envelopes decrease as the compact set increases. For local $(\delta,q)$-regularity, monotonicity gives only $F_{\delta,K_s,q}(a)\leq F_{\delta,K_r,q}(a)$ for $r<s$, and the definition requires every radius. By \cref{lem:relative-affine-zero-value}, $K$ is locally pluriregular at $a$ if and only if $h_{K_r,B(a,R)}^*(a)=-1$ for all small $0<r<R$.

\begin{lemma}\label{lem:affine-local-weighted-regularity}
Let $q$ be bounded and upper semicontinuous on $K$, and suppose that raw contact holds for the parameter $\delta>0$ and the obstacle $q$ on $K$, that is, $V_{\delta,q}=q$ on $K$. If $K$ is locally pluriregular at $a$, then $K$ is locally $(\eta,q)$-regular at $a$ for every $\eta\geq\delta$; in particular, $F_{\eta,K,q}(a)=q(a)$.
\end{lemma}
\begin{proof}
Raw contact also holds for every $\eta\geq\delta$, since $V_{\eta,q}$ increases with $\eta$ and is at most $q$ on $K$; so it suffices to treat $\eta=\delta$. Fix $s>0$. The compact set $K_s$ inherits raw contact: on $K_s$, its raw envelope is at least the raw envelope for $K$, which equals $q$, and at most the obstacle $q$. For $0<r<s$ put $M_r=\sup_{K_r}q$. Raw contact on $K_s$, monotonicity of envelopes with respect to the compact set, and \eqref{eq:affine-obstacle-sandwich} for $K_r$ give
\[
 q(a)\leq F_{\delta,K_s,q}(a)
 \leq F_{\delta,K_r,q}(a)
 \leq M_r+\delta V_{K_r}^*(a)=M_r.
\]
Since $q$ is upper semicontinuous and $a\in K_r$, we have $M_r\downarrow q(a)$ as $r\downarrow0$, which gives $F_{\delta,K_s,q}(a)=q(a)$.
\end{proof}

\begin{theorem}\label{thm:affine-local-strict-weight}
Let $a\in K$, $\tau>0$ and $\Psi\in\mathcal L_\tau\cap C(\mathbb C^n)$, and put $q=\Psi|_K$. Suppose that, for some $R,c>0$,
\[
 \ddc\Psi\geq c\omega_0\quad\text{on }B(a,R),
 \qquad\omega_0=\ddc\lVert z\rVert^2.
\]
For $0<r<R$ and $\delta\geq\tau$,
\begin{equation}\label{eq:affine-local-strict-weight-defect}
 F_{\delta,K,q}(a)-\Psi(a)
 \geq cr^2\bigl(1+h_{K_r,B(a,R)}^*(a)\bigr).
\end{equation}
Regular contact at $a$, that is, $F_{\delta,K,q}(a)=q(a)$, holds for some $\delta\geq\tau$ if and only if $K$ is locally pluriregular at $a$. In that case it holds for every $\delta\geq\tau$.
\end{theorem}
\begin{proof}
Set $B=B(a,R)$ and $E=K_r$. For a competitor $u$ for $h_{E,B}$, the function
\[
 H_u(z)=\Psi(z)-c\lVert z-a\rVert^2+cr^2(u(z)+1)
\]
is plurisubharmonic on $B$, because the assumption on $\ddc\Psi$ makes $\Psi-c\lVert z-a\rVert^2$ plurisubharmonic there. Since $u\leq0$, we have $H_u\leq\Psi$ where $\lVert z-a\rVert\geq r$. Define $W=\max(\Psi,H_u)$ on $B$ and $W=\Psi$ outside $B$. The two definitions agree near $\partial B$, so $W$ is plurisubharmonic on $\mathbb C^n$, and it belongs to $\mathcal L_\tau$ because it equals $\Psi$ outside $B$. On $E$, the equality $u=-1$ gives $H_u\leq\Psi$, and on $K\setminus E$ the distance bound gives the same inequality. Thus $W=q$ on $K$, so $W$ is a competitor for $F_{\delta,K,q}$ for each $\delta\geq\tau$. Since $W\geq H_u$ near $a$, taking the supremum over $u$, regularizing, and using the continuity of $\Psi$, we obtain \eqref{eq:affine-local-strict-weight-defect}.

The function $\Psi$ itself is a competitor, so raw contact holds for every $\delta\geq\tau$. If regular contact holds for one such $\delta$, the left-hand side of the estimate vanishes, so $h_{K_r,B(a,R)}^*(a)=-1$ for all $0<r<R$. By \cref{lem:relative-affine-zero-value}, every $K_r$ with $r<R$ is then non-pluripolar and satisfies $V_{K_r}^*(a)=0$; for $r\geq R$ this follows by monotonicity. Conversely, local pluriregularity and \cref{lem:affine-local-weighted-regularity} give regular contact at $a$ for every $\delta\geq\tau$.
\end{proof}
\index{Envelope!locally strict weight}

In this regular case, $V_K^*(a)\leq V_{K_r}^*(a)=0$ by monotonicity, and the right derivative $D_\tau(a)$ with respect to the growth allowance, for the weight $q$, vanishes by \cref{cor:affine-slope-regular-contact}, since regular contact persists.

\Needspace{10\baselineskip}
\begin{corollary}\label{cor:affine-weight-degenerate-hessian}\index{Contact set!weight Hessian degeneracy}
Let $\tau>0$, $\Psi\in\mathcal L_\tau\cap C^2(\mathbb C^n)$ and $a\in K$. If $F_{\tau,K,\Psi|_K}(a)=\Psi(a)$, but $F_{\delta,K,\Psi|_K}(a)>\Psi(a)$ for some $\delta>\tau$, then
\[
 (\ddc\Psi)^n(a)=0.
\]
\end{corollary}
\begin{proof}
The complex Hessian of $\Psi$ is semipositive. If its top power did not vanish at $a$, it would be positive definite at $a$, and by continuity $\ddc\Psi\geq c\omega_0$ on a neighborhood of $a$ for some $c>0$. \Cref{thm:affine-local-strict-weight} would then give regular contact at $a$ for every $\delta\geq\tau$, a contradiction.
\end{proof}
