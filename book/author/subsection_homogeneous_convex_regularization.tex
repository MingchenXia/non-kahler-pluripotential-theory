\subsection{Homogeneous convex regularization}
\label{subsec:homogeneous-convex-regularization}
\index{Regularization!homogeneous fiber norms}
\index{Vector bundle!continuous convex norms}
\index{Plurisubharmonic function!homogeneous regularization}
\index{Greene--Wu gluing}

\Cref{thm:convex-norm-deformation-obstruction} concerns smooth norms, whereas the norms produced by duality later in this section are in general only continuous. We show that a continuous homogeneous function with strict positivity margins can be approximated by smooth ones with the same margins, so the obstruction applies to it as well. The local step is Wu's convolution in projective coordinates, written as an average of holomorphic linear shears; the local approximations are glued by the argument of Greene and Wu. We give both steps in order to keep track of strict plurisubharmonicity and of strong convexity in the fibers; see \cref{sec:history-gauduchon} for attribution.

\Needspace{7\baselineskip}
Let $\pi:F\to X$ be a holomorphic vector bundle, and write $F^\times$ for the complement of its zero section. A continuous real function $Q$ on $F$ is \emph{complex homogeneous of degree two} if
\[
 Q(\lambda v)=|\lambda|^2Q(v)\qquad(\lambda\in\mathbb C).
\]
We call $Q$ \emph{locally strongly convex in the fibers} if, in a local bundle frame and near every point of $F^\times$, there is a constant $c>0$ such that $Q(z,v)-c|v|^2$ is convex on sufficiently small real fiber balls. For continuous functions, strict plurisubharmonicity means that subtracting a sufficiently small positive multiple of the Euclidean squared norm in local coordinates of the total space leaves a plurisubharmonic function. Both conditions are local, and on compact subsets of $F^\times$ they hold with uniform positive margins.

\begin{lemma}\label{lem:compact-sheaf-regularization}
Let $M$ be a compact smooth manifold and let $\mathcal S$ be a sheaf of germs of continuous real functions on $M$, whose elements we call sections. Suppose that $\mathcal S$ is closed under maxima and has the following two properties.

\emph{Stability:} if $f$ is a section near a compact set $K$, then $f+u$ is a section near $K$ whenever $u$ is smooth and sufficiently small in the $C^\infty$ topology on a neighborhood of $K$.

\emph{Local approximation:} every point has a neighborhood $U$ such that, for every compact $K\Subset U$, a section near $K$ can be approximated uniformly on $K$ by smooth sections near $K$, with arbitrarily small errors in any prescribed finite collection of derivative norms on compact subsets where the section is already smooth.

Then every global section can be approximated uniformly on $M$ by smooth global sections.
\end{lemma}
\begin{proof}
We first describe a gluing step. Let $f$ be a global section that is smooth near a compact set $L_1$, and let $L_2$ be a compact subset of one of the neighborhoods $U$ in the statement. Choose relatively compact open sets $W_3,W_4$ with $L_1\subset W_3$, $f$ smooth near $\overline W_3$, and $L_2\subset W_4\Subset U$. Choose a smooth function $0\leq\rho\leq1$ equal to one near $\overline W_3$, with compact support in the region where $f$ is smooth. Put $V=W_3\cup W_4$, and choose a smooth function $-1\leq\psi\leq1$ equal to one near $L_1\cup L_2$ and equal to minus one near $M\setminus V$.

By stability, for sufficiently small $\xi>0$ the function $f_\xi=f+\xi\psi$ is still a global section. Approximate $f_\xi$ on $\overline W_4$ by a smooth section $g$, with uniform error less than $\delta<\xi/2$, and with arbitrarily small derivative errors near the compact set $\overline W_4\cap\operatorname{supp}\rho$, where $f_\xi$ is smooth. On $V$ define
\[
 h=f_\xi\quad\hbox{on }W_3,\qquad
 h=\rho f_\xi+(1-\rho)g\quad\hbox{on }W_4.
\]
These definitions agree on the overlap, and $h$ is smooth. Where $\rho$ vanishes, $h$ is the section $g$. Where $\rho$ can be nonzero, $h-f_\xi=(1-\rho)(g-f_\xi)$, and stability applied to the smooth section $f_\xi$ makes $h$ a section if the derivative errors of $g-f_\xi$ are sufficiently small. Moreover, $|h-f_\xi|<\delta$.

Define $f_1=\max\{f,h\}$ on $V$ and $f_1=f$ outside $V$. Near $L_1\cup L_2$ we have $h>f$, and near the boundary of $V$ we have $h<f$. Thus $f_1$ is continuous, it is a section because $\mathcal S$ is closed under maxima, and it is smooth near $L_1\cup L_2$. Also $\|f_1-f\|_{C^0}<\xi+\delta$. Near $L_1$ it equals $f+\xi$, so the previous smoothness is preserved.

Now cover $M$ by finitely many compact sets $L_2$ of this kind and repeat the step, starting with $L_1$ empty and then taking for $L_1$ the union of the previously treated compact sets. At every step choose $\xi$ and $\delta$ so small that the sum of the finitely many uniform errors is below the prescribed tolerance. The final section is smooth everywhere and has the required uniform error.
\end{proof}

\begin{lemma}\label{lem:homogeneous-shear-regularization}
In a holomorphic bundle chart, write $v=(v',v_r)$ and use the affine projective coordinate $w=v'/v_r$ on the cone $v_r\ne0$. Suppose that a positive continuous complex homogeneous function $Q$ of degree two is strictly plurisubharmonic and locally strongly convex in the fibers on this cone, and put $q(z,w)=Q(z,(w,1))$. For a nonnegative smooth approximate identity $\eta_\delta(s,t)$ of integral one, define
\[
 q_\delta(z,w)=\int q(z-s,w-t)\eta_\delta(s,t)\,\mathrm ds\,\mathrm dt,
 \qquad Q_\delta(z,v)=|v_r|^2q_\delta(z,v'/v_r).
\]
On every relatively compact projective subchart, for sufficiently small $\delta$, $Q_\delta$ is a smooth positive complex homogeneous function of degree two that is strictly plurisubharmonic and locally strongly convex in the fibers. As $\delta\to0$, it converges to $Q$ uniformly on compact subsets of the cone consisting of vectors of norm one, and in every finite derivative norm wherever $Q$ is already smooth.
\end{lemma}
\begin{proof}
Complex homogeneity gives the identity
\[
 Q_\delta(z,v)
 =\int Q(z-s,T_tv)\eta_\delta(s,t)\,\mathrm ds\,\mathrm dt,
 \qquad T_t(v',v_r)=(v'-v_rt,v_r).
\]
For fixed $(s,t)$, the map $(z,v)\mapsto(z-s,T_tv)$ is holomorphic and invertible, and its restriction to each fiber is linear. On a compact subset of the cone consisting of vectors of norm one, all these maps with $(s,t)$ sufficiently small stay in a fixed compact subset of the original cone, and their differentials and inverses are uniformly bounded there. A positive local lower bound for the Levi form of $Q$, interpreted through quadratic barriers, therefore gives a positive lower bound for every pullback and for their average. The same argument with real quadratic barriers in the fibers preserves local strong convexity. Only fiber segments contained in the local cone neighborhood are used here.

Ordinary convolution makes $q_\delta$ smooth, and the formula for $Q_\delta$ makes it smooth on the cone and complex homogeneous. Uniform convergence, derivative convergence on compact regions where $Q$ is smooth, and positivity for small $\delta$ follow from the corresponding properties of convolution and from the positive minimum of $Q$ on compact sets of vectors of norm one.
\end{proof}

\begin{theorem}\label{thm:homogeneous-convex-regularization}
Let $Q:F\to[0,+\infty)$ be continuous, positive on $F^\times$, complex homogeneous of degree two, strictly plurisubharmonic on $F^\times$, and locally strongly convex in the fibers. Then there are continuous complex homogeneous functions $Q_j$ of degree two, smooth on $F^\times$, strictly plurisubharmonic there and locally strongly convex in the fibers, which converge to $Q$ uniformly on the unit sphere bundle of any fixed smooth Hermitian metric. Each $p_j=\sqrt{Q_j}$ is a smooth complex norm.
\end{theorem}
\begin{proof}
Fix a smooth Hermitian squared norm $G$ on $F$, and let $M=\mathbb P(F)$ be the bundle of complex lines in $F$. The quotient $m=Q/G$ is a positive continuous function on the compact manifold $M$. Define a sheaf $\mathcal S$ on $M$ by requiring that the corresponding function $Gm$ on the cone be positive, strictly plurisubharmonic, and locally strongly convex in the fibers.

This sheaf is closed under maxima. Indeed, multiplication by $G>0$ commutes with maxima. Near a point of the cone, subtract a common positive quadratic barrier from both functions; their maximum remains plurisubharmonic, and on sufficiently small fiber balls it remains convex. Positivity is also preserved.

For stability on a compact subset $K$ of $M$, restrict to vectors of norm one lying over $K$, and to a fixed compact annulus around them. Finitely many bundle charts provide positive common lower bounds for both barriers. If $u$ is a smooth function on $M$ that is sufficiently small in $C^2$ near $K$, the Levi form and the real fiber Hessian of $Gu$ are smaller than half these bounds on that annulus. The perturbed function is therefore still positive and has both strict margins. Homogeneity transports the local conditions to the rest of the cone over $K$. This proves the stability required in \cref{lem:compact-sheaf-regularization}.

On a sufficiently small projective coordinate chart, \cref{lem:homogeneous-shear-regularization} gives local approximation, including derivative control on compact subsets where $m$ is smooth; dividing its regularizations by $G$ gives local smooth sections of $\mathcal S$. By \cref{lem:compact-sheaf-regularization}, there are smooth positive global sections $m_j\to m$ uniformly. Put $Q_j=Gm_j$ off the zero section and $Q_j=0$ on the zero section. Homogeneity gives continuity at zero, uniformly over the compact base, as well as the asserted convergence on a fixed unit sphere bundle.

It remains to see that $p_j$ is a norm. Local convexity in the fibers away from zero implies convexity on the whole fiber. Indeed, the restriction of $Q_j$ to a real line avoiding zero is locally convex and hence convex, while on a real line through zero, homogeneity gives $Q_j(tv)=t^2Q_j(v)$ for real $t$, which is convex including at zero. Thus the set $B_x=\{v:Q_j(v)\leq1\}$ is convex, balanced and absorbing, and its gauge is $\sqrt{Q_j}$. If $v,w$ are nonzero, convexity applied to $v/p_j(v)$ and $w/p_j(w)$, with weights proportional to $p_j(v)$ and $p_j(w)$, gives the triangle inequality; the cases with a zero vector follow directly. Positivity, smoothness off zero and complex homogeneity give the remaining properties of a smooth complex norm.
\end{proof}

\begin{proposition}\label{prop:weak-homogeneous-convex-replacement}
Suppose that $F$ admits a smooth positive complex homogeneous function $H$ of degree two that is strictly plurisubharmonic on $F^\times$. If a positive continuous complex homogeneous function $Q$ of degree two is plurisubharmonic on $F^\times$ and locally strongly convex in the fibers, then $F$ admits a smooth complex norm with strictly plurisubharmonic square on $F^\times$.
\end{proposition}
\begin{proof}
For $\tau>0$, set $Q_\tau=Q+\tau H$; it is positive, complex homogeneous and strictly plurisubharmonic. On a compact annulus of vectors of norm close to one, local strong convexity of $Q$ in the fibers gives a positive uniform margin in finitely many frames, while the real fiber Hessian of $H$ is uniformly bounded there. Choose $\tau$ so small that the possible negative part of $\tau$ times this Hessian consumes less than half the margin. Homogeneity then gives local strong convexity of $Q_\tau$ in the fibers everywhere off zero. Apply \cref{thm:homogeneous-convex-regularization} to $Q_\tau$ and take one of its smooth regularizations; its square root is the required smooth complex norm.
\end{proof}

\begin{corollary}\label{cor:continuous-convex-norm-obstruction}
For every holomorphic structure in the neighborhood of \cref{thm:convex-norm-deformation-obstruction}, there is no positive continuous complex homogeneous function of degree two that is both strictly plurisubharmonic off zero and locally strongly convex in the fibers. If that structure admits a function $H$ as in \cref{prop:weak-homogeneous-convex-replacement}, there is also no positive continuous complex homogeneous function of degree two that is plurisubharmonic off zero and locally strongly convex in the fibers.
\end{corollary}
\begin{proof}
For the first assertion, \cref{thm:homogeneous-convex-regularization} would produce a smooth complex norm with plurisubharmonic square, contrary to \cref{thm:convex-norm-deformation-obstruction}. For the second, use \cref{prop:weak-homogeneous-convex-replacement} and the same obstruction.
\end{proof}

The second assertion requires the strictly plurisubharmonic homogeneous function $H$, which need not exist for an arbitrary holomorphic bundle; \cref{subsec:tautological-homogeneous-positivity} shows that it exists when the dual bundle is ample. Without a strong convexity margin in the fibers, the regularization gives no obstruction for continuous fiber norms.
