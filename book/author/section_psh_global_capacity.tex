\section{Hermitian capacity}
\label{sec:hermitian-capacity}\index{Capacity!Hermitian}

After a fixed rescaling, a local capacity test can be glued to a constant function and so becomes a global potential on $X$. This compares the local Bedford--Taylor capacity with a capacity defined by global potentials, and it transfers to $X$ both the volume--capacity estimate and an estimate for the sublevel sets of normalized potentials.

For a Borel set $E\subset X$, the \emph{Hermitian capacity} of $E$ is
\[
 \operatorname{Cap}_\omega(E)
 =\sup_{v\in\PSH(X,\omega),\ 0\leq v\leq1}
             \int_E(\omega+\ddc v)^n,
\]
where the products are those of \cref{prop:bounded-products}. Their local expansion and \eqref{eq:local-cln-range} show that $\operatorname{Cap}_\omega(X)<+\infty$. On relatively compact coordinate balls, a fixed smooth correction bounds the measures of global tests by a constant times the corresponding local capacity. Conversely, the gluing construction in the proof of \cref{lem:global-capacity-estimates} below bounds local capacity by global capacity. Consequently, the quasi-continuity of quasi-plurisubharmonic functions and the characterization of sets of capacity zero recalled in \cref{sec:psh-local-capacity} hold for $\operatorname{Cap}_\omega$. A sequence $u_j$ converges to $u$ \emph{in capacity} if
\[
 \operatorname{Cap}_\omega(\{|u_j-u|>\delta\})\longrightarrow0
 \quad\text{for every }\delta>0,
\]
where pole sets may be omitted. To compare with another Hermitian form $\omega'$, choose $C\geq1$ with $\omega'\leq C\omega$. Every test $0\leq v\leq1$ for $\operatorname{Cap}_{\omega'}$ gives a test $v/C$ for $\operatorname{Cap}_\omega$, and the positivity of mixed products gives
\[
 (\omega'+\ddc v)^n\leq C^n(\omega+\ddc(v/C))^n.
\]
Thus $\operatorname{Cap}_{\omega'}\leq C^n\operatorname{Cap}_\omega$. Interchanging the two forms shows that the capacities are equivalent up to fixed multiplicative constants, so convergence in capacity does not depend on the choice of $\omega$. Finally, the continuity statements for bounded products in \cref{sec:psh-local-capacity} pass to the products with smooth backgrounds, by their local expansion and a finite cover.

\subsection{Capacity estimates}

In this subsection we abbreviate $\operatorname{Cap}=\operatorname{Cap}_\omega$.

\begin{lemma}\label{lem:global-capacity-estimates}
There are constants $a,C>0$ such that
\begin{equation}\label{eq:global-volume-capacity}
 \operatorname{Vol}(E)\leq C\exp\bigl(-a\operatorname{Cap}(E)^{-1/n}\bigr)
\end{equation}
for every Borel set $E\subset X$. Moreover, for every $u\in\PSH(X,\omega)$ with $\sup_Xu=0$ and every $t>1$,
\begin{equation}\label{eq:global-capacity-tail}
 \operatorname{Cap}(\{u<-t\})\leq C/t.
\end{equation}
\end{lemma}
\begin{proof}
Cover $X$ by finitely many coordinate balls $B'_j$, with larger concentric balls $B_j$, $B'_j\Subset B_j$. A local capacity test $0\leq v\leq1$ on $B_j$ can be turned into a global test as follows. Choose a smooth function $\chi_j$ with $-2\leq\chi_j\leq0$, equal to zero near $\overline{B'_j}$ and to $-2$ near $\partial B_j$. Choose $\delta>0$ so small that
\[
 \omega+\delta\ddc\chi_j\geq0,\qquad 3\delta/2\leq1.
\]
\Needspace{6\baselineskip}
The function
\[
 V=\max\{\delta(v+\chi_j),-\delta/2\}+\delta/2
 \quad\hbox{on }B_j,
 \qquad V=0\quad\hbox{outside }B_j,
\]
is $\omega$-plurisubharmonic on $X$, because near $\partial B_j$ the constant branch is the larger one, and it takes values between zero and one. On $B'_j$ we have $\omega_V\geq\delta\ddc v$. By the positivity of bounded products,
\[
 \operatorname{cap}(E\cap B'_j,B_j)
 \leq\delta^{-n}\operatorname{Cap}(E).
\]
The local volume--capacity estimate, summed over the finite cover, proves \eqref{eq:global-volume-capacity}.

For the second estimate, choose on each $B_j$ a smooth negative quadratic function $r_j$ with $\ddc r_j\geq\omega$. For every global test $0\leq v\leq1$, the functions $u+r_j\leq0$ and $v+r_j$ are plurisubharmonic on $B_j$. Thus
\[
 \begin{split}
 \int_{\{u<-t\}}\omega_v^n
 &\leq t^{-1}\int_X(-u)\omega_v^n\\
 &\leq t^{-1}\sum_j\int_{B'_j}(-u-r_j)
                    \bigl(\ddc(v+r_j)\bigr)^n\\
 &\leq \frac Ct\left(1+\int_X|u|\,\mathrm{d}V\right).
 \end{split}
\]
Here the second inequality uses $\omega_v\leq\ddc(v+r_j)$, and the third is the weighted Chern--Levine--Nirenberg inequality, applied to local test functions with a common bound. The $L^1$ bound for normalized potentials in \cref{thm:normalized-compactness} completes the proof after taking the supremum over $v$.
\end{proof}

\Needspace{16\baselineskip}
\begin{corollary}\label{cor:lp-capacity-square}
Fix $p>1$. There is a constant $C_p$ such that every nonnegative $F\in L^p(X,\mathrm{d}V)$ satisfies
\[
 \int_E F\,\mathrm{d}V\leq C_p\|F\|_p\operatorname{Cap}(E)^2
\]
for every Borel set $E\subset X$.
\end{corollary}
\begin{proof}
By H\"older's inequality and \eqref{eq:global-volume-capacity}, the integral is at most
\[
 C\|F\|_p\exp(-a\operatorname{Cap}(E)^{-1/n}/p'),
 \qquad p'=p/(p-1).
\]
The function $x\mapsto x^{-2}\exp(-ax^{-1/n}/p')$ is bounded on $0<x\leq\operatorname{Cap}(X)<\infty$, and sets of capacity zero have zero volume.
\end{proof}

The same proof bounds the integral of an $L^p$ density by any fixed positive power of the capacity. The exponent two will be convenient for the iteration over sublevel sets in \cref{lem:backwards-capacity}.

\subsection{Convergence of potentials and measures}

For quasi-continuous obstacles, a decreasing lower semicontinuous approximation keeps the original bounds and confines the error to sets of vanishing capacity.

\begin{lemma}
\label{lem:quasicont-lsc-approximation}
Every bounded quasi-continuous function $f$ on $X$ admits bounded lower semicontinuous functions $f_j$ with common bounds such that $f_j\downarrow f$ quasi-everywhere and $f_j\to f$ in capacity.
\end{lemma}
\begin{proof}
Choose open sets $O_j$ of capacity at most $2^{-j}$ such that the restriction of $f$ to $X\setminus O_j$ is continuous, and set $G_j=\bigcup_{k\geq j}O_k$. Then the sets $G_j$ decrease, their capacities tend to zero, and $f$ is continuous on the closed complement of each $G_j$. Choose $M\geq\sup_Xf$ and define
\[
 f_j=f\ \hbox{on }X\setminus G_j,\qquad f_j=M\ \hbox{on }G_j.
\]
At a boundary point of $G_j$, the values of $f_j$ on the closed complement tend to the value of $f$ there, while the values inside $G_j$ are at least that value. Thus $f_j$ is lower semicontinuous. The functions $f_j$ decrease to $f$ outside $\bigcap_jG_j$, a set of capacity zero. Since $f_j-f$ is supported in $G_j$, they also converge to $f$ in capacity.
\end{proof}

Uniform bounds on potentials give a common capacity bound for their mixed measures.

\begin{lemma}\label{lem:bounded-mixed-capacity}
Let $A\geq0$ and let $0\leq k\leq n$ be an integer. There is a constant $C_A$ such that all the positive mixed measures
\[
 \omega_{v_1}\wedge\cdots\wedge\omega_{v_k}\wedge\omega^{n-k},
 \qquad v_i\in\PSH(X,\omega),
 \quad \|v_i\|_\infty\leq A,
\]
are bounded above on Borel sets by $C_A\operatorname{Cap}_\omega$. In particular, their total masses are uniformly bounded. If uniformly bounded functions tend to zero in capacity, then their integrals against these measures tend to zero uniformly in the $v_i$.
\end{lemma}
\begin{proof}
Translate the $v_i$ so that $0\leq v_i\leq2A$, and put
$q=(v_1+\cdots+v_k)/B$, where $B\geq\max\{2k,2kA,1\}$.
Then $0\leq q\leq1$ and
\[
 \omega_q=B^{-1}\sum_{i=1}^k\omega_{v_i}+(1-k/B)\omega,
\]
where all coefficients are positive. Expanding $\omega_q^n$ shows that the given mixed product is bounded by a constant multiple of $\omega_q^n$, and the definition of capacity proves the first claim. The case $k=0$ follows by testing with $v=0$. For a bounded function $z$, split its integral into the parts over $\{|z|>t\}$ and over its complement; first use convergence in capacity for fixed $t>0$, and then let $t\downarrow0$.
\end{proof}

The same control applies to the total variation of a product formed with a different smooth background.

\begin{lemma}\label{lem:signed-truncation-quasicont}
Let $\Omega$ be a Hermitian form and $\theta$ a smooth real $(1,1)$-form on $X$. Suppose that $\Omega$-plurisubharmonic functions $v_j$ have common finite upper and lower bounds and decrease to $v$. Then the signed measures
\[
 \nu_j=(\theta+\ddc v_j)^n
\]
converge to $(\theta+\ddc v)^n$ against every fixed bounded quasi-continuous function. Their total variations are bounded, uniformly in $j$, by a constant multiple of the local capacity on the balls of a fixed finite cover.
\end{lemma}
\begin{proof}
On a relatively compact coordinate ball, choose a fixed quadratic function $q$ such that the functions $v_j+q$ are plurisubharmonic; they still have common bounds. Expand
\[
 (\theta+\ddc v_j)^n
 =\sum_{\ell=0}^n\binom n\ell
   \bigl(\ddc(v_j+q)\bigr)^\ell
       \wedge(\theta-\ddc q)^{n-\ell}.
\]
The absolute value of the $\ell$-th summand is bounded by a fixed constant times
\[
 \bigl(\ddc(v_j+q)\bigr)^\ell\wedge\kappa_0^{n-\ell},
\]
where $\kappa_0$ is a fixed positive Euclidean form. This positive mixed measure is bounded, uniformly in $j$, by a constant times the local capacity: add a fixed strictly plurisubharmonic quadratic function to $v_j+q$, expand the top power of its Hessian, and rescale its uniformly bounded range into $[0,1]$ to apply the definition of capacity. This also covers $\ell=0$. Summing over a finite cover gives the asserted control of the total variations. The ordinary decreasing continuity of Bedford--Taylor products gives weak convergence term by term. For a bounded quasi-continuous coefficient, remove an open set of small capacity and extend the restriction of the coefficient to the compact complement to a continuous function on $X$. The variation bound controls the resulting error uniformly for the sequence and its limit, and letting the capacity of the removed set tend to zero proves the assertion.
\end{proof}

A finite non-pluripolar measure is uniformly absolutely continuous with respect to capacity, although it need not be bounded by a power of the capacity.

\Needspace{16\baselineskip}
\begin{lemma}\label{lem:npp-uniform-capacity}
Let $\mu$ be a finite non-pluripolar Borel measure on $X$. Then
\[
 \lim_{\delta\downarrow0}\sup\{\mu(E):E\text{ Borel},\
                      \operatorname{Cap}_\omega(E)\leq\delta\}=0.
\]
Consequently, convergence in capacity implies convergence in $\mu$-measure.
\end{lemma}
\begin{proof}
If the first assertion failed, there would be $\epsilon>0$ and Borel sets $E_j$ of capacity at most $2^{-j}$ with $\mu(E_j)\geq\epsilon$. By countable subadditivity of capacity,
\[
 \operatorname{Cap}_\omega\left(\limsup_j E_j\right)=0.
\]
Hence $\limsup_jE_j$ is pluripolar and has $\mu$-measure zero. On the other hand, since $\mu$ is finite, continuity from above applied to the sets $\bigcup_{k\geq j}E_k$ shows that $\limsup_jE_j$ has measure at least $\epsilon$, a contradiction. If $u_j\to u$ in capacity, applying the first assertion to the sets $\{|u_j-u|>\delta\}$ gives the consequence.
\end{proof}

\begin{lemma}\label{lem:monotone-psh-capacity}\index{Capacity!monotone convergence}
Let $u_j\in\PSH(X,\theta)$. Suppose either that $u_j$ decreases to a function $u\not\equiv-\infty$, or that $u_j$ increases, the sequence is uniformly bounded above, and $u=(\lim_j u_j)^*$. Then $u_j\to u$ in capacity.
\end{lemma}
\begin{proof}
In the decreasing case $u$ is a common lower bound for the sequence and its limit; in the increasing case $u_1$ is one. Choose a Hermitian form $\Omega\geq\theta$ and truncate all functions at $-T$. The truncations are bounded, monotone and $\Omega$-plurisubharmonic, so the convergence in capacity of locally bounded monotone sequences, recalled in \cref{sec:psh-local-capacity}, applies to them on the balls of a finite cover. Let $w$ be the common lower bound. Then $w-\sup_Xw$ is $\Omega$-plurisubharmonic with supremum zero, so \eqref{eq:global-capacity-tail}, applied with $\Omega$ in place of $\omega$, and the equivalence of the capacities $\operatorname{Cap}_\Omega$ and $\operatorname{Cap}_\omega$ show that the capacity of the sublevel set $\{w<-T\}$ tends to zero as $T\to\infty$. Since the functions agree with their truncations outside $\{w<-T\}$, letting $T\to\infty$ proves the lemma.
\end{proof}

A fixed bounded reference potential also controls a mixed trace in which one normalized potential varies.

\Needspace{9\baselineskip}
\begin{lemma}\label{lem:bounded-reference-trace}
Let $\theta\geq0$, and fix a bounded $\psi\in\PSH(X,\theta)$. There is a constant $C_\psi$ such that
\[
 \int_X\theta_v\wedge\theta_\psi^{n-1}\leq C_\psi
\]
for every bounded $v\in\PSH(X,\theta)$ with $\sup_Xv=0$.
\end{lemma}
\begin{proof}
For $n=1$, Stokes' theorem gives $\int_X\theta_v=\int_X\theta$, so assume $n\geq2$. Choose a fixed Hermitian form $\Omega\geq\theta$. All currents on the left-hand side are positive, and $\Omega_\psi\geq\theta_\psi$, so $\theta_v\wedge\theta_\psi^{n-1}\leq\theta_v\wedge\Omega_\psi^{n-1}$. Integrating and applying Stokes' theorem for bounded potentials to $\ddc v\wedge\Omega_\psi^{n-1}$, we obtain
\[
 \int_X\theta_v\wedge\theta_\psi^{n-1}
 \leq\int_X\theta\wedge\Omega_\psi^{n-1}
       +\int_Xv\,\ddc(\Omega_\psi^{n-1}).
\]
The first term on the right is bounded independently of $v$. To estimate the second, differentiate:
\[
 \ddc(\Omega_\psi^{n-1})
 =(n-1)\ddc\Omega\wedge\Omega_\psi^{n-2}
 +(n-1)(n-2)\mathrm{d}\Omega\wedge\dc\Omega\wedge\Omega_\psi^{n-3},
\]
where the second term is absent for $n=2$. Since $\ddc\Omega$ and $\mathrm d\Omega\wedge\dc\Omega$ are smooth forms, the absolute values of these terms are bounded by fixed multiples of the positive measures $\Omega^2\wedge\Omega_\psi^{n-2}$ and $\Omega^3\wedge\Omega_\psi^{n-3}$. The weighted local Chern--Levine--Nirenberg inequality of \cref{sec:psh-local-capacity} bounds the integrals of $|v|$ against these measures. Indeed, on each larger coordinate ball, add fixed smooth corrections to $v$ and $\psi$, shift the corrected $v$ so that it becomes negative, expand in the corrected positive Hessian of $\psi$, and apply that inequality on a smaller ball. The resulting bound is
\[
 C(1+\|\psi\|_\infty)^n(1+\|v\|_{L^1(X)}).
\]
By \cref{thm:normalized-compactness}, normalized $\theta$-plurisubharmonic functions have uniformly bounded $L^1$ norms, and summing the local bounds proves the assertion.
\end{proof}

\subsection{Vanishing area on curves}
\label{sec:curve-vanishing-area}

On a compact curve, the integral of the background is the total mass of every curvature current. When this integral vanishes, positivity forces the current to vanish. Normalized compactness and the local estimate of capacity by total variation make this uniqueness stable as the area tends to zero.

Throughout this subsection, $Y$ is a compact connected Riemann surface and $\eta$ is a smooth positive $(1,1)$-form on $Y$ with $\int_Y\eta=1$, which we use for $L^1$ norms and for the capacity $\operatorname{Cap}_\eta$ on $Y$.

\begin{lemma}\label{lem:curve-background-area}
For a smooth real $(1,1)$-form $\sigma$ on $Y$, put $A=\int_Y\sigma$. There is a smooth real function $h$ with $\sigma=A\eta+\ddc h$. The class $\PSH(Y,\sigma)$ is nonempty if and only if $A\geq0$. If $A=0$, it consists of the functions $-h+c$, $c\in\mathbb R$, and its unique member with supremum zero is $u_\sigma=-h+\min_Yh$.
\end{lemma}
\begin{proof}
The smooth form $\sigma-A\eta$ has integral zero, so the Poisson equation on the compact connected surface $Y$ has a smooth solution $h$, unique up to an additive constant. If $u\in\PSH(Y,\sigma)$, positivity and Stokes' theorem give
\[
 0\leq(\sigma+\ddc u)(Y)=A.
\]
Conversely, $\sigma+\ddc(-h)=A\eta$, so $-h\in\PSH(Y,\sigma)$ when $A\geq0$. If $A=0$, every positive curvature measure has total mass zero and therefore vanishes. Thus $\ddc(u+h)=0$ for every $u\in\PSH(Y,\sigma)$, and elliptic regularity and the uniqueness of harmonic functions on $Y$ show that $u+h$ is constant. Normalizing the supremum gives the stated formula.
\end{proof}

\Needspace{18\baselineskip}
\begin{theorem}\label{thm:curve-zero-area-collapse}
Let $\sigma_j$ be smooth real $(1,1)$-forms on $Y$ that converge uniformly to a smooth form $\sigma_0$, and suppose that
\[
 A_j=\int_Y\sigma_j\geq0,\qquad \int_Y\sigma_0=0.
\]
Let $u_0$ be the normalized $\sigma_0$-plurisubharmonic potential of \cref{lem:curve-background-area}. Then, for every $t>0$,
\[
 \begin{aligned}
 \sup_{u\in\PSH(Y,\sigma_j),\ \sup_Yu=0}\|u-u_0\|_{L^1(Y)}&\longrightarrow0,\\
 \sup_{u\in\PSH(Y,\sigma_j),\ \sup_Yu=0}
       \operatorname{Cap}_\eta(\{|u-u_0|>t\})&\longrightarrow0.
 \end{aligned}
\]
Moreover, the curvature measures $\sigma_j+\ddc u$ of these potentials tend to zero in total variation, uniformly in $u$, since each has total variation $A_j$.
\end{theorem}
\begin{proof}
Choose $C>0$ with $\sigma_j\leq C\eta$ for every $j$. Every sequence of normalized $\sigma_j$-plurisubharmonic functions then lies in the compact family of normalized $C\eta$-plurisubharmonic functions of \cref{thm:normalized-compactness}. Every $L^1$ limit $u$ of a subsequence is normalized and, by passing to the limit in the sense of distributions, satisfies $\sigma_0+\ddc u\geq0$. \Cref{lem:curve-background-area} gives $u=u_0$. If the first uniform limit failed, a sequence violating it would have a subsequence converging to this same limit, a contradiction.

For a potential $u$ in the $j$-th normalized family, put $T=\sigma_j+\ddc u$. Positivity and Stokes' theorem give $\|T\|_{\mathrm{TV}}=A_j$, and $A_j\to\int_Y\sigma_0=0$. Since $\sigma_0+\ddc u_0=0$,
\[
 \|\ddc(u-u_0)\|_{\mathrm{TV}}
 \leq A_j+\|\sigma_j-\sigma_0\|_{\mathrm{TV}}\longrightarrow0
\]
uniformly over that family. Apply \cref{lem:planar-variation-capacity} on a fixed finite cover of $Y$ by concentric coordinate disks. The first limit and the bound above make each local capacity tend to zero uniformly, and the comparison between local and global capacity at the beginning of this section proves the second limit. The identity $\|T\|_{\mathrm{TV}}=A_j$ proves the final assertion.
\end{proof}
