\section{Hermitian capacity}
\label{sec:hermitian-capacity}

Local capacity tests can be glued to the constant function on a compact manifold after a fixed rescaling. This compares local Bedford--Taylor capacity with a capacity defined by global potentials and transfers both volume estimates and normalized sublevel estimates to $X$.

For a Borel set $E\subset X$, put
\[
 \operatorname{Cap}_\omega(E)
 =\sup_{v\in\PSH(X,\omega),\ 0\leq v\leq1}
             \int_E(\omega+\ddc v)^n.
\]
The products are those of \cref{prop:bounded-products}. Their local expansion and \eqref{eq:local-cln-range} show that $\operatorname{Cap}_\omega(X)<+\infty$. On relatively compact coordinate balls, a fixed smooth correction bounds the measures of global tests by a constant times the corresponding local capacity. Conversely, the gluing construction below bounds local capacity by global capacity. Thus the quasi-continuity and capacity-zero characterizations in \cref{sec:psh-local-capacity} hold for $\operatorname{Cap}_\omega$. Convergence in this capacity means
\[
 \operatorname{Cap}_\omega(\{|u_j-u|>\delta\})\longrightarrow0
 \quad\text{for every }\delta>0,
\]
where pole sets may be omitted. The bounded-product continuity statements of \cref{sec:psh-local-capacity} pass to the Hermitian products by their local expansion and a finite cover.

For the following estimates, abbreviate $\operatorname{Cap}=\operatorname{Cap}_\omega$ and $\omega_v=\omega+\ddc v$, and let $\operatorname{Vol}$ denote the volume for the fixed smooth measure $\mathrm{d}V$.

\begin{lemma}\label{lem:global-capacity-estimates}
There are constants $a,C>0$ such that
\begin{equation}\label{eq:global-volume-capacity}
 \operatorname{Vol}(E)\leq C\exp\bigl(-a\operatorname{Cap}(E)^{-1/n}\bigr)
\end{equation}
for every Borel $E\subset X$. Also, for $u\in\PSH(X,\omega)$ with $\sup_Xu=0$ and $t>1$,
\begin{equation}\label{eq:global-capacity-tail}
 \operatorname{Cap}(\{u<-t\})\leq C/t.
\end{equation}
\end{lemma}
\begin{proof}
Take a finite cover by smaller coordinate balls $B'_j\Subset B_j$. A local capacity test $0\leq v\leq1$ on $B_j$ can be made into a global test as follows. Choose a smooth function $\chi_j$ equal to zero near $\overline{B'_j}$, equal to $-2$ near $\partial B_j$, and with $-2\leq\chi_j\leq0$. Choose $\delta>0$ small enough that
\[
 \omega+\delta\ddc\chi_j\geq0,\qquad 3\delta/2\leq1.
\]
\Needspace{6\baselineskip}
The function
\[
 V=\max\{\delta(v+\chi_j),-\delta/2\}+\delta/2
 \quad\hbox{on }B_j,
 \qquad V=0\quad\hbox{outside }B_j
\]
is globally $\omega$-plurisubharmonic and lies between zero and one. Near the boundary the constant branch is selected. On $B'_j$ we have $\omega_V\geq\delta\ddc v$. Positivity of bounded products therefore gives
\[
 \operatorname{cap}(E\cap B'_j,B_j)
 \leq\delta^{-n}\operatorname{Cap}(E).
\]
The local volume estimate, followed by summation over the finite cover, proves \eqref{eq:global-volume-capacity}.

For the second estimate, choose on each $B_j$ a smooth negative quadratic function $r_j$ with $\ddc r_j\geq\omega$. For every global test $0\leq v\leq1$, the functions $u+r_j\leq0$ and $v+r_j$ are plurisubharmonic. Thus
\[
 \begin{split}
 \int_{\{u<-t\}}\omega_v^n
 &\leq t^{-1}\int_X(-u)\omega_v^n\\
 &\leq t^{-1}\sum_j\int_{B'_j}(-u-r_j)
                    \bigl(\ddc(v+r_j)\bigr)^n\\
 &\leq \frac Ct\left(1+\int_X|u|\,\mathrm{d}V\right).
 \end{split}
\]
Here the second inequality uses $\omega_v\leq\ddc(v+r_j)$, and the third is the local weighted estimate; the local test functions have a common bounded range. The normalized $L^1$ bound in \cref{thm:normalized-compactness} finishes the proof after taking the supremum over $v$.
\end{proof}

\begin{corollary}\label{cor:lp-capacity-square}
Fix $p>1$. There is $C_p$ such that every nonnegative $F\in L^p(X,\mathrm{d}V)$ satisfies
\[
 \int_E F\,\mathrm{d}V\leq C_p\|F\|_p\operatorname{Cap}(E)^2
\]
for every Borel set $E$.
\end{corollary}
\begin{proof}
H\"older's inequality and \eqref{eq:global-volume-capacity} bound the integral by
\[
 C\|F\|_p\exp(-a\operatorname{Cap}(E)^{-1/n}/p'),
 \qquad p'=p/(p-1).
\]
The quotient of this exponential by $x^2$ is bounded on $0<x\leq\operatorname{Cap}(X)<\infty$. Capacity-zero sets have zero smooth volume.
\end{proof}

The same proof bounds an $L^p$ density by any fixed positive power of capacity. The choice of the exponent two will be convenient for the sublevel iteration in \cref{lem:backwards-capacity}.

For quasi-continuous obstacles, a decreasing lower semicontinuous approximation preserves the original bounds and confines the error to sets of vanishing capacity.

\begin{lemma}
\label{lem:quasicont-lsc-approximation}
Every bounded quasi-continuous function $f$ admits bounded lower semicontinuous functions $f_j$ with common bounds such that $f_j\downarrow f$ quasi-everywhere and $f_j\to f$ in capacity.
\end{lemma}
\begin{proof}
Choose open sets $O_j$ with capacity at most $2^{-j}$ such that $f$ is continuous on $X\setminus O_j$, and set $G_j=\bigcup_{k\geq j}O_k$. Then $G_j$ decreases, its capacity tends to zero, and $f$ is continuous on its closed complement. Choose $M\geq\sup_Xf$ and define
\[
 f_j=f\ \hbox{on }X\setminus G_j,\qquad f_j=M\ \hbox{on }G_j.
\]
At a boundary point of $G_j$, limits from the closed complement give $f$, while values from inside are at least that value. Thus $f_j$ is lower semicontinuous. The functions decrease to $f$ outside $\bigcap_jG_j$, a capacity-zero set. Their difference from $f$ is supported in $G_j$, which also proves convergence in capacity.
\end{proof}

Uniform bounds on potentials give a common capacity bound for their mixed measures.

\begin{lemma}\label{lem:bounded-mixed-capacity}
Fix a Hermitian metric $\omega$, a finite $A\geq0$ and an integer $0\leq k\leq n$. All positive mixed measures
\[
 \omega_{v_1}\wedge\cdots\wedge\omega_{v_k}\wedge\omega^{n-k},
 \qquad \|v_i\|_\infty\leq A,
 \quad v_i\in\PSH(X,\omega),
\]
are bounded above on Borel sets by $C_A\operatorname{Cap}_\omega$. In particular their total masses are bounded. Their integrals against uniformly bounded functions tending to zero in capacity tend uniformly to zero.
\end{lemma}
\begin{proof}
Translate the $v_i$ so that $0\leq v_i\leq2A$, and put
$q=(v_1+\cdots+v_k)/B$, where $B\geq\max\{2k,2kA,1\}$.
Then $0\leq q\leq1$ and
\[
 \omega_q=B^{-1}\sum_{i=1}^k\omega_{v_i}+(1-k/B)\omega.
\]
All coefficients are positive. Expanding $\omega_q^n$ bounds the indicated mixed product by a constant multiple of $\omega_q^n$. The definition of capacity proves the claim. The case $k=0$ follows by testing with zero. For a bounded function $z$, split its integral into $\{|z|>t\}$ and its complement; first use capacity convergence for fixed $t>0$, and then let $t\downarrow0$.
\end{proof}

The same control applies to the total variation of a product formed with a different smooth background.

\begin{lemma}\label{lem:signed-truncation-quasicont}
Let $\Omega$ be Hermitian and $\theta$ any smooth real $(1,1)$-form. Suppose $v_j$ are $\Omega$-plurisubharmonic, have common finite upper and lower bounds, and decrease to $v$. Then the signed measures
\[
 \nu_j=(\theta+\ddc v_j)^n
\]
converge to $(\theta+\ddc v)^n$ against every fixed bounded quasi-continuous function. Their total variations are uniformly controlled by local capacity on a fixed finite cover.
\end{lemma}
\begin{proof}
On a relatively compact coordinate ball, choose a fixed quadratic $q$ such that $v_j+q$ are plurisubharmonic. Their common bounds remain uniform. Expand
\[
 (\theta+\ddc v_j)^n
 =\sum_{\ell=0}^n\binom n\ell
   \bigl(\ddc(v_j+q)\bigr)^\ell
       \wedge(\theta-\ddc q)^{n-\ell}.
\]
The absolute value of the $\ell$th summand is bounded by a fixed constant times
\[
 \bigl(\ddc(v_j+q)\bigr)^\ell\wedge\kappa_0^{n-\ell},
\]
where $\kappa_0$ is a fixed positive Euclidean form. This mixed positive measure is uniformly bounded by a constant times local capacity: add a fixed strictly plurisubharmonic quadratic to $v_j+q$, expand its top power, and rescale its uniformly bounded range into $[0,1]$ for the capacity definition. This also handles $\ell=0$. Summing on a finite cover gives the asserted variation control. Ordinary decreasing Bedford--Taylor continuity gives weak convergence term by term. For a bounded quasi-continuous coefficient, remove an open set of small capacity and extend the coefficient continuously from its compact complement. The variation bound controls the error uniformly for the sequence and its limit. Letting that capacity tend to zero proves the assertion.
\end{proof}

A finite non-pluripolar measure is uniformly absolutely continuous with respect to capacity, although it need not satisfy a power bound.

\Needspace{16\baselineskip}
\begin{lemma}\label{lem:npp-uniform-capacity}
Let $\mu$ be a finite non-pluripolar Borel measure on $X$. Then
\[
 \lim_{\delta\downarrow0}\sup\{\mu(E):E\text{ Borel},\
                      \operatorname{Cap}_\omega(E)\leq\delta\}=0.
\]
Consequently, capacity convergence implies convergence in $\mu$-measure.
\end{lemma}
\begin{proof}
If the first assertion failed, there would be $\epsilon>0$ and Borel sets $E_j$ with capacity at most $2^{-j}$ and $\mu(E_j)\geq\epsilon$. Countable subadditivity of capacity gives
\[
 \operatorname{Cap}_\omega\left(\limsup_j E_j\right)=0.
\]
The limsup is therefore pluripolar and has zero $\mu$-measure. On the other hand, finiteness of $\mu$ and continuity from above applied to $\bigcup_{j\geq k}E_j$ show that the limsup has measure at least $\epsilon$, a contradiction. Apply the first assertion to the sets on which the difference of two converging potentials exceeds a fixed positive number to get the consequence.
\end{proof}

\begin{lemma}\label{lem:monotone-psh-capacity}
Let $u_j\in\PSH(X,\theta)$. If $u_j$ decreases to a genuine potential $u$, or increases with a common upper bound and $u=(\lim_j u_j)^*$, then $u_j\to u$ in capacity.
\end{lemma}
\begin{proof}
In the decreasing case $u$ is a common lower bound; in the increasing case $u_1$ is one. Choose a Hermitian $\Omega\geq\theta$ and truncate at $-T$. Bounded monotone capacity convergence applies to the truncations. Quasi-continuity of the common lower bound permits removal of an open set of arbitrarily small capacity containing its pole set and all its discontinuities. On the compact complement that lower bound is finite and continuous, hence bounded below. Its deep sublevels therefore have capacity tending to zero. Let $T\to\infty$.
\end{proof}

A fixed bounded reference also controls a mixed trace with one normalized, varying potential.

\begin{lemma}\label{lem:bounded-reference-trace}
Let $\theta\geq0$ and fix a bounded $\psi\in\PSH(X,\theta)$. There is a constant $C_\psi$ such that
\[
 \int_X\theta_v\wedge\theta_\psi^{n-1}\leq C_\psi
\]
for every bounded $v\in\PSH(X,\theta)$ with $\sup_Xv=0$.
\end{lemma}
\begin{proof}
Choose a fixed Hermitian metric $\Omega\geq\theta$. All currents in the left side are positive, and $\Omega_\psi\geq\theta_\psi$. Bounded Stokes gives
\[
 \int_X\theta_v\wedge\theta_\psi^{n-1}
 \leq\int_X\theta\wedge\Omega_\psi^{n-1}
       +\int_Xv\,\ddc(\Omega_\psi^{n-1}).
\]
The first term has a fixed bound. For $n\geq2$, the exact differentiation formula is
\[
 \ddc(\Omega_\psi^{n-1})
 =(n-1)\ddc\Omega\wedge\Omega_\psi^{n-2}
 +(n-1)(n-2)\mathrm{d}\Omega\wedge\dc\Omega\wedge\Omega_\psi^{n-3},
\]
where the second term is absent for $n=2$. Its absolute value is bounded by fixed multiples of the corresponding positive mixed measures with $\Omega^2$ and $\Omega^3$ in place of the smooth torsion factors. The weighted local Chern--Levine--Nirenberg inequality from \cref{sec:psh-local-capacity} bounds the integrals of $|v|$ against these measures. Indeed, on each larger coordinate ball add fixed smooth corrections to $v$ and $\psi$, shift the first to be negative, expand in the corrected positive Hessian of $\psi$, and apply that inequality on a smaller ball. The bound is
\[
 C(1+\|\psi\|_\infty)^n(1+\|v\|_{L^1(X)}).
\]
Normalized $\theta$-plurisubharmonic functions have uniformly bounded $L^1$ norms. Summing the local bounds proves the assertion. For $n=1$, the second integral is zero by Stokes, and the result is immediate.
\end{proof}

\subsection{Vanishing area on curves}
\label{sec:curve-vanishing-area}

On a compact curve, the integral of the background is the total mass of every curvature current. When this integral vanishes, positivity forces the current to vanish. Normalized compactness and the local variation-capacity estimate make this uniqueness stable as the area tends to zero.

Let $Y$ be a compact connected Riemann surface and fix a smooth positive form $\eta$ with $\int_Y\eta=1$. Use $\eta$ for $L^1$ norms and capacity on $Y$.

\begin{lemma}\label{lem:curve-background-area}
For a smooth real $(1,1)$-form $\sigma$ on $Y$, put $A=\int_Y\sigma$. There is a smooth real $h$ with $\sigma=A\eta+\ddc h$. The class $\PSH(Y,\sigma)$ is nonempty exactly when $A\geq0$. If $A=0$, it consists of the functions $-h+c$, $c\in\mathbb R$, and its unique supremum-zero member is $u_\sigma=-h+\min_Yh$.
\end{lemma}
\begin{proof}
The smooth form $\sigma-A\eta$ has integral zero. The scalar Poisson equation on the compact connected surface therefore has a smooth solution $h$, unique up to a constant. If $u\in\PSH(Y,\sigma)$, Stokes gives
\[
 (\sigma+\ddc u)(Y)=A\geq0.
\]
Conversely, $-h$ has curvature $A\eta$, so is admissible when $A\geq0$. If $A=0$, every positive curvature measure has mass zero and vanishes. Thus $\ddc(u+h)=0$; elliptic regularity and harmonic uniqueness make $u+h$ constant. Supremum normalization gives the stated formula.
\end{proof}

\begin{theorem}\label{thm:curve-zero-area-collapse}
Let smooth real forms $\sigma_j$ converge uniformly to a smooth form $\sigma_0$, and suppose
\[
 A_j=\int_Y\sigma_j\geq0,\qquad \int_Y\sigma_0=0.
\]
Let $u_0$ be the normalized $\sigma_0$-plurisubharmonic potential from \cref{lem:curve-background-area}. Then, for every $t>0$,
\[
 \begin{aligned}
 \sup_{u\in\PSH(Y,\sigma_j),\ \sup_Yu=0}\|u-u_0\|_{L^1(Y)}&\longrightarrow0,\\
 \sup_{u\in\PSH(Y,\sigma_j),\ \sup_Yu=0}
       \operatorname{Cap}_\eta(\{|u-u_0|>t\})&\longrightarrow0.
 \end{aligned}
\]
The full curvature measures tend uniformly to zero in total variation: each has norm $A_j$.
\end{theorem}
\begin{proof}
Choose $C>0$ with $\sigma_j\leq C\eta$ for every $j$. Any sequence of normalized $\sigma_j$-plurisubharmonic functions belongs to the compact normalized family for $C\eta$. Every $L^1$ cluster limit $u$ is normalized and satisfies $\sigma_0+\ddc u\geq0$, by passing to distributions. The preceding lemma gives $u=u_0$. A sequence violating the first uniform limit would have a convergent subsequence with this same limit, a contradiction.

For any potential in the $j$th normalized family, put $T=\sigma_j+\ddc u$. Positivity and Stokes give $\|T\|_{\mathrm{TV}}=A_j$. Since $\sigma_0+\ddc u_0=0$,
\[
 \|\ddc(u-u_0)\|_{\mathrm{TV}}
 \leq A_j+\|\sigma_j-\sigma_0\|_{\mathrm{TV}}\longrightarrow0
\]
uniformly over that family. Apply \cref{lem:planar-variation-capacity} on a fixed finite cover by concentric coordinate discs. The first limit and the displayed bound make each local capacity tend uniformly to zero. The local-global capacity comparison then proves the second limit. The current norm identity also proves the final assertion.
\end{proof}
