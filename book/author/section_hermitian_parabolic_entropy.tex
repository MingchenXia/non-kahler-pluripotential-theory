\subsection{Convex entropies along a Monge--Amp\`ere flow}
\label{subsec:hermitian-parabolic-entropy}\index{Entropy!Monge--Ampere flow@Monge--Amp\`ere flow}

The logarithmic Monge--Amp\`ere operator generates a parabolic flow whose linearization is the operator $L_u$ of \eqref{eq:linearized-ma}. Along this flow, the ratio of the Monge--Amp\`ere measure to a fixed target measure satisfies a heat-type equation. Integrating a convex function of this ratio by parts produces, besides the usual nonpositive gradient term, a term involving $\ddc(\omega_{u_t}^{n-1})$. If the forms along the flow are Gauduchon, this additional term vanishes and all convex entropies decrease. Requiring decrease for every smooth positive target at the initial time characterizes the Gauduchon condition.

Recall that $M=\int_X\omega^n$, and fix a smooth strictly positive probability volume form $\nu$ on $X$. Consider a smooth solution on an interval $[0,T)$ of the \emph{logarithmic Monge--Amp\`ere flow}
\begin{equation}\label{eq:hermitian-parabolic-entropy-flow}
 \dot u_t=\log\rho_t,\qquad
 \rho_t=\frac{\omega_{u_t}^n}{M\nu},\qquad
 u_0=0,\qquad \omega_{u_t}>0.
\end{equation}
The normalizing constant $M$ is the initial mass and stays fixed along the flow. Define the linearized operator $\Delta_t$ by
\[
 \Delta_t v\,\omega_{u_t}^n
 =n\ddc v\wedge\omega_{u_t}^{n-1};
\]
it is the operator $L_{u_t}$ of \eqref{eq:linearized-ma}. Logarithmic differentiation gives $\partial_t\log\rho_t=\Delta_t\dot u_t=\Delta_t\log\rho_t$. Integration by parts against $\omega_{u_t}^{n-1}$ then produces the same derivative of the background power that appears in the mass variation of \cref{prop:mass-variation}.

\begin{proposition}\label{prop:hermitian-parabolic-entropy-identity}
Let $\Phi\in C^2((0,+\infty),\R)$, and choose a function $G$ with $G'(r)=\Phi'(r)/r$. Along \eqref{eq:hermitian-parabolic-entropy-flow},
\begin{equation}\label{eq:hermitian-parabolic-entropy-identity}
\begin{split}
 \frac{\mathrm{d}}{\mathrm{d}t}\int_X\Phi(\rho_t)\nu
 ={}&\frac nM\int_XG(\rho_t)\ddc(\omega_{u_t}^{n-1})\\
 &-\frac nM\int_X\frac{\Phi''(\rho_t)}{\rho_t}
       \mathrm{d}\rho_t\wedge\dc\rho_t\wedge\omega_{u_t}^{n-1}.
\end{split}
\end{equation}
In particular, for $p>1$,
\begin{equation}\label{eq:hermitian-parabolic-power-entropy}
\begin{split}
 \frac{\mathrm{d}}{\mathrm{d}t}\int_X\rho_t^p\nu
 ={}&\frac{np}{(p-1)M}\int_X\rho_t^{p-1}\ddc(\omega_{u_t}^{n-1})\\
 &-\frac{np(p-1)}M\int_X\rho_t^{p-3}
       \mathrm{d}\rho_t\wedge\dc\rho_t\wedge\omega_{u_t}^{n-1}.
\end{split}
\end{equation}
\end{proposition}
\begin{proof}
Using $\partial_t\rho_t=\rho_t\Delta_t\log\rho_t$ and $\rho_t\nu=M^{-1}\omega_{u_t}^n$, we find that the derivative of the entropy is
\[
 \frac nM\int_X\Phi'(\rho_t)\ddc\log\rho_t
                  \wedge\omega_{u_t}^{n-1}.
\]
The chain rule and the definition of $G$ give
\[
 \Phi'(r)\ddc\log r
 =\ddc G(r)-\frac{\Phi''(r)}r\mathrm{d}r\wedge\dc r,
\]
and \cref{lem:ibp} applies to the first term. Adding a constant to $G$ does not change the result, since $\int_X\ddc(\omega_{u_t}^{n-1})=0$. Taking $\Phi(r)=r^p$ and $G(r)=p r^{p-1}/(p-1)$ proves the second formula.
\end{proof}

\begin{corollary}\label{cor:gauduchon-parabolic-entropies}\index{Gauduchon metric!convex entropy dissipation}
Suppose that every form $\omega_{u_t}$ in \eqref{eq:hermitian-parabolic-entropy-flow} is Gauduchon. Then $\int_X\rho_t\nu=1$, and $t\mapsto\int_X\Phi(\rho_t)\nu$ is nonincreasing for every convex function $\Phi\in C^2((0,+\infty),\R)$. In particular,
\[
 \|\rho_t\|_{L^p(\nu)}\leq\|\rho_s\|_{L^p(\nu)}
 \quad(0\leq s\leq t<T,\ p>1).
\]
If $\Phi''>0$, the derivative of the entropy vanishes at a time $t$ precisely when $\rho_t=1$, in which case $\dot u_t=0$.

The hypothesis holds on a Gauduchon surface, and in any dimension it holds if $\ddc(\omega^j)=0$ for $1\leq j\leq n-1$.
\end{corollary}
\begin{proof}
The derivative of the mass is
\[
 \frac{\mathrm{d}}{\mathrm{d}t}\int_X\rho_t\nu
 =\frac nM\int_X\dot u_t\ddc(\omega_{u_t}^{n-1})=0,
\]
and its initial value is one. In \eqref{eq:hermitian-parabolic-entropy-identity} the first term vanishes, while the second is nonpositive, because $\mathrm{d}\rho_t\wedge\dc\rho_t$ is a semipositive $(1,1)$-form and $\Phi''\geq0$. If $\Phi''>0$, the vanishing of this term forces $\mathrm{d}\rho_t=0$, and connectedness together with the mass normalization gives $\rho_t=1$. The converse and the assertion about $\dot u_t$ follow from the equation.

On a surface, $\ddc\omega_{u_t}=\ddc\omega=0$. Under the power identities, expand $\omega_{u_t}^{n-1}$ by the binomial theorem; since the factors $\ddc u_t$ are annihilated by $\mathrm d$ and $\dc$, each term has vanishing complex Hessian. These are also the persistence conditions of \cref{prop:gauduchon-potential-persistence,thm:mass-invariance-characterization}.
\end{proof}

For $\Phi(r)=r\log r$, the same formula gives the dissipation of the relative entropy. In higher dimension, the Gauduchon condition on the initial form alone need not persist along potential perturbations, as \cref{ex:gauduchon-persistence-torus} shows. The next criterion determines the role of this condition at the initial time, without assuming that the flow exists.

\begin{proposition}\label{prop:gauduchon-initial-entropy-criterion}\index{Gauduchon metric!initial entropy criterion}
Fix $p>1$. For each smooth strictly positive probability volume form $\nu$, put $\rho_\nu=\omega^n/(M\nu)$ and define
\[
 \mathcal D_p(\omega,\nu)
 :=\left.\frac{\mathrm{d}}{\mathrm{d}t}\right|_{t=0}
   \int_X\left(\frac{(\omega+t\ddc\log\rho_\nu)^n}{M\nu}\right)^p\nu.
\]
Then $\mathcal D_p(\omega,\nu)\leq0$ for every such $\nu$ if and only if $\omega$ is Gauduchon. If $\omega$ is not Gauduchon, there is such a $\nu$ with $\mathcal D_p(\omega,\nu)>0$.
\end{proposition}
\begin{proof}
The form in the definition is positive for sufficiently small $|t|$. The path $u_t=t\log\rho_\nu$ starts at $u_0=0$ with the velocity $\log\rho_\nu$ prescribed by \eqref{eq:hermitian-parabolic-entropy-flow} at time zero. The proof of \cref{prop:hermitian-parabolic-entropy-identity} at a given time uses only the potential and its velocity at that time, so \eqref{eq:hermitian-parabolic-power-entropy} holds for the derivative at $t=0$. If $\omega$ is Gauduchon, its first term vanishes and the derivative is nonpositive.

Conversely, suppose that $\lambda:=\ddc(\omega^{n-1})\ne0$. Write $\lambda=f\omega^n/M$, where $f$ is smooth and real. Stokes' theorem gives $\int_Xf\omega^n=0$, whereas
\[
 \int_Xf\lambda=\frac1M\int_Xf^2\omega^n>0.
\]
For small $\varepsilon>0$, set
\[
 c_\varepsilon=\frac1M\int_X\mathrm{e}^{-\varepsilon f}\omega^n,
 \qquad r_\varepsilon=c_\varepsilon\mathrm{e}^{\varepsilon f},
 \qquad \nu_\varepsilon=\frac{\omega^n}{M r_\varepsilon}.
\]
Then $\nu_\varepsilon$ is a smooth strictly positive probability volume form, and $\rho_{\nu_\varepsilon}=r_\varepsilon$. Since $c_\varepsilon=1+O(\varepsilon^2)$, $\int_X\lambda=0$ and $\mathrm{d}r_\varepsilon=\varepsilon r_\varepsilon\mathrm{d}f$, the two terms in \eqref{eq:hermitian-parabolic-power-entropy} give
\[
 \mathcal D_p(\omega,\nu_\varepsilon)
 =\frac{np}M\varepsilon\int_Xf\lambda+O(\varepsilon^2)>0
\]
for sufficiently small $\varepsilon>0$. This proves necessity and the last assertion.
\end{proof}

The criterion involves only the initial velocity: every smooth solution of \eqref{eq:hermitian-parabolic-entropy-flow} with target $\nu$ has the same initial entropy derivative, independently of its existence for later times or its convergence. Quantification over all targets is essential. For the target $\nu=\omega^n/M$, the flow $u_t=0$ is stationary for every Hermitian $\omega$, and all entropy derivatives vanish.
