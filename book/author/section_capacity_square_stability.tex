\subsection{Stability under a power capacity bound}
\label{subsec:capacity-square-stability}\label{subsec:power-capacity-stability}

The capacity lower bound of \cref{lem:power-backwards-capacity} turns an integral error against the solution measure into a uniform error. The argument uses only the power capacity bound on this measure and therefore applies to singular measures as well.

\begin{proposition}\label{prop:capacity-square-one-sided-stability}\label{prop:power-capacity-one-sided-stability}\index{Stability!capacity-dominated measures}\index{Stability!power capacity bounds}
Fix $A,\alpha>0$ and $H\geq1$. Let $u,v\in\PSH(X,\omega)\cap C(X)$ satisfy $\sup_Xu=0$ and $-H\leq v\leq0$, and suppose
\[
 \mu(E)\leq A\operatorname{Cap}_\omega(E)^{1+\alpha},
 \qquad \mu=\omega_u^n,
 \qquad E\subset X\text{ Borel}.
\]
There is a constant $C$, depending only on $(X,\omega),A,\alpha,H$, such that
\begin{equation}\label{eq:power-one-sided-stability}
 \sup_X(v-u)_+
 \leq C\left(\int_X(v-u)_+\,\mathrm d\mu\right)^{\sigma_\alpha},
 \qquad \sigma_\alpha=\frac{\alpha}{3n(1+\alpha)+\alpha}.
\end{equation}
For the square-capacity bound, $\alpha=1$, the exponent is $1/(6n+1)$.
\end{proposition}
\begin{proof}
Write $L=\int_X(v-u)_+\,\mathrm d\mu$ and $S=\sup_X(v-u)_+$. Apply \cref{lem:power-backwards-capacity} with competitor zero, $\varepsilon=1/2$ and one fixed $0<D<\delta_{1/2}$. It bounds $\operatorname{Cap}_\omega\{u<\inf_Xu+D\}$ below by a positive constant. The tail estimate \eqref{eq:global-capacity-tail} therefore bounds $-\inf_Xu$ in terms of $(X,\omega),A,\alpha$. Consequently $S$ has a fixed upper bound.

For $0<\varepsilon\leq1/2$, choose a fixed small $c>0$ and put $D=c\varepsilon^3$. The formula for $\delta_\varepsilon$ in \cref{lem:thin-sublevel-capacity} allows $c$ to be chosen so that $2D<\delta_\varepsilon$ and $4D\leq\varepsilon$. With
\[
 m=\inf_X(u-(1-\varepsilon)v),\qquad
 U_s=\{u<(1-\varepsilon)v+m+s\},
\]
we have $m\leq-S$ when $S>0$, since $v\leq0$. On $U_{2D}$,
\[
 v-u>S-\varepsilon H-2D.
\]
If $S>2\varepsilon H+4D$, this is greater than $S/2$. The backwards and thin-sublevel estimates give
\[
 c_0D^{n(1+1/\alpha)}
 \leq D^n\operatorname{Cap}_\omega(U_D)
 \leq C_H\mu(U_{2D})\leq\frac{2C_HL}{S},
\]
where $c_0>0$ depends only on the stated data. Hence $S\leq CL\varepsilon^{-3n(1+1/\alpha)}$ in this case. In the other case, $S\leq(2H+1)\varepsilon$.

For sufficiently small $L>0$, choose $\varepsilon=L^{\sigma_\alpha}$. The two alternatives give \eqref{eq:power-one-sided-stability}. The remaining positive values of $L$ are covered by the fixed upper bound for $S$. If $L=0$, the first alternative is impossible for every small $\varepsilon$, and the second gives $S=0$ on letting $\varepsilon\downarrow0$.
\end{proof}
