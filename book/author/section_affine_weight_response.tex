\subsubsection{Variation of continuous weights}
\label{sec:affine-weight-response}

Supporting probabilities also describe how the envelope responds to changes of the weight, jointly with changes of the growth allowance. Keep the non-pluripolar compact set $K\subset\mathbb C^n$ fixed. For $a\in\mathbb C^n$, $\delta>0$ and $q\in C(K,\mathbb R)$, let
\begin{equation}\label{eq:affine-active-supports}
 \mathcal J^{\mathrm{act}}_a(\delta,q)=
 \left\{(\mu,c)\in\mathcal J_a(K):
   \int_Kq\,\mathrm{d}\mu+\delta c=F_{\delta,q}(a)\right\}
\end{equation}
be the set of pairs that are active at $(\delta,q)$. We give the weights the uniform norm, the probability measures the weak topology, and the constants the usual topology of $\mathbb R$.

\begin{lemma}\label{lem:affine-active-support-stability}
The set $\mathcal J^{\mathrm{act}}_a(\delta,q)$ is nonempty, convex and compact. For $\delta,\eta\geq\epsilon>0$ and $p,q\in C(K,\mathbb R)$,
\begin{equation}\label{eq:affine-weight-growth-lipschitz}
 |F_{\eta,p}(a)-F_{\delta,q}(a)|
 \leq\|p-q\|_\infty+
 |\eta-\delta|\left(V_K^*(a)+
                    \frac{\operatorname{osc}_Kq}{\epsilon}\right).
\end{equation}
If $\delta_j\to\delta>0$, $q_j\to q$ uniformly and $(\mu_j,c_j)\in\mathcal J^{\mathrm{act}}_a(\delta_j,q_j)$, then these pairs have a convergent subsequence, and every subsequential limit belongs to $\mathcal J^{\mathrm{act}}_a(\delta,q)$.
\end{lemma}
\begin{proof}
Nonemptiness follows from \cref{thm:affine-supporting-probabilities}. Since the test functions are continuous, $\mathcal J_a(K)$ is closed and convex, and the equality defining active pairs is a closed affine condition. For an active pair, the defining equality and $\int_Kq\,\mathrm{d}\mu\geq\inf_Kq$ give the first upper bound below, and \eqref{eq:affine-obstacle-sandwich}, applied with the weight $q$, gives the second:
\[
 0\leq c\leq\frac{F_{\delta,q}(a)-\inf_Kq}{\delta}
 \leq V_K^*(a)+\frac{\operatorname{osc}_Kq}{\delta}.
\]
The pairs in $\mathcal J_a(K)$ with bounded constants form a compact set, as in the proof of \cref{thm:affine-supported-growth-derivative}; hence the set of active pairs is compact.

With $q$ fixed, \cref{lem:affine-growth-increment-bounds} and \eqref{eq:affine-obstacle-sandwich}, applied with the weight $q$, bound $|F_{\eta,q}(a)-F_{\delta,q}(a)|$ by $|\eta-\delta|(V_K^*(a)+\operatorname{osc}_Kq/\epsilon)$. Comparison of obstacles and translation by constants bound $|F_{\eta,p}(a)-F_{\eta,q}(a)|$ by $\|p-q\|_\infty$. Adding these bounds gives \eqref{eq:affine-weight-growth-lipschitz}.

For the sequence in the statement, the constants $c_j$ have a common bound, by the estimate above, because the weights $q_j$ are uniformly bounded and the parameters $\delta_j$ are bounded below by a positive constant. Weak compactness of probability measures gives a convergent subsequence of pairs, and its limit belongs to $\mathcal J_a(K)$, which is closed. The estimate \eqref{eq:affine-weight-growth-lipschitz} gives $F_{\delta_j,q_j}(a)\to F_{\delta,q}(a)$, while uniform convergence of the $q_j$ and weak convergence of the $\mu_j$ give $\int_Kq_j\,\mathrm{d}\mu_j\to\int_Kq\,\mathrm{d}\mu$. Passing to the limit in the equality defining active pairs proves the last assertion.
\end{proof}

\begin{proposition}\label{prop:affine-joint-directional-response}
For $r\in C(K,\mathbb R)$ and $s\in\mathbb R$, one has
\begin{equation}\label{eq:affine-joint-directional-response}
 \lim_{h\downarrow0}
 \frac{F_{\delta+hs,q+hr}(a)-F_{\delta,q}(a)}h
 =\min_{(\mu,c)\in\mathcal J^{\mathrm{act}}_a(\delta,q)}
       \left(\int_Kr\,\mathrm{d}\mu+sc\right).
\end{equation}
The same limit holds along any sequence $h_j\downarrow0$ if $s,r$ in the difference quotient are replaced by $s_j,r_j$, where $s_j\to s$ and $r_j\to r$ uniformly. Thus the one-sided directional derivative is stable under convergence of the directions.
\end{proposition}
\begin{proof}
By compactness of the set of active pairs, the minimum on the right is attained. For a fixed pair $(\mu,c)$ active at $(\delta,q)$, the representation \eqref{eq:affine-supporting-probability-minimum} at the perturbed parameters gives
\[
 F_{\delta+hs,q+hr}(a)
 \leq F_{\delta,q}(a)+h\left(\int_Kr\,\mathrm{d}\mu+sc\right).
\]
Choosing an active pair that attains the minimum on the right, we see that the upper limit of the difference quotients is at most this minimum. Conversely, choose a pair $(\mu_h,c_h)$ active at $(\delta+hs,q+hr)$. The representation at $(\delta,q)$ gives
\[
 \frac{F_{\delta+hs,q+hr}(a)-F_{\delta,q}(a)}h
 \geq\int_Kr\,\mathrm{d}\mu_h+sc_h.
\]
The estimate \eqref{eq:affine-weight-growth-lipschitz} bounds the difference quotients. Along a sequence of values of $h$ realizing their lower limit, \cref{lem:affine-active-support-stability} gives a subsequence of pairs converging to a pair active at $(\delta,q)$. For this pair, $\int_Kr\,\mathrm{d}\mu+sc$ is at least the minimum on the right. This proves \eqref{eq:affine-joint-directional-response}.

For convergent directions, the same two inequalities hold. The fixed active pair attaining the minimum gives an upper bound with the right limit. The active pairs at the perturbed parameters still have uniformly bounded constants, and their subsequential limits are active at $(\delta,q)$. Uniform convergence of the $r_j$ controls the integrals $\int_Kr_j\,\mathrm{d}\mu_{h_j}$, and $s_j\to s$ controls the terms $s_jc_{h_j}$. The lower bound therefore has the same limit. Note that all perturbed growth allowances $\delta+h_js_j$ are positive for small $h_j$.
\end{proof}

\begin{corollary}\label{cor:affine-two-sided-growth-slopes}
For fixed $a$ and $q$, the function $\delta\mapsto F_{\delta,q}(a)$ has finite left and right derivatives $D^-_{\delta,q}(a)$ and $D^+_{\delta,q}(a)$ at every positive parameter, with
\begin{equation}\label{eq:affine-two-sided-growth-slopes}
 D^-_{\delta,q}(a)=\max_{\mathcal J^{\mathrm{act}}_a(\delta,q)}c,
 \qquad
 D^+_{\delta,q}(a)=\min_{\mathcal J^{\mathrm{act}}_a(\delta,q)}c.
\end{equation}
It is differentiable exactly when all active pairs have the same constant $c$. For each fixed $a,q$, the set of positive parameters where it fails to be differentiable is at most countable.
\end{corollary}
\begin{proof}
Apply \eqref{eq:affine-joint-directional-response} with $r=0$ and $s=\pm1$. The direction $s=-1$ gives minus the left derivative, which proves \eqref{eq:affine-two-sided-growth-slopes}. Concavity gives
\[
 D^+_{\alpha,q}(a)\geq D^-_{\beta,q}(a)
 \qquad(0<\alpha<\beta).
\]
Hence the open intervals $(D^+_{\delta,q}(a),D^-_{\delta,q}(a))$, which are nonempty exactly at the parameters where differentiability fails, are pairwise disjoint. Choosing a rational number in each of them proves countability.
\end{proof}

\begin{corollary}\label{cor:affine-weight-differentiability}
For fixed $a$ and $\delta>0$, the map $q\mapsto F_{\delta,q}(a)$ on $C(K,\mathbb R)$ is G\^ateaux differentiable at $q$ if and only if all pairs active at $(\delta,q)$ have the same probability measure $\mu$. In that case its derivative is
\[
 r\longmapsto\int_Kr\,\mathrm{d}\mu.
\]
\end{corollary}
\begin{proof}
With $s=0$, \eqref{eq:affine-joint-directional-response} shows that the right derivative in the direction $r$ is the minimum of $\int_Kr\,\mathrm{d}\mu$ over active pairs, and the direction $-r$ shows that the left derivative is the maximum. If all active pairs have the same probability measure, these values agree and define a continuous linear derivative. Conversely, if the map is G\^ateaux differentiable, the minimum and the maximum agree for every continuous $r$. Continuous functions separate Borel probability measures on $K$, so all active pairs have the same probability measure; their constant is then determined by \eqref{eq:affine-active-supports}, since $\delta>0$.
\end{proof}
\index{Envelope!weight differentiability}

\begin{example}\label{ex:affine-quadratic-weight-corner}
Let $D\subset\mathbb C$ be the open unit disk, let $K=\overline D$, and fix $\delta>0$. For $q_t(z)=-t|z|^2$,
\begin{equation}\label{eq:affine-quadratic-weight-corner}
 F_{\delta,q_t}(0)=
 \begin{cases}
 -t,&t\geq0,\\
 0,&-\delta/2\leq t\leq0.
 \end{cases}
\end{equation}
Thus the map $q\mapsto F_{\delta,q}(0)$ is not G\^ateaux differentiable at $q=0$, and there are active pairs at $(\delta,0)$ with different probability measures.
\end{example}
\begin{proof}
For $t\geq0$, each competitor is at most $-t$ on the unit circle, so the maximum principle for subharmonic functions bounds it by $-t$ on $D$. This bound persists after regularization on $D$, and the constant $-t$ is a competitor. Hence $F_{\delta,q_t}(0)=-t$.

For $t=-b$ with $0\leq b\leq\delta/2$, define
\[
 u_b(z)=
 \begin{cases}
 b|z|^2,&|z|\leq1,\\
 b+\delta\log|z|,&|z|>1.
 \end{cases}
\]
This function is continuous and subharmonic: its Laplacian has density $4b$ inside the disk, density zero outside, and a nonnegative contribution on the circle coming from the jump $\delta-2b\geq0$ of the radial derivative. It belongs to $\mathcal L_\delta$ and equals $q_{-b}$ on $K$. Hence $F_{\delta,q_{-b}}(0)\geq u_b(0)=0$. On $D$ the raw envelope is at most the continuous function $b|z|^2$, so its regularization at zero is at most zero. This proves \eqref{eq:affine-quadratic-weight-corner}. In particular, the right and left derivatives in $t$ at $t=0$ are $-1$ and $0$, and \cref{cor:affine-weight-differentiability}, applied in the direction $r=-|z|^2$, gives the final assertion.
\end{proof}
