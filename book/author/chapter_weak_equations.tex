\chapter{Continuous solutions and normalization}
\label{chap:weak-equations}

Throughout this chapter, $X$ is a connected compact complex manifold of dimension $n\geq1$, $\omega$ is a Hermitian form and $\mathrm{d}V$ is a smooth positive volume form on $X$. As in \cref{chap:smooth-equations}, we write $\omega_u=\omega+\ddc u$; for a bounded $\omega$-plurisubharmonic function $u$, the power $\omega_u^n$ is the Bedford--Taylor product of \cref{prop:bounded-products}. We fix $p>1$ and a nonnegative $f\in L^p(X,\mathrm{d}V)$ with $\int_Xf\,\mathrm{d}V>0$. Norms and almost-everywhere statements refer to $\mathrm{d}V$ unless another measure is specified.

We look for continuous $\omega$-plurisubharmonic solutions of the equations $(\omega+\ddc u)^n=cf\,\mathrm{d}V$, with an unknown constant $c>0$, and $(\omega+\ddc u)^n=\mathrm{e}^{\lambda u}f\,\mathrm{d}V$, with $\lambda>0$. If $f$ is approximated by smooth positive densities, the preceding chapter gives smooth solutions, but normalized $L^1$ compactness of the potentials does not by itself identify the limit of their Monge--Amp\`ere measures. Two estimates close this gap. Integration against a Gauduchon metric bounds the normalization constants, and hence gives uniform $L^p$ control of the densities of the approximating equations. A modified comparison principle, which compensates for the variation of the total mass, then upgrades normalized $L^1$ convergence of the potentials to uniform convergence. For the exponential equation, the same Gauduchon trace controls the additive constants. Letting the exponent $\lambda$ tend to zero shows that the constant $c$ is determined by $f$, even though the normalized potential need not be unique.

On the Hopf threefold, the normalization constant satisfies a sharp lower bound, with equality exactly for densities pulled back from the projective plane. Finally, for concave fully nonlinear equations with a fixed admissible reference satisfying a strict asymptotic gap, an auxiliary Monge--Amp\`ere equation gives a uniform stability estimate. The resulting compactness provides the smooth comparison functions for a gradient bound.

\section{Continuous solvability}

There are two natural ways to normalize the prescribed density: by an unknown positive constant, or by an exponential of the potential.

\begin{theorem}[Ko{\l}odziej--Nguyen]\label{thm:kn-continuous}\index{Monge--Ampere equation@Monge--Amp\`ere equation!continuous solution}
There are $c>0$ and $u\in\PSH(X,\omega)\cap C(X)$ such that
\[
 (\omega+\ddc u)^n=c f\,\mathrm{d}V.
\]
The potential can be normalized by $\sup_Xu=0$.
\end{theorem}

The constant $c$ is uniquely determined by $f$, by \cref{prop:vanishing-exponent}. The normalized potential, however, need not be unique, even for a smooth density that vanishes on an open set; see \cref{thm:degenerate-hermitian-nonuniqueness}.

\Needspace{7\baselineskip}
\begin{theorem}[Nguyen]\label{thm:nguyen-exp}\index{Monge--Ampere equation@Monge--Amp\`ere equation!positive exponent}
For every $\lambda>0$, there is a unique continuous $\omega$-plurisubharmonic function $u_\lambda$ satisfying
\begin{equation}\label{eq:weak-exp}
 (\omega+\ddc u_\lambda)^n=\mathrm{e}^{\lambda u_\lambda}f\,\mathrm{d}V.
\end{equation}
\end{theorem}

For the exponential equation, a larger density gives a smaller solution, since the factor $\mathrm{e}^{\lambda u}$ has to compensate for the increase. Bounded exponential comparison gives this monotonicity even where the densities vanish.

\begin{lemma}\label{lem:exp-order}
Let $\lambda>0$, and let $f_a,f_b$ be nonnegative nonzero $L^p$ densities. Suppose that $a,b\in\PSH(X,\omega)\cap L^\infty(X)$ solve
\[
 \omega_a^n=\mathrm{e}^{\lambda a}f_a\,\mathrm{d}V,\qquad
 \omega_b^n=\mathrm{e}^{\lambda b}f_b\,\mathrm{d}V.
\]
If $f_a\leq f_b$ almost everywhere, then $b\leq a$ on $X$.
\end{lemma}

\begin{proof}
Since $f_a\leq f_b$, the two equations give
$\mathrm{e}^{-\lambda b}\omega_b^n\geq \mathrm{e}^{-\lambda a}\omega_a^n$. Hence $b\leq a$ by \cref{cor:bounded-exponential-comparison}, applied with $u=a$ and $v=b$.
\end{proof}

\begin{corollary}\label{cor:bounded-lp-continuity}
Every bounded $\omega$-plurisubharmonic solution of $\omega_u^n=f\,\mathrm{d}V$ is continuous.
\end{corollary}
\begin{proof}
Since $u$ is bounded, the density $\mathrm{e}^{-u}f$ is nonnegative, nonzero and belongs to $L^p$. By \cref{thm:nguyen-exp}, applied with $\lambda=1$ to this density, there is a continuous solution $r$ of
\[
 \omega_r^n=\mathrm{e}^{r-u}f\,\mathrm{d}V.
\]
The bounded potential $u$ solves the same exponential equation. Applying \cref{lem:exp-order} with $f_a=f_b=\mathrm{e}^{-u}f$ in both orders gives $u=r$, so $u$ is continuous.
\end{proof}

\section{The vanishing exponent and scalar uniqueness}

To show that the constant in \cref{thm:kn-continuous} is unique, we compare the solution $u_\lambda$ of \eqref{eq:weak-exp} with constant translates of a solution $u$ of the normalized equation. Translating $u$ by its maximum and by its minimum gives two barriers for the exponential equation, whose separation is the oscillation of $u$. After multiplication by $\lambda$, both barriers converge to the logarithm of the constant as $\lambda\downarrow0$. Since $u_\lambda$ depends only on $f$ and $\lambda$, the constant is determined by $f$.

\Needspace{16\baselineskip}
\begin{proposition}\label{prop:vanishing-exponent}\index{Normalization scalar!uniqueness}\index{Exponential approximation!vanishing exponent}
Suppose $u\in\PSH(X,\omega)\cap C(X)$ and $c>0$ satisfy $\omega_u^n=cf\,\mathrm{d}V$. Set $m=\inf_Xu$ and $M=\sup_Xu$. Then, for every $\lambda>0$, the solution of \eqref{eq:weak-exp} satisfies
\begin{equation}\label{eq:exp-barriers}
 u+\frac{\log c}{\lambda}-M
 \leq u_\lambda\leq
 u+\frac{\log c}{\lambda}-m.
\end{equation}
In particular,
\[
 \|\lambda u_\lambda-\log c\|_\infty
 \leq\lambda(M-m),
 \qquad
 \mathrm{e}^{\lambda u_\lambda}\longrightarrow c
 \quad\hbox{uniformly on }X\hbox{ as }\lambda\downarrow0.
\]
Consequently, $c$ is the only positive constant $c'$ for which the equation $\omega_v^n=c'f\,\mathrm{d}V$ has a solution $v\in\PSH(X,\omega)\cap C(X)$.
\end{proposition}
\begin{proof}
Set $a=u+\lambda^{-1}\log c-M$ and $b=u+\lambda^{-1}\log c-m$. These translates of $u$ have the same Monge--Amp\`ere measure as $u$, and since $m\leq u\leq M$,
\[
 \mathrm{e}^{\lambda a}f\,\mathrm{d}V
 \leq cf\,\mathrm{d}V=\omega_a^n=\omega_b^n
 \leq\mathrm{e}^{\lambda b}f\,\mathrm{d}V.
\]
Thus $\mathrm{e}^{-\lambda a}\omega_a^n\geq f\,\mathrm{d}V=\mathrm{e}^{-\lambda u_\lambda}\omega_{u_\lambda}^n\geq\mathrm{e}^{-\lambda b}\omega_b^n$. Applying \cref{cor:bounded-exponential-comparison} first to the pair $(u_\lambda,a)$ and then to the pair $(b,u_\lambda)$ gives $a\leq u_\lambda\leq b$, which is \eqref{eq:exp-barriers}. Multiplying these inequalities by $\lambda$ and using $m\leq u\leq M$ gives the norm bound, and the uniform convergence follows.

If $(v,c')$ is another such pair, the same argument applied to $v$ gives $\lambda u_\lambda\to\log c'$, because $u_\lambda$ depends only on $f$ and $\lambda$. Hence $c'=c$.
\end{proof}

The barriers also give $\operatorname{osc}_Xu_\lambda\leq2(M-m)$, and the densities $\mathrm{e}^{\lambda u_\lambda}f$ converge to $cf$ in $L^p$. By \cref{thm:lp-uniform-compactness}, the normalized potentials $u_\lambda-\sup_Xu_\lambda$ are therefore precompact in $C^0(X)$ as $\lambda\downarrow0$, and every cluster point solves the equation with constant $c$. Uniqueness of the constant does not, however, single out one cluster point; \cref{prop:symmetric-equation-clusters} describes the symmetries that these limits inherit.

For $C^2$ potentials, uniqueness of the constant also follows directly from the maximum principle: replacing one potential $v$ by $(1+\varepsilon)v$ leaves a strictly positive multiple of $\omega$ in reserve at an extremum.

\section{Normalization on the Hopf threefold}
\label{sec:hopf-lp-normalization}

In this section, $X$ is the Hopf threefold of \cref{chap:hopf}, with its Hermitian form $\omega$ and projection $\pi\colon X\to\mathbb P^2$. Recall that $\alpha=\pi^*\omega_{\FS}$ and $\beta=\omega-\alpha$, and that $\nu=\omega^3/6$ is a probability measure. We write $E$ for the elliptic fiber, $\tau_a$ for the translation by $a\in E$, $\mathrm{d}A_E$ for the probability Haar measure of $E$, and $A$ for the fiber-averaging operator of \cref{prop:hopf-fiber-average}.

The minimum-mass characterization in that proposition gives a lower bound for the normalization constant, and equality forces every solution to descend to the projective base. The uniform $L^p$ control of this chapter makes the equality case stable under limits of the data, although the densities themselves need only converge weakly.

\Needspace{9\baselineskip}
Here we take $\mathrm{d}V=\omega^3$; the density $f$ is still a nonnegative nonzero function in $L^p$. By \cref{thm:kn-continuous,prop:vanishing-exponent}, there is a unique positive constant $c_f$ for which the equation
\[
 (\omega+\ddc u)^3=c_f f\,\omega^3
\]
has a continuous $\omega$-plurisubharmonic solution. Put $I_f=\int_Xf\,\mathrm{d}\nu$.

\begin{proposition}\label{prop:hopf-lp-normalization}
One has $c_f I_f\geq1$, and every continuous solution $u$ satisfies
\begin{equation}\label{eq:hopf-lp-average}
 \|u-Au\|_{L^2(X,\nu)}^2\leq4(c_f I_f-1).
\end{equation}
Moreover, $c_f I_f=1$ if and only if $f=\pi^*g$ almost everywhere with respect to $\nu$, for some nonnegative $g\in L^p(\mathbb P^2,\omega_{\FS}^2)$. In this case every continuous solution is the pullback of a continuous $\omega_{\FS}$-plurisubharmonic function.
\end{proposition}
\begin{proof}
Integrating the equation gives $\int_X\omega_u^3=6c_f I_f$. Both the inequality $c_fI_f\geq1$ and \eqref{eq:hopf-lp-average} therefore follow from \eqref{eq:hopf-average-mass}, whose left-hand side is nonnegative.

Suppose $c_f I_f=1$. By \cref{prop:hopf-fiber-average}, $u=\pi^*q$ for a bounded $\omega_{\FS}$-plurisubharmonic function $q$; local holomorphic sections of $\pi$ show that $q$ is continuous. Thus $u\circ\tau_a=u$ for every $a\in E$. Since bounded products are natural under biholomorphisms and $\omega$ is invariant under the translations, it follows that
\[
 \tau_a^*(f\nu)=f\nu\qquad(a\in E).
\]
As $f$ is defined only almost everywhere, we use this invariance as an identity of measures.

By Fubini's theorem, the fiber average $Af$ is defined almost everywhere. It equals $\pi^*g$ for a measurable function $g$ on the base, because local trivializations differ by translations of the Haar variable. Jensen's inequality gives
\[
 \int_{\mathbb P^2}|g|^p\omega_{\FS}^2
 \leq\int_X|f|^p\,\mathrm{d}\nu.
\]
For every continuous real test function $\chi$ on $X$, the invariance of the measure gives $\int_X\chi f\,\mathrm{d}\nu=\int_X(\chi\circ\tau_a)f\,\mathrm{d}\nu$. Integrating in $a$, and using Fubini's theorem together with the invariance of Haar measure under $a\mapsto-a$, we obtain $\int_X\chi f\,\mathrm{d}\nu=\int_X\chi Af\,\mathrm{d}\nu$. Hence $f=Af=\pi^*g$ almost everywhere, with $g\geq0$.

Conversely, suppose $f=\pi^*g$ almost everywhere. Disintegration gives $I_f=\int_{\mathbb P^2}g\omega_{\FS}^2$. Applying \cref{thm:kn-continuous} on the base, we find a continuous $\omega_{\FS}$-plurisubharmonic function $q$ and a constant $k>0$ with $(\omega_{\FS}+\ddc q)^2=kg\,\omega_{\FS}^2$. Bounded decreasing regularization and Stokes' theorem give $\int_{\mathbb P^2}(\omega_{\FS}+\ddc q)^2=1$, so $k=I_f^{-1}$.

Put $u=\pi^*q$. To compute its bounded product, take smooth strictly $\omega_{\FS}$-plurisubharmonic functions $q_j\downarrow q$ by \cref{thm:bounded-exact-regularization}. Their pullbacks decrease to $u$ and are strictly $\omega$-plurisubharmonic. Since $\alpha^3=\beta^2=0$, the smooth cube identity gives, for each $j$,
\[
 (\omega+\ddc\pi^*q_j)^3
 =3\pi^*(\omega_{\FS}+\ddc q_j)^2\wedge\beta.
\]
\Needspace{11\baselineskip}
In a local trivialization, the right-hand side is $6\mathrm{d}A_E$ times the base measure $(\omega_{\FS}+\ddc q_j)^2$: the horizontal components of $\beta$ vanish against the base top-degree form, and its vertical restriction is $2\mathrm{d}A_E$. Continuity of bounded products on the base and on $X$, tested in these trivializations and patched by a partition of unity, therefore gives
\[
 (\omega+\ddc u)^3=\frac{f}{I_f}\,\omega^3.
\]
Uniqueness of the constant implies $c_f=I_f^{-1}$. Every continuous solution then has mass $6$, so the mass characterization in \cref{prop:hopf-fiber-average} makes it a pullback from the base, and the descended potential is continuous by local sections, as above.
\end{proof}

When $c_fI_f$ is close to one, compactness gives uniform convergence of subsequences of potentials and weak convergence of their densities, and the equality case identifies the limits.

\begin{proposition}\label{prop:hopf-near-equality-compactness}\index{Hopf manifold!near-minimum mass}
Fix $1<p<\infty$ and $M\geq1$. Suppose $f_j\geq0$ satisfy
\[
 \int_X f_j\,\mathrm{d}\nu=1,\qquad
 \|f_j\|_{L^p(X,\nu)}\leq M,\qquad c_{f_j}\longrightarrow1.
\]
For each $j$, let $u_j$ be any continuous solution of $\omega_{u_j}^3=c_{f_j}f_j\omega^3$ normalized by $\sup_Xu_j=0$. Then
\[
 \|u_j-Au_j\|_{C^0(X)}\longrightarrow0,\qquad
 f_j-Af_j\rightharpoonup0\quad\hbox{weakly in }L^p(X,\nu).
\]
Every subsequence has a further subsequence with $u_j\to\pi^*q$ uniformly and $f_j\rightharpoonup\pi^*g$ weakly in $L^p$, where $q$ is a continuous $\omega_{\FS}$-plurisubharmonic function, $g\geq0$, $\int_{\mathbb P^2}g\omega_{\FS}^2=1$, and
\[
 (\omega_{\FS}+\ddc q)^2=g\omega_{\FS}^2.
\]
\end{proposition}
\begin{proof}
Since the constants $c_{f_j}$ converge, the densities $c_{f_j}f_j$ have a common $L^p$ bound. By \cref{thm:lp-uniform-compactness}, every subsequence has a further subsequence on which $u_j\to u$ uniformly. Since $L^p$ is reflexive, a further extraction gives $f_j\rightharpoonup f$ weakly in $L^p$. Positivity, the unit integral and the bound $\|f\|_p\leq M$ pass to the limit, and continuity of bounded products under uniform convergence gives
\[
 (\omega+\ddc u)^3=f\omega^3.
\]
Thus $c_f=1$. Since also $I_f=1$, \cref{prop:hopf-lp-normalization} gives $u=\pi^*q$ and $f=\pi^*g$. The bounded cube identity proved there, tested against pullbacks of continuous functions on the base, gives the asserted base equation. In particular, both limits are fixed by $A$.

Haar averaging is a contraction on $C^0(X)$ and on $L^p(X,\nu)$. It is also self-adjoint with respect to $\nu$: Fubini's theorem and the invariance of $\nu$ under the fiber translations give $\int\chi Af\,\mathrm{d}\nu=\int A\chi f\,\mathrm{d}\nu$ for $\chi\in L^{p'}$. Along the extracted subsequence, therefore, $u_j-Au_j\to0$ uniformly and $f_j-Af_j\rightharpoonup0$ in $L^p$. If either convergence failed for the original sequence, some subsequence would stay away from the limit (for the weak convergence, when tested against one fixed function in $L^{p'}$), and a further subsequence of the kind just constructed would give a contradiction. This proves both conclusions for the whole sequence.
\end{proof}

\begin{example}\label{ex:hopf-oscillating-near-minimizers}\index{Hopf manifold!oscillating densities}
\enlargethispage{2\baselineskip}
Choose $0<a<1/2$, and use the function $s=\log|z|^2$ on the covering space $\mathbb C^3\setminus\{0\}$ from \cref{chap:hopf}. Since $s$ changes by $-2$ under the defining contraction, every $2$-periodic function of $s$ descends to $X$. For an integer $m\geq1$, define the smooth normalized potential
\[
 u_m=\frac{a}{\pi^2m^2}\bigl(\cos(\pi m s)-1\bigr),
 \qquad \|u_m-Au_m\|_{C^0}=\frac{a}{\pi^2m^2}\longrightarrow0.
\]
Primes below denote derivatives with respect to $s$. Since $\ddc s=\alpha$ and $\mathrm{d}s\wedge\dc s=\beta$, the chain rule gives
\[
 \omega+\ddc u_m=(1+u_m')\alpha+(1+u_m'')\beta,
 \quad u_m'=-\frac{a}{\pi m}\sin(\pi m s),\quad
 u_m''=-a\cos(\pi m s).
\]
Both coefficients are bounded below by positive constants independent of $m$. The rank identities $\alpha^3=\beta^2=0$ imply
\[
 (\omega+\ddc u_m)^3=H_m\omega^3,\qquad
 H_m=\left(1-\frac{a}{\pi m}\sin(\pi m s)\right)^2
          \bigl(1-a\cos(\pi m s)\bigr).
\]
On each fiber, $s$ is twice the real Haar coordinate plus a constant depending on the base point, so its distribution modulo $2$ is uniform. When the periodic expression $H_m=(1+u_m')^2(1+u_m'')$ is averaged, the terms $u_m'$, $u_m''$, $u_m'u_m''$ and $(u_m')^2u_m''$ contribute zero, since they are derivatives of periodic functions. Hence
\[
 I_m:=\int_XH_m\,\mathrm{d}\nu=AH_m
      =1+\frac{a^2}{2\pi^2m^2}.
\]
Set $f_m=H_m/I_m$. These smooth densities have unit integral and common strictly positive lower and finite upper bounds. Uniqueness of the constant gives $c_{f_m}=I_m\to1$, and $Af_m=1$. Nevertheless,
\[
 \|f_m-1+a\cos(\pi m s)\|_{C^0(X)}=O(m^{-1}),
 \qquad
 \lim_{m\to\infty}\|f_m-Af_m\|_{L^1(X,\nu)}=\frac{2a}{\pi}>0.
\]
Here the constant in $O(m^{-1})$ depends only on $a$, and the last identity uses the fact that the average of $|\cos|$ over a period is $2/\pi$. Thus near equality does not imply strong $L^1$ convergence to the fiber average, even for uniformly positive bounded smooth densities. For any pullback $g$ of a function on the base, the contraction property gives $\|f_m-Af_m\|_1\leq2\|f_m-g\|_1$, so approximation by such pullbacks also fails in $L^1$.
\end{example}
