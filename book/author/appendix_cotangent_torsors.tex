\chapter{Cotangent torsors on split elliptic ruled surfaces}
\label{app:cotangent-torsors}
\begingroup
\predisplaypenalty=10000

A class in $H^1(M,\Omega^1_M)$ on a complex manifold $M$ determines a holomorphic extension of $\mathcal O_M$ by the cotangent bundle and an affine bundle modeled on the cotangent bundle, its cotangent torsor; dually, a K\"ahler class determines an extension of the tangent bundle by the trivial line bundle, its canonical extension. This appendix computes these objects on the split ruled surface $M=\mathbb P_{\mathrm{lines}}(\mathcal O_E\oplus A)$ over an elliptic curve $E$, where $A$ is a line bundle of positive degree. The tautological line bundle of the canonical extension is big, but its section algebra is not finitely generated, and the cotangent torsor is never Stein, although its universal cover is Stein when the class is nonzero on the ruling fibers.

These results give a second instance of the contrast found for the nine-point blowups in \cref{subsec:flat-divisor-complements}: a background with a well-defined order of degeneration can have very different holomorphic geometry near its null locus. No other part of the book depends on this appendix.

\Cref{subsec:elliptic-affine-quotients} constructs the relevant quotients, and \cref{subsec:elliptic-cotangent-torsor-classes} identifies them with the cotangent torsors. \Cref{subsec:split-elliptic-symmetric-tensors,subsec:elliptic-cotangent-extension-algebra} compute the symmetric tensors of the tangent bundle and the section algebras of the dual extensions whose class is nonzero on the ruling fibers, and \cref{subsec:elliptic-zero-ruling-extension} treats the classes that vanish there.

\section{Punctured line bundles and elliptic quotients}
\label{subsec:elliptic-affine-quotients}

In this section we construct the relevant quotients and show that they are not Stein, although their universal covers are. The obstruction comes from fiberwise Laurent expansions on the punctured total space of a line bundle of nonzero degree, rather than from the flat circle levels used in \cref{cor:flat-divisor-nonstein}.

\begin{lemma}\label{lem:positive-line-puncture-hull}
\index{Holomorphic hull!punctured line bundle}
\index{Line bundle!punctured positive degree}
Let $C$ be a compact connected Riemann surface, and let $A\to C$ be a holomorphic line bundle of positive degree. Write $A^\times$ for its total space with the zero section removed. Then $A^\times$ is not holomorphically convex. More precisely, put $B=A^*$ and choose a smooth Hermitian norm on $B$. Every holomorphic function on $B^\times$ extends holomorphically to $B$, and the holomorphic hull in $B^\times$ of
\[
 K_r=\{v\in B:|v|=r\},\qquad r>0,
\]
contains $\{v:0<|v|\leq r\}$ and is not compact.
\end{lemma}
\begin{proof}
In a local holomorphic frame of $B$, a function $f\in\mathcal O(B^\times)$ has a fiberwise Laurent expansion
\[
 f(z,t)=\sum_{k\in\mathbb Z}a_k(z)t^k.
\]
By the Cauchy integral formula, each coefficient is holomorphic in $z$, and under a change of frame the coefficients $a_k$ transform as the components of global sections of $B^{-k}$. For $k<0$ this line bundle has negative degree, so it has no nonzero section: the zero divisor of a nonzero holomorphic section has nonnegative degree. Thus all negative coefficients vanish. On a smaller base chart, Cauchy estimates on a fixed fiber circle give locally uniform convergence of the remaining power series at $t=0$. These extensions agree on overlaps and define a holomorphic function on all of $B$.

The maximum modulus principle on the disk in each fiber gives
\[
 |f(v)|\leq\max_{K_r}|f|\qquad(0<|v|\leq r).
\]
The circle bundle $K_r$ is compact because $C$ is compact. A sequence in one fiber whose norms tend to zero lies in the hull and has no convergent subsequence in $B^\times$, so the hull is not compact. Finally, fiberwise inversion identifies $A^\times$ biholomorphically with $B^\times$: a nonzero $v\in A$ corresponds to the unique $\ell\in A^*$ with $\ell(v)=1$. This proves the assertion for $A^\times$.
\end{proof}

The following quotient isolates this obstruction in an affine-bundle model.

Let $\Lambda=\mathbb Z+\mathbb Z\tau\subset\mathbb C$, with $\operatorname{Im}\tau>0$, and put $E=\mathbb C/\Lambda$. A \emph{factor of automorphy} is a family of nowhere-zero holomorphic functions $b_\lambda(z)$, $\lambda\in\Lambda$, satisfying
\[
 b_{\lambda+\gamma}(z)=b_\lambda(z+\gamma)b_\gamma(z).
\]
This cocycle identity says that $\lambda\cdot(z,t)=(z+\lambda,b_\lambda(z)t)$ is an action of $\Lambda$ on $\mathbb C_z\times\mathbb C_t$, and the quotient
\[
 A=(\mathbb C_z\times\mathbb C_t)/\Lambda
\]
is a holomorphic line bundle over $E$. Suppose that $A$ has positive degree.

\begin{example}\label{ex:positive-elliptic-automorphy}
\index{Factor of automorphy!degree-one elliptic bundle}
The factors
\[
 b_{m\tau+n}(z)=\exp\bigl(-\pi\mathrm{i}(m^2\tau+2mz)\bigr)
\]
define a line bundle of degree one on $E$.
\end{example}
\begin{proof}
For $\gamma=r\tau+s$, the cocycle identity follows from
\[
 \begin{aligned}
 b_{m\tau+n}(z+\gamma)b_{r\tau+s}(z)
       &=b_{(m+r)\tau+n+s}(z)\exp(-2\pi\mathrm{i}ms)\\
       &=b_{(m+r)\tau+n+s}(z).
 \end{aligned}
\]
Put $v=\operatorname{Im}\tau$. The squared norm
\[
 |t|^2\exp\bigl(-2\pi(\operatorname{Im}z)^2/v\bigr)
\]
is invariant under the action, because $\operatorname{Im}(z+m\tau+n)=\operatorname{Im}z+mv$ and
\[
 |b_{m\tau+n}(z)|^2=\exp\bigl(2\pi(m^2v+2m\operatorname{Im}z)\bigr).
\]
Its first Chern form is
\[
 \frac{\mathrm{i}}{2\pi}\partial\bar\partial
              \left(\frac{2\pi(\operatorname{Im}z)^2}{v}\right)
       =\frac{\mathrm{i}}{2v}\,\mathrm{d}z\wedge\mathrm{d}\bar z.
\]
This positive form integrates to one over a fundamental parallelogram, which has area $v$. Thus the degree is one.
\end{proof}

On
\[
 Q=\{([x_0:x_1],[w_0:w_1])\in\mathbb P^1\times\mathbb P^1:
                    \Delta=x_0w_1-x_1w_0\ne0\},
\]
the complement of the diagonal, define the holomorphic function $\sigma=x_1w_0/\Delta$.

\Needspace{16\baselineskip}
\begin{proposition}\label{prop:elliptic-affine-quotient-nonstein}
\index{Stein manifold!elliptic affine quotient}
Fix $\kappa\in\mathbb C^*$ and $K\in\mathbb C$. For $\lambda=m\tau+n\in\Lambda$, the formula
\[
 \begin{aligned}
 \lambda\cdot(z,u,[x],[w])={}&
 \bigl(z+\lambda,u+m(K+\kappa\sigma),\\
       &[x_0:b_\lambda(z)x_1],[w_0:b_\lambda(z)w_1]\bigr)
 \end{aligned}
\]
defines a free and properly discontinuous holomorphic action of $\Lambda$ on $\mathbb C^2\times Q$. The quotient $Z$ is not Stein, and $\sigma$ descends to a holomorphic function on $Z$. If $K\notin\{0,\kappa\}$, the fiber
\[
 \sigma^{-1}(-K/\kappa)\simeq\mathbb C\times A^\times
\]
is a closed submanifold of $Z$ that is not holomorphically convex. If $K=0$ or $K=\kappa$, the quotient contains a compact elliptic curve. In every case the universal cover of $Z$ is the Stein manifold $\mathbb C^2\times Q$.
\end{proposition}
\begin{proof}
\emph{The action.} Multiplying $x_1$ and $w_1$ simultaneously by $b_\lambda(z)$ multiplies $\Delta$ and $x_1w_0$ by the same factor, so $\sigma$ is invariant. This invariance and the cocycle identity for $b$ give the group law, including the additivity of the integer $m$ in the $u$-coordinate. Translation in the $z$-coordinate makes the action free and properly discontinuous: only finitely many lattice translates of the projection of one compact set meet another fixed compact set. Hence the quotient $Z$ is a complex manifold, and $\sigma$ descends to a holomorphic function on $Z$.

\emph{The fiber of $\sigma$.} Set $a=-K/\kappa$. If $a\notin\{0,-1\}$, every point of $\{\sigma=a\}\subset Q$ has $x_0x_1\ne0$, since $\sigma=0$ where $x_1=0$ and $\sigma=-1$ where $x_0=0$. In the chart
\[
 (t,y)\longmapsto([1:t],[y:ty+1]),
\]
we have $\Delta=1$, and the equation $\sigma=a$ becomes $ty=a$. Hence this fiber of $\sigma$ on $Q$ is $\mathbb C^*$, with parameter $t$, and the group action on its preimage is
\[
 \lambda\cdot(z,u,t)=(z+\lambda,u,b_\lambda(z)t).
\]
This gives the asserted biholomorphism with $\mathbb C\times A^\times$. The submanifold $\{u=0\}$ of this fiber is closed in the fiber, and therefore closed in $Z$; under fiberwise inversion it is $B^\times$, with $B=A^*$ as in \cref{lem:positive-line-puncture-hull}. Let $K_r$ be the compact circle bundle of that lemma. Every holomorphic function on $Z$ restricts to a holomorphic function on $B^\times$, so the hull of $K_r$ in $Z$ contains the same sequence with fiber norms tending to zero. This sequence has no convergent subsequence in the closed submanifold, and hence none in $Z$. Thus neither the fiber nor $Z$ is holomorphically convex.

\emph{The exceptional parameters.} For $K=0$, take $q_+=([1:0],[0:1])$, where $\sigma=0$. The invariant submanifold $\{u=0,\ q=q_+\}$ descends to a copy of $E$. For $K=\kappa$, use $q_-=([0:1],[1:0])$, where $\sigma=-1$, with the same conclusion. Every holomorphic function on $Z$ is constant on this compact curve by the maximum modulus principle, so $Z$ is not holomorphically separable, and again not Stein.

\emph{The universal cover.} The map
\[
 Q\longrightarrow\{(s,c,d)\in\mathbb C^3:cd=s(s-1)\},
 \qquad ([x],[w])\longmapsto
 \left(\frac{x_0w_1}{\Delta},\frac{x_0w_0}{\Delta},\frac{x_1w_1}{\Delta}\right)
\]
is an isomorphism. Its inverse recovers $[x]$ and $[w]$ as the column space and the row space of the rank-one matrix
\[
 \begin{pmatrix}c&s\\s-1&d\end{pmatrix}.
\]
This matrix never vanishes, and the hypersurface is smooth: simultaneous vanishing of the three partial derivatives of $cd-s(s-1)$ would give $c=d=0$ and $s=1/2$, contradicting its equation. Thus $Q$ is an affine variety and hence Stein. Projection to $[x]\in\mathbb P^1$ makes $Q$ a locally trivial bundle of affine lines. The chart above and the chart over $\{x_1\ne0\}$, given by $(v,y_\infty)\mapsto([v:1],[vy_\infty+1:y_\infty])$, are related by
\[
 v=1/t,\qquad y_\infty=-t^2y-t.
\]
Since the fibers are contractible, $\pi_1(Q)=\pi_1(\mathbb P^1)=0$. Therefore $\mathbb C^2\times Q$ is simply connected and Stein, and it is the universal cover of $Z$.
\end{proof}

\section{Cotangent extensions and their torsors}
\label{subsec:elliptic-cotangent-torsor-classes}

We now show that the quotients of \cref{prop:elliptic-affine-quotient-nonstein} are the cotangent torsors of a family of extensions on the ruled surface $M=\mathbb P_{\mathrm{lines}}(\mathcal O_E\oplus A)$, a family that contains the canonical extensions of all K\"ahler classes. An extension of the trivial line bundle by the cotangent bundle is determined, up to isomorphism, by a class in $H^1(M,\Omega^1_M)$. This group is two-dimensional: one component comes from the elliptic base and one from the ruling fibers. In the quotient, the base component appears as the constant $K$ and the ruling component as the coefficient $\kappa$ of $\sigma$.

Keep $E=\mathbb C/\Lambda$ with $\Lambda=\mathbb Z+\mathbb Z\tau$, and let $\vartheta_E$ be the holomorphic one-form on $E$ whose pullback to $\mathbb C$ is $\mathrm{d}z$. On a complex manifold $Y$, a holomorphic extension of vector bundles
\[
 0\longrightarrow\Omega_Y^1\longrightarrow W
       \xrightarrow{p_W}\mathcal O_Y\longrightarrow0
\]
defines the affine bundle $T_W=p_W^{-1}(1)$, modeled on $\Omega_Y^1$, which we call its cotangent torsor. Choosing local holomorphic sections $s_i$ of $W$ with $p_W(s_i)=1$, the differences $s_j-s_i$ form a \v{C}ech cocycle with values in $\Omega^1_Y$; its class in $H^1(Y,\Omega_Y^1)$ is the class of the extension, and it determines the extension up to isomorphism. Dually, a class $\alpha\in H^1(Y,\Omega^1_Y)$ determines an extension $0\to\mathcal O_Y\to V\to T_Y\to0$; when $\alpha$ is the class of a K\"ahler form, $V$ is the \emph{canonical extension} of that K\"ahler class.

\begin{lemma}\label{lem:elliptic-additive-class}
\index{Cech cocycle!elliptic additive class}
Choose local inverses $z_i\colon U_i\to\mathbb C$ of the projection $\mathbb C\to E$, and on each component of an overlap write
\[
 z_i-z_j=m_{ij}\tau+n_{ij},\qquad m_{ij},n_{ij}\in\mathbb Z.
\]
The cocycle $\{m_{ij}\vartheta_E\}$ defines a nonzero class $\eta\in H^1(E,\Omega_E^1)$, which spans this one-dimensional space.
\end{lemma}
\begin{proof}
Consider the vector bundle obtained from $\mathbb C_z\times\mathbb C_a\times\mathbb C_b$ by the action
\[
 (m\tau+n)\cdot(z,a,b)=(z+m\tau+n,a+mb,b).
\]
The coordinate $b$ gives a map to $\mathcal O_E$, which defines an extension of $\mathcal O_E$ by $\Omega_E^1$, the kernel coordinate $a$ corresponding to $a\vartheta_E$. The local lifts $s_i=(z_i,0,1)$ satisfy $s_j-s_i=m_{ij}\vartheta_E$, as one sees by expressing $s_i$ in the coordinates over $z_j$. Thus the extension has class $\eta$.

If it split, the pullback of a holomorphic lift of $1$ to $\mathbb C$ would have the form $(z,g(z),1)$ with $g$ entire and
\[
 g(z+1)=g(z),\qquad g(z+\tau)=g(z)+1.
\]
The derivative $g'$ would be lattice periodic, hence constant by the maximum principle on the compact torus. The first identity then forces this constant to be zero, contradicting the second. Hence $\eta\ne0$. Serre duality on the compact connected curve gives $H^1(E,\Omega_E^1)\simeq H^0(E,\mathcal O_E)^*\simeq\mathbb C$, which proves the spanning assertion.
\end{proof}

\begin{lemma}\label{lem:elliptic-degree-automorphy-normalization}
\index{Factor of automorphy!positive-degree normalization}
Every holomorphic line bundle $A$ of positive degree $d$ on $E$ is defined by factors of automorphy of the form
\[
 b_{m\tau+n}(z)=\chi_{m\tau+n}
       \exp\bigl(-\pi\mathrm{i}d(m^2\tau+2mz)\bigr),
\]
where $\chi\colon\Lambda\to\mathrm U(1)$ is a character. In particular,
\[
 \frac{b_{m\tau+n}'(z)}{b_{m\tau+n}(z)}=-2\pi\mathrm{i}dm.
\]
\end{lemma}
\begin{proof}
Let $A_d$ be the $d$th tensor power of the degree-one bundle of \cref{ex:positive-elliptic-automorphy}. Its factors are the stated exponential factors, and its degree is $d$. The bundle $F=A\otimes A_d^*$ has degree zero. Start with a smooth Hermitian metric on $F$; its Chern curvature has integral zero. Solving the scalar Poisson equation on the flat compact torus gives a smooth real function whose addition to the weight of the metric makes the curvature vanish. The resulting Chern connection is flat and unitary.

On the simply connected cover $\mathbb C$, parallel transport gives a global parallel frame of unit length of the pullback of $F$. It is holomorphic, because the $(0,1)$-part of the Chern connection is the $\bar\partial$-operator of $F$. Each lattice element changes this frame by a constant unitary factor, and these factors form a character. Tensoring with the given frame of $A_d$ gives factors of the stated form for $A$, and differentiation gives the last formula.
\end{proof}

Let $M=\mathbb P_{\mathrm{lines}}(\mathcal O_E\oplus A)$ be the bundle of lines in $\mathcal O_E\oplus A$, and let $f\colon M\to E$ be its projection. Let $C_\infty=\mathbb P_{\mathrm{lines}}(A)$ be the section given by the lines in $A$. Over $U_i$ the line spanned by $(1,t_ie_i)$, where $e_i$ is the local frame of $A$, defines an affine coordinate $t_i$ away from $C_\infty$, and $v_i=1/t_i$ is a coordinate away from the other section $\mathbb P_{\mathrm{lines}}(\mathcal O_E)$. Over $U_i\cap U_j$,
\[
 t_i=b_{ij}t_j,\quad v_i=b_{ij}^{-1}v_j,
 \qquad b_{ij}=b_{m_{ij}\tau+n_{ij}}(z_j).
\]
Define $\xi\in H^1(M,\Omega_M^1)$ as the logarithmic cocycle of $\mathcal O_M(-C_\infty)$. More precisely, use the cover by the charts with indices $(i,0)$ and $(i,\infty)$, the local defining functions $q_{i,0}=1$ and $q_{i,\infty}=v_i$ of $C_\infty$, and the cocycle $\mathrm{d}\log(q_{j,b}/q_{i,a})$. This convention contains no factor $1/(2\pi\mathrm{i})$.

\begin{lemma}\label{lem:split-elliptic-cotangent-cohomology}
\index{Ruled surface!cotangent cohomology}
There is a direct sum decomposition
\[
 H^1(M,\Omega_M^1)=\mathbb C\xi\oplus\mathbb C f^*\eta.
\]
The restriction of a class $c=\rho\xi+Kf^*\eta$ to any ruling fiber is nonzero if and only if $\rho\ne0$.
\end{lemma}
\begin{proof}
The relative cotangent sequence is
\[
 0\longrightarrow f^*\Omega_E^1\longrightarrow\Omega_M^1
      \longrightarrow\Omega_{M/E}^1\longrightarrow0.
\]
The two affine charts of a projective line give the direct images
\[
 f_*\mathcal O_M=\mathcal O_E,\quad R^1f_*\mathcal O_M=0,
 \qquad f_*\Omega_{M/E}^1=0,\quad
 R^1f_*\Omega_{M/E}^1=\mathcal O_E.
\]
For the last identity, a Laurent cocycle $g(z,t)\,\mathrm{d}t$ can be reduced, by subtracting forms holomorphic on the two charts, to its coefficient of $\mathrm{d}t/t$. Its residue is unchanged by $t_i=b_{ij}t_j$, with differentials taken relative to the base, so the local generators glue to the trivial line bundle. For functions, the two halves of a Laurent series exhaust the cocycle, which gives $R^1f_*\mathcal O_M=0$. The direct images in degree zero follow because $\mathbb P^1$ has no nonconstant holomorphic functions and no nonzero holomorphic one-forms.

The Leray spectral sequences and the fact that the base is a curve now give
\[
 H^1(M,f^*\Omega_E^1)\simeq H^1(E,\Omega_E^1),\quad
 H^2(M,f^*\Omega_E^1)=0,\quad
 H^1(M,\Omega_{M/E}^1)\simeq\mathbb C,
\]
together with $H^0(M,\Omega^1_{M/E})=0$. The cohomology sequence of the relative cotangent sequence is therefore the exact sequence
\[
 0\longrightarrow\mathbb C f^*\eta
 \longrightarrow H^1(M,\Omega_M^1)\longrightarrow\mathbb C\longrightarrow0.
\]
On a fiber, the class $\xi$ is represented on the two affine charts by $\mathrm{d}\log v=-\mathrm{d}t/t$, whose residue is nonzero, so $\xi$ splits the sequence. The pulled-back class $f^*\eta$ restricts to zero on every fiber. This proves both assertions.
\end{proof}

\begin{theorem}\label{thm:split-elliptic-cotangent-torsor}
\index{Canonical extension!split elliptic ruled surface}
\index{Affine bundle!cotangent torsor}
Let $W$ be a holomorphic extension of $\mathcal O_M$ by $\Omega_M^1$ with class $c=\rho\xi+Kf^*\eta$, where $\rho\ne0$. Then $T_W$ is biholomorphic to the quotient of \cref{prop:elliptic-affine-quotient-nonstein}, with the factors of \cref{lem:elliptic-degree-automorphy-normalization} and
\[
 \kappa=2\pi\mathrm{i}d\rho.
\]
In particular, $T_W$ is not Stein, while its universal cover is Stein. This applies to the cotangent torsor of the canonical extension of every K\"ahler class on $M$.
\end{theorem}
\begin{proof}
\emph{Step 1: a model torsor.} Over $\mathbb C_z\times\mathbb P^1$, glue two affine cotangent charts by
\[
 v=1/t,\qquad u_\infty=u,\qquad
 y_\infty=-t^2y-\rho t.
\]
Here $(u,y)$ are the coefficients of the covector $u\,\mathrm{d}z+y\,\mathrm{d}t$, and the linear part of the gluing is the change of these coefficients under $v=1/t$. If $s_0$ and $s_\infty$ are the zero sections of the two charts, the affine term says that
\[
 s_\infty-s_0=\rho\,\mathrm{d}v/v.
\]
For $\lambda=m\tau+n$, give the first chart the lattice action
\[
 (z,t,u,y)\longmapsto
 \bigl(z+\lambda,b_\lambda(z)t,
 u+Km-(b_\lambda'/b_\lambda)ty,y/b_\lambda(z)\bigr).
\]
Its linear part is the transport of covectors under $(z,t)\mapsto(z+\lambda,b_\lambda(z)t)$, and the cocycle identity for the factors and the invariance of $ty$ give the group law. In the other chart the action becomes
\[
 (z,v,u_\infty,y_\infty)\longmapsto
 \bigl(z+\lambda,b_\lambda^{-1}v,
 u_\infty+Km+(b_\lambda'/b_\lambda)(v y_\infty+\rho),
 b_\lambda y_\infty\bigr),
\]
which is holomorphic also at $v=0$, and the two formulas agree on the overlap. Translation in $z$ makes the action free and properly discontinuous.

\emph{Step 2: its class.} We compute the class of the quotient torsor using the local lifts $z_i$. On the two types of overlaps over the base, the zero sections satisfy
\[
 \begin{aligned}
 s_{j,0}-s_{i,0}&=Km_{ij}f^*\vartheta_E,\\
 s_{j,\infty}-s_{i,\infty}
   &=Km_{ij}f^*\vartheta_E+\rho\,\mathrm{d}\log(v_j/v_i).
 \end{aligned}
\]
For a mixed overlap, these formulas and $s_{j,\infty}-s_{j,0}=\rho\,\mathrm{d}v_j/v_j$ give
\[
 s_{j,\infty}-s_{i,0}=Km_{ij}f^*\vartheta_E+\rho\,\mathrm{d}v_j/v_j.
\]
These are exactly the cocycles defining $Kf^*\eta+\rho\xi$. Two extensions with the same class have splitting cocycles that differ by a coboundary, and changing their local lifts by that coboundary identifies their inverse images of $1$. Thus the quotient torsor is $T_W$.

\emph{Step 3: identification with the quotient.} Since $\rho\ne0$, set $Y=y/\rho$ and $Y_\infty=y_\infty/\rho$. Their transition is $Y_\infty=-t^2Y-t$. The maps
\[
 (t,Y)\longmapsto([1:t],[Y:tY+1]),\qquad
 (v,Y_\infty)\longmapsto([v:1],[vY_\infty+1:Y_\infty])
\]
agree on overlaps and identify the glued charts with $Q$. In the first chart $\sigma=tY=ty/\rho$. Substituting $b_\lambda'/b_\lambda=-2\pi\mathrm{i}dm$ turns the lattice action into the action of \cref{prop:elliptic-affine-quotient-nonstein} with $\kappa=2\pi\mathrm{i}d\rho$. That proposition gives the assertions on the Stein property.

\emph{Step 4: K\"ahler classes.} The restriction of a K\"ahler class to a ruling fiber is nonzero, because its integral over the fiber is positive. For the canonical extension $0\to\mathcal O_M\to V\to T_M\to0$ of a K\"ahler class, the dual extension of $\mathcal O_M$ by $\Omega^1_M$ has the negative of that class, which still has nonzero restriction to the fibers; by \cref{lem:split-elliptic-cotangent-cohomology} its ruling component $\rho$ is nonzero. This proves the final assertion.
\end{proof}

For a holomorphic vector bundle $F$ on a complex manifold $Y$ we use the quotient convention for projectivization: $\mathbb P(F)$ parametrizes the one-dimensional quotients of the fibers of $F$, that is, the lines in $F^*$, and the projection $\pi\colon\mathbb P(F)\to Y$ satisfies $\pi_*\mathcal O_{\mathbb P(F)}(r)=\operatorname{Sym}^rF$. For the canonical extension $V$ of a K\"ahler class, the complement $\mathbb P(V)\setminus\mathbb P(T_M)$ consists of the lines in $V^*$ not contained in $\Omega^1_M$, and it identifies with the cotangent torsor $T_W=p_W^{-1}(1)$ of $W=V^*$ by normalizing a nonzero functional to take the value one on the distinguished copy of $\mathcal O_M$.

\section{Symmetric tensors on split elliptic ruled surfaces}
\label{subsec:split-elliptic-symmetric-tensors}

To compute the section algebra of a canonical extension, we first compute that of the tangent bundle of $M$. A holomorphic section of $\operatorname{Sym}^rT_M$ is the same as a holomorphic function on the total space of the cotangent bundle that is a homogeneous polynomial of degree $r$ on each fiber. The $\mathbb C^*$-action on the rulings lifts to the cotangent bundle and decomposes these polynomials into weights. Each weight space is the space of sections of an explicit vector bundle on $E$, built from powers of $A$ and symmetric powers of a nonsplit bundle $U$ of rank two. Riemann--Roch then gives exact dimensions: the tautological line bundle on $\mathbb P(T_M)$ is big, while a square-zero quotient shows that the section algebra is not finitely generated.

Keep $E$, the line bundle $A$ of degree $d>0$, the surface $M$ and the normalized factors $b_\lambda$ of \cref{lem:elliptic-degree-automorphy-normalization}, and put
\[
 q_\lambda=-b_\lambda'/b_\lambda=2\pi\mathrm{i}dm,
 \qquad \lambda=m\tau+n.
\]
Write $u\,\mathrm{d}z+y\,\mathrm{d}t$ for a covector in the finite ruling chart. Evaluating covectors on the Euler field $t\partial/\partial t$ of the rulings gives the global fiberwise linear function $h=ty$ on the cotangent bundle. The dimensions of the weight spaces measure how many symmetric tensors there are, and their products detect the failure of finite generation.

Let $U$ be the rank-two vector bundle on $E$ whose sections, pulled back to $\mathbb C$, are the linear polynomials $F(z,h,u)$ in formal variables $h,u$ with entire coefficients satisfying
\[
 F(z+\lambda,h,u)=F(z,h,u-q_\lambda h).
\]
The invariant polynomial $h$ and the coefficient of $u$ give an exact sequence $0\to\mathcal O_Eh\to U\to\mathcal O_E\to0$.

\begin{lemma}\label{lem:split-elliptic-tensor-weights}
\index{Weight decomposition!cotangent polynomials}
\index{Symmetric tensor!elliptic ruled surface}
For every integer $r\geq0$, the space $H^0(M,\operatorname{Sym}^rT_M)$ is the direct sum of finitely many weight spaces, with weights $-r\leq\ell\leq r$. The weight-zero space is $H^0(E,\operatorname{Sym}^rU)$. For $1\leq k\leq r$, the space of weight $-k$ is
\[
 H^0(E,A^k\otimes\operatorname{Sym}^{r-k}U),
\]
and the space of weight $k$ is $H^0(E,A^{-k}\otimes\operatorname{Sym}^{r-k}U)$. Multiplication adds tensor degrees and weights.
\end{lemma}
\begin{proof}
On the cover $\mathbb C\times\mathbb P^1$, the cotangent coordinates of the two ruling charts are related by
\[
 v=1/t,\qquad u_\infty=u,\qquad y_\infty=-t^2y,
\]
and the lattice acts by
\[
 (z,t,u,y)\longmapsto
 (z+\lambda,b_\lambda t,u+q_\lambda ty,y/b_\lambda);
\]
both formulas follow by preserving the one-form $u\,\mathrm{d}z+y\,\mathrm{d}t$. In the other ruling chart the action is
\[
 (z,v,u_\infty,y_\infty)\longmapsto
 (z+\lambda,b_\lambda^{-1}v,
 u_\infty-q_\lambda v y_\infty,b_\lambda y_\infty),
\]
which is holomorphic at $v=0$ and agrees with the formula in the finite chart. Both formulas preserve $h=ty=-v y_\infty$.

A function that is homogeneous of degree $r$ on the fibers is a sum of terms $g_b(z,t)y^bu^{r-b}$, $0\leq b\leq r$, with $g_b$ holomorphic on $\mathbb C^2$. Since $y=-v^2y_\infty$, holomorphy at $v=0$ says that $v^{2b}g_b(z,1/v)$ is holomorphic there; there is no cancellation between different $b$, because the monomials in $u_\infty,y_\infty$ are linearly independent. Expanding in $t$ therefore gives
\[
 g_b(z,t)=\sum_{a=0}^{2b}c_{a,b}(z)t^a,
\]
with entire coefficients.

Give $t$ weight $1$, $y$ weight $-1$ and $u$ weight $0$, so that $t^ay^bu^{r-b}$ has weight $\ell=a-b$. The chart changes and the lattice action preserve this grading; equivalently, it is the grading by the cotangent lift of the global action $t\mapsto ct$, $c\in\mathbb C^*$. The bounds $0\leq a\leq2b$ and $b\leq r$ give $-r\leq\ell\leq r$.

For $\ell=-k<0$, a homogeneous function of weight $\ell$ can be written uniquely as $y^kF(z,h,u)$ with $F$ homogeneous of degree $r-k$. Its invariance is equivalent to
\[
 F(z+\lambda,h,u)
       =b_\lambda(z)^kF(z,h,u-q_\lambda h),
\]
which is the transformation rule for sections of $A^k\otimes\operatorname{Sym}^{r-k}U$. Conversely, each such polynomial defines a function that is holomorphic in both ruling charts, because its exponents satisfy $0\leq a\leq2b$. For $\ell=k>0$, the condition $a\leq2b$ shows that a function of weight $\ell$ can be written uniquely as $t^kh^kF(z,h,u)$, with $F$ of the same remaining degree $r-k$. Its invariance involves the factor $b_\lambda^{-k}$, which gives $A^{-k}\otimes\operatorname{Sym}^{r-k}U$. Weight zero consists exactly of the homogeneous polynomials $F(z,h,u)$ with the transformation rule of $\operatorname{Sym}^rU$. Products of polynomial functions add degrees and weights, which proves the last assertion.
\end{proof}

\begin{lemma}\label{lem:elliptic-unipotent-symmetric-cohomology}
\index{Unipotent bundle!elliptic symmetric powers}
The extension $U$ does not split. For every $s\geq0$,
\[
 H^0(E,\operatorname{Sym}^sU)=\mathbb C h^s.
\]
For $k\geq1$ and $s\geq0$,
\[
 \begin{aligned}
 H^0(E,A^{-k}\otimes\operatorname{Sym}^sU)&=0,\\
 H^1(E,A^k\otimes\operatorname{Sym}^sU)&=0,\\
 h^0(E,A^k\otimes\operatorname{Sym}^sU)&=kd(s+1),
 \end{aligned}
\]
where $h^0=\dim H^0$. For $s\geq1$, taking the coefficient of $u^s$ defines a surjection
\[
 H^0(E,A\otimes\operatorname{Sym}^sU)\longrightarrow H^0(E,A),
\]
whose kernel is $hH^0(E,A\otimes\operatorname{Sym}^{s-1}U)$.
\end{lemma}
\begin{proof}
Write an invariant polynomial of degree $s$ as
\[
 F(z,h,u)=\sum_{j=0}^s a_j(z)h^{s-j}u^j.
\]
If its largest nonzero $u$-degree is $j>0$, then $a_j$ is lattice periodic, hence a nonzero constant. Comparing the next coefficients gives
\[
 a_{j-1}(z+1)=a_{j-1}(z),\qquad
 a_{j-1}(z+\tau)=a_{j-1}(z)-j(2\pi\mathrm{i}d)a_j,
\]
so the derivative $a_{j-1}'$ is lattice periodic, hence constant on the compact torus. The first identity forces this constant to vanish, and then $a_{j-1}$ is constant, contradicting the second identity. Hence $F=a_0h^s$ with $a_0$ constant; this also covers $s=0$. For $s=1$ this shows that $U$ does not split, since a splitting would give an invariant section with $u$-coefficient one.

The powers of the invariant line $\mathcal O_Eh$ give a filtration of $\operatorname{Sym}^sU$ whose $s+1$ successive quotients are trivial line bundles. A line bundle of negative degree on a compact curve has no nonzero section, since the zero divisor of a nonzero section has nonnegative degree. Hence the filtration twisted by $A^{-k}$ has no global sections. Serre duality and $\Omega_E^1\simeq\mathcal O_E$ give $H^1(E,A^k)=0$, and the cohomology sequences of the filtration twisted by $A^k$ then give $H^1(E,A^k\otimes\operatorname{Sym}^sU)=0$. Riemann--Roch on the curve of genus one, for a bundle of rank $s+1$ and degree $kd(s+1)$, gives the dimension formula.

Finally, the leading coefficient gives the exact sequence of bundles
\[
 0\longrightarrow A\otimes\operatorname{Sym}^{s-1}U
 \xrightarrow{\ h\ }A\otimes\operatorname{Sym}^sU
 \longrightarrow A\longrightarrow0.
\]
Indeed, translating $u$ by a multiple of $h$ does not change the leading coefficient, and a polynomial with zero leading coefficient is divisible by $h$. The vanishing of $H^1(E,A\otimes\operatorname{Sym}^{s-1}U)$ just proved gives the asserted surjectivity and kernel on global sections.
\end{proof}

For a line bundle $\mathcal L$ on a projective manifold $Y$ of dimension $N$, recall that its volume is $\operatorname{vol}(\mathcal L)=\limsup_{r\to\infty}N!\,h^0(Y,\mathcal L^r)/r^N$, and that $\mathcal L$ is big if and only if $\operatorname{vol}(\mathcal L)>0$.

\begin{theorem}\label{thm:split-elliptic-symmetric-dimensions}
\index{Vector bundle!big tangent bundle}
\index{Volume!tautological line bundle}
For every integer $r\geq0$,
\[
 h^0(M,\operatorname{Sym}^rT_M)
       =1+\frac{d\,r(r+1)(r+2)}6.
\]
The tautological line bundle $\mathcal O_{\mathbb P(T_M)}(1)$ is big and has volume $d$.
\end{theorem}
\begin{proof}
By \cref{lem:split-elliptic-tensor-weights,lem:elliptic-unipotent-symmetric-cohomology}, the positive weights contribute nothing, the weight zero contributes one dimension, and the weight $-k$ contributes $kd(r-k+1)$ dimensions. Thus
\[
 h^0(M,\operatorname{Sym}^rT_M)
 =1+d\sum_{k=1}^r k(r-k+1)
 =1+\frac{d\,r(r+1)(r+2)}6.
\]
The elliptic curve and its holomorphic line bundle $A$ are algebraic, so $M$ and $\mathbb P(T_M)$ are projective. The identity $\pi_*\mathcal O_{\mathbb P(T_M)}(r)=\operatorname{Sym}^rT_M$ identifies the section spaces. Since $\mathbb P(T_M)$ has dimension three,
\[
 \operatorname{vol}\bigl(\mathcal O_{\mathbb P(T_M)}(1)\bigr)
 =\limsup_{r\to\infty}\frac{6h^0(M,\operatorname{Sym}^rT_M)}{r^3}=d>0,
\]
so the tautological line bundle is big.
\end{proof}

\begin{theorem}\label{thm:split-elliptic-tangent-ring-nonfinite}
\index{Section ring!nonfinite generation}
The complex algebra
\[
 R=\bigoplus_{r\geq0}H^0(M,\operatorname{Sym}^rT_M)
\]
is not finitely generated.
\end{theorem}
\begin{proof}
Write $R_{r,k}$ for the component of tensor degree $r$ and weight $-k$. By \cref{lem:split-elliptic-tensor-weights,lem:elliptic-unipotent-symmetric-cohomology}, only $k\geq0$ occurs, multiplication adds both indices, and the subalgebra of weight zero is $\mathbb C[h]$. Let $I$ be the ideal formed by all components with $k>0$, and put
\[
 S=R/(hR+I^2).
\]
Its component of weight zero is $\mathbb C$, and its ideal $J$ of positive weight index satisfies $J^2=0$, so $S=\mathbb C\oplus J$ as a vector space.

For every $r\geq1$, products in $I^2$ have weight index at least two. Consequently the component of $S$ of degree $r$ and weight index one is
\[
 R_{r,1}/hR_{r-1,1}\simeq H^0(E,A).
\]
For $r=1$ this holds because $R_{0,1}=0$, and for $r\geq2$ it is the last assertion of \cref{lem:elliptic-unipotent-symmetric-cohomology} with $s=r-1$. Each of these components has dimension $d>0$, and they lie in distinct degrees, so $J$ is infinite dimensional.

If $R$ were finitely generated, its quotient $S$ would be finitely generated too. Write a finite set of algebra generators of $S$ as $c_i+j_i$, $1\leq i\leq N$, with $c_i\in\mathbb C$ and $j_i\in J$. Because $J^2=0$, every polynomial in these generators lies in the finite-dimensional span of $1,j_1,\ldots,j_N$. This contradicts the infinite dimension of $J$.
\end{proof}

\section{Section algebras of cotangent extensions}
\label{subsec:elliptic-cotangent-extension-algebra}

We now pass from the tangent bundle to the dual $V$ of an extension $W$ of $\mathcal O_M$ by $\Omega^1_M$ with nonzero ruling component, for instance the canonical extension of a K\"ahler class. Sections of $\operatorname{Sym}^rV$ are fiber-homogeneous polynomial functions on $W$. Compared with the cotangent bundle there is one more fiber coordinate $w$, the coordinate of the quotient $\mathcal O_M$, and the affine term of the torsor becomes a quadratic relation among the fiber-polynomial functions over the ruling. The weight decomposition of the preceding section still applies and shows that the section algebra of $V$ is a polynomial ring in one variable over the tangent algebra. In particular, $\mathcal O_{\mathbb P(V)}(1)$ is big and the section algebra of $V$ is not finitely generated.

Keep the notation of the two preceding sections. Let $W$ be an extension of $\mathcal O_M$ by $\Omega_M^1$ with class $\rho\xi+Kf^*\eta$, where $\rho\ne0$, and put $V=W^*$. Adding the coordinate $w$ of the quotient $\mathcal O_M$ to the charts in the proof of \cref{thm:split-elliptic-cotangent-torsor}, the fiber coordinates over the elliptic cover change by
\[
 v=1/t,\qquad u_\infty=u,\qquad
 y_\infty=-t^2y-\rho t w,\qquad w_\infty=w.
\]
For $\lambda=m\tau+n$, the lattice acts by
\[
 (z,t,u,y,w)\longmapsto
 (z+\lambda,b_\lambda t,u+m(Kw+q_0ty),y/b_\lambda,w),
 \qquad q_0=2\pi\mathrm{i}d,
\]
with the normalized factors $b_\lambda$. Fiber-homogeneous polynomial functions on $W$ represent symmetric tensors of $V$. In the ruling coordinates they satisfy a quadratic relation coming from the affine term of the torsor, while the elliptic monodromy determines which weights descend to $M$.

\begin{lemma}\label{lem:elliptic-extension-fiber-ring}
\index{Quadric!cotangent-extension functions}
\index{Weight decomposition!cotangent extension}
Over $\mathbb C\times\mathbb P^1$, the graded algebra of fiber-homogeneous polynomial functions on the pullback of $W$ is
\[
 \mathcal O(\mathbb C)
 [c_-,c_+,h,w,u]/\bigl(c_-c_+-h(h+\rho w)\bigr),
\]
where all the variables have degree one and, in the finite ruling chart,
\[
 c_-=y,\qquad c_+=t(ty+\rho w),\qquad h=ty.
\]
The lattice fixes $h$ and $w$, multiplies $c_-$ by $b_\lambda^{-1}$ and $c_+$ by $b_\lambda$, and sends $u$ to $u+m(q_0h+Kw)$.
\end{lemma}
\begin{proof}
Set $s_1=y$ and $s_2=ty+\rho w$ in the finite chart. Since $\rho\ne0$, these are linear coordinates that can replace $y,w$, with $w=(s_2-ts_1)/\rho$. In the infinite chart put
\[
 (s_{1,\infty},s_{2,\infty})
       =(-v y_\infty-\rho w,-y_\infty).
\]
The transition is $(s_{1,\infty},s_{2,\infty})=t(s_1,s_2)$, so $s_1,s_2$ are the fiber coordinates of $\mathcal O_{\mathbb P^1}(-1)^{\oplus2}$, while $u$ is the coordinate of a trivial line bundle.

By holomorphy in the infinite chart, a polynomial that is homogeneous of degree $j$ in $s_1,s_2$ has coefficients of degree at most $j$ in $t$. Distributing the exponent of $t$ among the $j$ fiber factors, every admissible monomial is a product of $s_1,s_2,ts_1,ts_2$. These four generators satisfy the determinant relation
\[
 s_1(ts_2)-(ts_1)s_2=0,
\]
and there are no further relations. Indeed, counting the $j+1$ monomials in $s_1,s_2$ and the $j+1$ admissible powers of $t$, the space of functions of degree $j$ has dimension $(j+1)^2$. The polynomial algebra in four variables modulo this nonzero quadratic relation has the same dimension
\[
 \binom{j+3}{3}-\binom{j+1}{3}=(j+1)^2
\]
in degree $j$, where the second term is zero for $j<2$. The natural surjective graded map is therefore an isomorphism in every degree. Including $u$ adjoins a polynomial variable. The argument is uniform in $z$, and all coefficient functions are entire.

Identify the four generators as $c_-=s_1$, $h=ts_1$, $h+\rho w=s_2$ and $c_+=ts_2$; the determinant relation becomes the stated quadratic relation. These functions are polynomial also in the infinite chart, where
\[
 c_-=-v^2y_\infty-\rho v w,\qquad
 c_+=-y_\infty,\qquad h=-v y_\infty-\rho w.
\]
The lattice formulas follow by substitution in the finite chart.
\end{proof}

\begin{theorem}\label{thm:elliptic-extension-polynomial-ring}
\index{Section ring!canonical extension}
\index{Canonical extension!section algebra}
There is a degree-preserving isomorphism of complex algebras
\[
 \bigoplus_{r\geq0}H^0(M,\operatorname{Sym}^rV)
 \simeq
 \left(\bigoplus_{r\geq0}H^0(M,\operatorname{Sym}^rT_M)\right)[w],
\]
where the variable $w$ on the right has degree one.
\end{theorem}
\begin{proof}
\emph{Step 1: weights.} Give $c_-$ weight $-1$, $c_+$ weight $1$, and $h,w,u$ weight zero. These weights come from the ruling action and are compatible with the quadratic relation and the lattice action. Using the relation to eliminate products $c_-c_+$, a polynomial of weight $-k$, $k\geq1$, can be written as $c_-^kF(z,h,w,u)$, and a polynomial of weight $k$ as $c_+^kF(z,h,w,u)$, where $F$ has degree $r-k$ in tensor degree $r$; weight zero consists of the polynomials $F(z,h,w,u)$. These expressions are unique. Indeed, the substitution $c_+\mapsto h(h+\rho w)/c_-$ embeds the algebra of \cref{lem:elliptic-extension-fiber-ring} into
\[
 \mathcal O(\mathbb C)[h,w,u,c_-,c_-^{-1}],
\]
the reduced expressions involve distinct powers of $c_-$, and their coefficients vanish only if the original coefficients vanish, because $h(h+\rho w)$ is a nonzero polynomial. This proves that the weight decomposition is a direct sum.

\emph{Step 2: coefficient bundles.} Let $U_3$ be the rank-three bundle on $E$ of linear polynomials in $h,w,u$ with the transformation rule
\[
 F(z+\lambda,h,w,u)
       =F(z,h,w,u-m(q_0h+Kw)).
\]
By the transformation of $c_-$ and $c_+$ in \cref{lem:elliptic-extension-fiber-ring}, the block of weight $-k$ is $H^0(E,A^k\otimes\operatorname{Sym}^{r-k}U_3)$, and the block of weight $k$ is $H^0(E,A^{-k}\otimes\operatorname{Sym}^{r-k}U_3)$; the inverse translation of $u$ appears when invariance is written out. The signs of the twists agree with those in \cref{lem:split-elliptic-tensor-weights}.

\emph{Step 3: reduction to the tangent bundle.} Since $q_0\ne0$, put $H=h+(K/q_0)w$. Then the translation of $u$ is $u\mapsto u-mq_0H$, and consequently
\[
 U_3\simeq U\oplus\mathcal O_Ew,
\]
where the variable $h$ of the bundle $U$ defined above is replaced by $H$. The symmetric powers of $U_3$ are the direct sums of the bundles $w^j\operatorname{Sym}^{s-j}U$, $0\leq j\leq s$. The vanishing for negative twists in \cref{lem:elliptic-unipotent-symmetric-cohomology} eliminates every positive weight, and the same lemma identifies the algebra of sections of weight zero with $\mathbb C[H,w]$.

In each negative weight, expanding in powers of $w$ identifies the remaining spaces with the corresponding blocks for the tangent bundle in \cref{lem:split-elliptic-tensor-weights}, with a polynomial variable $w$ adjoined. Sending the Euler variable $h$ of those blocks to $H$ and their leading factor $y^k$ to $c_-^k$ gives bijections in every tensor degree and weight. Products of surviving blocks multiply their coefficients and add their powers of $c_-$, and the eliminated positive weights contribute no products. The bijections therefore preserve multiplication and give the asserted isomorphism of graded algebras.
\end{proof}

\begin{corollary}\label{cor:elliptic-extension-symmetric-dimensions}
\index{Volume!canonical-extension tautological line}
For every integer $r\geq0$,
\[
 h^0(M,\operatorname{Sym}^rV)
       =r+1+d\binom{r+3}{4}.
\]
The tautological line bundle on $\mathbb P(V)$ is big, with volume $d$, and the section algebra of $V$ is not finitely generated. These conclusions hold for the canonical extension of every K\"ahler class on $M$.
\end{corollary}
\begin{proof}
The isomorphism of \cref{thm:elliptic-extension-polynomial-ring} and \cref{thm:split-elliptic-symmetric-dimensions} give
\[
 h^0(M,\operatorname{Sym}^rV)
 =\sum_{j=0}^r h^0(M,\operatorname{Sym}^jT_M)
 =r+1+d\sum_{j=0}^r\binom{j+2}{3}
 =r+1+d\binom{r+3}{4},
\]
where $\binom{2}{3}=0$. The surface $M$ is projective, and by coherent GAGA the holomorphic bundle $V$ is algebraic. Thus $\mathbb P(V)$ is a projective fourfold, and the volume of its tautological line bundle is the limit of $24h^0(M,\operatorname{Sym}^rV)/r^4$, which equals $d>0$. Hence this line bundle is big. If the section algebra of $V$ were finitely generated, so would be its quotient by $w$, contradicting \cref{thm:split-elliptic-tangent-ring-nonfinite}.

For the canonical extension $0\to\mathcal O_M\to V\to T_M\to0$ of a K\"ahler class, the dual extension of $\mathcal O_M$ by $\Omega^1_M$ has nonzero ruling component, as shown in the proof of \cref{thm:split-elliptic-cotangent-torsor}. Its class therefore has $\rho\ne0$ in the conventions used here, and the same argument applies.
\end{proof}

\section{Cotangent extensions with zero ruling component}
\label{subsec:elliptic-zero-ruling-extension}

Every K\"ahler class has a nonzero ruling component, but the extensions with $\rho=0$, that is, with class $Kf^*\eta$, can be treated in the same way. Their section algebras are the same as for $\rho\neq0$, but their torsors behave differently: they contain compact rational curves, and their universal covers are not Stein.

Keep the notation of the preceding sections, with the class $\eta$ and the form $\vartheta_E$ of \cref{lem:elliptic-additive-class}. Let $W$ be an extension of $\mathcal O_M$ by $\Omega_M^1$ with class $Kf^*\eta$, and put $V=W^*$. The homogeneous charts used above, now with zero ruling component, have transitions
\[
 v=1/t,\qquad u_\infty=u,\qquad
 y_\infty=-t^2y,\qquad w_\infty=w.
\]
For $\lambda=m\tau+n$, the lattice acts by
\[
 (z,t,u,y,w)\longmapsto
 (z+\lambda,b_\lambda t,u+m(q_0ty+Kw),y/b_\lambda,w),
 \qquad q_0=2\pi\mathrm{i}d.
\]
For the differences $z_i-z_j=m_{ij}\tau+n_{ij}$ of local lifts, the zero sections of these charts differ by $Km_{ij}f^*\vartheta_E$ on overlaps over the base and agree on the two ruling charts, so they give the class $Kf^*\eta$. The linear parts are the cotangent transitions, and in the infinite ruling chart the lattice acts holomorphically by $u_\infty\mapsto u_\infty+m(Kw-q_0v y_\infty)$. Thus the same cocycle computation constructs $W$ also when its ruling component is zero. The fiberwise change of coordinates used for $\rho\ne0$ is not available, but the polynomial functions on the cotangent bundle give the homogeneous algebra directly.

\begin{lemma}\label{lem:elliptic-zero-ruling-quadric}
\index{Quadric!zero-ruling cotangent extension}
Over $\mathbb C\times\mathbb P^1$, the graded algebra of fiber-homogeneous polynomial functions on the pullback of $W$ is
\[
 \mathcal O(\mathbb C)[c_-,c_+,h,u,w]/(c_-c_+-h^2),
 \qquad c_-=y,\quad c_+=t^2y,\quad h=ty,
\]
where all five variables have degree one. The lattice fixes $h$ and $w$, sends $c_-$ to $b_\lambda^{-1}c_-$ and $c_+$ to $b_\lambda c_+$, and translates $u$ by $m(q_0h+Kw)$.
\end{lemma}
\begin{proof}
As in the proof of \cref{lem:split-elliptic-tensor-weights}, holomorphy in the infinite ruling chart bounds the degree in $t$ of the coefficient of $y^j$ by $2j$; the variable $w$ does not change between the charts. Distributing the exponent $a$ among $j$ integers in $\{0,1,2\}$, every monomial $t^ay^j$ with $0\leq a\leq2j$ is a product of $y$, $ty$ and $t^2y$. These generators satisfy $y(t^2y)=(ty)^2$.

In degree $j$, the space of functions of $y$ and $t$ has dimension $2j+1$. The polynomial algebra in three variables modulo the nonzero quadratic relation has the same dimension
\[
 \binom{j+2}{2}-\binom{j}{2}=2j+1,
\]
where the second term is zero for $j<2$, so the natural surjective graded map is an isomorphism. Adjoining the two trivial fiber coordinates $u,w$ gives the stated algebra, with entire coefficients in $z$. In the other chart,
\[
 c_-=-v^2y_\infty,\qquad c_+=-y_\infty,
 \qquad h=-v y_\infty,
\]
so all generators are holomorphic there. Substitution gives their lattice transformations.
\end{proof}

\begin{proposition}\label{prop:elliptic-zero-ruling-section-ring}
\index{Section ring!zero-ruling cotangent extension}
There is a degree-preserving isomorphism of algebras
\[
 \bigoplus_{r\geq0}H^0(M,\operatorname{Sym}^rV)
 \simeq
 \left(\bigoplus_{r\geq0}H^0(M,\operatorname{Sym}^rT_M)\right)[w],
 \qquad \deg w=1.
\]
Consequently the dimension formula, the bigness and volume of the tautological line bundle, and the failure of finite generation in \cref{cor:elliptic-extension-symmetric-dimensions} hold also for this extension.
\end{proposition}
\begin{proof}
The proof of \cref{thm:elliptic-extension-polynomial-ring} uses \cref{lem:elliptic-extension-fiber-ring} only through its description of the algebra as $\mathcal O(\mathbb C)[c_-,c_+,h,w,u]/(c_-c_+-P)$, with the nonzero polynomial $P=h(h+\rho w)$, and through the lattice action on these generators. \Cref{lem:elliptic-zero-ruling-quadric} gives the same description with $P=h^2$ and the same lattice action, so that proof applies verbatim and gives the isomorphism. Summing the dimensions for the tangent bundle and taking the quotient by $w$ gives the remaining assertions, exactly as in the proof of \cref{cor:elliptic-extension-symmetric-dimensions}.
\end{proof}

\begin{proposition}\label{prop:elliptic-zero-ruling-torsor-cover}
\index{Stein manifold!zero-ruling torsor obstruction}
\index{Universal cover!zero-ruling cotangent torsor}
The cotangent torsor $T_W$ is not Stein. Its universal cover is
\[
 \mathbb C^2\times\operatorname{Tot}
                      (\mathcal O_{\mathbb P^1}(-2)),
\]
where $\operatorname{Tot}$ denotes the total space of a line bundle; this cover is not Stein either.
\end{proposition}
\begin{proof}
Set $w=1$. The subset $\{y=0\}$ of the finite ruling chart agrees with $\{y_\infty=0\}$ in the other chart, and the lattice acts on it by $u\mapsto u+Km$. It therefore descends to a submanifold of $T_W$ that is a $\mathbb P^1$-bundle over
\[
 B_K=\mathbb C^2_{z,u}/\Lambda,\qquad
 (m\tau+n)\cdot(z,u)=(z+m\tau+n,u+Km);
\]
this quotient is a complex manifold because translation in $z$ gives a free and properly discontinuous action. Every fiber of this bundle is a compact connected curve in $T_W$, on which holomorphic functions are constant. Thus $T_W$ is not holomorphically separable, and hence not Stein.

Before the lattice quotient, the affine charts have no affine term in the ruling direction, and they glue to $\mathbb C^2_{z,u}\times\operatorname{Tot}(\mathcal O_{\mathbb P^1}(-2))$, since $\mathcal O_{\mathbb P^1}(-2)$ is the cotangent bundle of $\mathbb P^1$. This space is simply connected: contracting the fibers of the line bundle retracts the last factor onto $\mathbb P^1$, and $\mathbb C^2$ is contractible. The lattice acts freely and properly discontinuously, so this space is the universal cover. Its zero section contains a compact projective line, which again obstructs holomorphic separability and hence the Stein property.
\end{proof}

Thus the classes with zero and with nonzero ruling component have isomorphic graded section algebras, although \cref{thm:split-elliptic-cotangent-torsor} gives a Stein universal cover only when the ruling component is nonzero.
