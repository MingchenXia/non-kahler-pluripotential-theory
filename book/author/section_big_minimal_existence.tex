\section{Minimal singularities in an arbitrary big background}
\label{sec:big-minimal-existence}

We solve the equation in the type of $V_\theta$. Resolve strict analytic approximants to reduce first to a semipositive big background. Their solutions have uniform bounds relative to the approximating singularities; upper and lower envelopes then recover a solution of minimal type. For the normalized equation, a common interval below $V_\theta$ allows us to apply the preceding compactness theorem directly.

The fixed barrier satisfies $\rho\leq V_\theta\leq0$. We use the resolution form of analytic regularization in \cref{thm:resolvable-regularization-input}.

\Needspace{10\baselineskip}
\begin{lemma}\label{lem:big-resolved-equation}
Let $\psi\in\PSH(X,\theta)$ satisfy $\theta_\psi\geq a\omega$ for some $a>0$ and have the resolved smooth-remainder form of \cref{thm:resolvable-regularization-input}. For every nonzero $0\leq f\in L^p(\mathrm{d}V)$, $p>1$, and $\lambda>0$, there exists $u\in\PSH(X,\theta)$ with $u\sim\psi$ such that
\[
 \mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}f\,\mathrm{d}V.
\]
\end{lemma}
\begin{proof}
Resolve $\psi$ by $\pi:Y\to X$. The pulled-back current has the form
\[
 \pi^*\theta+\ddc\pi^*\psi=\beta+[D],
\]
where $D$ is an effective real divisor, $[D]$ is its current of integration, and $\beta$ is a smooth real $(1,1)$-form on $Y$. Off $D$ one has $\beta\geq a\pi^*\omega$, so this inequality holds everywhere by smoothness. Thus $\beta$ is semipositive.

We check its bigness. For a blowup with smooth center and exceptional divisor $E$, choose a metric on $\mathcal O(-E)$ with curvature positive in the directions tangent to the projective fibers. The pullback of a Hermitian metric is positive in the complementary directions along $E$. Consequently its sum with a sufficiently small positive multiple of this curvature is a Hermitian metric: first check positivity along $E$, then in a neighborhood, and finally on the compact complement, where the pullback is already positive. Iterating over the blowups gives an effective real exceptional divisor $F$, with logarithmic potential $\zeta$, and a smooth Hermitian form $\kappa$ such that
\[
 \pi^*\omega+\ddc\zeta=\kappa+[F].
\]
For the iteration, pull back the previous smooth positive form, apply the one-blowup construction, and add the pulled-back previous divisor to the new positive multiple of $E$. Hence
$\beta+a\ddc\zeta\geq a\kappa$, proving bigness of $\beta$. This argument concerns Hermitian metrics and never requires a closed positive form on $Y$.

Fix a smooth positive volume form $\mathrm{d}V_Y$, and write $\pi^*\mathrm{d}V=J\,\mathrm{d}V_Y$. The function $J$ is smooth, nonnegative and bounded, and is positive off the exceptional locus. Define
\[
 H=\mathrm{e}^{\lambda\pi^*\psi}(f\circ\pi)J.
\]
It is nonzero and belongs to $L^p(\mathrm{d}V_Y)$. Indeed $\psi$ is bounded above, and change of variables gives
\[
 \int_Y|(f\circ\pi)J|^p\,\mathrm{d}V_Y
 \leq\|J\|_\infty^{p-1}\int_X|f|^p\,\mathrm{d}V.
\]
Values on the analytic exceptional sets do not affect these integrals. The semipositive big-background theorem \cref{thm:big-bounded-exponential} produces a bounded $s\in\PSH(Y,\beta)$ with
\[
 \mathrm{MA}_\beta(s)=\mathrm{e}^{\lambda s}H\,\mathrm{d}V_Y.
\]
The function $\pi^*\psi+s$ is $\pi^*\theta$-plurisubharmonic, because its current is $\beta_s+[D]\geq0$, where $\beta_s=\beta+\ddc s$. By \cref{thm:hermitian-space-descent} it descends to $u\in\PSH(X,\theta)$. Boundedness of $s$ gives $u\sim\psi$. Off the exceptional and polar sets, change of variables proves the required equation. Both its extended left-hand side and its smooth-volume right-hand side ignore those analytic sets. The equation therefore holds on all of $X$.
\end{proof}

\Needspace{9\baselineskip}
\begin{theorem}\label{thm:big-minimal-exponential}
Let $\theta$ be any smooth big real $(1,1)$-form and let $0\leq f\in L^p(\mathrm{d}V)$, $p>1$, be nonzero. For every $\lambda>0$ there is a unique $u\in\PSH(X,\theta)$ such that
\[
 u\sim V_\theta,\qquad
 \mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}f\,\mathrm{d}V.
\]

\end{theorem}
\begin{proof}
Apply \cref{thm:resolvable-regularization-input} to $V_\theta$. Set $\delta_j=2^{-j}$ and take a subsequence whose curvature errors are at most $\delta_j$. Denote its potentials by $\widetilde\psi_j\downarrow V_\theta$; then
$\theta_{\widetilde\psi_j}\geq-\delta_j\omega$. Set
\[
 a_j=(1-\delta_j)\sup_X\widetilde\psi_j\geq0,
 \qquad
 \psi_j=(1-\delta_j)\widetilde\psi_j+\delta_j\rho-a_j.
\]
Normalized compactness, or monotonicity and Hartogs' lemma, gives $a_j\to0$. Resolving the singularities of $\widetilde\psi_j$ and $\rho$ together shows that $\psi_j$ has a smooth remainder after resolution. Moreover,
\begin{equation}\label{eq:big-minimal-approximation}
 \theta_{\psi_j}\geq\delta_j^2\omega,
 \qquad
 (1-\delta_j)V_\theta+\delta_j\rho-a_j
 \leq\psi_j\leq V_\theta.
\end{equation}
The upper inequality follows because $\psi_j$ is $\theta$-plurisubharmonic and at most zero. The explicit subtraction of $a_j$ keeps the normalization separate from the fixed pole barrier.

By the resolved equation lemma there are $u_j\sim\psi_j$ satisfying
\[
 \mathrm{MA}_\theta(u_j)=\mathrm{e}^{\lambda u_j}f\,\mathrm{d}V.
\]
Each $u_j$ is bounded below by $\rho$ up to a constant depending on $j$. The Gauduchon trace proposition gives a uniform upper bound for $\sup_Xu_j$.

There is also a uniform lower estimate. Translate the fixed $\rho$ down by $1+\sup_j a_j$ when applying \cref{lem:big-reference-subsolution}, and take its reference to be $\psi_j-1$. The lemma gives $\psi_j-1\leq v_j\leq\psi_j\leq0$ and $\mathrm{MA}_\theta(v_j)\geq mf\,\mathrm{d}V$, with $m>0$ independent of $j$. Put $w_j=v_j+\lambda^{-1}\log m$. Then
\[
 \mathrm{MA}_\theta(w_j)\geq \mathrm{e}^{\lambda w_j}f\,\mathrm{d}V,
 \qquad u_j\sim w_j\sim\psi_j.
\]
The exponential order in \cref{cor:big-barrier-orders} gives $u_j\geq w_j$. Thus for a uniform $C$,
\begin{equation}\label{eq:big-minimal-solution-bounds}
 \psi_j-C\leq u_j\leq V_\theta+C.
\end{equation}
After extraction, $u_j$ converges in $L^1$ and almost everywhere to a genuine $u$. The preceding bounds give $V_\theta-C\leq u\leq V_\theta+C$.

\Needspace{14\baselineskip}
We verify its equation using envelopes. Put $A_j=\sup_{k\geq j}a_k$ and, for $N\geq j$, define
\[
 p_{j,N}=P_\theta(u_j,\ldots,u_N),\qquad
 t_{j,N}=\max_{j\leq k\leq N}u_k.
\]
Both functions lie above the genuine competitor
\[
 b_j=(1-\delta_j)V_\theta+\delta_j\rho-C-A_j.
\]
In particular every envelope is genuine. All these functions are uniformly locally bounded on $\Omega$, also as $j$ varies. Contact support from \cref{thm:big-general-envelope-contact} and a disjoint partition by the first contact index give
\[
 \mathrm{MA}_\theta(p_{j,N})\leq \mathrm{e}^{\lambda p_{j,N}}f\,\mathrm{d}V.
\]
There is no factor $N$ here: on each piece the envelope equals one $u_k$, and ordered contact bounds its measure by that single measure. Locality and ordered contact for a finite maximum similarly give
\[
 \mathrm{MA}_\theta(t_{j,N})\geq \mathrm{e}^{\lambda t_{j,N}}f\,\mathrm{d}V.
\]
Let $N\to\infty$, setting $p_j=\lim_Np_{j,N}$ and $t_j=(\lim_Nt_{j,N})^*$. Local bounded monotone continuity passes both inequalities to these limits. Next $p_j$ increases to a regularized limit $p$, while $t_j$ decreases to $u$ by Hartogs' lemma. The bound by $b_j$ gives $p\sim V_\theta$, and $p\leq u$. A second use of local bounded monotone continuity gives
\[
 \mathrm{MA}_\theta(p)\leq \mathrm{e}^{\lambda p}f\,\mathrm{d}V,
 \qquad \mathrm{MA}_\theta(u)\geq \mathrm{e}^{\lambda u}f\,\mathrm{d}V.
\]
These inequalities hold on $\Omega$ and extend across $Z$ because all measures involved ignore it. Thus
$\mathrm{MA}_\theta(p)\leq \mathrm{e}^{\lambda(p-u)}\mathrm{MA}_\theta(u)$. The exponential order gives $p\geq u$, so $p=u$ and the equation follows. Applying that order in both directions also proves uniqueness. Every passage of the Monge--Amp\`ere operator above was along a locally bounded monotone sequence on $\Omega$.
\end{proof}

\begin{lemma}\label{lem:big-exponential-uniform-type}
Under the hypotheses of \cref{thm:big-minimal-exponential}, let
$u_\lambda$ be its solution and put
\[
 v_\lambda=u_\lambda-\sup_Xu_\lambda,
 \qquad c_\lambda=\mathrm{e}^{\lambda\sup_Xu_\lambda}.
\]
There are constants $B,C,m>0$, independent of $0<\lambda\leq1$, such that
\begin{equation}\label{eq:big-normalized-uniform-type}
 V_\theta-B\leq v_\lambda\leq V_\theta,
 \qquad m\leq c_\lambda\leq C.
\end{equation}
\end{lemma}
\begin{proof}
The equation for $v_\lambda$ is
\[
 \mathrm{MA}_\theta(v_\lambda)=c_\lambda\mathrm{e}^{\lambda v_\lambda}
                         f\,\mathrm{d}V.
\]
\Needspace{7\baselineskip}
Fix a Gauduchon metric $\gamma$ and write
$\gamma^n=G\,\mathrm{d}V$. Since $v_\lambda\leq0$ and
$0<\lambda\leq1$, \eqref{eq:big-singular-gauduchon-trace} gives
\[
 c_\lambda^{1/n}
 \int_X\mathrm{e}^{v_\lambda/n}f^{1/n}G^{1-1/n}\,\mathrm{d}V
 \leq\int_X\theta\wedge\gamma^{n-1}.
\]
The integrals on the left have a positive uniform lower bound.
Otherwise quasi-plurisubharmonic compactness would give an almost
everywhere convergent subsequence of the normalized $v_\lambda$ with a
genuine limit, and dominated convergence would give a positive limit
integral, since $f$ is nonzero. Thus $c_\lambda\leq C$.

To control the singularity type, uniform Skoda integrability gives
fixed $\varepsilon>0$ and $1<q<p$ with
$\|\mathrm{e}^{-\varepsilon v_\lambda}f\|_q$ uniformly bounded.
Apply \cref{lem:big-reference-subsolution} with reference
$V_\theta-1$ and barrier $\rho-1$. It supplies
\[
 V_\theta-1\leq\eta_\lambda\leq V_\theta,
 \qquad \mathrm{MA}_\theta(\eta_\lambda)\geq
 m\mathrm{e}^{-\varepsilon v_\lambda}f\,\mathrm{d}V
 \geq mf\,\mathrm{d}V
\]
for a fixed $m>0$. Scalar order, together with
$\mathrm{MA}_\theta(v_\lambda)\leq c_\lambda f\,\mathrm{d}V$, gives
$c_\lambda\geq m$. Set
$D=\varepsilon^{-1}\max\{0,\log(C/m)\}$ and
$w_\lambda=\eta_\lambda-D$. Then
\[
 \mathrm{e}^{\varepsilon(w_\lambda-v_\lambda)}\mathrm{MA}_\theta(v_\lambda)
 \leq m\mathrm{e}^{-\varepsilon v_\lambda}f\,\mathrm{d}V
 \leq\mathrm{MA}_\theta(w_\lambda),
\]
since $\lambda v_\lambda\leq0$ and $\eta_\lambda\leq0$.
Exponential order gives $v_\lambda\geq w_\lambda$, proving the uniform
bounds with $B=1+D$.
\end{proof}

\Needspace{12\baselineskip}
\begin{theorem}\label{thm:big-minimal-normalized}
Under the assumptions of \cref{thm:big-minimal-exponential}, there are $v\in\PSH(X,\theta)$, a constant $C>0$ and a uniquely determined $c>0$ such that
\[
 \sup_Xv=0,
 \qquad V_\theta-C\leq v\leq V_\theta,
 \qquad \mathrm{MA}_\theta(v)=cf\,\mathrm{d}V.
\]
Uniqueness of the normalized potential is not asserted.
\end{theorem}
\begin{proof}
Apply \cref{lem:big-exponential-uniform-type} with $\lambda=1/j$ and
write $v_j=v_{1/j}$, $c_j=c_{1/j}$. Extract a subsequence with
$v_j\to v$ in $L^1$ and almost everywhere, and $c_j\to c\in[m,C]$.
The limit is normalized, genuine and of minimal ordinary type.
The densities $c_j\mathrm{e}^{v_j/j}f$ have a common $L^p$ bound.
Hence \cref{cor:big-reference-uniform-convergence}, with reference
$V_\theta$, upgrades the convergence to uniform convergence of
differences on $\Omega$ and identifies the weak $L^p$ limit of these
densities with the density of $\mathrm{MA}_\theta(v)$.

At almost every point $v_j$ converges to a finite value, so $v_j/j\to0$.
Since $v_j\leq0$ and $c_j\leq C$, dominated convergence gives
$c_j\mathrm{e}^{v_j/j}f\to cf$ in $L^p$. Therefore
$\mathrm{MA}_\theta(v)=cf\,\mathrm{d}V$. Scalar uniqueness is
\cref{cor:big-barrier-orders}.
\end{proof}
