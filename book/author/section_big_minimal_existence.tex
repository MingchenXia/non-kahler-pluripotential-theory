\section{Minimal singularities in an arbitrary big background}
\label{sec:big-minimal-existence}

We now solve the equations in the ordinary singularity type of $V_\theta$. Demailly's regularization approximates $V_\theta$ by strict potentials with analytic singularities. On a resolution of these singularities, the corresponding equations become equations in a semipositive big background, which we have already solved. The solutions satisfy bounds relative to the approximating singularities that are uniform in the approximation, and upper and lower envelopes then produce a solution with minimal singularities. For the normalized equation, the normalized exponential solutions lie in a common interval below $V_\theta$, so the compactness theorem of the preceding section applies directly.

Since $\rho\leq0$ is $\theta$-plurisubharmonic, it is a competitor for $V_\theta$, and hence $\rho\leq V_\theta\leq0$. We use the analytic regularization of \cref{thm:resolvable-regularization-input} in its resolved form.

\Needspace{10\baselineskip}
\begin{lemma}\label{lem:big-resolved-equation}
Let $\psi\in\PSH(X,\theta)$ satisfy $\theta_\psi\geq a\omega$ for some $a>0$, and suppose that $\psi$ has a smooth remainder after resolution, as in \cref{thm:resolvable-regularization-input}. For every nonzero $0\leq f\in L^p(\mathrm{d}V)$, $p>1$, and every $\lambda>0$, there exists $u\in\PSH(X,\theta)$ with $u\sim\psi$ such that
\[
 \mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}f\,\mathrm{d}V.
\]
\end{lemma}
\begin{proof}
Let $\pi:Y\to X$ be a modification resolving the singularities of $\psi$. The pulled-back current has the form
\[
 \pi^*\theta+\ddc\pi^*\psi=\beta+[D],
\]
where $D$ is an effective real divisor, $[D]$ is its current of integration, and $\beta$ is a smooth real $(1,1)$-form on $Y$. Off $D$ we have $\beta\geq a\pi^*\omega$, and by continuity this inequality holds everywhere. In particular, $\beta$ is semipositive.

We check that $\beta$ is big. For a blowup with smooth center and exceptional divisor $E$, choose a metric on $\mathcal O(-E)$ whose curvature is positive in the directions tangent to the projective fibers. Along $E$, the pullback of a Hermitian metric is positive in the complementary directions. Consequently, its sum with a sufficiently small positive multiple of this curvature form is a Hermitian metric: positivity holds first along $E$, then on a neighborhood of $E$, and finally on the compact complement, where the pullback is already positive. Iterating over the blowups, we obtain an effective real exceptional divisor $F$, with logarithmic potential $\zeta$, and a smooth Hermitian form $\kappa$ such that
\[
 \pi^*\omega+\ddc\zeta=\kappa+[F].
\]
At each step of the iteration, we pull back the smooth positive form obtained so far, apply the construction for one blowup, and add the pullback of the previous divisor to the new positive multiple of $E$. Hence
$\beta+a\ddc\zeta\geq a\kappa$, so $\beta$ is big. This argument only involves Hermitian metrics and never requires a closed positive form on $Y$.

Fix a smooth positive volume form $\mathrm{d}V_Y$ on $Y$, and write $\pi^*\mathrm{d}V=J\,\mathrm{d}V_Y$. The function $J$ is smooth, nonnegative and bounded, and it is positive off the exceptional locus. Define
\[
 H=\mathrm{e}^{\lambda\pi^*\psi}(f\circ\pi)J.
\]
This function is nonzero and belongs to $L^p(\mathrm{d}V_Y)$. Indeed, $\psi$ is bounded above, and the change of variables formula gives
\[
 \int_Y|(f\circ\pi)J|^p\,\mathrm{d}V_Y
 \leq\|J\|_\infty^{p-1}\int_X|f|^p\,\mathrm{d}V;
\]
the analytic exceptional sets do not affect these integrals. \Cref{thm:big-bounded-exponential}, applied on $Y$ in the semipositive big background $\beta$, gives a bounded $s\in\PSH(Y,\beta)$ with
\[
 \mathrm{MA}_\beta(s)=\mathrm{e}^{\lambda s}H\,\mathrm{d}V_Y.
\]
The function $\pi^*\psi+s$ is $\pi^*\theta$-plurisubharmonic, because the corresponding current is $\beta_s+[D]\geq0$, where $\beta_s=\beta+\ddc s$. By \cref{thm:hermitian-space-descent} it descends to some $u\in\PSH(X,\theta)$, and $u\sim\psi$ because $s$ is bounded. Off the exceptional set and the pole set of $\psi$, the change of variables formula gives the required equation. Neither the extended measure on the left nor the measure $\mathrm{e}^{\lambda u}f\,\mathrm{d}V$ on the right charges these analytic sets, so the equation holds on all of $X$.
\end{proof}

\Needspace{9\baselineskip}
\begin{theorem}\label{thm:big-minimal-exponential}
Let $\theta$ be any smooth big real $(1,1)$-form and let $0\leq f\in L^p(\mathrm{d}V)$, $p>1$, be nonzero. For every $\lambda>0$ there is a unique $u\in\PSH(X,\theta)$ such that
\[
 u\sim V_\theta,\qquad
 \mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}f\,\mathrm{d}V.
\]
\end{theorem}
\begin{proof}
\emph{Step 1: approximating singularity types.} Apply \cref{thm:resolvable-regularization-input} to $V_\theta$. Set $\delta_j=2^{-j}$ and pass to a subsequence whose curvature errors are at most $\delta_j$. Denote its potentials by $\widetilde\psi_j$; thus $\widetilde\psi_j\downarrow V_\theta$ and
$\theta_{\widetilde\psi_j}\geq-\delta_j\omega$. Set
\[
 a_j=(1-\delta_j)\sup_X\widetilde\psi_j\geq0,
 \qquad
 \psi_j=(1-\delta_j)\widetilde\psi_j+\delta_j\rho-a_j.
\]
By Hartogs' lemma, $\sup_X\widetilde\psi_j\to\sup_XV_\theta=0$, so $a_j\to0$. A common resolution of the singularities of $\widetilde\psi_j$ and $\rho$ shows that $\psi_j$ has a smooth remainder after resolution. Moreover,
\begin{equation}\label{eq:big-minimal-approximation}
 \theta_{\psi_j}\geq\delta_j^2\omega,
 \qquad
 (1-\delta_j)V_\theta+\delta_j\rho-a_j
 \leq\psi_j\leq V_\theta.
\end{equation}
The upper inequality holds because $\psi_j$ is $\theta$-plurisubharmonic and nonpositive. Subtracting $a_j$ explicitly keeps the normalization separate from the fixed barrier $\rho$.

\emph{Step 2: uniform bounds.} By \cref{lem:big-resolved-equation} there are $u_j\sim\psi_j$ satisfying
\[
 \mathrm{MA}_\theta(u_j)=\mathrm{e}^{\lambda u_j}f\,\mathrm{d}V.
\]
Each $u_j$ is bounded below by $\rho$ minus a constant depending on $j$, so \cref{prop:big-singular-gauduchon-trace} gives a uniform upper bound for $\sup_Xu_j$.

There is also a uniform lower bound. We apply \cref{lem:big-reference-subsolution} with the barrier $\rho$ replaced by $\rho-1-\sup_ja_j$ and with the reference $\psi_j-1$, which lies between this barrier and zero. The lemma gives $v_j$ with $\psi_j-1\leq v_j\leq\psi_j\leq0$ and $\mathrm{MA}_\theta(v_j)\geq mf\,\mathrm{d}V$, where $m>0$ is independent of $j$. Put $w_j=v_j+\lambda^{-1}\log m$. Then
\[
 \mathrm{MA}_\theta(w_j)\geq \mathrm{e}^{\lambda w_j}f\,\mathrm{d}V,
 \qquad u_j\sim w_j\sim\psi_j,
\]
and the exponential order in \cref{cor:big-barrier-orders} gives $u_j\geq w_j$. Thus, for a uniform constant $C$,
\begin{equation}\label{eq:big-minimal-solution-bounds}
 \psi_j-C\leq u_j\leq V_\theta+C.
\end{equation}
After extraction, $u_j$ converges in $L^1$ and almost everywhere to a $\theta$-plurisubharmonic function $u$, and the preceding bounds give $V_\theta-C\leq u\leq V_\theta+C$.

\Needspace{14\baselineskip}
\emph{Step 3: the equation for the limit.} We identify the equation using envelopes. Put $A_j=\sup_{k\geq j}a_k$ and, for $N\geq j$, define
\[
 p_{j,N}=P_\theta(\min\{u_j,\ldots,u_N\}),\qquad
 t_{j,N}=\max_{j\leq k\leq N}u_k.
\]
Both functions lie above the $\theta$-plurisubharmonic function
\[
 b_j=(1-\delta_j)V_\theta+\delta_j\rho-C-A_j,
\]
which is a competitor for every $p_{j,N}$; in particular, no envelope is identically $-\infty$. All these functions are locally bounded on $\Omega$, uniformly in $j$ and $N$. Contact support (\cref{thm:big-general-envelope-contact}), together with a partition of the contact set according to the first index of contact, gives
\[
 \mathrm{MA}_\theta(p_{j,N})\leq \mathrm{e}^{\lambda p_{j,N}}f\,\mathrm{d}V.
\]
No factor $N$ appears here: on each piece the envelope equals a single $u_k$, and ordered contact bounds its measure by the measure of that $u_k$. Similarly, locality and ordered contact for a finite maximum give
\[
 \mathrm{MA}_\theta(t_{j,N})\geq \mathrm{e}^{\lambda t_{j,N}}f\,\mathrm{d}V.
\]
Now let $N\to\infty$, and set $p_j=\lim_Np_{j,N}$ and $t_j=(\lim_Nt_{j,N})^*$. Continuity of local bounded products along monotone sequences passes both inequalities to these limits. Next, $p_j$ increases to a regularized limit $p$, while $t_j$ decreases to $u$ by Hartogs' lemma. The bound by $b_j$ gives $p\sim V_\theta$, and $p\leq u$. A second application of the monotone continuity gives
\[
 \mathrm{MA}_\theta(p)\leq \mathrm{e}^{\lambda p}f\,\mathrm{d}V,
 \qquad \mathrm{MA}_\theta(u)\geq \mathrm{e}^{\lambda u}f\,\mathrm{d}V.
\]
These inequalities hold on $\Omega$, and they extend across $Z$ because no measure involved charges $Z$. Thus
$\mathrm{MA}_\theta(p)\leq \mathrm{e}^{\lambda(p-u)}\mathrm{MA}_\theta(u)$, and the exponential order in \cref{cor:big-barrier-orders} gives $p\geq u$. Hence $p=u$, and the equation follows. Applying the same order in both directions proves uniqueness. Note that the Monge--Amp\`ere operator was only ever applied along monotone sequences that are locally bounded on $\Omega$.
\end{proof}

\begin{lemma}\label{lem:big-exponential-uniform-type}
Under the hypotheses of \cref{thm:big-minimal-exponential}, let
$u_\lambda$ be the solution given there, and put
\[
 v_\lambda=u_\lambda-\sup_Xu_\lambda,
 \qquad c_\lambda=\mathrm{e}^{\lambda\sup_Xu_\lambda}.
\]
There are constants $B,C,m>0$, independent of $0<\lambda\leq1$, such that
\begin{equation}\label{eq:big-normalized-uniform-type}
 V_\theta-B\leq v_\lambda\leq V_\theta,
 \qquad m\leq c_\lambda\leq C.
\end{equation}
\end{lemma}
\begin{proof}
The normalized potential satisfies
\[
 \mathrm{MA}_\theta(v_\lambda)=c_\lambda\mathrm{e}^{\lambda v_\lambda}
                         f\,\mathrm{d}V.
\]
\Needspace{7\baselineskip}
Fix a Gauduchon metric $\gamma$ and write
$\gamma^n=G\,\mathrm{d}V$. Since $v_\lambda\leq0$ and
$0<\lambda\leq1$, \eqref{eq:big-singular-gauduchon-trace} gives
\[
 c_\lambda^{1/n}
 \int_X\mathrm{e}^{v_\lambda/n}f^{1/n}G^{1-1/n}\,\mathrm{d}V
 \leq\int_X\theta\wedge\gamma^{n-1}.
\]
The integrals on the left have a positive lower bound independent of $\lambda$.
Otherwise, the compactness of normalized quasi-plurisubharmonic functions would give an almost
everywhere convergent sequence of the normalized $v_\lambda$ whose integrals tend to zero, with a
$\theta$-plurisubharmonic limit; since $f$ is nonzero, dominated convergence would give a positive limit
integral. Thus $c_\lambda\leq C$.

To control the singularity type, we use uniform exponential integrability (\cref{lem:uniform-exponential}) and H\"older's inequality to choose
$\varepsilon>0$ and $1<q<p$ such that
$\|\mathrm{e}^{-\varepsilon v_\lambda}f\|_q$ is uniformly bounded.
\Cref{lem:big-reference-subsolution}, applied with the reference
$V_\theta-1$ and the barrier $\rho-1$, gives $\eta_\lambda\in\PSH(X,\theta)$ with
\[
 V_\theta-1\leq\eta_\lambda\leq V_\theta,
 \qquad \mathrm{MA}_\theta(\eta_\lambda)\geq
 m\mathrm{e}^{-\varepsilon v_\lambda}f\,\mathrm{d}V
 \geq mf\,\mathrm{d}V
\]
for a fixed $m>0$. The scalar order in \cref{cor:big-barrier-orders}, together with
$\mathrm{MA}_\theta(v_\lambda)\leq c_\lambda f\,\mathrm{d}V$, gives
$c_\lambda\geq m$. Set
$D=\varepsilon^{-1}\max\{0,\log(C/m)\}$ and
$w_\lambda=\eta_\lambda-D$. Then
\[
 \mathrm{e}^{\varepsilon(w_\lambda-v_\lambda)}\mathrm{MA}_\theta(v_\lambda)
 \leq m\mathrm{e}^{-\varepsilon v_\lambda}f\,\mathrm{d}V
 \leq\mathrm{MA}_\theta(w_\lambda),
\]
since $\lambda v_\lambda\leq0$ and $\eta_\lambda\leq0$.
The exponential order in \cref{cor:big-barrier-orders} gives $v_\lambda\geq w_\lambda$, which proves the uniform
bounds with $B=1+D$.
\end{proof}

\Needspace{12\baselineskip}
\begin{theorem}\label{thm:big-minimal-normalized}\index{Monge--Ampere equation@Monge--Amp\`ere equation!minimal singularities}
Under the assumptions of \cref{thm:big-minimal-exponential}, there are $v\in\PSH(X,\theta)$, a constant $C>0$ and a uniquely determined $c>0$ such that
\[
 \sup_Xv=0,
 \qquad V_\theta-C\leq v\leq V_\theta,
 \qquad \mathrm{MA}_\theta(v)=cf\,\mathrm{d}V.
\]
\end{theorem}

As in the semipositive case, only the constant $c$ is asserted to be unique, not the normalized potential.

\begin{proof}
Apply \cref{lem:big-exponential-uniform-type} with $\lambda=1/j$, and
write $v_j=v_{1/j}$ and $c_j=c_{1/j}$. Extract a subsequence with
$v_j\to v$ in $L^1$ and almost everywhere, and with $c_j\to c\in[m,C]$.
The limit is normalized, $\theta$-plurisubharmonic, and has minimal singularities.
The densities $c_j\mathrm{e}^{v_j/j}f$ have a common $L^p$ bound.
Hence \cref{cor:big-reference-uniform-convergence}, applied with the reference
$V_\theta$, upgrades the convergence to uniform convergence of
differences on $\Omega$ and identifies the weak $L^p$ limit of these
densities with the density of $\mathrm{MA}_\theta(v)$.

At almost every point, $v_j$ converges to a finite value, so $v_j/j\to0$.
Since $v_j\leq0$ and $c_j\leq C$, dominated convergence gives
$c_j\mathrm{e}^{v_j/j}f\to cf$ in $L^p$. Therefore
$\mathrm{MA}_\theta(v)=cf\,\mathrm{d}V$. The uniqueness of the constant follows from
\cref{cor:big-barrier-orders}.
\end{proof}
