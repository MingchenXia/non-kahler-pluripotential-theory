\section{Second derivatives and ellipticity}

The equation controls the product of the eigenvalues of $\omega_u$ relative to $\omega$, so an upper bound for their sum gives uniform ellipticity. We obtain this trace bound by completing a square in the differentiated equation. The remaining torsion term is of first order in the trace, and it can be absorbed at the maximum of a suitable function of the trace and the potential.

In holomorphic coordinates $(z^1,\ldots,z^n)$, subscripts denote coordinate derivatives, inverse matrices carry upper indices, and repeated indices run from $1$ to $n$. Write
\[
 \omega=\frac{\mathrm i}{2\pi}g_{i\bar j}\,\mathrm{d}z^i\wedge \mathrm{d}\bar z^j,
 \qquad \omega_u=\frac{\mathrm i}{2\pi}h_{i\bar j}\,\mathrm{d}z^i\wedge \mathrm{d}\bar z^j,
 \qquad h_{i\bar j}=g_{i\bar j}+u_{i\bar j}.
\]
For a smooth real function $v$, set $\Delta_gv=g^{i\bar j}v_{i\bar j}$ and $\Delta_hv=h^{i\bar j}v_{i\bar j}$. Thus $\Delta_h=L_u$ and $\Delta_hu=n-\operatorname{tr}_hg$. Write also $|\partial v|_h^2=h^{i\bar j}v_i v_{\bar j}$. Let $\nabla$ be the Chern connection of the fixed metric $g$, with
\[
 \Gamma_{ik}^p=g^{p\bar q}\partial_i g_{k\bar q},\qquad
 T_{ik}^p=\Gamma_{ik}^p-\Gamma_{ki}^p,\qquad
 \tau_k=\sum_iT_{ik}^i.
\]
Here $T$ is the torsion of $\nabla$ and $\tau$ its trace; all bounds for this connection, its torsion and its curvature depend only on $g$.

\Needspace{13\baselineskip}
\begin{lemma}\label{lem:hermitian-trace-inequality}
Suppose $h=g+(u_{i\bar j})>0$, and put
\[
 S=\operatorname{tr}_gh,\qquad B=\operatorname{tr}_hg,
 \qquad G=\log\frac{\det h}{\det g}.
\]
There is $C_g$ such that, at every point where $S\geq1$,
\begin{equation}\label{eq:hermitian-trace-inequality}
 \Delta_h\log S\geq
 \frac2{S^2}\operatorname{Re}\bigl(h^{k\bar q}\tau_k S_{\bar q}\bigr)
 +\frac{\Delta_gG}{S}-C_gB.
\end{equation}
\end{lemma}
\Needspace{7\baselineskip}
\begin{proof}
The identity $\mathrm{d}\omega_u=\mathrm{d}\omega$ is equivalent to
\begin{equation}\label{eq:trace-torsion-switch}
 \nabla_i h_{k\bar j}-\nabla_k h_{i\bar j}
 =T_{ik}^p(g_{p\bar j}-h_{p\bar j}),
\end{equation}
together with its complex conjugate; this can also be checked directly, since the third derivatives of $u$ cancel.

For tensors with two holomorphic and one antiholomorphic lower index, we use the norm
\[
 \|A\|^2=g^{i\bar\ell}h^{k\bar q}h^{p\bar j}
 A_{ik\bar j}\overline{A_{\ell q\bar p}}.
\]
Set $A_{ik\bar j}=\nabla_i h_{k\bar j}$ and
$D_{ik\bar j}=T_{ik}^p h_{p\bar j}$. We first claim
\begin{equation}\label{eq:trace-before-square}
 \Delta_hS=\|A+D\|^2+\Delta_gG+E,
 \qquad |E|\leq C_g(1+S)B.
\end{equation}
To prove this, start with
$\Delta_hS=h^{i\bar j}g^{k\bar\ell}\nabla_i\nabla_{\bar j}h_{k\bar\ell}$.
Switch $\bar j,\bar\ell$ by the conjugate of \eqref{eq:trace-torsion-switch}, commute $\nabla_i,\nabla_{\bar\ell}$, and then switch $i,k$ by \eqref{eq:trace-torsion-switch}. The terms with a derivative of $h$ are
\[
 h^{i\bar j}g^{k\bar\ell}\nabla_{\bar\ell}\nabla_kh_{i\bar j}
 -2\operatorname{Re}\bigl(h^{i\bar j}g^{k\bar\ell}
             T_{ik}^p\nabla_{\bar\ell}h_{p\bar j}\bigr).
\]
The first switch also produces the term
$-h^{i\bar j}g^{k\bar\ell}\overline{T_{j\ell}^q}\nabla_i h_{k\bar q}$.
Applying \eqref{eq:trace-torsion-switch} to it gives one of the two conjugate derivative terms displayed above, together with
\[
 h^{i\bar j}g^{k\bar\ell}T_{ik}^p\overline{T_{j\ell}^q}
                   (h_{p\bar q}-g_{p\bar q}).
\]
The part of this expression containing $h_{p\bar q}$ is $\|D\|^2$. Differentiating the logarithmic determinant twice gives
\[
 \Delta_gG=h^{i\bar j}g^{k\bar\ell}
                \nabla_{\bar\ell}\nabla_kh_{i\bar j}-\|A\|^2.
\]
By the antisymmetry of $T$, the two remaining derivative terms combine to $2\operatorname{Re}\langle A,D\rangle$, where $\langle\cdot,\cdot\rangle$ is the inner product of the norm above. These identities give \eqref{eq:trace-before-square}.

To bound the remainder $E$ explicitly, let
\[
 \mathcal C_{i\bar\ell k\bar j}
  =(\nabla_i\nabla_{\bar\ell}-\nabla_{\bar\ell}\nabla_i)h_{k\bar j}.
\]
\Needspace{7\baselineskip}
The remainder is the contraction by $h^{i\bar j}g^{k\bar\ell}$ of
\[
 \begin{split}
 &(\nabla_i\overline{T_{j\ell}^q})(g_{k\bar q}-h_{k\bar q})
 +(\nabla_{\bar\ell}T_{ik}^p)(g_{p\bar j}-h_{p\bar j})\\
 &\hspace{15mm}+\mathcal C_{i\bar\ell k\bar j}
              -T_{ik}^p\overline{T_{j\ell}^q}g_{p\bar q}.
 \end{split}
\]
The commutator $\mathcal C$ is the curvature of $\nabla$ acting linearly on $h$, as one sees by inserting $\nabla=\partial+\Gamma$ and canceling the derivatives of $h$. In a $g$-unitary frame diagonalizing $h$, every coefficient in this remainder is bounded by $C_g(1+S)$, so the contraction is bounded by $C_g(1+S)\sum_i h^{i\bar i}=C_g(1+S)B$, as asserted. In particular, no derivative of $h$ remains in $E$.

It remains to control the gradient term introduced by the logarithm. Define
\[
 H_{ik\bar j}=h_{i\bar j}\frac{S_k}{S},\qquad
 K=\|A+D-H\|^2\geq0.
\]
By \eqref{eq:trace-torsion-switch},
\[
 g^{i\bar\ell}(A+D)_{ik\bar\ell}=S_k+\tau_k.
\]
Moreover $\|H\|^2=|\partial S|_h^2/S$. Expanding the square therefore gives
\[
 K=\|A+D\|^2-\frac{|\partial S|_h^2}{S}
       -\frac2S\operatorname{Re}(h^{k\bar q}\tau_kS_{\bar q}).
\]
Combining this with \eqref{eq:trace-before-square} and with $\Delta_h\log S=\Delta_hS/S-|\partial S|_h^2/S^2$, we obtain the exact identity
\[
 \Delta_h\log S=\frac KS+
 \frac2{S^2}\operatorname{Re}(h^{k\bar q}\tau_kS_{\bar q})
 +\frac{\Delta_gG}{S}+\frac ES.
\]
Since $K\geq0$ and $S\geq1$, and since $E/S\geq-2C_gB$, enlarging the constant proves the lemma.
\end{proof}

\Needspace{12\baselineskip}
\begin{theorem}\label{thm:hermitian-second-estimate}
Fix $D,K>0$ and a smooth real function $G$ on $X$. Suppose $u$ is smooth, $\omega_u>0$, and
\[
 \operatorname{osc}_Xu\leq D,\qquad
 \omega_u^n=\mathrm{e}^G\omega^n,\qquad
 |G|\leq K,\qquad \Delta_gG\geq-K.
\]
There is a constant $C=C(\omega,D,K)$ such that
\[
 C^{-1}\omega\leq\omega_u\leq C\omega.
\]
\end{theorem}
\begin{proof}
Use $h=g+(u_{i\bar j})$, $S=\operatorname{tr}_g h$, and $B=\operatorname{tr}_h g$ as in \cref{lem:hermitian-trace-inequality}. Write $s=u-\inf_Xu+1$, so that $1\leq s\leq D+1$, and consider the maximum of
\[
 Q=\log S-Au+s^{-1}
\]
on $X$, where $A\geq1$ will be fixed below. Let $x$ be a maximum point. If $S(x)<1$, an upper bound for $S$ on $X$ already follows from the oscillation bound and the definition of $Q$, as at the end of the proof. We may therefore assume $S(x)\geq1$ and use \cref{lem:hermitian-trace-inequality}.

\Needspace{6\baselineskip}
Since the eigenvalues of $h$ relative to $g$ are positive, $SB\geq n^2$, and hence $\Delta_gG/S\geq-KB/n^2$. Moreover, $\partial Q=0$ at $x$ gives
\[
 \frac{S_k}{S}=(A+s^{-2})u_k.
\]
\Needspace{7\baselineskip}
Because $|\tau|_h^2\leq C_gB$, the absolute value of the torsion term is bounded by
\begin{equation}\label{eq:trace-torsion-absorption}
 \begin{split}
 \left|\frac2{S^2}\operatorname{Re}(h^{k\bar q}\tau_kS_{\bar q})\right|
 &\leq\frac{C_g(A+1)}S\sqrt B\,|\partial u|_h\\
 &\leq\frac{|\partial u|_h^2}{s^3}
          +\frac{C_g(A+1)^2s^3}{S^2}B.
 \end{split}
\end{equation}
If $S(x)^2\leq(A+1)^2(D+1)^3$, then $S(x)$ is bounded once $A$ is fixed, and again so is $S$ on $X$. In the remaining case the coefficient of $B$ in the last term is at most $C_g$.

On the other hand, direct computation gives
\[
 \Delta_hQ=\Delta_h\log S+(A+s^{-2})(B-n)
                         +2s^{-3}|\partial u|_h^2.
\]
The trace inequality and \eqref{eq:trace-torsion-absorption} imply at $x$
\[
 0\geq\Delta_hQ\geq(A-C_g-K/n^2)B-n(A+1),
\]
after a fixed enlargement of $C_g$. Choosing $A>C_g+K/n^2+1$, we conclude that $B(x)$ is bounded. If $\lambda_1,\ldots,\lambda_n$ are the eigenvalues of $h$ relative to $g$, then $\prod_i\lambda_i=\mathrm{e}^G$, and for $n\geq2$
\[
 S=\sum_i\lambda_i
 =\mathrm{e}^G\sum_i\prod_{j\ne i}\lambda_j^{-1}
 \leq n \mathrm{e}^K B^{n-1}.
\]
For $n=1$, directly $S=\mathrm{e}^G$. Thus $S(x)$ is bounded in all cases. For every $y\in X$, the maximality of $Q$ at $x$ gives
\[
 \log S(y)\leq\log S(x)+A(u(y)-u(x))+s(x)^{-1}-s(y)^{-1}
             \leq\log S(x)+AD+1.
\]
This proves $S\leq C$ on $X$. The lower bound for the determinant then gives
$\lambda_i\geq \mathrm{e}^{-K}C^{-(n-1)}$, which proves the uniform equivalence.
\end{proof}

\begin{corollary}\label{cor:classical-higher-estimate}
Fix $\omega$. For smooth solutions of
\[
 (\omega+\ddc u)^n=\mathrm{e}^G\omega^n,\qquad\omega+\ddc u>0,
\]
a uniform $C^0$ bound for $u$ and uniform bounds for the smooth real logarithmic density $G$ in every $C^k$ norm imply uniform bounds for $u$ in every $C^k$ norm.
\end{corollary}
\begin{proof}
\Cref{thm:hermitian-second-estimate} gives uniform ellipticity and a bound for $\Delta_gu=S-n$. The linear elliptic $W^{2,p}$ estimates for the fixed operator $\Delta_g$ first bound $u$ in $C^{1,\alpha}$ for every $0<\alpha<1$. The local Evans--Krylov estimate for the concave logarithmic determinant then gives a uniform $C^{2,\beta}$ bound for some $0<\beta<1$. To apply it in its real form, project real symmetric Hessians to their complex Hermitian part and extend the logarithmic determinant concavely from the compact set of positive matrices given by the preceding theorem. The extension is uniformly elliptic on real symmetric matrices because the projection preserves positivity and the trace, so the usual Evans--Krylov theorem applies. Finally, differentiating the equation gives linear uniformly elliptic equations with H\"older coefficients, and Schauder estimates and induction give all higher bounds. No regularity results beyond these standard local elliptic estimates are needed.
\end{proof}

The same conclusion holds for equations $\omega_u^n=\mathrm{e}^{a u+F}\omega^n$ with $a$ in a fixed bounded interval of nonnegative numbers, $u$ uniformly bounded, and $F$ uniformly bounded in every $C^k$ norm. Indeed, with $G=au+F$,
\[
 \Delta_g(a u+F)=a(S-n)+\Delta_gF\geq-an+\Delta_gF,
\]
so \cref{thm:hermitian-second-estimate} applies without any prior bound for derivatives of $u$. The estimate for the fixed operator $\Delta_g$ then gives a $C^{1,\alpha}$ bound, after which the same local regularity and bootstrapping apply.
