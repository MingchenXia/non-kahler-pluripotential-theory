\section{Stability above a strict barrier}
\label{sec:big-reference-stability}

The subsolutions of \cref{lem:big-reference-subsolution} stay within a bounded distance of a given reference. With them, the proof of \cref{thm:bounded-one-sided-stability} extends to unbounded potentials lying above $\rho$, and the resulting estimate is uniform over all references between $\rho$ and zero. For a fixed reference, it upgrades $L^1$ compactness of solutions to uniform convergence of their differences on $\Omega$.

\begin{theorem}\label{thm:big-reference-one-sided-stability}\index{Stability!varying references}
Fix $C_0,L>0$, $p>1$, and
\[
 0<\gamma<\frac{p-1}{np+p-1}.
\]
Let $\eta\in\PSH(X,\theta)$ satisfy $\rho\leq\eta\leq0$, and suppose that
$u\in\PSH(X,\theta)$ satisfies
\[
 \eta-C_0\leq u\leq\eta,\qquad
 \mathrm{MA}_\theta(u)=f\,\mathrm{d}V,\qquad 0\leq f\in L^p(\mathrm{d}V),
 \quad\|f\|_p\leq L.
\]
There is a constant $C$, depending only on the fixed geometry and on
$\rho,C_0,L,p,\gamma$, such that every $v\in\PSH(X,\theta)$ with
$v\leq\eta$ satisfies
\begin{equation}\label{eq:big-reference-stability}
 \sup_\Omega(v-u)_+
 \leq C\left(\int_\Omega(v-u)_+\,\mathrm{d}V\right)^\gamma.
\end{equation}
\end{theorem}

In particular, the constant does not depend on the reference $\eta$, and the comparison potential $v$ need not have the ordinary singularity type of $\eta$. The difference $v-u$ is evaluated on $\Omega$, where $u$ and $\eta$ are locally bounded; since $Z$ has zero volume, the integral is the same as over $X$. Since $Z$ is pluripolar, the resulting inequality $v\leq u+C\bigl(\int_\Omega(v-u)_+\,\mathrm{d}V\bigr)^\gamma$ on $\Omega$ extends to the canonical representatives on all of $X$.

\begin{proof}
Replacing $v$ by $\max(v,u)$, we may assume
$\eta-C_0\leq u\leq v\leq\eta$, so that $0\leq v-u\leq C_0$ on $\Omega$.
\Needspace{5\baselineskip}
Fix $q\in(1,p)$ and $\delta>0$, and put
\[
 E_\delta=\{u<v-\delta\}\cap\Omega,\qquad
 g_\delta=\mathbf1_{E_\delta}f,\qquad m_\delta=\|g_\delta\|_q.
\]
When $m_\delta>0$, \cref{lem:big-reference-subsolution}, followed by
subtracting one from its potential, gives $h\in\PSH(X,\theta)$ and a
fixed $b>0$ such that
\[
 \eta-1\leq h\leq\eta,\qquad
 \mathrm{MA}_\theta(h)\geq b\,g_\delta/m_\delta\,\mathrm{d}V.
\]
The constant $b$ is uniform in $\eta$, because the auxiliary density has
$L^q$ norm one and the strict barrier is fixed.

If $0<m_\delta<b/2$, set
\[
 t=(2m_\delta/b)^{1/n},\qquad
 w=(1-t)v+th-\delta-C_0t.
\]
Since $h-v\leq C_0$, we have $w\leq v-\delta$, so the set $\{u<w\}\cap\Omega$ is contained in
$E_\delta$. On this set the local expansion into positive mixed products gives
$\mathrm{MA}_\theta(w)\geq t^n\mathrm{MA}_\theta(h)\geq2\mathrm{MA}_\theta(u)$. Both potentials are bounded below
by $\rho$ minus a constant, and $w\leq u+C_0$, so
\cref{thm:big-barrier-domination} applies and gives $u\geq w$. It follows
that
\[
 v-u\leq\delta+C_0t+t(v-h)\leq\delta+(C_0+1)t.
\]
If $m_\delta=0$, we apply the same domination theorem with coefficient zero
to $u$ and $v-\delta$. If $m_\delta\geq b/2$, we use $v-u\leq C_0$.
In all cases,
\[
 \sup_\Omega(v-u)\leq\delta+C_1m_\delta^{1/n}.
\]
Write $I=\int_\Omega(v-u)\,\mathrm{d}V$. If $I=0$, the canonical
potentials agree. Otherwise H\"older's and Chebyshev's inequalities give
\[
 m_\delta\leq L(I/\delta)^{(p-q)/(pq)}.
\]
For $0<I\leq1$, the choice
$\delta=I^{(p-q)/(npq+p-q)}$ yields the exponent
$\gamma_q=(p-q)/(npq+p-q)$. We choose $q>1$ so close to one that
$\gamma_q>\gamma$. The case $I>1$ follows from the bound $v-u\leq C_0$.
This proves \eqref{eq:big-reference-stability}.
\end{proof}

\Needspace{15\baselineskip}
\begin{corollary}\label{cor:big-reference-uniform-convergence}
Fix $\eta\in\PSH(X,\theta)$ with $\rho\leq\eta\leq0$, constants $C_0,L>0$, and $p>1$. Consider the family of potentials
$u\in\PSH(X,\theta)$ satisfying
\[
 \eta-C_0\leq u\leq\eta,\qquad
 \mathrm{MA}_\theta(u)=f_u\,\mathrm{d}V,\qquad0\leq f_u\in L^p(\mathrm{d}V),
 \quad\|f_u\|_p\leq L.
\]
This family is compact for uniform convergence of differences on $\Omega$.
If $u_j$ belong to the family and $u_j\to u$ in $L^1$, then $u$ belongs to the family and
\[
 \sup_\Omega|u_j-u|\leq C\|u_j-u\|_{L^1(\mathrm{d}V)}^\gamma
 \longrightarrow0
\]
for every exponent $\gamma$ as in \cref{thm:big-reference-one-sided-stability}.
Moreover, $f_{u_j}$ converges weakly to $f_u$ in $L^p$.
\end{corollary}
\begin{proof}
Applying the theorem in both orders to $u_j$ and $u_k$, with the common reference
$\eta$, we see that an $L^1$-Cauchy sequence in the family is uniformly Cauchy in
differences on $\Omega$. Its uniform limit on $\Omega$ is locally
$\theta$-plurisubharmonic and agrees almost everywhere, hence everywhere
for the canonical representatives, with the $L^1$ limit $u$. The bounds
$\eta-C_0\leq u\leq\eta$ hold on all of $X$. Letting $k\to\infty$ in
the pairwise estimate proves the quantitative bound, before any equation for $u$ is used.

The densities form a bounded set in the reflexive space $L^p$. Let $f$ be the weak limit of a subsequence $f_{u_{j_k}}$. Since the potentials are locally bounded and converge locally uniformly on $\Omega$, Bedford--Taylor continuity, which holds also for the smooth nonclosed background, gives $\mathrm{MA}_\theta(u)=f\,\mathrm{d}V$ on $\Omega$. Neither measure charges the
analytic set $Z$, so this equality holds on $X$. Positivity and
weak lower semicontinuity of the norm give $f\geq0$ and $\|f\|_p\leq L$.
Thus every weak cluster point equals the density $f_u$ of $\mathrm{MA}_\theta(u)$, which proves
weak convergence of the whole sequence. Finally, the common bounds by
$\eta$ and $\eta-C_0$ give $L^1$ compactness of the family, by the compactness of quasi-plurisubharmonic functions, and the preceding argument upgrades each convergent subsequence
to uniform convergence of differences on $\Omega$.
\end{proof}

For solutions with minimal singularities, normalized by $\sup_Xu=0$, we take $\eta=V_\theta$. Then $\rho\leq V_\theta\leq0$, and every normalized comparison potential $v\in\PSH(X,\theta)$ satisfies $v\leq V_\theta$, as the theorem requires.
