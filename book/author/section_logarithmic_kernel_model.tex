\section{A logarithmic kernel scale in two dimensions}
\label{sec:logarithmic-kernel-model}

In the curve models above, the metric weight has a single homogeneous leading term. In two dimensions, a mixed monomial in the weight can compete with the pure monomials and introduce a logarithm into the scale of the kernel. For an explicit metric on $\mathbb P^2$, the relevant global section norm can be computed directly.

Write $\mathrm dA$ for Euclidean area on $\mathbb C$. We first isolate the integral responsible for the logarithm.
\begin{lemma}\label{lem:mixed-laplace-logarithm}
Set $W(a,b)=ab+a^3+b^3$ on $[0,\infty)^2$. As $t\to\infty$,
\[
 \begin{gathered}
 J(t):=\int_0^\infty\int_0^\infty
                   \mathrm e^{-tW(a,b)}\,\mathrm da\,\mathrm db
       =\frac{\log t}{3t}+O(t^{-1}),
 \\
 \int_0^\infty\int_0^\infty(a+b)\mathrm e^{-tW(a,b)}
                   \,\mathrm da\,\mathrm db=O(t^{-4/3}).
 \end{gathered}
\]
\end{lemma}
\begin{proof}
Put $A=t^{-2/3}$ and $B=t^{-1/3}$; the logarithm comes from the interval $A\leq a\leq B$. For $a>0$, the substitution $u=tab$ gives
\[
 H_t(a):=\int_0^\infty\mathrm e^{-t(ab+b^3)}\,\mathrm db
       =\frac1{ta}\int_0^\infty
                   \mathrm e^{-u-u^3/(t^2a^3)}\,\mathrm du.
\]
Since $0\leq1-\mathrm e^{-v}\leq v$ for $v\geq0$,
\[
 0\leq\frac1{ta}-H_t(a)\leq\frac6{t^3a^4}.
\]
Over $A\leq a\leq B$, the integral of this error is $O(t^{-1})$. Replacing $\mathrm e^{-ta^3}$ by one in $\int_A^B\mathrm e^{-ta^3}/(ta)\,\mathrm da$ costs at most $\int_A^B a^2\,\mathrm da=O(t^{-1})$ as well. Thus the middle interval contributes
\[
 \int_A^B\mathrm e^{-ta^3}H_t(a)\,\mathrm da
       =\frac1t\log\frac BA+O(t^{-1})
       =\frac{\log t}{3t}+O(t^{-1}).
\]
For $a<A$, the bound $H_t(a)\leq\int_0^\infty\mathrm e^{-tb^3}\,\mathrm db=O(t^{-1/3})$ gives a contribution $O(t^{-1})$. For $a>B$, use $H_t(a)\leq1/(ta)$ and substitute $v=ta^3$ to obtain the bound
\[
 \int_B^\infty\frac{\mathrm e^{-ta^3}}{ta}\,\mathrm da
       =\frac1{3t}\int_1^\infty\frac{\mathrm e^{-v}}v\,\mathrm dv
       =O(t^{-1}).
\]
Finally, $aH_t(a)\leq1/t$ gives
\[
 \int_0^\infty\int_0^\infty a\mathrm e^{-tW(a,b)}
                         \,\mathrm da\,\mathrm db
       \leq\frac1t\int_0^\infty\mathrm e^{-ta^3}\,\mathrm da
       =O(t^{-4/3}).
\]
Symmetry proves the same bound for $b$.
\end{proof}

On $L=\mathcal O_{\mathbb P^2}(1)$, define a Hermitian metric by
\[
 |S|_h^2([Z])=
 \frac{|S(Z)|^2}
 {\bigl(|Z_0|^6+3|Z_0Z_1Z_2|^2+3|Z_1|^6+3|Z_2|^6\bigr)^{1/3}}
 \qquad(S\text{ a linear form}).
\]
The denominator is positive away from zero and homogeneous of degree two in $|Z|$, so the metric is well defined and smooth. Its local weight is one third of the logarithm of a sum of squared absolute values of holomorphic cubic functions, so its curvature is semipositive. In the chart $Z_0\ne0$, put $z_j=Z_j/Z_0$ and $p=[1:0:0]$. The frame $e=Z_0$ has
\begin{equation}\label{eq:logarithmic-projective-weight}
 |e|_h^2=\mathrm e^{-\phi},\qquad
 \phi(z)=\tfrac13\log\bigl(1+3W(|z_1|^2,|z_2|^2)\bigr).
\end{equation}
In particular the curvature vanishes at $p$.

\begin{proposition}\label{prop:projective-logarithmic-kernel}\index{Bergman kernel!logarithmic scale}\index{Bergman kernel!mixed monomials}
Let $\nu$ be a smooth positive volume form on $\mathbb P^2$, invariant under the rotations of $z_1$ and of $z_2$, and write
\[
 \nu=\rho(z)\,\mathrm dA(z_1)\,\mathrm dA(z_2),\qquad \rho_0=\rho(0)>0
\]
in the chart above. Give $H^0(\mathbb P^2,L^m)$ the norm $\int_{\mathbb P^2}|s|_{h^m}^2\,\mathrm d\nu$, and let $B_m$ be its kernel density. Then
\[
 B_m(p)=\frac{3}{\pi^2\rho_0}\frac m{\log m}
                   \left(1+O((\log m)^{-1})\right).
\]
For the Fubini--Study probability volume, the leading constant is $3/2$.
\end{proposition}
\Needspace{4\baselineskip}
\begin{proof}
The homogeneous monomials form an orthogonal basis, because the metric and the volume are invariant under the two independent rotations and distinct monomials transform by distinct characters. At $p$, only $Z_0^m$ has a nonzero value, equal to one in the frame $e^{\otimes m}$. Consequently $B_m(p)=I_m^{-1}$, where
\[
 I_m=\int_{\mathbb P^2}|Z_0^m|_{h^m}^2\,\mathrm d\nu
       =\int_{\mathbb C^2}(1+3W(|z_1|^2,|z_2|^2))^{-m/3}
                       \rho(z)\,\mathrm dA(z_1)\,\mathrm dA(z_2).
\]
Here the omitted projective line $\{Z_0=0\}$ has zero volume. By rotational invariance, the first derivatives of $\rho$ vanish at zero, so $\rho(z)=\rho_0+O(|z_1|^2+|z_2|^2)$ near zero. In the variables $a=|z_1|^2$ and $b=|z_2|^2$, polar integration has Jacobian $\pi^2\,\mathrm da\,\mathrm db$.

Choose a fixed small $\delta>0$. On $W\leq\delta$, set $\ell(W)=\tfrac13\log(1+3W)$. For constants $c,C>0$ independent of $m$,
\[
 cW\leq\ell(W)\leq W,\qquad
 0\leq\mathrm e^{-m\ell(W)}-\mathrm e^{-mW}
       \leq CmW^2\mathrm e^{-cmW}
       \leq\frac C m\mathrm e^{-cmW/2}.
\]
The first estimate of \cref{lem:mixed-laplace-logarithm} bounds the integral of this error in the weight by $O(m^{-2}\log m)$, and the second bounds the error from replacing $\rho$ by $\rho_0$ by $O(m^{-4/3})$. On the complement $W>\delta$, the contribution to $I_m$ is at most $\nu(\mathbb P^2)(1+3\delta)^{-m/3}$, and the corresponding tail of the exponential model is at most $\mathrm e^{-m\delta/2}J(m/2)$. Therefore
\[
 I_m=\pi^2\rho_0J(m)+O(m^{-4/3})+O(m^{-2}\log m)
       =\frac{\pi^2\rho_0}{3m}\log m+O(m^{-1}).
\]
Inverting proves the first assertion. The Fubini--Study probability volume has the local expression
\[
 \nu=\frac2{\pi^2}(1+|z_1|^2+|z_2|^2)^{-3}
                     \,\mathrm dA(z_1)\,\mathrm dA(z_2).
\]
It has mass one, since $2\int_0^\infty\int_0^\infty(1+a+b)^{-3}\,\mathrm da\,\mathrm db=1$, and its density at zero is $2/\pi^2$. The leading constant is therefore $3/2$.
\end{proof}

Thus the full kernel, without any selection, already has a logarithmic factor at this curvature zero, with a coefficient depending on the volume density at $p$. A transition for growing vanishing ideals would require control of the norms of the selected monomials on the moving scale, as in the curve case.
