\section{General measures with a model reference}
\label{sec:big-model-general-measures}

We now prove the existence statements of \cref{thm:li-general-measure}. Suppose first that $\mu$ is dominated by the Monge--Amp\`ere measure of a bounded potential. The supersolution corollary gives $Q\in\mathcal E_\varphi$ with a nonzero weighted measure $\mathrm{e}^{-Q}\mu_Q\leq\mu$. Among the potentials $w\in\mathcal E_\varphi$ whose weighted measure $\mathrm{e}^{-w}\mu_w$ lies between this measure and $\mu$, we maximize the weighted mass. If a maximizer left part of $\mu$ uncovered, weighted addition of a supersolution for the remainder would increase the weighted mass; hence the maximizer solves the equation. Truncating densities then treats every finite non-pluripolar measure.

\begin{theorem}\label{thm:big-model-exponential}
Let $\mu$ be a nonzero finite non-pluripolar measure and let $\lambda>0$. There is a unique $u\in\mathcal E_\varphi$ such that
\[
 \mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}\mu.
\]
\end{theorem}
\begin{proof}
We first take $\lambda=1$ and suppose that
\[
 \mu\leq A\sigma,\qquad A>0,\qquad
 \sigma=\mathrm{MA}_\omega(\psi),\quad
 \psi\in\PSH(X,\omega)\cap L^\infty(X).
\]
\Cref{cor:big-relative-supersolution} gives $Q\in\mathcal E_\varphi$ such that
\[
 \nu_0=\mathrm{e}^{-Q}\mu_Q\leq\mu
\]
is a finite nonzero measure. Consider
\[
 \mathcal T=\{w\in\mathcal E_\varphi:
              \nu_0\leq \mathrm{e}^{-w}\mu_w\leq\mu\}.
\]
The family contains $Q$. For $w\in\mathcal T$, exponential order (\cref{cor:big-relative-orders}) gives $w\leq Q$. Also
\[
 \ell\leq\mu_w(X)\leq \mathrm{e}^{\sup_Xw}\mu(X),
 \qquad
 \sup_Xw\geq\log\frac{\ell}{\mu(X)}.
\]
These bounds prevent the collapse of any sequence in $\mathcal T$.

\Needspace{6\baselineskip}
Let
\[
 \alpha=\sup_{w\in\mathcal T}\int_X\mathrm{e}^{-w}\,\mu_w\leq\mu(X),
\]
and choose $w_j\in\mathcal T$ whose weighted masses tend to $\alpha$. Extract an $L^1$ limit $w\not\equiv-\infty$. Since
\[
 \mu_{w_j}\leq \mathrm{e}^{\sup_XQ}\mu,
\]
the fixed-dominator theorem \cref{thm:big-relative-fixed-continuity} gives $w\in\mathcal E_\varphi$, convergence in capacity, and $\mu_{w_j}\rightharpoonup\mu_w$. Put $\nu_j=\mathrm{e}^{-w_j}\mu_{w_j}$. After a further extraction, these finite measures converge weakly to a measure $\nu$, and
\[
 \nu_0\leq\nu\leq\mu,\qquad \nu(X)=\alpha.
\]

We claim that $\mu_w=\mathrm{e}^w\nu$. By \cref{lem:fixed-npp-integrals},
\[
 \int_X|\mathrm{e}^{w_j}-\mathrm{e}^w|\,\mathrm{d}\mu\longrightarrow0.
\]
Let $f$ be continuous. Since $\nu_j\leq\mu$, replacing $\mathrm{e}^{w_j}$ by $\mathrm{e}^w$ in $\int_Xf \mathrm{e}^{w_j}\,\mathrm{d}\nu_j$ changes the integral by an amount tending to zero. For the remaining integral, approximate the fixed bounded measurable function $f \mathrm{e}^w$ in $L^1(\mu)$ by continuous functions. Since $\nu_j,\nu\leq\mu$, the approximation error is uniform in $j$, while the integrals of each continuous approximation converge by weak convergence. Thus
\[
 \int_Xf\,\mu_{w_j}
 =\int_Xf \mathrm{e}^{w_j}\,\mathrm{d}\nu_j
 \longrightarrow\int_Xf \mathrm{e}^w\,\mathrm{d}\nu.
\]
This proves the claim. The measure $\nu$ does not charge the poles of $w$, so $\mathrm{e}^{-w}\mu_w=\nu$. Hence $w\in\mathcal T$ attains the maximum $\alpha$.

Suppose that $\mu-\nu$ were nonzero. Then \cref{cor:big-relative-supersolution}, applied to $\mu-\nu$, would give $v\in\mathcal E_\varphi$ with
\[
 0\ne\nu_v=\mathrm{e}^{-v}\mu_v\leq\mu-\nu.
\]
In particular $\mu_v\leq \mathrm{e}^{\sup_Xv}\mu\leq A \mathrm{e}^{\sup_Xv}\sigma$, so the bounded-carrier hypothesis of \cref{thm:big-weighted-addition} holds. That theorem, with $\lambda=1$, gives $a\in\mathcal E_\varphi$ with
\[
 \mathrm{e}^{-a}\mu_a=\mathrm{e}^{-w}\mu_w+\mathrm{e}^{-v}\mu_v=\nu+\nu_v.
\]
The right-hand side lies between $\nu_0$ and $\mu$, so $a\in\mathcal T$, but its mass exceeds $\alpha$. This contradicts maximality. Therefore $\nu=\mu$, and $w$ solves the equation.

\Needspace{8\baselineskip}
Now let $\mu$ be any nonzero finite non-pluripolar measure. The bounded-carrier decomposition \cref{lem:bounded-measure-carrier}, applied to $\omega$, writes
\[
 \mu=f\sigma,\qquad
 \sigma=\mathrm{MA}_\omega(\psi),\quad
 \psi\in\PSH(X,\omega)\cap L^\infty(X),\quad
 0\leq f\in L^1(\sigma).
\]
\Needspace{8\baselineskip}
Set $\mu_k=\min(f,k)\sigma$. These measures are nonzero, and the first part gives $u_k\in\mathcal E_\varphi$ with $\mu_{u_k}=\mathrm{e}^{u_k}\mu_k$. As in the last part of the proof of \cref{thm:big-minimal-general-measures}, now with $\mathcal E_\varphi$ and $\ell$ in place of $\mathcal E_V$ and $\ell_V$, the $u_k$ decrease to some $u\in\mathcal E_\varphi$ with $\mu_u=\mathrm{e}^u\mu$.

For general $\lambda>0$, apply the result for exponent one in the background $\lambda\theta$, with model $\lambda\varphi$ and datum $\lambda^n\mu$, and divide the resulting potential by $\lambda$. The scaling of the relative class and of the Monge--Amp\`ere measure gives the equation. Uniqueness follows by applying \cref{cor:big-relative-orders} to two solutions in both directions.
\end{proof}

\Needspace{15\baselineskip}
\begin{theorem}\label{thm:big-model-normalized}
For every nonzero finite non-pluripolar measure $\mu$, there are $u\in\mathcal E_\varphi$ and $c>0$ such that
\[
 \sup_Xu=0,\qquad \mathrm{MA}_\theta(u)=c\mu.
\]
The scalar $c$ is unique and satisfies
\[
 \frac{\ell}{\mu(X)}\leq c\leq\frac{M}{\mu(X)}.
\]
\end{theorem}
\begin{proof}
For each integer $j\geq1$, the preceding theorem gives $u_j\in\mathcal E_\varphi$ with $\mu_{u_j}=\mathrm{e}^{u_j/j}\mu$. The zero-exponent part of the proof of \cref{thm:general-measure-minimal} applies to these $u_j$, with the comparison potential $s=u_1$, with \cref{cor:big-relative-orders,thm:big-relative-fixed-continuity} in place of \cref{cor:relative-extended-exponential-order,thm:relative-fixed-measure-continuity}, and with the lower bound $\ell$ of \eqref{eq:big-relative-mass-interval} in place of $v_-(\theta)$; \cref{lem:normalized-exponential-npp} holds for every smooth background. It gives $u\in\mathcal E_\varphi$ and $c>0$ with $\sup_Xu=0$ and $\mu_u=c\mu$. Integrating this equation and using \eqref{eq:big-relative-mass-interval} gives the stated bounds for $c$, and the scalar comparison in \cref{cor:big-relative-orders} gives the uniqueness of $c$.
\end{proof}

For $L^p$ densities, \cref{sec:big-model-lp} proves the stronger conclusion $u\sim\varphi$.

\subsection{Stability under changes of reference}
\label{subsec:big-equation-reference-stability}

By \cref{thm:big-varying-model-closure}, convergence of the models for $d_\theta$ keeps limits in the relative class of the limiting model. For equations, convergence of the densities provides, after extraction, the fixed dominating measure required there. For a positive exponent, uniqueness then identifies the limiting potential; at exponent zero, uniqueness of the scalar identifies the limiting measure.

\begin{theorem}\label{thm:big-varying-equation-stability}\index{Stability!varying model equations}
Let $\varphi_j,\varphi\in\mathcal M_\delta(\theta)$ for some $\delta>0$, with $d_\theta(\varphi_j,\varphi)\to0$. Let $\nu$ be a finite non-pluripolar measure and let $f_j,f$ be nonnegative nonzero functions in $L^1(\nu)$ with $f_j\to f$ in $L^1(\nu)$. Put $\mu_j=f_j\nu$ and $\mu=f\nu$.
\begin{enumerate}
\item For a fixed $\lambda>0$, the solutions
\[
 u_j\in\mathcal E_{\varphi_j},\qquad
 \mathrm{MA}_\theta(u_j)=\mathrm{e}^{\lambda u_j}\mu_j
\]
converge in $L^1(X)$ and in capacity to the solution $u\in\mathcal E_\varphi$ with datum $\mu$. Their Monge--Amp\`ere measures converge to $\mathrm{MA}_\theta(u)$ in total variation.
\item Let $u_j\in\mathcal E_{\varphi_j}$ satisfy $\sup_Xu_j=0$ and $\mathrm{MA}_\theta(u_j)=c_j\mu_j$. Then $c_j\to c$, where $c$ is the scalar for the reference $\varphi$ and the datum $\mu$, and
\[
 \mathrm{MA}_\theta(u_j)\longrightarrow c\mu
 \quad\text{in total variation}.
\]
The potentials form a relatively compact family in $L^1$. Every $L^1$ cluster limit $u$ lies in $\mathcal E_\varphi$, has supremum zero and solves $\mathrm{MA}_\theta(u)=c\mu$. Each subsequence converging in $L^1$ converges to its limit in capacity.
\end{enumerate}
\end{theorem}
\begin{proof}
Write $M_0=\overline{\operatorname{vol}}_\theta(V_\theta)$. Every solution above has mass between $\delta$ and $M_0$. In the first case let $s_j=\sup_Xu_j$. Then
\[
 s_j\geq\lambda^{-1}\log\frac{\delta}{\mu_j(X)},
\]
which is a uniform lower bound since $\mu_j(X)\to\mu(X)>0$. To bound $s_j$ from above, suppose that $s_j\to+\infty$ along a subsequence. By normalized compactness, after a further extraction, $v_j=u_j-s_j\to v$ in $L^1$ with $\sup_Xv=0$. The fixed-measure lemma \cref{lem:fixed-npp-integrals} and $f_j\to f$ in $L^1(\nu)$ imply
\[
 \int_X\mathrm{e}^{\lambda v_j}f_j\,\mathrm{d}\nu
 \longrightarrow\int_X\mathrm{e}^{\lambda v}f\,\mathrm{d}\nu>0.
\]
The limit is positive because $f\nu$ is nonzero and does not charge the poles of $v$. But the left-hand side equals $\mathrm{e}^{-\lambda s_j}\mu_{u_j}(X)\leq M_0\mathrm{e}^{-\lambda s_j}$, which tends to zero, a contradiction. Thus $s_j$ is bounded, and the $u_j$ form a relatively compact family in $L^1$ whose limits are not identically $-\infty$.

Start with any subsequence and extract $u_j\to u$ in $L^1$. Extract again so that $\sum_j\|f_j-f\|_{L^1(\nu)}<\infty$. The function
\[
 g=f+\sum_j|f_j-f|\in L^1(\nu)
\]
dominates every $f_j$ of this subsequence. If $C$ is a common upper bound for the $u_j$, then $\mu_{u_j}\leq\mathrm{e}^{\lambda C}g\nu$. By \cref{thm:big-varying-model-closure}, $u\in\mathcal E_\varphi$, the potentials converge in capacity, and the measures converge weakly to $\mu_u$. On the other hand,
\[
 \mathrm{e}^{\lambda u_j}f_j\longrightarrow
 \mathrm{e}^{\lambda u}f\quad\text{in }L^1(\nu).
\]
Indeed, the error coming from $f_j-f$ is at most $\mathrm{e}^{\lambda C}\|f_j-f\|_{L^1(\nu)}$, and \cref{lem:fixed-npp-integrals} for the fixed measure $f\nu$ controls the remaining term. Hence $\mu_u=\mathrm{e}^{\lambda u}\mu$, with convergence in total variation. By the uniqueness in \cref{thm:big-model-exponential}, every cluster limit is this solution, and the subsequence criterion gives all three convergences for the original sequence.

In the second case, integrating the equation gives
\[
 \frac{\delta}{\mu_j(X)}\leq c_j\leq\frac{M_0}{\mu_j(X)}.
\]
Since $\mu_j(X)\to\mu(X)>0$, these scalars stay in a compact subinterval of $(0,+\infty)$. After extraction, normalized compactness gives $u_j\to u$ in $L^1$ with $\sup_Xu=0$, and $c_j\to c'>0$. Extract a summable density error as above. Then $\mu_{u_j}\leq Cg\nu$ for a fixed $C$, so \cref{thm:big-varying-model-closure} gives $u\in\mathcal E_\varphi$, convergence in capacity and weak convergence to $\mu_u$. Since $c_jf_j\to c'f$ in $L^1(\nu)$, the limiting equation is $\mu_u=c'\mu$, and the uniqueness of the scalar in \cref{thm:big-model-normalized} gives $c'=c$. Hence the whole sequence of scalars converges to $c$, and $c_j\mu_j\to c\mu$ in total variation. Applying the same argument within any $L^1$-convergent subsequence proves its convergence in capacity and the equation for its limit.
\end{proof}
