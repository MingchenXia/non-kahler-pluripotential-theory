The ordinary envelope $P_\theta[u]$ is in general less singular than $u$. When the pole set of $u$ is a compact set of Hausdorff dimension zero, however, the full local Monge--Amp\`ere product of \cref{thm:small-pole-local-products} is defined, and the ordinary envelope preserves its restriction to the pole set. Bounded mass controls the local approximations in every degree, and the products of lower degree put no mass on the pole set.

For potentials that are locally bounded off a compact set of Hausdorff dimension zero, the bounded truncations have full weak limits in every degree. This does not use the bounded mass property.

\Needspace{16\baselineskip}
\begin{lemma}\label{lem:small-pole-full-measures}\index{Monge--Ampere measure@Monge--Amp\`ere measure!full measure at small poles}
Let $\omega$ be a Hermitian form on $X$, let $S\subset X$ be a compact set of Hausdorff dimension zero, and let $v\in\PSH(X,\omega)$ be locally bounded on $X\setminus S$. Put $v_k=\max(v,-k)$. As $k\to\infty$, the currents $(\ddc v_k)^j$, $1\leq j\leq n$, converge weakly to currents $W_j(v)$ of order zero, and the measures $(\omega+\ddc v_k)^n$ converge weakly to a finite positive measure $\Phi_v$, which we call the \emph{full Monge--Amp\`ere measure} of $v$. One has $\int_XW_n(v)=0$,
\begin{equation}\label{eq:small-pole-full-total}
 J(v):=\Phi_v(X)
 =\int_X\omega^n+
       \sum_{j=1}^{n-1}\binom nj
              \int_XW_j(v)\wedge\omega^{n-j},
\end{equation}
and $v_-(\omega)\leq J(v)\leq v_+(\omega)$. The measure $\Phi_v$ equals $\mathrm{MA}_\omega(v)$ on $X\setminus S$. On each coordinate chart, $\mathbf1_S\Phi_v=\mathbf1_S(\ddc(v+q))^n_{\mathrm{full}}$ for every smooth function $q$ making $v+q$ plurisubharmonic there; in particular, this restriction does not depend on $q$.
\end{lemma}
\begin{proof}
On each relatively compact chart choose a smooth $q_1$ with $\ddc q_1\geq\omega$, so that the functions $v_k+q_1$ are plurisubharmonic and decrease to $v+q_1$. \Cref{thm:small-pole-local-products} then gives the full weak limits in every degree. Indeed, $\ddc v_k=\ddc(v_k+q_1)-\ddc q_1$, and the powers can be expanded in the closed positive powers of $\ddc(v_k+q_1)$ and smooth forms. The local limits patch together because the approximating bounded currents are global. The limits are currents of order zero with finite mass on the compact manifold $X$, and $\Phi_v$ is positive. Stokes' theorem for the bounded potentials $v_k$ gives $\int_XW_n(v)=0$, and the binomial expansion of $(\omega+\ddc v_k)^n$ gives \eqref{eq:small-pole-full-total}. Every truncation $v_k$ has mass between $v_-(\omega)$ and $v_+(\omega)$, which bounds $J(v)$. On every compact subset of $X\setminus S$, the truncations eventually agree with $v$, so $\Phi_v$ equals $\mathrm{MA}_\omega(v)$ there.

Every closed positive power of degree less than $n$ in this expansion has coefficient measures vanishing on $S$, by \cref{cor:closed-current-hausdorff-null}. Thus only the top closed power contributes to $\mathbf1_S\Phi_v$, which identifies $\mathbf1_S\Phi_v$ with $\mathbf1_S(\ddc(v+q_1))^n_{\mathrm{full}}$. If $v+q$ is plurisubharmonic for another smooth $q$, choose $C>0$ with $\ddc(q+C|z|^2)\geq\omega$ on a smaller chart. The same argument applies to $q+C|z|^2$, and by multilinearity the full top power of $\ddc(v+q+C|z|^2)$ differs from that of $\ddc(v+q)$ by full powers of lower degree wedged with smooth forms, which vanish on $S$. These assertions are local on each chart and do not require a single chart containing $S$.
\end{proof}

\Needspace{14\baselineskip}
\begin{theorem}\label{thm:small-pole-envelope-residual}\index{Residual mass!ordinary envelope}
Let $\omega$ be a Hermitian form on $X$, and let $S\subset X$ be a compact set of Hausdorff dimension zero. Suppose $u\in\PSH(X,\omega)$ is locally bounded on $X\setminus S$ and has pole set $\{u=-\infty\}=S$. Put $p=P_\omega[u]$ and $p_m=P_\omega(u+m,0)$ for real $m>0$. Then $p$ and $p_m$ are locally bounded on $X\setminus S$, their full Monge--Amp\`ere measures and that of $u$ (\cref{lem:small-pole-full-measures}) satisfy
\[
 \mathbf1_S\Phi_p=\mathbf1_S\Phi_u,
\]
and $\Phi_{p_m}\to\Phi_p$ weakly as $m\to\infty$.
\end{theorem}
\begin{proof}
\emph{Step 1: the rooftops.} Adding a constant to $u$ changes neither $p$ nor the family $(p_m)_{m>0}$, apart from a shift of the parameter $m$. We may therefore assume $u\leq0$. Then
\[
 u\leq p_m\leq\min(u+m,0),\qquad p_m\uparrow p,
\]
where the increasing limit is understood after upper semicontinuous regularization. Thus $p_m\sim u$, and $p_m$ and $p$ are locally bounded off $S$. By \cref{lem:monotone-psh-capacity}, $p_m\to p$ in capacity. On compact subsets of $X\setminus S$ the potentials have a common bounded range, so all bounded local products converge there, by the bounded product theory of \cref{sec:psh-local-capacity}.

\emph{Step 2: full measures.} By Step~1 and \cref{lem:small-pole-full-measures}, the full Monge--Amp\`ere measures $\Phi_v$, the currents $W_j(v)$ and the total masses $J(v)$ of \eqref{eq:small-pole-full-total} are defined for $v=u$, $v=p$ and $v=p_m$.

\emph{Step 3: convergence of the lower-degree powers.} We need local trace bounds for the closed powers of $p_m$ that are uniform in $m$. Fix a relatively compact coordinate neighborhood and write $\kappa=\ddc|z|^2$. Choose $B>0$ with $B\kappa\geq\omega$ there, and $A>1$ with $A\omega-B\kappa\geq\kappa$. For $p_{m,k}=\max(p_m,-k)$, the closed current
$T_{m,k}=B\kappa+\ddc p_{m,k}$ is positive and
\[
 Q_{m,k}:=A\omega+\ddc p_{m,k}\geq T_{m,k}+\kappa.
\]
Expanding $Q_{m,k}^n$ with the smooth positive difference $Q_{m,k}-T_{m,k}\geq\kappa$ gives, for $1\leq j\leq n$,
\[
 \binom nj T_{m,k}^j\wedge\kappa^{n-j}
       \leq Q_{m,k}^n,
 \qquad
 \int_XQ_{m,k}^n\leq A^n v_+(\omega).
\]
The last bound holds because $Q_{m,k}=A(\omega+\ddc(p_{m,k}/A))$ and $p_{m,k}/A$ is a bounded $\omega$-plurisubharmonic function. Letting $k\to\infty$ gives trace bounds, independent of $m$, for every full closed power of $\ddc(p_m+B|z|^2)$.

For $j<n$, every subsequential weak limit of these closed positive $(j,j)$-currents puts no mass on $S$, again by \cref{cor:closed-current-hausdorff-null}. Off $S$ the bounded capacity convergence identifies the limit with the corresponding full power of $\ddc(p+B|z|^2)$, which also puts no mass on $S$. Hence the limit is unique. Expanding back in the smooth correction and using a finite coordinate cover yields
\[
 W_j(p_m)\longrightarrow W_j(p),\qquad 1\leq j<n.
\]
Hence \eqref{eq:small-pole-full-total} gives $J(p_m)\to J(p)$.

\emph{Step 4: the residual measures.} Let $R=\mathbf1_S\Phi_u$. Since $p_m-u$ is bounded, the equality case of \cref{lem:small-pole-residual-invariance}, applied on each chart after a common smooth correction, gives $\mathbf1_S\Phi_{p_m}=R$ for every $m$. The total masses $J(p_m)$ are bounded by $v_+(\omega)$, so along any sequence $m_i\to\infty$ a subsequence of $\Phi_{p_{m_i}}$ converges weakly to a measure $\eta$. Since $\Phi_{p_m}-R$ is positive, so is $\eta-R$. Off $S$, $\eta$ equals $\mathrm{MA}_\omega(p)$, by the bounded convergence of Step~1, and its total mass is $\lim_mJ(p_m)=J(p)$ by Step~3. The measure $\Phi_p$ has the same total mass and the same restriction to $X\setminus S$, so
\[
 \Phi_p(S)=\eta(S)\geq R(S).
\]
On the other hand, $u\leq p$, so the inequality in \cref{lem:small-pole-residual-invariance} gives $\mathbf1_S\Phi_p\leq R$ on each chart. Together with the previous inequality, this forces $\Phi_p(S)=R(S)$, hence $\mathbf1_S\Phi_p=R$, which is the asserted equality of residual measures. It also gives $\mathbf1_S\eta=R$, since $\eta-R\geq0$ and $\eta(S)=R(S)$. As $\eta$ and $\Phi_p$ also agree off $S$, we conclude that $\eta=\Phi_p$. Every weak subsequential limit of $\Phi_{p_m}$ is therefore $\Phi_p$, which proves the convergence $\Phi_{p_m}\to\Phi_p$.
\end{proof}
