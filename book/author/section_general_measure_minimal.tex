\section{General measures in the minimal envelope class}
\label{sec:general-measure-minimal}

We first take the reference zero. It has minimal singularities because $\theta\geq0$, and it satisfies \eqref{eq:als-model-condition} for every $\varepsilon\in(0,1)$; we call $\mathcal E_0$ the minimal envelope class. By \cref{lem:bounded-measure-carrier}, a finite non-pluripolar measure has an integrable density with respect to the Monge--Amp\`ere measure of a bounded potential. Truncating that density gives bounded solutions whose measures are dominated by one fixed finite measure, and dominated continuity passes to the full density. Letting the exponent tend to zero then produces the normalized equation.

\Needspace{15\baselineskip}
\begin{theorem}\label{thm:general-measure-minimal}
Let $\mu$ be a finite non-pluripolar measure with $\mu(X)>0$.
For every $\lambda>0$ there is a unique $u\in\mathcal E_0$ satisfying
\[
 \mathrm{MA}_\theta^{\mathrm{ext}}(u)=\mathrm{e}^{\lambda u}\mu.
\]
There also exist $v\in\mathcal E_0$, normalized by $\sup_Xv=0$, and a unique positive scalar $c$ such that
\[
 \mathrm{MA}_\theta^{\mathrm{ext}}(v)=c\mu.
\]
\end{theorem}
\begin{proof}
For the reference zero, $P_{\epsilon\theta}(0)=0$ for every $\epsilon>0$. Letting $\epsilon\downarrow0$ in \cref{thm:relative-uniform-mass-lower} therefore gives
\begin{equation}\label{eq:minimal-lower-mass}
 \mathrm{MA}^{\mathrm{ext}}_\theta(w)(X)\geq v_-(\theta)
 \qquad(w\in\mathcal E_0).
\end{equation}

\emph{Positive exponent.} By \cref{lem:bounded-measure-carrier}, write $\mu=f\nu$ with $\nu=\mathrm{MA}_\theta(\psi)$, where $\psi\in\PSH(X,\theta)$ is bounded and $0\leq f\in L^1(\nu)$. For positive integers $j$, let $f_j=\min(f,j)$. These densities are bounded, and the measures $f_j\nu$ are nonzero. By \cref{thm:bounded-subsolution} there are bounded $u_j\in\PSH(X,\theta)$ with
\[
 \mathrm{MA}_\theta(u_j)=\mathrm{e}^{\lambda u_j}f_j\nu.
\]
These bounded potentials belong to $\mathcal E_0$ by \cref{prop:relative-elementary}. Since $f_j$ increases with $j$, exponential order (\cref{cor:relative-extended-exponential-order}) shows that $u_j$ decreases; in particular $u_j\leq u_1$. On the other hand, since $f_j\leq f$,
\[
 v_-(\theta)\leq\mathrm{MA}_\theta(u_j)(X)
       \leq \mathrm{e}^{\lambda\sup_Xu_j}\mu(X),
\]
so the suprema of the $u_j$ have a common lower bound. Hence the decreasing limit $u$ is a $\theta$-plurisubharmonic function, not identically $-\infty$, and $u_j\to u$ in $L^1(X)$. Moreover
\[
 \mathrm{MA}_\theta(u_j)\leq \mathrm{e}^{\lambda\sup_Xu_1}\mu.
\]
Applying \cref{thm:relative-fixed-measure-continuity} with reference zero, we obtain $u\in\mathcal E_0$ and weak convergence of the Monge--Amp\`ere measures to $\mathrm{MA}^{\mathrm{ext}}_\theta(u)$. The right-hand sides converge in total variation to $\mathrm{e}^{\lambda u}f\nu$, since the functions $\mathrm{e}^{\lambda u_j}f_j$ converge $\nu$-almost everywhere and are bounded by $\mathrm{e}^{\lambda\sup_Xu_1}f\in L^1(\nu)$. This proves the equation, and \cref{cor:relative-extended-exponential-order} gives uniqueness.

\Needspace{10\baselineskip}
\emph{Zero exponent.} For each positive integer $j$, let $u_j\in\mathcal E_0$ be the solution just constructed with exponent $1/j$. To bound the normalization, we compare with the solution $s\in\mathcal E_0$ for exponent one, so that $\mathrm{MA}^{\mathrm{ext}}_\theta(s)=\mathrm{e}^s\mu$. Set $H=\max(0,\sup_Xs)$ and $w_j=s+(j-1)H$. Then $w_j/j\geq s$, and
\[
 \mathrm{e}^{-w_j/j}\mathrm{MA}^{\mathrm{ext}}_\theta(w_j)
 \leq \mathrm{e}^{-s}\mathrm{MA}^{\mathrm{ext}}_\theta(s)
 =\mu
 =\mathrm{e}^{-u_j/j}\mathrm{MA}^{\mathrm{ext}}_\theta(u_j).
\]
Exponential order, now with exponent $1/j$, gives $u_j\leq w_j$. Put
\[
 M_j=\sup_Xu_j,\qquad v_j=u_j-M_j,
 \qquad c_j=\mathrm{e}^{M_j/j}.
\]
\Needspace{7\baselineskip}
Then $\sup_Xv_j=0$ and $0<c_j\leq \mathrm{e}^H$, because $M_j\leq\sup_Xw_j\leq jH$. By \eqref{eq:minimal-lower-mass},
\[
 v_-(\theta)\leq c_j\int_X \mathrm{e}^{v_j/j}\,\mathrm{d}\mu
                 \leq c_j\mu(X),
\]
so $c_j\geq v_-(\theta)/\mu(X)>0$. After extraction, \cref{thm:normalized-compactness} gives $v_j\to v$ in $L^1(X)$, where $v\not\equiv-\infty$ and $\sup_Xv=0$, and $c_j\to c>0$.

By \cref{lem:normalized-exponential-npp}, $a_j=\mathrm{e}^{v_j/j}\to1$ in $L^1(\mu)$. Consequently
\[
 \mathrm{MA}^{\mathrm{ext}}_\theta(v_j)=c_ja_j\mu
       \longrightarrow c\mu
 \quad\hbox{in total variation},\qquad
 \mathrm{MA}^{\mathrm{ext}}_\theta(v_j)\leq \mathrm{e}^H\mu.
\]
Applying \cref{thm:relative-fixed-measure-continuity} again, we obtain $v\in\mathcal E_0$ and identify the limit as $\mathrm{MA}^{\mathrm{ext}}_\theta(v)$; thus $\mathrm{MA}^{\mathrm{ext}}_\theta(v)=c\mu$. The scalar assertion of \cref{cor:relative-extended-exponential-order}, applied in both directions to two solutions, proves that $c$ is unique.
\end{proof}

The solutions for the reference zero also bound from above the approximating solutions for the reference $\varphi$. Indeed, every $\theta$-plurisubharmonic potential is $\preceq0$, so exponential order with reference zero compares them with arbitrary $\theta$-plurisubharmonic potentials. This comparison controls the normalization in \cref{sec:general-measure-relative}.
