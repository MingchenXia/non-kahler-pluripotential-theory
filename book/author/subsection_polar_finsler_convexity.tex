\subsection{Polar norms of semipositive Finsler metrics}
\label{subsec:polar-finsler-convexity}
\index{Kobayashi curvature!polar norm}
\index{Finsler metric!semipositive curvature}

For a holomorphic vector bundle that is not assumed to carry a positively curved Hermitian metric, positivity can be measured by Finsler metrics and their Kobayashi curvature. Polar duality converts a Finsler metric on $E$ with semipositive Kobayashi curvature into a continuous norm on $E^*$ whose square is plurisubharmonic and uniformly convex in the fibers. With the regularization of \cref{subsec:homogeneous-convex-regularization}, \cref{thm:convex-norm-deformation-obstruction} therefore excludes such Finsler metrics on an ample bundle whose dual is close to a split bundle with positive Gauduchon degrees.

Let $E\to X$ be a holomorphic vector bundle of positive rank; in this subsection $n\geq1$. Write $E^\times$ for the complement of its zero section. A \emph{squared Finsler metric} on $E$ is a positive smooth function $G$ on $E^\times$ with
\[
 G(x,tv)=|t|^2G(x,v),\qquad t\in\mathbb C\setminus\{0\},
\]
extended continuously by zero to the zero section. It is \emph{strongly pseudoconvex} if its vertical complex Hessian $(G_{i\bar j})$ is positive definite; this does not require convexity in the real fiber variables.

In local holomorphic base and fiber coordinates $(z,v)$, with Greek indices for base coordinates, Latin indices for fiber coordinates and summation over repeated indices, put
\[
 \begin{gathered}
 S_{\alpha\bar\beta}
 =G_{\alpha\bar\beta}
  -G_{\alpha\bar j}G^{\bar j i}G_{i\bar\beta},\\
 \mathcal K_G=-\frac{\mathrm i}{2\pi G}
 S_{\alpha\bar\beta}\,\mathrm dz^\alpha\wedge\mathrm d\bar z^\beta,
 \end{gathered}
\]
where $(G^{\bar j i})$ is the inverse of the vertical Hessian. We use this convention for the \emph{Kobayashi curvature} $\mathcal K_G$, a horizontal $(1,1)$-form at each point of $E^\times$. Thus $\mathcal K_G\geq0$ means $S\leq0$, and $\mathcal K_G>0$ means that $S$ is negative definite.

\Needspace{6\baselineskip}
\begin{lemma}\label{lem:polar-finsler-osculating-section}
\index{Holomorphic section!osculating quadratic jet}
Let $G$ be a strongly pseudoconvex squared Finsler metric. If $\mathcal K_G>0$, then for every $v_0\in E_{x_0}^\times$ there is a local holomorphic section $\sigma$ with $\sigma(x_0)=v_0$ and
\[
 G(z,\sigma(z))\leq G(x_0,v_0)
\]
near $x_0$. Moreover, without any curvature assumption, on a coordinate ball, for $\varphi=|z|^2$ and $\varepsilon>0$, the metric $G_\varepsilon=\mathrm e^{-\varepsilon\varphi}G$ is strongly pseudoconvex and satisfies
\[
 \mathcal K_{G_\varepsilon}=\mathcal K_G+\varepsilon\ddc\varphi.
\]
\end{lemma}
\begin{proof}
Since the vertical Hessian is positive, the Levi-orthogonal complement $H_{G,v}$ of the vertical tangent space at $v$ projects isomorphically onto the base tangent space. The Levi form of $G$ on $H_{G,v}$ is the Schur complement $S$, which is negative definite when $\mathcal K_G>0$. For the vertical radial vector $R=v_i\partial_{v_i}$, complex homogeneity gives
\[
 \partial G(R)=G,\qquad
 \mathcal L_G(a,\bar R)=\partial G(a)
\]
for every tangent vector $a$, where $\mathcal L_G$ is the Levi form. Since $R$ is vertical, the second identity implies $\partial G|_{H_{G,v}}=0$.

Choose a holomorphic section $\sigma_0$ through $v_0$ whose tangent space at $x_0$ is $H_{G,v_0}$. In coordinates centered at $x_0$, the function $G\circ\sigma_0$ has zero first derivative and Taylor expansion
\[
 G(x_0,v_0)+P(z)+\overline{P(z)}
 +S_{\alpha\bar\beta}z^\alpha\bar z^\beta+O(|z|^3),
\]
where $P$ is a homogeneous holomorphic quadratic polynomial. The identity $\partial_vG(v_0)(v_0)=G(v_0)>0$ gives a vertical vector $b$ with $\partial_vG(v_0)(b)=1$. Set $\sigma=\sigma_0-bP$ in the local frame. This changes only the pure quadratic terms and cancels $P+\bar P$. The remaining quadratic form is negative definite, and therefore dominates the cubic remainder on a sufficiently small ball. The section stays nonzero there and satisfies the asserted inequality.

For the last assertion, differentiate $\mathrm e^{-\varepsilon\varphi}G$ in the Schur complement. Euler's identities give $G_{\bar j}=v_iG_{i\bar j}$ and $G_\alpha=G_{\alpha\bar j}\bar v_j$. Hence the terms containing $\partial\varphi$, including its square, cancel between the base Hessian and the mixed block product, leaving
\[
 S(G_\varepsilon)=\mathrm e^{-\varepsilon\varphi}
       \bigl(S(G)-\varepsilon G(\varphi_{\alpha\bar\beta})\bigr).
\]
Dividing by $G_\varepsilon$ gives the curvature formula. The vertical Hessian of $G_\varepsilon$ is the positive multiple $\mathrm e^{-\varepsilon\varphi}(G_{i\bar j})$, so $G_\varepsilon$ is again strongly pseudoconvex.
\end{proof}

\Needspace{12\baselineskip}
\begin{proposition}\label{prop:semipositive-finsler-polar-psh}
\index{Polar norm!plurisubharmonic square}
Let $G$ be a strongly pseudoconvex squared Finsler metric with $\mathcal K_G\geq0$. Its \emph{polar square}
\[
 Q(x,\xi)=\max_{[v]\in\mathbb P(E_x)}
                \frac{|\xi(v)|^2}{G(x,v)},\qquad \xi\in E_x^*,
\]
is continuous and plurisubharmonic on the total space of $E^*$, positive off the zero section and complex homogeneous of degree two. The function $\log Q$ is plurisubharmonic off the zero section, and $\sqrt Q$ is a complex norm on each fiber.
\end{proposition}
\begin{proof}
The ratio is invariant under complex rescaling of $v$, so the maximum can be taken over the compact unit sphere in a local frame, which proves continuity. Positive upper and lower bounds for $G$ on that sphere compare $Q$ with a Hermitian squared dual norm; this gives positivity off zero and continuity at zero, and complex homogeneity is clear. Moreover,
\[
 \sqrt{Q(x,\xi)}=\sup_{v\ne0}\frac{|\xi(v)|}{\sqrt{G(x,v)}}.
\]
Each term satisfies the triangle inequality in $\xi$, and so does their supremum.

On a coordinate ball, set $G_\varepsilon=\mathrm e^{-\varepsilon|z|^2}G$. By \cref{lem:polar-finsler-osculating-section}, its Kobayashi curvature is strictly positive, and its polar square is $Q_\varepsilon=\mathrm e^{\varepsilon|z|^2}Q$. At a point $(x_0,\xi_0)$ with $\xi_0\ne0$, choose a maximizing line and a vector $v_0$ on it. The lemma, applied to $G_\varepsilon$, gives a holomorphic section $\sigma$ such that
\[
 \log Q_\varepsilon(z,\xi)
 \geq\log|\xi(\sigma(z))|^2-\log G_\varepsilon(x_0,v_0),
\]
with equality at $(x_0,\xi_0)$. The pairing $\xi(\sigma(z))$ is nonzero near that point, so the right-hand side is pluriharmonic there. Restriction to sufficiently small holomorphic disks centered at the point therefore gives the submean inequality for $\log Q_\varepsilon$. This argument applies at every point off the zero section, and together with continuity it shows that $\log Q_\varepsilon$ is plurisubharmonic there.

As $\varepsilon\downarrow0$, the functions $\log Q_\varepsilon=\log Q+\varepsilon|z|^2$ converge locally uniformly off the zero section. Passing to the limit in the submean inequalities on disks proves that $\log Q$ is plurisubharmonic off the zero section. Since the exponential is increasing and convex, Jensen's inequality shows that $Q=\exp(\log Q)$ is plurisubharmonic there. At a point of the zero section, $Q=0$ and $Q\geq0$, so every centered disk satisfies the submean inequality there. Together with continuity and the submean inequalities at nonzero points, this proves plurisubharmonicity on the whole total space of $E^*$.
\end{proof}

\Needspace{12\baselineskip}
\begin{lemma}\label{lem:polar-finsler-uniform-strong-convexity}
\index{Legendre duality!strong fiber convexity}
Let $k$ be a smooth Hermitian metric on $E$, and let $G$ be a positive smooth complex homogeneous function of degree two on $E^\times$, extended by zero. Then its polar square $Q$, defined as in \cref{prop:semipositive-finsler-polar-psh}, has a constant $c>0$ such that
\[
 Q(x,\xi)-c|\xi|_{k^*}^2
\]
is real convex on every dual fiber. Neither strong pseudoconvexity nor real convexity of $G$ is required here.
\end{lemma}
\begin{proof}
The real fiber Hessian of $G$ is bounded above by $2L$ times $k$ on the compact unit sphere bundle of $k$, for some $L>0$, and homogeneity extends this bound to every nonzero vector. Also, $G$ is $C^1$ at zero with derivative zero, since its value is $O(|v|^2)$ and its first fiber derivative is $O(|v|)$, uniformly over the base. Integrating the Hessian estimate along a segment gives
\begin{equation}\label{eq:polar-finsler-upper-quadratic}
 G(v+w)\leq G(v)+\mathrm d_vG(w)+L|w|_k^2;
\end{equation}
a segment passing through zero is handled by splitting it there and using the continuity of the first derivative.

Complex homogeneity and completing the square on each complex line give the identity
\begin{equation}\label{eq:polar-finsler-legendre}
 Q(x,\xi)=\max_{v\in E_x}
                  \{2\operatorname{Re}\xi(v)-G(x,v)\}.
\end{equation}
The maximum exists because $G(v)\geq a|v|_k^2$ for a uniform constant $a>0$. Fix a maximizer $v_0$ for $\xi_0$. Differentiation, including the case $\xi_0=v_0=0$, gives $\mathrm dG(v_0)=2\operatorname{Re}\xi_0$. Evaluate the expression in \eqref{eq:polar-finsler-legendre} for $\xi_0+\zeta$ at $v_0+w$ and apply \eqref{eq:polar-finsler-upper-quadratic}. The terms involving $\xi_0(w)$ cancel, and we obtain
\[
 Q(\xi_0+\zeta)\geq Q(\xi_0)+2\operatorname{Re}\zeta(v_0)
                 +2\operatorname{Re}\zeta(w)-L|w|_k^2.
\]
Taking the supremum over $w$ yields
\[
 Q(\xi_0+\zeta)\geq Q(\xi_0)+2\operatorname{Re}\zeta(v_0)
                         +L^{-1}|\zeta|_{k^*}^2.
\]
Set $c=L^{-1}$ and $\Phi=Q-c|\cdot|_{k^*}^2$. After subtracting the square, this inequality gives a real affine function on the whole dual fiber that lies below $\Phi$ and equals $\Phi$ at $\xi_0$. Hence $\Phi$ is the supremum of such affine functions, and therefore convex. Neither uniqueness of the maximizer nor real convexity of $G$ has been used.
\end{proof}

\Needspace{6\baselineskip}
\begin{corollary}\label{cor:ample-semipositive-finsler-smooth-dual}
If $E$ is ample and has a strongly pseudoconvex squared Finsler metric with semipositive Kobayashi curvature, then $E^*$ admits a smooth complex norm whose square is strictly plurisubharmonic off zero and locally strongly convex in the real fibers.
\end{corollary}
\begin{proof}
By \cref{prop:semipositive-finsler-polar-psh}, the polar square is a positive continuous plurisubharmonic complex homogeneous function of degree two, and by \cref{lem:polar-finsler-uniform-strong-convexity} it is locally strongly convex in the fibers, since in a local frame the quadratic form $|\cdot|_{k^*}^2$ dominates a positive multiple of the Euclidean one over compact sets. Applying \cref{prop:ample-dual-strict-homogeneous-function} to $F=E^*$, whose dual $F^*=E$ is ample, gives the function $H$ on $E^*$ required in \cref{prop:weak-homogeneous-convex-replacement}, and that proposition gives the asserted smooth norm.
\end{proof}

\Needspace{6\baselineskip}
\begin{corollary}\label{cor:gauduchon-semipositive-finsler-obstruction}
\index{Vector bundle!semipositive curvature obstruction}
Let $F$ be the underlying smooth bundle of $F_0$ in \cref{thm:convex-norm-deformation-obstruction}, with a holomorphic structure in the neighborhood given there. If $F^*$ is ample, then $F^*$ admits no strongly pseudoconvex squared Finsler metric with semipositive Kobayashi curvature.
\end{corollary}
\begin{proof}
Apply \cref{cor:ample-semipositive-finsler-smooth-dual} to $E=F^*$. Such a metric would produce a smooth complex norm on $F$ with plurisubharmonic square, contrary to \cref{thm:convex-norm-deformation-obstruction}.
\end{proof}

The corollary applies only near a split bundle whose line summands have positive Gauduchon degrees, and it is not vacuous only if such a neighborhood contains holomorphic structures with ample dual. On Hirzebruch surfaces, \cref{thm:hirzebruch-stable-ample-approximation} gives stable ample bundles whose duals lie in this neighborhood. Since the polar square need not be smooth, the regularization of \cref{subsec:homogeneous-convex-regularization} is an essential step of the argument.
