\subsection{Elliptic surface volumes and order spectra}\label{sec:nef-surface-orders}

On a product of two elliptic curves, a semipositive background whose coefficients depend on one factor only has an explicit volume. Solving the smooth equation reduces to a Poisson equation on that factor, and the volume of each Hermitian perturbation is governed by the harmonic mean of the perturbed vertical coefficient. The rate at which this coefficient vanishes therefore determines the order of degeneration. Zeros of finite order give fractional orders or logarithmic decay, while coefficients that vanish to infinite order can make the logarithmic rate oscillate.

Let $E=\mathbb C/(\mathbb Z+\mathrm{i}\mathbb Z)$, write $X=E_z\times E_w$, and fix
\[
 \omega_1=\frac{\mathrm{i}}{2}\mathrm{d} z\wedge\mathrm{d}\bar z,\qquad
 \omega_2=\frac{\mathrm{i}}{2}\mathrm{d} w\wedge\mathrm{d}\bar w,\qquad
 \mu=\omega_1\wedge\omega_2,\qquad h=\omega_1+\omega_2.
\]
Both area forms have integral one. Let $a,q,r$ be smooth functions of $z$ alone, with $a,q$ real and nonnegative, $r$ complex-valued, and $|r|^2\leq aq$. Set
\[
 \theta=\frac{\mathrm{i}}{2}\bigl(a\,\mathrm{d} z\wedge\mathrm{d}\bar z+
 r\,\mathrm{d} z\wedge\mathrm{d}\bar w+\bar r\,\mathrm{d} w\wedge\mathrm{d}\bar z+
 q\,\mathrm{d} w\wedge\mathrm{d}\bar w\bigr).
\]
This is a smooth semipositive form, and in particular it is nef.

\begin{proposition}\label{prop:nef-elliptic-exact-volume}
For $\epsilon>0$, put $g=q+\epsilon$ and $I=\int_Eg^{-1}\omega_1$. Then
\begin{equation}\label{eq:nef-elliptic-volume}
 \mathcal V_\mu(\theta+\epsilon h)
 =\frac{2}{I}\left(\int_E(a+\epsilon)\omega_1-
 \int_E\frac{|r|^2}{g}\omega_1\right),
\end{equation}
and the infimum defining this volume is attained by a smooth positive Gauduchon metric. In particular, if $a$ is a constant $b>0$ and $r=0$, then $\mathcal V_\mu(\theta+\epsilon h)=2(b+\epsilon)/I$. If $|r|^2=aq$, then
\begin{equation}\label{eq:nef-rank-one-elliptic-volume}
 \mathcal V_\mu(\theta+\epsilon h)=\frac{2\epsilon(1+K)}{I},
 \qquad K=\int_E\frac{a}{q+\epsilon}\omega_1.
\end{equation}
\end{proposition}

\begin{proof}
Define
\[
 c=I^{-1}\left(\int_E(a+\epsilon)\omega_1-
 \int_E\frac{|r|^2}{g}\omega_1\right),\qquad
 A=\frac{c+|r|^2}{g}.
\]
Since $(a+\epsilon)g-|r|^2>0$, we have $c>0$, and the definition of $c$ makes $A-a-\epsilon$ have zero mean with respect to $\omega_1$. Dividing its nonconstant Fourier coefficients by the corresponding nonzero eigenvalues of the flat Laplacian produces a smooth real function $u$ on $E$ with $\ddc u=(A-a-\epsilon)\omega_1$. Pull it back to $X$. The form $\Omega=\theta+\epsilon h+\ddc u$ has coefficient matrix $\left(\begin{smallmatrix}A&r\\\bar r&g\end{smallmatrix}\right)$, which has determinant $c$ and positive diagonal entries. Hence $\Omega>0$ and $\Omega^2=2c\mu$, and \cref{thm:gaud-volume-dual-formula} gives \eqref{eq:nef-elliptic-volume}.

The metric $\gamma_*=\Omega/g$ has constant coefficient $1$ in the fiber direction, and its remaining coefficients depend only on $z$. Applying $\ddc$ to its base term or its mixed terms produces a repeated $\mathrm{d} z$ or $\mathrm{d}\bar z$, so $\ddc\gamma_*=0$. Thus $\gamma_*$ is a smooth Gauduchon metric conformal to $\Omega$, and by \cref{thm:gaud-volume-dual-formula} it attains the infimum. The two special cases follow by direct substitution.
\end{proof}

\begin{proposition}\label{prop:nef-diagonal-fractional-and-zero}\index{Nef degeneration!fractional and zero orders}
For each integer $m>1$ there is a smooth semipositive nonclosed form on $X$ with common order $1-1/m$. For each $b>0$ there is also one with
\[
 \mathcal V_\mu(\theta+\epsilon h)\sim\frac{2b}{\pi\log(1/\epsilon)}
 \quad(\epsilon\downarrow0);
\]
it has volume zero and both orders zero.
\end{proposition}

\begin{proof}
In \cref{prop:nef-elliptic-exact-volume} take $a=b>0$ and $r=0$. Fix a point $z_0\in E$ and a flat coordinate $\zeta=z-z_0$ on a small disk about $z_0$. Choose $q$ smooth, positive outside $z_0$, and equal to $|\zeta|^{2m}$ on that disk; a positive smooth extension exists away from the disk. For $m=1$, polar coordinates give
\[
 I=\int_E\frac{\omega_1}{q+\epsilon}
   =\pi\log(1/\epsilon)+O(1),
\]
and \eqref{eq:nef-elliptic-volume} gives the asymptotic formula of the statement. If $m>1$, the substitution $s=|\zeta|^2=\epsilon^{1/m}t$ gives
\[
 I\sim C_m\epsilon^{1/m-1},\qquad
 C_m=\pi\int_0^\infty\frac{\mathrm{d} t}{1+t^m}>0,
\]
so the common order is $1-1/m$. In the logarithmic case, \cref{thm:gaud-volume-nef-limit} gives $\mathcal V_\mu(\theta)=\lim_{\epsilon\downarrow0}\mathcal V_\mu(\theta+\epsilon h)=0$. Moreover, the quotient of the volume by $\epsilon^\delta$ diverges for every $\delta>0$, so both orders vanish. Finally, $\mathrm{d}\theta=\mathrm{d} q\wedge\omega_2\ne0$, so these backgrounds are not closed.
\end{proof}

\begin{proposition}\label{prop:nef-oscillatory-orders}\index{Nef degeneration!unequal orders}
For every $0<\alpha<\beta<1$ there is a smooth semipositive nonclosed background on $X$, with the same $\mu$ and $h$, such that
\[
 \bigl(\underline a(\theta),\overline a(\theta)\bigr)
 =(1+\alpha,1+\beta).
\]
For every $0<\sigma<1$ there is such a background with common order $1+\sigma$.
\end{proposition}

\begin{proof}
We use the rank-one formula \eqref{eq:nef-rank-one-elliptic-volume}, with coefficients that vanish to infinite order at one point at a prescribed oscillating rate.

\emph{Step 1: the coefficients.} For all sufficiently large $t$, define
\[
 p(t)=\frac{\alpha+\beta}{2}
 +\frac{\beta-\alpha}{2}\sin(\log\log t),\qquad P(t)=tp(t).
\]
Choose $T$ so large that $0<c_0\leq P'(t)\leq d_0<1$ for $t\geq T$; this is possible because
\[
 P'(t)=p(t)+\frac{\beta-\alpha}{2\log t}\cos(\log\log t)
\]
and $\alpha\leq p\leq\beta$. Fix $z_0\in E$ and a flat coordinate $\zeta$ centered at $z_0$. On $E\setminus\{z_0\}$ choose a smooth function $t(z)\geq T$ that equals $|\zeta|^{-2}$ near $z_0$ and is bounded outside every neighborhood of $z_0$. Put
\[
 q=\mathrm{e}^{-t},\qquad a=\mathrm{e}^{-P(t)},\qquad
 r=\mathrm{e}^{-(P(t)+t)/2},
\]
and set all three equal to zero at $z_0$. These functions are smooth and vanish to infinite order at $z_0$: we have $P(t)\geq\alpha t$, and every fixed derivative of $P$ grows at most polynomially in $t$, so by the chain rule each coordinate derivative is bounded by a polynomial in $|\zeta|^{-1}$ times $\mathrm{e}^{-c/|\zeta|^2}$ for some $c>0$. Since $|r|^2=aq$, the resulting $\theta$ is semipositive. It is not closed: the component $\mathrm{d} q\wedge\omega_2$ of $\mathrm{d}\theta$ contains both fiber differentials, and derivatives of the remaining terms cannot cancel it.

\emph{Step 2: asymptotics of $I$ and $K$.} Write $\epsilon=\mathrm{e}^{-L}$, and let $T_0\geq T$ be such that $t=|\zeta|^{-2}$ on the disk $\{|\zeta|^2<1/T_0\}$. Outside this disk the integrands of $I$ and $K$ in \eqref{eq:nef-rank-one-elliptic-volume} are bounded, so their contributions are $O(1)$, while polar coordinates on the disk give
\[
 I_{\mathrm{loc}}=\pi\int_{T_0}^\infty
 \frac{t^{-2}\,\mathrm{d} t}{\mathrm{e}^{-t}+\mathrm{e}^{-L}},\qquad
 K_{\mathrm{loc}}=\pi\int_{T_0}^\infty
 \frac{\mathrm{e}^{-P(t)}t^{-2}\,\mathrm{d} t}{\mathrm{e}^{-t}+\mathrm{e}^{-L}}.
\]
The part of $I_{\mathrm{loc}}$ over $t\geq L$ is at least a fixed multiple of $\mathrm{e}^L/L$. For an upper bound, use $(\mathrm{e}^{-t}+\mathrm{e}^{-L})^{-1}\leq\mathrm{e}^t$ for $t\leq L$ and $(\mathrm{e}^{-t}+\mathrm{e}^{-L})^{-1}\leq\mathrm{e}^L$ for $t\geq L$, and split $[T_0,L]$ at $L/2$. Thus
\begin{equation}\label{eq:nef-elliptic-I-asymptotic}
 \log I=L-\log L+O(1).
\end{equation}
The derivatives of $P$ and of $t-P(t)$ are positive and bounded away from zero. Splitting the integral for $K$ at $t=L$ therefore gives $K\leq C\mathrm{e}^{L-P(L)}$. On $[L,L+1]$ the denominator is at most $2\mathrm{e}^{-L}$ and $P(t)\leq P(L)+1$, so
\[
 c_1L^{-2}\mathrm{e}^{L-P(L)}\leq K\leq c_2\mathrm{e}^{L-P(L)};
\]
the complementary $O(1)$ term is absorbed because $P(L)\leq\beta L$. Consequently $\log K=L-P(L)+O(\log L)$, and $K\to\infty$.

\emph{Step 3: the orders.} Combining these estimates with \eqref{eq:nef-rank-one-elliptic-volume} and \eqref{eq:nef-elliptic-I-asymptotic} yields
\begin{equation}\label{eq:nef-oscillatory-log-volume}
 \log \mathcal V_\mu(\theta+\mathrm{e}^{-L}h)=-L-P(L)+O(\log L),
 \qquad
 \frac{\log \mathcal V_\mu(\theta+\epsilon h)}{\log\epsilon}
 =1+p(L)+o(1).
\end{equation}
Since $\sin(\log\log L)$ oscillates between $-1$ and $1$, the lower and upper limits of $p(L)$ are $\alpha$ and $\beta$, which proves the first assertion. If instead $P(t)=\sigma t$, the same smoothness and integral estimates give the common order $1+\sigma$. In particular, $\sigma=\sqrt2-1$ gives an irrational common order.
\end{proof}
