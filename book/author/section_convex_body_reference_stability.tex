\section{Convex-body criteria for varying references}
\label{sec:convex-body-reference-stability}
\index{Reference stability!convex-body criterion}

When the singularities of a torus-invariant reference are described by a convex body, the closed mass formula of \cref{thm:normalization-mass-quotient} turns convergence of the convex bodies into convergence of the functionals. A translated simplex contained in all the bodies keeps the limiting reference solvable, and the Hausdorff distance then controls the functionals uniformly over all continuous tests, even though the ordinary types vary.

Let $n\geq2$, and let $\omega_{\mathrm{FS}}$ be the Fubini--Study form on $\mathbb P^n$, normalized by $\int_{\mathbb P^n}\omega_{\mathrm{FS}}^n=1$. In homogeneous coordinates $[Z_0:\cdots:Z_n]$, consider the torus $T=\{Z_0\cdots Z_n\ne0\}$, and put
\[
 x_i=\log|Z_i/Z_0|^2,\qquad
 L(x)=\log\bigl(1+\textstyle\sum_i\mathrm{e}^{x_i}\bigr),\qquad
 \Sigma=\{p_i\geq0:\textstyle\sum_i p_i\leq1\},
\]
where $1\leq i\leq n$ and $\Sigma\subset\mathbb R^n$ is the standard simplex. Functions of $x=(x_1,\ldots,x_n)$ are regarded as functions on $T$; thus $\omega_{\mathrm{FS}}=\ddc L$ on $T$. Define the continuous function $\ell$ on $\Sigma$ by
\[
 \ell(p)=\sum_i p_i\log p_i+
       (1-\textstyle\sum_i p_i)\log(1-\textstyle\sum_i p_i),
 \qquad 0\log0=0.
\]
For a convex body $K\subset\Sigma$, that is, a compact convex set with nonempty interior, set
\[
 h_K(x)=\sup_{p\in K}p\cdot x,\qquad
 g_K(x)=\sup_{p\in K}\{p\cdot x-\ell(p)\}.
\]

\subsection{Canonical models and their mass}

\begin{lemma}\label{lem:convex-body-model}
The functions $h_K-L$ and $g_K-L$ on $T$ extend uniquely to $\omega_{\mathrm{FS}}$-plurisubharmonic functions $\psi_K,\Phi_K$ on $\mathbb P^n$. They satisfy
\[
 \begin{aligned}
  &\psi_K\leq\Phi_K\leq\psi_K+\log(n+1),\qquad\sup\Phi_K=0,\\
  &P_{\omega_{\mathrm{FS}}}[\psi_K]=\Phi_K
       =P_{\omega_{\mathrm{FS}}}[\Phi_K].
 \end{aligned}
\]
If $p_*+\delta\Sigma\subset K$ for some $p_*\in\mathbb R^n$ and $0<\delta<1$, then $P_{(1-\delta)\omega_{\mathrm{FS}}}(\Phi_K)\not\equiv-\infty$.
\end{lemma}
\begin{proof}
The Legendre transform of $L$ is $L^*=\ell$ on $\Sigma$ and $+\infty$ outside it. Since $-\log(n+1)\leq\ell\leq0$,
\[
 h_K\leq g_K\leq h_K+\log(n+1),\qquad
 g_K\leq L,\qquad h_K\leq h_\Sigma\leq L.
\]
A finite convex function of the logarithmic moduli is plurisubharmonic: for a smooth convex function the complex Hessian is positive, and convolution gives the general case. On $T\cap\{Z_j\ne0\}$, add the local Fubini--Study potential $\log\sum_i|Z_i/Z_j|^2$; the resulting functions are again convex functions of the logarithmic moduli of the affine coordinates, and they are bounded above by that smooth potential because $h_K,g_K\leq L$. \Cref{lem:qpsh-removable-analytic} therefore gives the extensions across the coordinate hyperplanes and the stated bounds. If $p\in\operatorname{int}K$, then $p\in\operatorname{int}\Sigma$, and at $x_i=\log(p_i/(1-\sum_k p_k))$ the supremum defining $L=\ell^*$ is attained at $p\in K$, so $g_K(x)=L(x)$. This proves $\sup\Phi_K=0$.

For $C\geq\log(n+1)$, let $v_C=P_{\omega_{\mathrm{FS}}}(\psi_K+C,0)$, the envelope of the obstacle $\min(\psi_K+C,0)$. Since $\Phi_K$ is a competitor, $\Phi_K\leq v_C\leq0$. The obstacle is upper semicontinuous, so $v_C\leq\psi_K+C$ everywhere. The compact torus acting by rotations of the coordinates preserves the obstacle, so $v_C$ is invariant, and on $T$ the function $f=v_C+L$ is a finite convex function of $x$ with $f\leq L$ and $f\leq h_K+C$. The second inequality shows that the Legendre transform $f^*$ is $+\infty$ outside $K$, and the first gives $f^*\geq\ell$. Thus
\[
 f(x)=\sup_{p\in\operatorname{dom}f^*}\{p\cdot x-f^*(p)\}
       \leq g_K(x).
\]
Hence $v_C\leq\Phi_K$ on $T$, and since both functions are quasi-plurisubharmonic and $T$ is the complement of an analytic set, $v_C=\Phi_K$ on $\mathbb P^n$. Letting $C\to\infty$ gives $P_{\omega_{\mathrm{FS}}}[\psi_K]=\Phi_K$. Since $\psi_K\leq\Phi_K\leq\psi_K+\log(n+1)$, the envelopes of $\min(\Phi_K+C,0)$ and $\min(\psi_K+C,0)$ interlace as $C\to\infty$, so also $P_{\omega_{\mathrm{FS}}}[\Phi_K]=\Phi_K$.

For the final assertion put $\varepsilon=1-\delta$, $b=\sum_i p_{*,i}$ and $\chi=p_*\cdot x-\varepsilon L$ on $T$. The inclusion $p_*+\delta\Sigma\subset\Sigma$ implies $p_{*,i}\geq0$ and $b\leq\varepsilon$, and the extension of $\chi$ to $\mathbb P^n$ satisfies
\[
 \varepsilon\omega_{\mathrm{FS}}+\ddc\chi
 =\sum_i p_{*,i}[Z_i=0]+(\varepsilon-b)[Z_0=0]\geq0.
\]
Since $h_K\geq p_*\cdot x+\delta h_\Sigma$ and $L-h_\Sigma\leq\log(n+1)$, one has $\chi-\delta\log(n+1)\leq\Phi_K$. Thus $\chi-\delta\log(n+1)$ is an $\varepsilon\omega_{\mathrm{FS}}$-plurisubharmonic competitor for $P_{\varepsilon\omega_{\mathrm{FS}}}(\Phi_K)$, which proves the claim.
\end{proof}

In the rest of this section, either $X=\mathbb P^n$ and $\pi=\mathrm{id}$, or $\pi\colon X=\operatorname{Bl}_q\mathbb P^n\to\mathbb P^n$ is the blowup at a fixed point $q$, with exceptional divisor $E_q$; put $\Theta=\pi^*\omega_{\mathrm{FS}}$.

\begin{lemma}\label{lem:convex-body-mass}\index{Convex body!model mass}
The manifold $X$ is K\"ahler, and $\Theta$ is smooth, closed, semipositive and big. For $r>0$, the reference $r\pi^*\Phi_K$ satisfies $P_{r\Theta}[r\pi^*\Phi_K]=r\pi^*\Phi_K$, has small unbounded locus, and satisfies
\begin{equation}\label{eq:convex-body-mass}
 \int_X\mathrm{MA}_{r\Theta}(r\pi^*\Phi_K)
       =r^n n!\operatorname{Vol}_n(K).
\end{equation}
If $p_*+\delta\Sigma\subset K$ with $0<\delta<1$, then $P_{(1-\delta)r\Theta}(r\pi^*\Phi_K)\not\equiv-\infty$.
\end{lemma}
\begin{proof}
On $\mathbb P^n$, the envelope identity is \cref{lem:convex-body-model}. In the blowup case, every $\pi^*\omega_{\mathrm{FS}}$-plurisubharmonic function descends to $\mathbb P^n$. Indeed, off $E_q$ add a local potential of $\omega_{\mathrm{FS}}$; the resulting plurisubharmonic function is bounded above near $q$, because the original function is bounded above near the compact fiber $E_q$, and therefore extends across $q$. The pullback of the extension agrees with the original function off $E_q$, hence everywhere. Pullback is therefore a bijection from $\PSH(\mathbb P^n,\omega_{\mathrm{FS}})$ onto $\PSH(X,\Theta)$ preserving inequalities and envelopes, and positive scaling preserves them as well.

On projective space, $\Theta=\omega_{\mathrm{FS}}$ is K\"ahler. For bigness in the blowup case, choose unitary homogeneous coordinates $[W_0:\cdots:W_n]$, used only in this paragraph, in which $q=[1:0:\cdots:0]$, set
\[
 \psi_q=\log\frac{\sum_{i=1}^n|W_i|^2}{\sum_{i=0}^n|W_i|^2},
\]
and let $Q\colon X\to\mathbb P^{n-1}$ be the map induced by the linear projection from $q$. For $0<t<1$,
\[
 \Theta+\ddc(t\pi^*\psi_q)
 =(1-t)\pi^*\omega_{\mathrm{FS}}+tQ^*\omega_{\mathrm{FS},n-1}+t[E_q].
\]
The smooth part is a K\"ahler form, since $(\pi,Q)$ embeds $X$ as the incidence graph in $\mathbb P^n\times\mathbb P^{n-1}$ and both coefficients are positive. This proves the assertions about the manifold and the form $\Theta$.

For a smooth convex $g$, the coordinates $z_i=\mathrm{e}^{x_i/2+\mathrm{i}\vartheta_i}$ give
\[
 \mathrm{d}^{\mathrm c}g=\frac1{2\pi}\sum_i g_i\,\mathrm{d}\vartheta_i,
 \qquad
 \int_T(\ddc g)^n=n!\int_{\mathbb R^n}\det(D^2g)\,\mathrm{d}x.
\]
For a finite convex $g$, convolution and the continuity of bounded Monge--Amp\`ere products under local uniform convergence identify $(\ddc g)^n$ on $T$ in the same way: its image under $z\mapsto x$ is $n!M_g$, where $M_g$ is the Alexandrov measure, which assigns to a Borel set of $\mathbb R^n$ the Euclidean volume of its subgradient image. For $g_K$, the bound $g_K\leq h_K+\log(n+1)$ and a separation argument put every subgradient in $K$. Conversely, if $p\in\operatorname{int}K$ and $p+c\mathbb B\subset K$, where $\mathbb B$ is the closed unit ball of $\mathbb R^n$ and $c>0$, then $g_K(x)-p\cdot x\geq c|x|$; this function attains a minimum, so $p$ is a subgradient of $g_K$ at some point. Consequently
\[
 \operatorname{int}K\subset\partial g_K(\mathbb R^n)\subset K,
 \qquad M_{g_K}(\mathbb R^n)=\operatorname{Vol}_n(K),
\]
since the boundary of a convex body has zero volume. The analytic complement of $T$ receives no non-pluripolar mass, so this is the full mass. Locality off the exceptional divisor proves the formula after pullback, since the divisor and its image carry no non-pluripolar mass, and homogeneity gives the factor $r^n$. The reference is locally bounded off the coordinate hyperplanes and their inverse image, an analytic set, hence has small unbounded locus. Finally, scaling by $r$ and pulling back the competitor found at the end of the proof of \cref{lem:convex-body-model} gives the last assertion.
\end{proof}

\subsection{Hausdorff control of the functional}

\begin{theorem}\label{thm:convex-body-reference-stability}
Let convex bodies $P_j,P\subset\Sigma$ contain the same translated simplex $p_*+\delta\Sigma$, with $0<\delta<1$, and suppose that their Hausdorff distance $d_j=d_H(P_j,P)$ tends to zero. Let $r_j>0$ tend to $r>0$. On the fixed $X$ above, set
\[
 (\theta_j,\varphi_j)=(r_j\Theta,r_j\pi^*\Phi_{P_j}),
 \qquad (\theta,\varphi)=(r\Theta,r\pi^*\Phi_P).
\]
For every bounded real Borel function $F$, the equation \eqref{eq:relative-normalization} is solvable in the ordinary type of each $\varphi_j$ and of $\varphi$; these types have masses $m_j=r_j^n n!\operatorname{Vol}_n(P_j)$ and $m=r^n n!\operatorname{Vol}_n(P)>0$. Fix a smooth positive volume form $\mathrm{d}V$ on $X$, put $Z_F=\int_X\mathrm{e}^F\,\mathrm{d}V$, and let $B_j,B$ and $\mathcal A_j,\mathcal A$ be the corresponding normalization scalars and supporting sets. Then
\[
 B_j(F)=\log m_j-\log Z_F,\qquad B(F)=\log m-\log Z_F.
\]
Writing $s=\delta/(n+\sqrt n)$, one has
\begin{equation}\label{eq:convex-body-functional-bound}
 \sup_{G\in C(X,\mathbb R)}|B_j(G)-B(G)|
 \leq n|\log(r_j/r)|+n\log(1+d_j/s).
\end{equation}
Moreover, $n!\operatorname{Vol}_n(P_j)\geq\delta^n$ and $n!\operatorname{Vol}_n(P)\geq\delta^n$, the backgrounds converge smoothly, and the references converge in $L^1(X)$ and locally uniformly on $\pi^{-1}(T)$. For continuous real $F$, every supporting set at $F$ equals $\{\nu_F\}$, where $\nu_F=\mathrm{e}^F\mathrm{d}V/Z_F$. If continuous $F_j$ converge uniformly to $F$ and $\nu_j\in\mathcal A_j(F_j)$, then
\begin{equation}\label{eq:convex-body-support-bound}
 \int_X\left|\frac{\mathrm{d}\nu_j}{\mathrm{d}V}
          -\frac{\mathrm{d}\nu_F}{\mathrm{d}V}\right|\mathrm{d}V
 \leq\mathrm{e}^{2\|F_j-F\|_\infty}-1.
\end{equation}
\end{theorem}
\begin{proof}
By the two preceding lemmas, each $\varphi_j$ and $\varphi$ is an envelope fixed point with small unbounded locus and the stated positive mass, and it satisfies the smaller-background condition. \Cref{thm:normalization-mass-quotient} therefore gives solvability and the formulas for the scalars and supporting sets.

The common simplex contains a ball $c+s\mathbb B$ of radius $s$, the inradius of $\delta\Sigma$, where $\mathbb B$ is the closed unit ball. The Hausdorff distance gives $P_j\subset P+d_j\mathbb B$, while $s\mathbb B\subset P-c$ and convexity give
\[
 P+d_j\mathbb B\subset c+(1+d_j/s)(P-c).
\]
Exchanging the roles of $P$ and $P_j$ bounds both volume ratios by $(1+d_j/s)^n$. Subtracting the scalar formulas and including the factor $r_j^n/r^n$ gives \eqref{eq:convex-body-functional-bound}. The volume bound holds because $\delta\Sigma$ has volume $\delta^n/n!$.

Let $\omega_\ell$ be the modulus of uniform continuity of $\ell$ on $\Sigma$. Matching the maximizers in the definitions of $g_{P_j}$ and $g_P$ by points at distance at most $d_j$ gives
\[
 |g_{P_j}(x)-g_P(x)|\leq d_j|x|+\omega_\ell(d_j).
\]
This proves local uniform convergence on $T$. The common lower bound $\chi-\delta\log(n+1)\leq\Phi_{P_j}$ obtained at the end of the proof of \cref{lem:convex-body-model} is integrable on $X$, also after pullback, so boundedness of $r_j$ and dominated convergence off the coordinate hyperplanes give $L^1$ convergence. Finally, the ratio of the normalized exponential densities lies in $[\mathrm{e}^{-2\eta},\mathrm{e}^{2\eta}]$, where $\eta=\|F_j-F\|_\infty$. This proves \eqref{eq:convex-body-support-bound} and hence convergence in total variation.
\end{proof}
