The same directional positivity yields an exhaustion on the tangent projective complement.

\Needspace{12\baselineskip}
\begin{corollary}\label{cor:ricci-flat-tangent-complement-convexity}
\index{RC-positivity!tangent complement}\index{Projective bundle!Ricci-flat tangent complement}
For a connected compact complex surface $X$ admitting a nonflat Ricci-flat K\"ahler metric, the complement
\[
 \mathbb P(\mathcal O_X\oplus T^{1,0}X)\setminus\mathbb P(\mathcal O_X)
\]
is $3$-convex. The tangent bundle admits no Hermitian metric with positive definite mean curvature relative to any Hermitian form.
\end{corollary}
\begin{proof}
By \cref{thm:ricci-flat-surface-rc-positive}, the tangent bundle has an RC-positive Hermitian metric. Its directional curvature form has a positive eigenvalue for every nonzero fibre vector. Apply \cref{thm:bundle-curvature-complement-convexity} with $n=r=2$ and $s=1$. The mean-curvature obstruction is the second assertion of \cref{thm:ricci-flat-surface-rc-positive}.
\end{proof}
