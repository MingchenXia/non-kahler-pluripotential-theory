\subsection{Warped product cones and flat completion}
\label{sec:warped-gauduchon-completion}

The invariant cones above have complete unit-volume hypersurfaces. A product with a curve gives nonclosed cones whose hypersurfaces can instead complete to strips or half-planes. Their volume is a reciprocal integral, and the variance of the same reciprocal coefficient measures the distance to the boundary. The corresponding probability is a trace of an actual minimizing balanced metric.

Let $(B,\kappa)$ be a connected compact K\"ahler manifold of dimension $m\geq2$, with $\int_B\kappa^m=1$. Let $C$ be a connected compact complex curve with a smooth positive form $w$ of integral one. Fix a smooth strictly positive probability form $\sigma$ on $B$ and a smooth nonconstant real function $a$ with $\min a=0$ and $\max a=M>0$. On $X=B\times C$, of dimension $n=m+1$, put $\mu=\sigma\wedge w$ and suppress pullbacks. In the real span of $\kappa,aw,w$ write
\[
 \theta_{b,c,e}=b\kappa+Lw,\qquad L=ca+e.
\]

\begin{theorem}\label{thm:warped-gauduchon-volume}\index{Gauduchon variational volume!warped product cones}
This span has no nonzero smooth $\ddc$ direction. Its entire cone of backgrounds admitting a smooth positive potential representative is
\[
 \mathscr C=\{b>0,\ e>0,\ e+cM>0\}.
\]
For $\theta\in\mathscr C$, let $H(c,e)=\int_B\sigma/L$. Then
\begin{equation}\label{eq:warped-gauduchon-volume}
 \mathcal V_\mu(\theta)=\frac{nb^m}{H(c,e)}.
\end{equation}
An actual minimizing metric is
\[
 \gamma=L^{-1/m}(A+Lw),\qquad
 A=b\kappa+\ddc\phi>0,\qquad
 A^m=\frac{b^m}{H}\frac{\sigma}{L}.
\]
It is balanced. Backgrounds with $c\ne0$ are nonclosed.
\end{theorem}
\begin{proof}
A smooth positive representative paired with the closed semipositive form $\kappa^{m-1}\wedge w$ gives $b>0$. Its integral over the compact fiber $\{y\}\times C$ is $L(y)>0$, by smooth Stokes. These are precisely the stated inequalities; conversely, they make $\theta$ positive. The same tests on a smooth $\ddc$ direction give $b=0$ and $ca+e=0$ everywhere. Nonconstancy of $a$ then gives $c=e=0$. Also $\mathrm{d}\theta=c\,\mathrm{d}a\wedge w$, which is not identically zero when $c\ne0$.

Apply \cref{thm:tw-smooth} on the closed positive base background $b\kappa$. Closed Stokes fixes the scalar because the displayed target has integral $b^m$. This supplies $A$. Put $\Omega=A+Lw$. Binomial expansion gives
\begin{equation}\label{eq:warped-balanced-cofactor}
 \Omega^n=\frac{nb^m}{H}\mu,\qquad
 \gamma^m=L^{-1}A^m+mA^{m-1}\wedge w.
\end{equation}
The first summand is a base top-degree form and the second is closed. Hence $\gamma$ is balanced. For every smooth positive Gauduchon metric $\eta$, Stokes and matrix arithmetic--geometric mean give
\[
 \int_X\theta\wedge\eta^m
 =\int_X\Omega\wedge\eta^m
 \geq\left(\frac{nb^m}{H}\right)^{1/n}D_\mu(\eta).
\]
For $\gamma$ the numerator is $nb^m$, while
$D_\mu(\gamma)=(nb^m/H)^{m/n}H$. Its quotient attains the lower bound, proving \eqref{eq:warped-gauduchon-volume} for the full infimum.
\end{proof}

The minimizing metric gives the probability identity
\begin{equation}\label{eq:warped-minimizer-trace}
 \frac{Lw\wedge\gamma^m}{b^m}
       =\frac{\sigma\wedge w}{HL}.
\end{equation}
We use its base marginal to compute the metric. Set
\begin{align*}
 v&=c/e\in(-1/M,\infty),
 &Q(v)&=\int_B\frac{\sigma}{1+va},\\
 \mathrm{d}\nu_v&=\frac{\sigma}{Q(v)(1+va)},
 &\ell_v&=\frac{a}{1+va}.
\end{align*}

\begin{proposition}\label{prop:warped-flat-unit-metric}\index{Hessian metric!flat warped cones}
On $\mathscr S=\{\mathcal V_\mu=1\}$, the global coordinates $U=\log b$ and $v$ give
\begin{equation}\label{eq:warped-flat-unit-metric}
 \begin{aligned}
 g_{\mathscr S}&=m\,\mathrm{d}U^2+\rho(v)\,\mathrm{d}v^2,\\
 \rho(v)&=\frac{Q''/Q-2(Q'/Q)^2}{n}
       =\frac2n\operatorname{Var}_{\nu_v}(\ell_v)>0.
 \end{aligned}
\end{equation}
For any $v_*\in(-1/M,\infty)$, the coordinates
\[
 Y=\sqrt m\,U,\qquad
 R(v)=\int_{v_*}^v\sqrt{\rho(t)}\,\mathrm{d}t
\]
are a global isometry onto $\mathbb R\times I$, where $I$ is an open interval. The intrinsic distance is the Euclidean distance in $(Y,R)$.
\end{proposition}
\begin{proof}
For an affine coefficient variation, write $F=\delta L/L$ and $\mathrm{d}\nu=\sigma/(HL)$. Differentiating under the integral gives
\[
 D^2\log H(\delta,\delta)
   =2\mathbb E_\nu F^2-(\mathbb E_\nu F)^2.
\]
It is positive unless $\delta L=0$ everywhere, which forces $\delta c=\delta e=0$. Thus
\[
 g=-D^2\log\lambda_\mu
   =\frac mn\left(\frac{\mathrm{d}b}{b}\right)^2
                  +\frac1nD^2\log H
\]
is positive. On the unit surface,
\[
 e=\frac{Q(v)}{n\exp(mU)},\qquad c=ve.
\]
To pull back the affine Hessian, put $t=\mathrm{d}e/e$ and $z=Q'/Q$. Along an affine path $v''=-2tv'$, so $\log H=-\log e+\log Q(v)$ gives
\[
 D^2\log H=t^2-2zt\,\mathrm{d}v
                   +(Q''/Q-z^2)\,\mathrm{d}v^2.
\]
Substitute $t=z\,\mathrm{d}v-m\,\mathrm{d}U$ to obtain \eqref{eq:warped-flat-unit-metric}. Since $Q'/Q=-\mathbb E_{\nu_v}\ell_v$ and $Q''/Q=2\mathbb E_{\nu_v}\ell_v^2$, its variance formula follows. Nonconstancy of $a$ makes this variance positive. The resulting interval product is convex, so straight segments prove the distance assertion.
\end{proof}

\subsubsection{Endpoint trace estimates}

Assume each extreme set $\{a=0\}$ and $\{a=M\}$ is a finite disjoint union of smooth compact real submanifolds $Z_i$ of positive codimension. Near each component suppose, respectively,
\[
 a\asymp\operatorname{dist}(Z_i)^{k_i},\qquad
 M-a\asymp\operatorname{dist}(Z_i)^{k_i},\qquad k_i>0,
\]
with positive two-sided constants. The corresponding function is bounded below off disjoint small tubes. Define separately
\[
 r_0=\min_{\{a=0\}}
       \frac{\operatorname{codim}_{\mathbb R}Z_i}{k_i},\qquad
 r_M=\min_{\{a=M\}}
       \frac{\operatorname{codim}_{\mathbb R}Z_i}{k_i}.
\]

\Needspace{15\baselineskip}
\begin{lemma}\label{lem:warped-endpoint-length}\index{Gauduchon metric!trace variance}
The end $v\to\infty$ has finite length exactly when $r_0>1$, and the end $v\to-1/M$ exactly when $r_M>1$. More precisely, write $f=a$ at the first end and $f=M-a$ at the second, and put
\[
 H_f=\int_B\frac{\sigma}{f+\epsilon},\quad
 J_f=\int_B\frac{\sigma}{(f+\epsilon)^2},\quad
 K_f=\int_B\frac{\sigma}{(f+\epsilon)^3}.
\]
In the corresponding endpoint coordinate,
\begin{equation}\label{eq:warped-endpoint-variance}
 \rho(v)\,\mathrm{d}v^2=G_f(\epsilon)\,\mathrm{d}\epsilon^2,
 \quad
 G_f=\frac2n\left[\frac{K_f}{H_f}
                    -\left(\frac{J_f}{H_f}\right)^2\right].
\end{equation}
If $r$ is the appropriate ratio, then $G_f\asymp\epsilon^{-2}$ for $r<1$ and $G_f\asymp\epsilon^{-2}/\log(1/\epsilon)$ for $r=1$. For $r>1$, $\sqrt{G_f}$ is integrable near zero.
\end{lemma}
\begin{proof}
At the right end use $\epsilon=1/v$. Then
\[
 \ell_v=\epsilon-\frac{\epsilon^2}{a+\epsilon},\qquad
 \mathrm{d}v=-\epsilon^{-2}\mathrm{d}\epsilon.
\]
At the left end use $\epsilon=-1/v-M$, obtaining
\[
 \ell_v=\frac{(M+\epsilon)^2}{M-a+\epsilon}-(M+\epsilon),
 \qquad \mathrm{d}v=(M+\epsilon)^{-2}\mathrm{d}\epsilon.
\]
Both give \eqref{eq:warped-endpoint-variance} as the variance of $1/(f+\epsilon)$ under $\sigma/[H_f(f+\epsilon)]$. This is the marginal in \eqref{eq:warped-minimizer-trace}, since positive scaling of $L$ leaves it unchanged.

Tubular integration, splitting the normal radius at $\epsilon^{1/k_i}$, gives
\[
 H_f\asymp
 \begin{cases}\epsilon^{r-1}&r<1,\\
 \log(1/\epsilon)&r=1,\\1&r>1,
 \end{cases}
 \qquad
 K_f\asymp
 \begin{cases}\epsilon^{r-3}&r<3,\\
 \log(1/\epsilon)&r=3,\\1&r>3.
 \end{cases}
\]
The analogous threshold for $J_f$ is two. A finite sum of tube integrals is governed by the smallest ratio.

If $r<1$, choose a component attaining $r$ and two normal annuli at fixed multiples of $\epsilon^{1/k_i}$. Choose the multiples sufficiently far apart that the two-sided bounds separate their $f/\epsilon$ ranges. Each annulus has probability bounded below, since its weighted mass and $H_f$ have order $\epsilon^{r-1}$. Their reciprocal values differ by at least $c\epsilon^{-1}$. The identity
\[
 \operatorname{Var}_\nu T
   =\frac12\iint(T(x)-T(y))^2\,\mathrm{d}\nu(x)\mathrm{d}\nu(y)
\]
gives the lower bound; $T\leq\epsilon^{-1}$ gives the upper bound.

If $r=1$, the near annulus has probability at least $c/\log(1/\epsilon)$. In the same tube, normal radii between $\epsilon^{1/(2k_i)}$ and a fixed small radius carry probability bounded below, by the logarithmic integral. Their reciprocals are at most $C\epsilon^{-1/2}$, while the near-annulus reciprocals are at least $c\epsilon^{-1}$. The variance identity and $G_f\leq2K_f/(nH_f)$ prove the claimed logarithmic bound. Both subcritical and critical square-root integrals diverge.

For $r>1$, $H_f$ tends to a finite positive integral and $G_f\leq C K_f$. For $1<r<3$ its square root is bounded by $C\epsilon^{(r-3)/2}$, whose exponent exceeds $-1$. At $r=3$ the bound is $C\sqrt{\log(1/\epsilon)}$, and for $r>3$ it is bounded. All these bounds are integrable.
\end{proof}

\begin{theorem}\label{thm:warped-unit-volume-completion}\index{Metric completion!warped unit-volume hypersurface}
Under the stated extreme-set hypotheses, the metric completion of $\mathscr S$ is a closed Euclidean strip if $r_0,r_M>1$, a closed Euclidean half-plane if exactly one exceeds one, and the whole Euclidean plane otherwise. Every finite end is parameterized by $b>0$. Its representatives are
\begin{align*}
 \theta_+(b)&=b\kappa+\frac{H_a}{nb^m}a w,
 &H_a&=\int_B\frac{\sigma}{a}\quad(r_0>1),\\
 \theta_-(b)&=b\kappa+\frac{H_-}{nb^m}(1-a/M)w,
 &H_-&=\int_B\frac{\sigma}{1-a/M}\quad(r_M>1).
\end{align*}
They are smooth semipositive, nef, nonclosed and nonbig, and have full variational volume one. There are no other Cauchy boundary points in this specified cone.
\end{theorem}
\begin{proof}
The completion of the convex interval product in \cref{prop:warped-flat-unit-metric} is its Euclidean closure. A Cauchy sequence has finite $Y$, hence a finite positive limit of $b$, and its $R$ coordinate reaches only an interior point or a finite end. \Cref{lem:warped-endpoint-length} gives the three alternatives.

At the right end, $vQ(v)\to H_a$ gives the displayed coefficient limit. At the left end, write $1+va=1-a/M+(v+1/M)a$; dominated convergence gives $Q(v)\to H_-$. These are all coefficient limits at finite ends. Their forms are semipositive and nef by positive Hermitian perturbation, and are nonclosed. Increasing their vertical coefficient by $\epsilon$ makes them positive; \eqref{eq:warped-gauduchon-volume} gives a limit of one. Monotonicity gives that their full infima are at most this limit, whereas continuity of each fixed test quotient gives the reverse inequality after taking its infimum. The same argument applies to decreasing Hermitian regularization.

Suppose an arbitrary quasi-psh $u$ gave
\[
 \theta_\pm+\ddc u\geq\delta(\kappa+w),\qquad\delta>0.
\]
For every smooth nonnegative base top-degree form $\eta$, its pullback is $\ddc$-closed by dimension. Distributional Stokes therefore gives
\[
 \int_B(L_\pm-\delta)\eta\geq0.
\]
But $L_\pm$ vanishes somewhere. A nonzero $\eta$ supported where $L_\pm<\delta$ contradicts this inequality. Thus no strict current representative exists; no singular potential has been restricted to a fiber.
\end{proof}

\subsubsection{Projective and torus examples}

\begin{example}\label{ex:warped-projective-strip}
On $B=\mathbb P^2$, take normalized Fubini--Study $\kappa$, $\sigma=\kappa^2$, $C=\mathbb P^1$, and
\[
 a([Z])=\frac{|Z_1|^2/2+|Z_2|^2}
                  {|Z_0|^2+|Z_1|^2+|Z_2|^2}.
\]
Then $M=1$, $r_0=r_M=2$, and the completion is a closed strip with finite faces
\[
 b\kappa+\frac{4\log2}{3b^2}a w,\qquad
 b\kappa+\frac{4\log2}{3b^2}(1-a)w.
\]
For $v\ne0$,
\[
 Q(v)=\frac4{v^2}\bigl[(v+1)\log(1+v)
                      -(v+2)\log(1+v/2)\bigr],
 \qquad Q(0)=1.
\]
Its width is the positive finite integral
\[
 W=\int_{-1}^{\infty}
       \sqrt{\frac{Q''/Q-2(Q'/Q)^2}{3}}\,\mathrm{d}v.
\]
\end{example}
\begin{proof}
The extrema are the points $[1:0:0]$ and $[0:0:1]$. Their affine expressions have positive quadratic normal behavior, so both real codimensions are four and both orders are two. To compute the law of $a$, polar integration in the $Z_0$ chart gives $2\,\mathrm{d}s_1\mathrm{d}s_2/(1+s_1+s_2)^3$, where $s_j=|z_j|^2$. The transformation $u_j=s_j/(1+s_1+s_2)$ gives uniform density two on $u_1,u_2\geq0$, $u_1+u_2\leq1$. For $0\leq s\leq1/2$, the triangle $u_1/2+u_2\leq s$ has area $s^2$. Thus the law has density $4s$ on $[0,1/2]$; exchanging $u_0$ and $u_2$ gives density $4(1-s)$ on $[1/2,1]$. Integrating these two densities gives $H_a=H_-=4\log2$ and the displayed $Q$. Its integral definition proves the removable extension at zero. The completion theorem gives the faces and finite width, with Euclidean coordinates $Y=\sqrt2\log b$ and $R(v)$.
\end{proof}

\begin{example}\label{ex:warped-torus-plane}
Let $E=\mathbb C/(\mathbb Z+i\mathbb Z)$, $B=E^m$, and write $z_j=u_j+i\tau_j$. Put
\[
 \eta_j=\frac i2\mathrm{d}z_j\wedge\mathrm{d}\bar z_j,
 \qquad s=(m!)^{-1/m},\qquad
 \kappa=s\sum_j\eta_j,\qquad\sigma=\kappa^m.
\]
For $a=\sin^2(\pi u_1)$, the unit-volume hypersurface is already a complete Euclidean plane. Its exact coordinates are
\[
 Y=\sqrt m\log b,\qquad
 R=\frac{\log(1+v)}{2\sqrt n},\qquad v>-1.
\]
Its minimizing balanced metrics can be constructed by periodic integration.
\end{example}
\begin{proof}
The forms descend and $\sigma=\prod_j\eta_j$ is a probability. Both extreme sets are real hypersurfaces with quadratic normal behavior, so both ratios are $1/2$. Splitting the period and substituting $t=\tan(\pi u)$ gives
\begin{align*}
 Q(v)&=\int_0^1\frac{\mathrm{d}u}{1+v\sin^2(\pi u)}\\
     &=\frac2\pi\int_0^\infty\frac{\mathrm{d}t}{1+(1+v)t^2}
      =\frac1{\sqrt{1+v}}.
\end{align*}
Let $F_v(u)=[Q(v)(1+v\sin^2(\pi u))]^{-1}$, which has period mean one. Twice integrating the zero-mean equation
\[
 \phi_v''=4\pi bs(F_v-1)
\]
gives a smooth periodic potential: choose the first integration constant to make its derivative have mean zero. Our convention gives $\ddc\phi_v=\phi_v''\eta_1/(4\pi)$, hence
\[
 A=bs\left(F_v\eta_1+\sum_{j=2}^m\eta_j\right)>0,
 \qquad A^m=b^mF_v\sigma=\frac{b^m}{H}\frac{\sigma}{L}.
\]
Thus \eqref{eq:warped-balanced-cofactor} constructs the actual minimizer without an existence theorem in this example. Differentiating $Q$ gives
\[
 \rho(v)=\frac1{4n(1+v)^2},\qquad
 g_{\mathscr S}=m\,\mathrm{d}U^2+
             \frac1{4n}\left(\frac{\mathrm{d}v}{1+v}\right)^2.
\]
The displayed $(Y,R)$ range over all of $\mathbb R^2$. Both coefficient faces therefore have infinite distance on the unit surface.
\end{proof}

The fixed cone and product probability determine these distances. The finite disjoint power-stratum hypotheses provide the endpoint criterion; intersecting extreme strata and infinite-order vanishing require further estimates. The trace variances describe actual minimizing tests, without asserting a pointwise limit of their metrics at a finite end.
