\section{A convex functional on probability measures}\label{sec:normalization-dual}

To pass to the limit in the supporting inequality, we need to control the mass that supporting probabilities can put on small sets. The convex dual of $B_\varphi$ provides this control: the measures in each of its sublevel sets are uniformly absolutely continuous. This allows the logarithmic density and the supporting measure to vary simultaneously.

Let $\mathcal P(X)$ denote the set of Borel probability measures on $X$, with the weak topology. Define
\index{Normalization scalar!convex dual}
\begin{equation}\label{eq:normalization-dual-definition}
 H_\varphi(\nu)
 =\sup_{G\in C(X)}\left(B_\varphi(G)+\int_XG\,\mathrm{d}\nu\right),
 \qquad \nu\in\mathcal P(X).
\end{equation}
The value $+\infty$ is allowed, while taking $G=0$ shows that
$H_\varphi\geq B_\varphi(0)>-\infty$.

\begin{proposition}\label{prop:normalization-dual}
The functional $H_\varphi$ is convex and weakly lower semicontinuous. For every $F\in C(X)$,
\begin{equation}\label{eq:normalization-dual-minimum}
 B_\varphi(F)
 =\min_{\nu\in\mathcal P(X)}
       \left(H_\varphi(\nu)-\int_XF\,\mathrm{d}\nu\right).
\end{equation}
The minimizers are exactly $\mathcal A_\varphi(F)$.
\end{proposition}
\begin{proof}
Each function of $\nu$ in the supremum defining $H_\varphi$ is affine and weakly continuous, so their supremum is convex and lower semicontinuous. By definition,
$H_\varphi(\nu)-\int_XF\,\mathrm{d}\nu\geq B_\varphi(F)$, with equality precisely when
\[
 B_\varphi(G)+\int_XG\,\mathrm{d}\nu
 \leq B_\varphi(F)+\int_XF\,\mathrm{d}\nu
 \quad\text{for every }G\in C(X).
\]
This is the defining inequality of $\mathcal A_\varphi(F)$, and the minimum is attained because $\mathcal A_\varphi(F)$ is nonempty by \cref{thm:weak-adjoint-probabilities}.
\end{proof}

\Needspace{14\baselineskip}
\begin{theorem}\label{thm:normalization-dual-compactness}
For every finite real $L$, the set
\[
 \mathcal K_L=\{\nu\in\mathcal P(X):H_\varphi(\nu)\leq L\}
\]
is weakly compact. Its measures are uniformly absolutely continuous with respect to $\mathrm{d}V$: for every $\epsilon>0$ there is a $\delta>0$ such that
\begin{equation}\label{eq:normalization-dual-uniform-ac}
 \mathrm{d}V(E)<\delta\quad\Longrightarrow\quad
 \sup_{\nu\in\mathcal K_L}\nu(E)<\epsilon
\end{equation}
for every Borel set $E$. In particular, writing $\mathrm{d}\nu=h_\nu \mathrm{d}V$, the densities in this set are uniformly integrable in $L^1(\mathrm{d}V)$. On $\mathcal K_L$, weak convergence of measures also implies weak convergence of their densities in $L^1(\mathrm{d}V)$.
\end{theorem}
\Needspace{9\baselineskip}
\begin{proof}
Weak compactness follows from lower semicontinuity and compactness of $\mathcal P(X)$. We may assume that the sublevel set is nonempty, so that $L\geq B_\varphi(0)$.
Fix $t>0$. By \cref{thm:normalization-bounded-lone}, there is $\delta_t>0$ such that
\[
 0\leq\chi\leq1,\quad \chi\in C(X),\quad
 \int_X\chi\,\mathrm{d}V<\delta_t
 \quad\Longrightarrow\quad
 B_\varphi(t\chi)>B_\varphi(0)-1.
\]
Indeed, otherwise there would be a sequence of such functions $\chi$ tending to zero in $L^1$ with $B_\varphi(t\chi)\leq B_\varphi(0)-1$, contrary to that theorem. For $\nu\in\mathcal K_L$, the defining supremum gives
\[
 t\int_X\chi\,\mathrm{d}\nu
 \leq L-B_\varphi(t\chi)
 <L-B_\varphi(0)+1.
\]
Choose $t$ so large that $(L-B_\varphi(0)+1)/t<\epsilon/2$. If $\mathrm{d}V(E)<\delta_t/2$, outer regularity gives an open neighborhood $O$ of $E$ with $\mathrm{d}V(O)<\delta_t$. For any compact $K\subset E$, choose a continuous cutoff $0\leq\chi\leq1$ equal to one on $K$ and supported in $O$. The last inequality gives $\nu(K)<\epsilon/2$, uniformly in $\nu\in\mathcal K_L$, and inner regularity yields $\nu(E)\leq\epsilon/2<\epsilon$. This proves \eqref{eq:normalization-dual-uniform-ac}.

In particular, every measure in the sublevel set is absolutely continuous, with a density of integral one. Together with the uniform small-set estimate, this is exactly uniform integrability on the finite measure space $(X,\mathrm{d}V)$.

Finally, suppose that $\nu_j\to\nu$ weakly within the sublevel set. For a bounded measurable $G$, choose continuous $G_k$ with
$|G_k|\leq\|G\|_\infty$ and $G_k\to G$ in $\mathrm{d}V$-measure; such approximations exist by Lusin's theorem. Uniform absolute continuity gives
\[
 \sup_{\sigma\in\mathcal K_L}\int_X|G_k-G|\,\mathrm{d}\sigma\longrightarrow0.
\]
To see this, split $X$ into $\{|G_k-G|\leq a\}$ and its complement; the integral over the first set is at most $a$, and the integral over the second is at most $2\|G\|_\infty$ times a uniformly small measure. For fixed $k$, weak convergence applies to $G_k$. Letting first $j\to\infty$ and then $k\to\infty$ proves convergence of the integrals against $G$, which is exactly weak $L^1(\mathrm{d}V)$ convergence of the densities. In particular, every sequence in the sublevel set has a subsequence whose densities converge weakly in $L^1$.
\end{proof}

\section{Bounded measurable densities and stability}

The supporting probabilities of continuous approximations of a bounded measurable logarithmic density lie in one sublevel set of $H_\varphi$. Uniform integrability then allows both the density and the supporting measure to vary when we pass to the limit in the supporting inequality.

\Needspace{18\baselineskip}
\begin{theorem}\label{thm:normalization-measurable-adjoint}
For every $F\in L^\infty(\mathrm{d}V)$ there is a nonempty set of absolutely continuous probability measures, again denoted $\mathcal A_\varphi(F)$, characterized by
\begin{equation}\label{eq:measurable-adjoint-support}
 B_\varphi(G)\leq B_\varphi(F)-\int_X(G-F)\,\mathrm{d}\nu
 \qquad\text{for every }G\in L^\infty(\mathrm{d}V).
\end{equation}
For continuous $F$, this set agrees with the one defined by \eqref{eq:adjoint-probability-set}. Formula \eqref{eq:normalization-dual-minimum} also holds for bounded measurable $F$, with these measures as its minimizers; there the objective is interpreted as $+\infty$ whenever $H_\varphi(\nu)=+\infty$, and otherwise $\nu$ is absolutely continuous, so that the integral of the $L^\infty$ class $F$ is well defined.

Suppose that $F_j,F\in L^\infty(\mathrm{d}V)$ are uniformly bounded and $F_j\to F$ in $L^1(\mathrm{d}V)$, and let $\nu_j\in\mathcal A_\varphi(F_j)$. Then the densities of the $\nu_j$ are uniformly integrable, and every weak $L^1$ cluster point belongs to $\mathcal A_\varphi(F)$. If the latter is a singleton, the whole sequence converges to it weakly in $L^1(\mathrm{d}V)$.
\end{theorem}
\begin{proof}
First, if $H_\varphi(\nu)<\infty$, the measure $\nu$ is absolutely continuous by \cref{thm:normalization-dual-compactness}. A bounded measurable $G$ admits uniformly bounded continuous approximations $G_k\to G$ in $L^1(\mathrm{d}V)$, and absolute continuity together with the common bound gives $\int_XG_k\,\mathrm{d}\nu\to\int_XG\,\mathrm{d}\nu$. Combined with \cref{thm:normalization-bounded-lone}, this proves
\begin{equation}\label{eq:dual-measurable-tests}
 H_\varphi(\nu)
 =\sup_{G\in L^\infty(\mathrm{d}V)}
       \left(B_\varphi(G)+\int_XG\,\mathrm{d}\nu\right)
\end{equation}
for such $\nu$.

To construct supporting probabilities at $F$, choose continuous $F_j$ with $|F_j|\leq M=\|F\|_\infty$ and $F_j\to F$ in $L^1(\mathrm{d}V)$, and choose $\nu_j\in\mathcal A_\varphi(F_j)$. The minimum formula of \cref{prop:normalization-dual} and the Lipschitz bound give
\[
 H_\varphi(\nu_j)
 =B_\varphi(F_j)+\int_XF_j\,\mathrm{d}\nu_j
 \leq B_\varphi(0)+2M.
\]
By \cref{thm:normalization-dual-compactness}, a subsequence converges weakly in $L^1$ to a probability measure $\nu$. Uniform absolute continuity gives
\[
 \int_X|F_j-F|\,\mathrm{d}\nu_j\longrightarrow0,
\]
by splitting again at a fixed small positive threshold: the sets where $|F_j-F|$ exceeds it have $\mathrm{d}V$-measure tending to zero. Since $F$ is bounded and measurable, weak $L^1$ convergence applies to its integral. Lower semicontinuity of $H_\varphi$ and continuity of $B_\varphi$ yield
\[
 H_\varphi(\nu)
 \leq\liminf_j H_\varphi(\nu_j)
 =B_\varphi(F)+\int_XF\,\mathrm{d}\nu.
\]
The reverse inequality follows from \eqref{eq:dual-measurable-tests} with $G=F$. Equality, combined with the same supremum formula, proves \eqref{eq:measurable-adjoint-support} for $\nu$ and the minimum formula.

For continuous $F$, every measure satisfying \eqref{eq:adjoint-probability-set} has finite dual value by \cref{prop:normalization-dual}, so \eqref{eq:dual-measurable-tests} extends its supporting inequality to all bounded measurable tests; the converse inclusion is trivial. Thus the two definitions agree. Finally, an absolutely continuous probability satisfying \eqref{eq:measurable-adjoint-support} has finite dual value and attains the minimum, so the characterization is exact.

For the stability assertion, we start from the given $F_j$ and $\nu_j$ instead of an approximating construction. The common uniform bound again places all $\nu_j$ in a single finite sublevel set of $H_\varphi$, and the same compactness and limiting argument shows that every weak $L^1$ cluster point attains the minimum for $F$. If the set of minimizers is a singleton, compactness forces the whole sequence to converge to its member.
\end{proof}

\section{Entropy and the closed case}\label{sec:normalization-entropy}

In a closed background on a compact K\"ahler manifold, all potentials of an ordinary singularity type have the same mass, and integrating the equation determines the scalar explicitly; its convex dual is then the relative entropy. The variational formula for the entropy also gives estimates in nonclosed backgrounds.

For a Borel probability measure $\nu$ absolutely continuous with respect to $\mathrm{d}V$, let $h$ be its nonnegative density and define the \emph{relative entropy}
\[
 \operatorname{Ent}_{\mathrm{d}V}(\nu)=\int_Xh\log h\,\mathrm{d}V,
 \qquad 0\log0=0,
\]
and set $\operatorname{Ent}_{\mathrm{d}V}(\nu)=+\infty$ if $\nu$ is not absolutely continuous. The integral is well defined in $(-\infty,+\infty]$, since $h\log h\geq-\mathrm{e}^{-1}$. By Jensen's inequality it is at least $-\log\mathrm{d}V(X)$, so it can be negative only when $\mathrm{d}V(X)>1$.

Suppose now that $X$ is K\"ahler and $\theta$ is closed. By the closed-background theory of \cite{SGPT}, all potentials of the ordinary type of $\varphi$ have the same non-pluripolar mass $m_\varphi$, which is positive under the standing hypotheses. For $F\in L^\infty(\mathrm{d}V)$, integrating the equation gives
\begin{equation}\label{eq:normalization-closed-formula}
 B_\varphi(F)=\log m_\varphi-\log\int_X\mathrm{e}^F\,\mathrm{d}V,
 \qquad
 \mathcal A_\varphi(F)
 =\left\{\frac{\mathrm{e}^F\,\mathrm{d}V}{\int_X\mathrm{e}^F\,\mathrm{d}V}\right\}.
\end{equation}
For the second identity, write $\nu_F$ for the normalized exponential measure. By the first identity, the supporting inequality \eqref{eq:measurable-adjoint-support} for $\nu_F$ and $G\in L^\infty(\mathrm{d}V)$ is Jensen's inequality $\log\int_X\mathrm{e}^{G-F}\,\mathrm{d}\nu_F\geq\int_X(G-F)\,\mathrm{d}\nu_F$. Conversely, testing the supporting inequality of any $\nu\in\mathcal A_\varphi(F)$ with $G=F\pm th$, $h\in L^\infty(\mathrm{d}V)$, and letting $t\downarrow0$ gives $\int_Xh\,\mathrm{d}\nu=\int_Xh\,\mathrm{d}\nu_F$, so $\nu=\nu_F$.

\begin{proposition}\label{prop:normalization-dual-entropy}
If $X$ is K\"ahler and $\theta$ is closed, then
\[
 H_\varphi(\nu)=\log m_\varphi+\operatorname{Ent}_{\mathrm{d}V}(\nu).
\]
Thus the minimum formula \eqref{eq:normalization-dual-minimum} becomes the usual variational formula for the logarithmic integral in \eqref{eq:normalization-closed-formula}.
\end{proposition}
\begin{proof}
Substitute \eqref{eq:normalization-closed-formula} into the defining supremum and apply \cref{lem:entropy-variational}.
\end{proof}

The dual functional depends on the reference volume. If $\mathrm{d}V$ is replaced by $\mathrm{d}V'=\mathrm{e}^a \mathrm{d}V$ for a continuous real $a$, the corresponding functionals satisfy
\[
 B'_\varphi(F)=B_\varphi(F+a),\qquad
 H'_\varphi(\nu)=H_\varphi(\nu)-\int_Xa\,\mathrm{d}\nu.
\]
The first identity follows from the equation, and the second, which also holds when the values are infinite, from the change of variable $G\mapsto G+a$ in the defining supremum. In a nonclosed background, $H_\varphi$ is still defined by the same supremum, but the total mass need not be constant on an ordinary type, so the entropy computation no longer determines it.
