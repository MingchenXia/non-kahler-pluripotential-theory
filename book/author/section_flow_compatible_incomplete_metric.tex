\subsection{An incomplete metric with compatible flow data}
\label{sec:flow-compatible-incomplete-metric}

Exponential equations of this kind arise as auxiliary elliptic equations in the smoothing of singular initial data by geometric flows; the completeness question for them is discussed in \cref{sec:history-completeness}. In that setting the class, the divisor and the volume form are not arbitrary: they are determined by the initial potential through a log resolution. The following example shows that incompleteness persists under these compatibility conditions.

Let $(X,\omega)$ be a compact K\"ahler manifold and $\phi_0\in\PSH(X,\omega)$ an initial potential with analytic singularities. \emph{Compatible data} on a log resolution $\pi:Y\to X$ satisfy
\begin{gather}
 \pi^*\phi_0=\sum_jm_j\log|\sigma_j|^2+\ell_0,\qquad
 \pi^*(\omega+\ddc\phi_0)=\sum_jm_j[D_j]+R_0,
 \label{eq:compatible-initial-resolution}\\
 K_Y=\pi^*K_X+\sum_jb_jD_j,\qquad
 \pi^*\mathrm{d}V_X=\prod_j|\sigma_j|^{2b_j}\mathrm{d}V_Y.
 \label{eq:compatible-volume-resolution}
\end{gather}
Here $D=\sum_jD_j$ is a divisor with simple normal crossings, $\sigma_j$ is a defining section of $\mathcal O_Y(D_j)$ with a smooth Hermitian metric, the numbers $m_j$ and $b_j$ are the coefficients of the pulled-back singularity and of the relative canonical divisor, $\ell_0$ is smooth, $R_0$ is a smooth closed semipositive form, and $\mathrm{d}V_X$, $\mathrm{d}V_Y$ are smooth positive volume forms. If $\Theta_j$ denotes the curvature form of the metric on $\mathcal O_Y(D_j)$, put
\[
 \lambda=\max_j\frac{m_j}{1+b_j},\qquad
 \Theta=\sum_j(1+b_j)\Theta_j,\qquad
 \theta=\lambda\Theta+R_0,\qquad
 |\sigma|^2=\prod_j|\sigma_j|^2.
\]
The number $\lambda$ is the inverse of the exponential integrability index of $\phi_0$: by \eqref{eq:compatible-initial-resolution} and \eqref{eq:compatible-volume-resolution}, $\mathrm{e}^{-q\phi_0}$ is integrable exactly when $q\lambda<1$. The metric in the next theorem is the solution of the auxiliary elliptic equation \eqref{eq:compatible-incomplete-equation} in this background.

\Needspace{18\baselineskip}
\begin{theorem}\label{thm:flow-compatible-incomplete-metric}\index{Metric completeness!compatible initial data}
There are compatible data \eqref{eq:compatible-initial-resolution}--\eqref{eq:compatible-volume-resolution} on smooth projective surfaces, with $\mathrm{d}V_X=\omega^2$, $R_0$ K\"ahler and $D$ a disjoint union of two smooth curves, such that the equation
\begin{equation}\label{eq:compatible-incomplete-equation}
 \langle(\theta+\ddc\rho)^2\rangle
 =\mathrm{e}^{\rho}|\sigma|^{-2}\,\mathrm{d}V_Y
\end{equation}
has a unique solution in $\mathcal E^1(Y,\theta)$. The solution has the Lelong numbers of a potential with minimal singularities, but every dense Zariski open set on which its current is smooth and strictly positive is incomplete. The measure $|\sigma|^{-2}\mathrm{d}V_Y$ has a nonintegrable pole with smooth strictly positive coefficient along both components of $D$.
\end{theorem}

The class $\{\theta\}$ in this example is not nef. Thus this theorem and \cref{thm:nef-incomplete-cusp} show separately that neither the compatibility identities nor nefness of the class implies completeness. We do not know whether completeness holds when the class is nef and the compatibility identities hold. In neither example is the auxiliary metric identified with the metric of the maximal flow.

The cusp in the example comes from a rotation-invariant solution of an exponential equation on $\mathbb P^1$ whose measure has a simple pole.

\Needspace{12\baselineskip}
\begin{lemma}\label{lem:fixed-volume-curve-cusp}
On $\mathbb P^1$, let $z=Z_1/Z_0$, $r=|z|^2$, $\alpha=\ddc\log(1+r)$ and $h=r/(1+r)$, so that $\int_{\mathbb P^1}\alpha=1$. For every $c>2$ there is a $c\alpha$-plurisubharmonic function $u$, depending only on $|z|$ and smooth off $0$, such that
\begin{equation}\label{eq:fixed-volume-curve-cusp}
 g_1:=c\alpha+\ddc u=\mathrm{e}^{u}h^{-1}c\alpha.
\end{equation}
It has zero Lelong number at $0$, mass $\int g_1=c$, and
\[
 \int_{\mathbb P^1}|u|\alpha+\int_{\mathbb P^1}|u|g_1<\infty.
\]
Moreover, with $t=\log r$ and $s=-\log r$, there is $A\in\R$ such that, as $s\to\infty$,
\[
 \begin{aligned}
 u&=\log(2/c)-2\log(s+A)+O(\mathrm{e}^{-s}),\\
 u_t&=\frac{2}{s+A}+O(\mathrm{e}^{-s}),\qquad
 u_{tt}=\frac{2}{(s+A)^2}+O(\mathrm{e}^{-s}).
 \end{aligned}
\]
\end{lemma}

\begin{proof}
\emph{Barriers and the radial inverse.}
Set $L=1-\log h=1+\log(1+1/r)$ and $w=-2\log L$. Differentiation gives
\[
 \ddc w=\left(-\frac2L+\frac2{rL^2}\right)\alpha,\qquad
 Q(r):=\frac{c\alpha+\ddc w}{\mathrm{e}^{w}h^{-1}c\alpha}
 =\frac{crL^2-2rL+2}{c(1+r)}.
\]
The function $Q$ is positive, since $L\geq1$ and $c>2$. Its limits are $2/c$ at $0$ and $1-2/c$ at infinity, so $0<q_-\leq Q\leq q_+<\infty$ for some constants $q_\pm$. Choose $a_0\leq\min(0,\log q_-)$ and $b_0\geq\max(0,\log q_+)$. Then $l=w+a_0$ and $H=w+b_0$ satisfy
\[
 c\alpha+\ddc l\geq\mathrm{e}^{l}h^{-1}c\alpha,\qquad
 c\alpha+\ddc H\leq\mathrm{e}^{H}h^{-1}c\alpha,\qquad l\leq w\leq H.
\]

Exhaust $\mathbb P^1\setminus\{0\}$ by the disks $|\zeta|<R$ in the coordinate $\zeta=1/z$, and solve the equation on them with boundary value $w$. For $q=|\zeta|^2$, the equation becomes
\[
 \Delta u=f(q)\mathrm{e}^{u}-g(q),\qquad
 f(q)=\frac{4c}{1+q},\qquad g(q)=\frac{4c}{(1+q)^2},
\]
where $\Delta$ is the Euclidean real Laplacian.

For $k>0$, the regular radial solution $v=V(q)$ of $(-\Delta+k)v=F(q)$ satisfies
\begin{equation}\label{eq:radial-cusp-volterra}
 V(q)=A_0+\frac14\int_0^q\log(q/t)\bigl(kV(t)-F(t)\bigr)\,\mathrm{d}t.
\end{equation}
The $j$-fold iterate of the integral kernel applied to the constant $1$ is $q^j/(4^j(j!)^2)$, so successive approximation converges uniformly on every finite interval. The coefficient of $A_0$ in the solution is $\sum_{j\geq0}(kq/4)^j/(j!)^2>0$, so exactly one value of $A_0$ gives any prescribed value at $q=R^2$. Moreover,
\[
 V'(q)=\frac14\int_0^1\bigl(kV(qt)-F(qt)\bigr)\,\mathrm{d}t,
\]
and this formula gives smoothness at $q=0$ by induction. The maximum principle for $-\Delta+k$ shows that this radial inverse is unique and preserves order.

Choose $k\geq\sup f\mathrm{e}^{H}$ on the fixed disk, and let $T(v)$ be the radial solution of $(-\Delta+k)T(v)=kv-f\mathrm{e}^{v}+g$ with boundary value $w$; the right side is nondecreasing in $v$ on $[l,H]$. The barrier inequalities give $l\leq T(l)$ and $T(H)\leq H$, so the iterates of $T$ starting from $l$ increase and stay below $H$. The derivative formula gives a uniform bound for their first derivatives on the fixed interval. Compactness and monotonicity give uniform convergence; passing to the limit in \eqref{eq:radial-cusp-volterra} proves the equation, and the averaged derivative formula bootstraps the limit to smoothness.

The same barriers bound the solutions on every fixed disk in $\zeta$ as $R\to\infty$. For an exact solution $U(q)$ of the nonlinear equation,
\[
 U'(q)=\frac14\int_0^1\bigl(f(qt)\mathrm{e}^{U(qt)}-g(qt)\bigr)\,\mathrm{d}t,
\]
and induction gives bounds for all derivatives on compact intervals. A diagonal limit is therefore smooth on the $\zeta$-plane, including the point $z=\infty$, and satisfies $u=-2\log(-\log r)+O(1)$ at $0$.

\emph{The asymptotics, mass and energy.}
In the variable $t=\log r$, the equation reads
\[
 u_{tt}=\frac{c\mathrm{e}^{u}}{1+\mathrm{e}^t}
              -\frac{c\mathrm{e}^t}{(1+\mathrm{e}^t)^2}.
\]
Put $v(t)=u(t)+c\log(1+\mathrm{e}^t)$ and $V(s)=v(-s)$. Then
\[
 V''=k_0(s)\mathrm{e}^{V},\qquad
 k_0(s)=c(1+\mathrm{e}^{-s})^{-(c+1)},\qquad
 V=-2\log s+O(1).
\]
Convexity and this sublinear decay imply $V'<0$ and $V'\to0$ at infinity. The function $J=(V')^2-2k_0\mathrm{e}^{V}$ satisfies $J'=-2k_0'\mathrm{e}^{V}$ and tends to zero. Since $k_0'=O(\mathrm{e}^{-s})$ and $\mathrm{e}^{V}\asymp s^{-2}$,
\[
 J(s)=2\int_s^\infty k_0'(\tau)\mathrm{e}^{V(\tau)}\,\mathrm{d}\tau
       =O(\mathrm{e}^{-s}s^{-2}).
\]
Thus $(\mathrm{e}^{-V/2})'=\sqrt{c/2}(1+O(\mathrm{e}^{-s}))$, and integration gives $\mathrm{e}^{-V/2}=\sqrt{c/2}(s+A)+O(\mathrm{e}^{-s})$. This identity and the differential equation give all three asymptotic formulas.

The local potential $u+c\log(1+r)$ extends subharmonically across $0$ by \cref{lem:qpsh-removable-analytic}. Its radial slope tends to zero at the puncture, so the extension adds no point mass. The density of $g_1$ is asymptotic to
\[
 \frac{\mathrm{i}}{2\pi}\frac{2}{r s^2}
          \,\mathrm{d}z\wedge\mathrm{d}\bar z.
\]
Stokes' theorem for currents gives mass $c$. The energy integral at the cusp is bounded by $C\int^\infty(\log s)/s^2\,\mathrm{d}s$, and the integral against the smooth form $\alpha$ is finite as well.
\end{proof}

\Needspace{12\baselineskip}
\begin{proof}[Proof of \cref{thm:flow-compatible-incomplete-metric}]
\emph{The initial data and the resolution.}
Take $X=\mathbb P^1_Z\times\mathbb P^1_W$, with the pulled-back normalized Fubini--Study forms $\alpha,\beta$ of the two factors, and put
\[
 \omega=6(\alpha+\beta),\qquad \mathrm{d}V_X=\omega^2=72\alpha\wedge\beta.
\]
Let $H=(Z_1=0)$ and $p=([0:1],[1:0])=(\infty,0)$. Define
\[
 \begin{gathered}
 h_H=\frac{|Z_1|^2}{|Z_0|^2+|Z_1|^2},\qquad
 \eta=\frac{|Z_0|^2}{|Z_0|^2+|Z_1|^2}
              +\frac{|W_1|^2}{|W_0|^2+|W_1|^2},\\
 \phi_0=2\log h_H+2\log\eta.
 \end{gathered}
\]
The zero sets of $h_H$ and $\eta$ are $H$ and $\{p\}$, which are disjoint. Let $\pi:Y\to X$ be the blowup at $p$, write $E$ for the exceptional curve, and set $H_Y=\pi^{-1}H$ and $D=H_Y+E$. The surface $Y$ is smooth, compact, connected and projective, and $\pi$ is an isomorphism off $p$.

Use $|\sigma_H|^2=\pi^*h_H$ and $|\sigma_E|^2=\pi^*\eta$. Near $p$, put $x=Z_0/Z_1$ and $y=W_1/W_0$. In the blowup chart $x=a$, $y=ab$,
\[
 \pi^*\eta=|a|^2A(a,b),\qquad
 A(a,b)=\frac1{1+|a|^2}+\frac{|b|^2}{1+|ab|^2}>0.
\]
Thus $\pi^*\eta$ defines a smooth Hermitian metric on $\mathcal O_Y(E)$ across $E$. The second chart is obtained by interchanging $x$ and $y$, and the metric factors satisfy the transition rule.

The sections $Z_0W_0$, $\sqrt2Z_0W_1$, $Z_1W_1$ of $\mathcal O(1,1)$ have summed squared Fubini--Study norm $\eta$, and near $p$ they generate the ideal $(x,\sqrt2xy,y)=(x,y)$. Dividing their pullbacks by $\sigma_E$ gives a morphism $F:Y\to\mathbb P^2$ with
\begin{equation}\label{eq:compatible-curvature-map}
 F^*\gamma=\pi^*(\alpha+\beta)-\Theta_E,\qquad
 \Theta_H=\pi^*\alpha,
\end{equation}
where $\gamma$ is the normalized Fubini--Study form on $\mathbb P^2$. In the first chart $F=[1:\sqrt2ab:b]$, so $F|_E$ is an isomorphism onto a line.

Near $p$,
\[
 \eta=\frac{|x|^2+|y|^2+2|xy|^2}{(1+|x|^2)(1+|y|^2)}.
\]
Thus $\phi_0$ has analytic singularities. In the convention $\frac1{2m}\log\sum|f_j|^2$, the coefficient $2$ is obtained by raising the numerator to the fourth power and expanding it as a sum of squared holomorphic monomials. The resolved identities are
\begin{align}\label{eq:compatible-kahler-residual}
 \pi^*\phi_0&=2\log|\sigma_H|^2+2\log|\sigma_E|^2,\\
 R_0&=\pi^*\omega-2\Theta_H-2\Theta_E
       =2\pi^*\alpha+4\pi^*\beta+2F^*\gamma.\notag
\end{align}
The first two terms of $R_0$ are positive away from $E$, and along $E$ their kernel is $TE$, on which $F^*\gamma$ is positive. Hence $R_0$ is K\"ahler. The resolved current is $2[H_Y]+2[E]+R_0$, which shows that $\phi_0$ is $\omega$-plurisubharmonic away from $p$; since it is bounded above, it extends across $p$. In the notation of \eqref{eq:compatible-initial-resolution}, $m_H=m_E=2$ and $\ell_0=0$.

The Jacobian determinant of the blowup in the chart is $a$, so $b_E=1$ and $b_H=0$. Directly, $\mathrm{e}^{-q\phi_0}$ is integrable near $H$ exactly when $q<1/2$ and near $p$ exactly when $q<1$. Consequently $\lambda=\max\{2,1\}=2$, and
\begin{equation}\label{eq:compatible-auxiliary-background}
 \Theta=\Theta_H+2\Theta_E,\qquad
 \theta=2\Theta+R_0=\pi^*\omega+2\Theta_E.
\end{equation}
Since $\Theta_E|_E=-\omega_{\mathrm{FS}}$, the class $\{\theta\}$ has degree $-2$ on $E$, so it is not nef. It is big, since the potential $3\log|\sigma_E|^2$ gives the current
\[
 (\pi^*\omega-\Theta_E)+3[E],\qquad
 \pi^*\omega-\Theta_E=5\pi^*(\alpha+\beta)+F^*\gamma>0.
\]

\emph{The prescribed volume.}
Define $\mathrm{d}V_Y=\pi^*\omega^2/|\sigma_E|^2$ off $E$. In the chart above,
\[
 \mathrm{d}V_Y=\frac{72\,\mathrm{d}A_a\mathrm{d}A_b}
 {\pi^2A(a,b)(1+|a|^2)^2(1+|ab|^2)^2},
\]
since the factor $|a|^2$ from the Jacobian cancels exactly. The second chart gives the extension on the rest of $E$, and the extensions agree on overlaps. Thus $\mathrm{d}V_Y$ is a smooth strictly positive volume form with $\pi^*\mathrm{d}V_X=|\sigma_E|^2\mathrm{d}V_Y$, as required in \eqref{eq:compatible-volume-resolution}. Since $H_Y$ and $E$ are disjoint, the measure $|\sigma_H\sigma_E|^{-2}\mathrm{d}V_Y$ has a nonintegrable pole with smooth positive coefficient along each component.

\emph{The solution and the minimal reference.}
Take $u$ from \cref{lem:fixed-volume-curve-cusp} with $c=6$, pulled back to $X$ from the first factor, and put $g_X=(6\alpha+\ddc u)+6\beta$. This form is smooth and strictly positive near $p$, and
\[
 g_X^2=\mathrm{e}^{u}h_H^{-1}\omega^2.
\]
Set $P=2\log|\sigma_E|^2$ and $\rho=P+\pi^*u$. Then
\[
 \theta+\ddc P=2[E]+\pi^*\omega,\qquad
 \theta+\ddc\rho=2[E]+\pi^*g_X.
\]
Off $D$, the equation follows from the identity
\[
 \mathrm{e}^{\rho}|\sigma_H\sigma_E|^{-2}\mathrm{d}V_Y
 =\mathrm{e}^{\pi^*u}|\sigma_H|^{-2}|\sigma_E|^2\mathrm{d}V_Y
 =\pi^*g_X^2.
\]
Both measures extend by zero on $D$, which proves the global non-pluripolar equation.

Let $\psi$ be $\theta$-plurisubharmonic. Off $E$, the function $\psi-P$ descends to an $\omega$-plurisubharmonic function on $X\setminus\{p\}$, which by \cref{lem:isolated-puncture-extension} extends across $p$ and is bounded above on $X$. Thus $\psi\leq P+C_\psi$ off $E$, and \cref{lem:qpsh-almost-everywhere-order} gives the inequality everywhere. Since $P$ itself is $\theta$-plurisubharmonic, it has minimal singularities.

The current of $P$ has non-pluripolar mass $\int_X\omega^2=72$, which is therefore the volume of the class, whereas the cup-product square of the class is $72+4E^2=68$. The cusp has mass $6$ and no atom, so the current of $\rho$ also has mass $\int_Xg_X^2=72$, and $\rho$ has full mass. Near $E$, $\pi^*u$ is smooth, so $\rho$ has the Lelong numbers of $P$; along $H_Y$ the cusp has zero Lelong number. The solution $\rho$ does not itself have minimal singularities, because $u$ is unbounded below near $H$.

\emph{Relative energy and uniqueness.}
Choose $M$ with $u-M\leq0$ and use the potential $P+M$, which has minimal singularities, as reference. The canonical approximants of $\rho$ are
\[
 \rho_k=\max(\rho,P+M-k)=P+M+\pi^*u_k,\qquad u_k=\max(u-M,-k),
\]
and their difference from the reference off $E$ is $\pi^*u_k$. On the curve let $b(t)=6\log(1+\mathrm{e}^t)$ and $V(t)=b(t)+u(t)$. For large $k$ there is a unique level $t=T_k$ in the cusp with $u(T_k)-M=-k$. If $S_k=-T_k$, the asymptotics of \cref{lem:fixed-volume-curve-cusp} give $S_k\asymp\mathrm{e}^{k/2}$.

The local potential of the truncated curve current is $b+u_k=\max(V-M,b-k)$. On $\{t<T_k\}$ its mass is $b'(T_k)$, and at $t=T_k$ its derivative jumps upward by $u_t(T_k)$. Smooth convex approximation and continuity of Bedford--Taylor products of bounded potentials therefore give total mass $V'(T_k)=O(S_k^{-1})$ on the cutoff region $\{t\leq T_k\}$. There $u_k=-k$, so the absolute energy contribution of this region is
\[
 kV'(T_k)=O(k\mathrm{e}^{-k/2})\longrightarrow0.
\]
Outside it the density and the potential are unchanged, and $\int|u|g_1$ is finite. Consequently $\int|u_k|g_{1,k}$ has a finite limit, where $g_{1,k}=6\alpha+\ddc u_k$, and $\int|u_k|\alpha$ converges by dominated convergence.

The mixed measures of the product current and of its reference are
\[
 \omega^2=72\alpha\wedge\beta,\quad
 (g_{1,k}+6\beta)\wedge\omega
     =6g_{1,k}\wedge\beta+36\alpha\wedge\beta,\quad
 (g_{1,k}+6\beta)^2=12g_{1,k}\wedge\beta.
\]
Their pullbacks are exactly the mixed non-pluripolar measures on $Y$, which do not charge $E$. All three relative energy integrals of the canonical approximants therefore have finite limits. This proves $\rho\in\mathcal E^1(Y,\theta)$; replacing $P+M$ by the normalized envelope $V_\theta$ changes the reference by a bounded amount and does not affect this conclusion.

Uniqueness follows from the full-mass comparison argument in the proof of \cref{thm:incomplete-divisorial-metric}. The measure $|\sigma|^{-2}\mathrm{d}V_Y$ has infinite mass, but the two exponential densities being compared have mass $72$. Comparison on the strict sublevel sets forces equality almost everywhere, and hence equality of the quasi-plurisubharmonic potentials everywhere.

\emph{The missing metric boundary.}
The set on which the current is smooth and strictly positive is $Y\setminus(H_Y\cup E)$: the cusp coefficient diverges at $H_Y$, and the nonzero current $2[E]$ excludes smoothness along $E$. The smooth part $\pi^*g_X$ also degenerates along $TE$.

Let $U$ be a dense Zariski open set on which the current is smooth and strictly positive. As in the proof of \cref{thm:nef-incomplete-cusp}, choose a point of $E$ avoiding every other component of the analytic set $Y\setminus U$; in a small chart around it the complement of $U$ is exactly $E$. A transverse path $a=\xi>0$, $b=b_0$ lies in $U$ for small $\xi$ and approaches that point. Since $g_X$ is smooth near $p$ and $\pi$ is smooth, the coefficients of $\pi^*g_X$ are bounded in the chart, so the path has bounded speed and finite remaining length. Its tails give an intrinsic Cauchy sequence with no limit in $U$, since its only ambient limit is the removed point and a smooth positive Riemannian metric induces the manifold topology. Hence every such $U$ is incomplete.
\end{proof}
