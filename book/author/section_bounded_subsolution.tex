\section{A bounded subsolution theorem}

The bounded exponential equation remains solvable for right-hand sides that are singular with respect to smooth volume, provided they have a bounded density with respect to the Monge--Amp\`ere measure of a bounded potential. We regularize this potential to obtain equations with bounded densities with respect to smooth volume; their solutions are controlled by a moving dominating measure. When the density is allowed to vanish, the normalizations are bounded by the mixed density inequality of \cref{lem:nonclosed-mixed-density} and the trace estimate for a fixed reference in \cref{lem:bounded-reference-trace}.

\Needspace{15\baselineskip}
\begin{theorem}\label{thm:bounded-subsolution}\index{Subsolution!bounded}
Let $\theta$ be semipositive and big. Suppose
\[
 \mu=f\,\mathrm{MA}_\theta(\psi),\qquad
 \psi\in\PSH(X,\theta)\cap L^\infty(X),\qquad
 0\leq f\in L^\infty(\mathrm{MA}_\theta(\psi)),
\]
and $\mu(X)>0$. For every $\lambda>0$ there is a unique bounded $u\in\PSH(X,\theta)$ satisfying
\[
 \mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}\mu.
\]
\end{theorem}
\begin{proof}
Write $\nu=\mathrm{MA}_\theta(\psi)$, a finite non-pluripolar measure.

\emph{Step 1: smooth strictly positive density.} First suppose $f$ is smooth and strictly positive. Choose a constant $A\geq1$ with $\Omega=A\omega\geq\theta$ and, by \cref{thm:bounded-exact-regularization}, smooth $\Omega$-plurisubharmonic functions $\psi_j\downarrow\psi$. The envelopes
\[
 q_j=P_\theta(\psi_j)
\]
lie between $\psi$ and $\psi_j$, so they are uniformly bounded and decrease to $\psi$. Contact support and ordered contact in the background $\Omega$ give
\[
 \mathrm{MA}_\theta(q_j)
 \leq\mathbf1_{\{q_j=\psi_j\}}\Omega_{\psi_j}^n.
\]
In particular, each $\mathrm{MA}_\theta(q_j)$ has a bounded density with respect to smooth volume. Moreover $\mathrm{MA}_\theta(q_j)(X)>0$ by \cref{prop:big-noncollapsing}. \Cref{thm:big-bounded-exponential} therefore gives bounded $u_j\in\PSH(X,\theta)$ with
\[
 \mathrm{MA}_\theta(u_j)=\mathrm{e}^{\lambda u_j}f\,
                                     \mathrm{MA}_\theta(q_j).
\]
Since $f$ is bounded above and away from zero, we can choose $C$ such that $\mathrm{e}^{\lambda(q_j-C)}f\leq1\leq \mathrm{e}^{\lambda(q_j+C)}f$ for all $j$. Then $q_j-C$ and $q_j+C$ are a subsolution and a supersolution, and bounded exponential comparison gives $q_j-C\leq u_j\leq q_j+C$.

Extract an $L^1$ limit $u$, which is again bounded. The moving domination
\[
 \mathrm{MA}_\theta(u_j)\leq C'\Omega_{\psi_j}^n
\]
and \cref{cor:common-barrier-continuity}, applied with the reference zero and with the uniformly bounded $\omega$-plurisubharmonic functions $w_j=\psi_j/A$, for which $\Omega_{\psi_j}^n=A^n\omega_{w_j}^n$, give convergence in capacity. Continuity of bounded products under convergence in capacity, with the quasi-continuous weights $\mathrm{e}^{\lambda u_j}f$, then gives
\[
 \mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}f\nu.
\]

\emph{Step 2: bounded measurable density.} Choose smooth strictly positive functions $f_j$ with a common upper bound $L>0$ such that $f_j\to f$ in $L^1(\nu)$. Such an approximation exists. Continuous functions are dense in $L^1(\nu)$, since $\nu$ is a finite Borel measure on the compact manifold $X$, and continuous functions are uniform limits of smooth ones. Truncating to a fixed bounded interval and adding a positive constant tending to zero preserves the bounds and the convergence. Step 1 gives bounded $u_j$ with
\[
 \mathrm{MA}_\theta(u_j)=\mathrm{e}^{\lambda u_j}f_j\nu.
\]
For a fixed $C$ we have $\mathrm{e}^{\lambda(\psi-C)}f_j\leq1$ for all $j$, so $\psi-C$ is a subsolution and $u_j\geq\psi-C$. The upper bound requires a separate argument, because $\inf_Xf_j$ may tend to zero.

The construction in Step 1 also gives the mixed inequality
\begin{equation}\label{eq:bounded-subsolution-mixed}
 \theta_{u_j}\wedge\theta_\psi^{n-1}
 \geq \mathrm{e}^{\lambda u_j/n}f_j^{1/n}\nu.
\end{equation}
To prove it without a mixed inequality for arbitrary singular measures, we use the approximants $u_{j,k}$ and $q_k$ of Step 1, for which
\[
 \mathrm{MA}_\theta(u_{j,k})
     =\mathrm{e}^{\lambda u_{j,k}}f_j\mathrm{MA}_\theta(q_k),
\]
and both Monge--Amp\`ere measures have densities with respect to smooth volume. \Cref{lem:nonclosed-mixed-density} applies to the pair $(u_{j,k},q_k)$. For fixed $j$, both sequences are uniformly bounded and converge in capacity to $(u_j,\psi)$, so continuity of bounded products against quasi-continuous weights passes the mixed inequality to the limit and yields \eqref{eq:bounded-subsolution-mixed}.

Put $M_j=\sup_Xu_j$, $v_j=u_j-M_j$ and $c_j=\mathrm{e}^{\lambda M_j}$. Suppose the $M_j$ were unbounded above. After extraction, $M_j\to+\infty$ and $v_j\to v$ in $L^1(X)$, where $v\in\PSH(X,\theta)$ is normalized. By \cref{lem:fixed-npp-integrals}, $\mathrm{e}^{\lambda v_j/n}\to \mathrm{e}^{\lambda v/n}$ in $L^1(\nu)$. Together with the inequalities
\[
 |f_j^{1/n}-f^{1/n}|^n\leq|f_j-f|,
 \qquad0\leq f_j\leq L,
\]
this gives
\[
 \int_X \mathrm{e}^{\lambda v_j/n}f_j^{1/n}\,\mathrm{d}\nu
 \longrightarrow\int_X\mathrm{e}^{\lambda v/n}f^{1/n}\,\mathrm{d}\nu>0;
\]
the last integral is positive because $\mu(X)>0$ and $\nu$ does not charge the pole set of $v$. Integrating \eqref{eq:bounded-subsolution-mixed} and applying \cref{lem:bounded-reference-trace}, we find some $a>0$ such that, along the extraction,
\[
 a c_j^{1/n}\leq
 \int_X\theta_{v_j}\wedge\theta_\psi^{n-1}\leq C_\psi,
\]
contradicting $M_j\to+\infty$. Thus the $u_j$ are uniformly bounded above as well as below.

Extract an $L^1$ limit $u$. Now $\mathrm{MA}_\theta(u_j)\leq C''\nu$, so \cref{cor:common-barrier-continuity} gives convergence in capacity and weak convergence of the Monge--Amp\`ere measures. The right-hand sides converge in total variation, by the convergence of exponentials against the fixed measure $\nu$ and the estimate
\[
 \int_X|\mathrm{e}^{\lambda u_j}f_j-\mathrm{e}^{\lambda u}f|\,\mathrm{d}\nu
 \leq C\|f_j-f\|_{L^1(\nu)}
       +\|f\|_\infty\int_X|\mathrm{e}^{\lambda u_j}-\mathrm{e}^{\lambda u}|\,\mathrm{d}\nu.
\]
Consequently $\mathrm{MA}_\theta(u)=\mathrm{e}^{\lambda u}\mu$. Uniqueness follows from bounded exponential comparison.
\end{proof}
