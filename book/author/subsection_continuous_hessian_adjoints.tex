\subsection{Continuous positive complex Hessians}
\label{subsec:continuous-hessian-adjoints}

The recovery theorem \cref{thm:normalization-gauduchon-recovery} extends
to a bounded solution $u$ for which the form $\omega+\ddc u$ is continuous
and everywhere positive. In this case one can vary the potential in both
directions, and the resulting densities force every supporting probability
to be \emph{stationary} for the linearized operator $L$, that is, to satisfy
$\int_XLh\,\mathrm d\nu=0$ for every smooth $h$. The description of the range
of $L$ in Sobolev spaces, \cref{lem:continuous-hermitian-fredholm}, then
gives uniqueness.

\Needspace{14\baselineskip}
\begin{theorem}\label{thm:continuous-hessian-adjoint-uniqueness}\index{Adjoint measure!continuous positive Hessian}
Let $n\geq2$, let $\theta=\omega$ be a Hermitian form, and let $\varphi=0$.
Suppose that $F\in C(X)$ and that a bounded solution $u\in\PSH(X,\omega)$
of \eqref{eq:relative-normalization} has the property that the current
\[
 \sigma=\omega+\ddc u
\]
is a continuous form, positive everywhere on $X$. Then $\mathcal A_0(F)$
consists of one probability $\nu_F$. With the operator
$L_\sigma=\operatorname{tr}_\sigma\ddc$ of
\cref{lem:continuous-hermitian-fredholm}, it satisfies
\begin{align*}
 \int_X L_\sigma h\,\mathrm d\nu_F&=0\quad(h\in C^\infty(X)),\\
 DB_0(F)[H]&=-\int_XH\,\mathrm d\nu_F\quad(H\in C(X)).
\end{align*}
The derivative is taken in the G\^ateaux sense.
\end{theorem}
\begin{proof}
In local coordinates, the Euclidean trace of $\ddc u$ is a constant
multiple of the Laplacian of $u$, and it is continuous. Local
constant-coefficient elliptic regularity gives $u\in W^{2,p}$ for every
finite $p$. Indeed, subtract the Newtonian potential of a cutoff of this
trace. The second derivatives of that potential are given by the
order-zero Fourier multipliers $\xi_i\xi_j/|\xi|^2$, so they lie in every
$L^p$, $p<\infty$; the remaining distribution is harmonic. Sobolev
embedding gives a $C^{1,\alpha}$ representative for every $\alpha<1$,
which agrees with the quasi-psh representative.

On a coordinate ball choose smooth $\rho$ with $\ddc\rho\geq\omega$.
The radial regularizations of the bounded psh function $u+\rho$ decrease
to it, and their complex Hessians converge uniformly on smaller balls.
Continuity of bounded products and the finite smooth-background
expansion identify $(\omega+\ddc u)^n$ with $\sigma^n$. For smooth real
$h$, the same argument applies to $u+th$, which is admissible for all
sufficiently small signed $t$ by uniform positivity. Define
\[
 F_t=F+\log\frac{(\sigma+t\ddc h)^n}{\sigma^n}.
\]
Then $F_t\in C(X)$ and
$(\omega+\ddc(u+th))^n=c_0(F)\mathrm e^{F_t}\mathrm dV$, so uniqueness of
the constant gives $B_0(F_t)=B_0(F)$. The supporting inequality
\eqref{eq:adjoint-probability-set} with $G=F_t$ gives
$\int_X(F_t-F)\,\mathrm d\nu\leq0$ for each
$\nu\in\mathcal A_0(F)$. Since
$F_t-F=tL_\sigma h+O(t^2)$ uniformly, using both signs of $t$ yields
$\int_XL_\sigma h\,\mathrm d\nu=0$.

Choose $p=4n+1>2n$ and $q=p/(p-1)=1+1/(4n)$. By
\cref{cor:gaud-adjoint-lq}, every supporting probability has an $L^q$
density, since $q<n/(n-1)$. Smooth functions are dense in $W^{2,p}(X)$,
and $L_\sigma\colon W^{2,p}(X)\to L^p(\mathrm dV)$ is bounded, so this
identity extends to every $h\in W^{2,p}(X)$. Thus the densities of all
supporting probabilities annihilate the range of $L_\sigma$, and by
\cref{lem:continuous-hermitian-fredholm} they lie in a one-dimensional
subspace of $L^q(\mathrm dV)$. Their integrals equal one, so they coincide.
Nonemptiness is \cref{thm:weak-adjoint-probabilities}, and the derivative
formula follows from \cref{prop:normalization-directional}.
\end{proof}

\subsection{Products with a Hermitian factor}

An explicit product construction shows that the hypothesis of
\cref{thm:continuous-hessian-adjoint-uniqueness} allows densities with an
arbitrary continuous modulus. Let $(Y,\kappa)$ be a connected
compact Hermitian $k$-fold, $k\geq1$, with $\int_Y\kappa^k=1$.
Denote by $\nu_\kappa$ the smooth stationary probability of
$L_\kappa=\operatorname{tr}_\kappa\ddc$, given by the Gauduchon multiplier
when $k\geq2$ (\cref{prop:adjoint-gauduchon}); for $k=1$ it is $\kappa$.
Write $\omega_1$ for the Fubini--Study form of mass one on $\mathbb P^1$.
All factor forms and functions below are pulled back to
$X=\mathbb P^1\times Y$.

\begin{proposition}\label{prop:continuous-product-adjoint}\index{Adjoint measure!Hermitian product}
Let $a>0$, let $b$ be smooth and positive on $\mathbb P^1$, and let $A$
be continuous and positive there, with $\int_{\mathbb P^1}A\omega_1=a$.
For
\[
 \omega=a\omega_1+b\kappa,\qquad \mathrm dV=\omega_1\wedge\kappa^k,
 \qquad F=\log((k+1)Ab^k),
\]
one has
\[
 c_0(F)=1,\qquad
 \mathcal A_0(F)=\left\{\frac Aa\,\omega_1\otimes\nu_\kappa\right\}.
\]
\end{proposition}
\begin{proof}
The Green solution of $\ddc u=(A-a)\omega_1$ on $\mathbb P^1$ has
continuous complex Hessian and is bounded. Its pullback yields
$\sigma=A\omega_1+b\kappa>0$, and
$\sigma^{k+1}=(k+1)Ab^k\omega_1\wedge\kappa^k$.
Thus uniqueness of the constant gives $c_0(F)=1$, and
\cref{thm:continuous-hessian-adjoint-uniqueness} gives a unique
supporting probability. For every smooth real $h$ on $X$, the block
decomposition
\[
 L_\sigma h=A^{-1}L_{\omega_1}h+b^{-1}L_\kappa h,
\]
in which $L_{\omega_1}$ and $L_\kappa$ act along the two factors, shows
that $(A/a)\omega_1\otimes\nu_\kappa$ is stationary: the first term
integrates to zero by Stokes' theorem on $\mathbb P^1$, and the second by
stationarity on $Y$ and Fubini's theorem. This candidate is a probability
with a bounded density with respect to $\mathrm dV$, so its stationarity
extends to $W^{2,p}$ tests as in the proof of
\cref{thm:continuous-hessian-adjoint-uniqueness}. The one-dimensional
annihilator of \cref{lem:continuous-hermitian-fredholm} and probability
normalization therefore identify it with the supporting probability.
\end{proof}

\begin{example}\label{ex:non-dini-hopf-product-adjoint}\index{Dini continuity!failure with unique adjoint}\index{Hopf manifold!adjoint product example}
There is a non-K\"ahler threefold with a nonclosed Hermitian background
and a positive continuous density whose logarithm is neither Dini
continuous nor H\"older continuous of any positive order, but whose
supporting probability is explicit and unique.
\end{example}
\begin{proof}
Take the Hopf surface
$Y=(\mathbb C^2\setminus\{0\})/\langle z\mapsto2z\rangle$ and its
unit-volume metric $\kappa=c|z|^{-2}\ddc|z|^2$, where $c>0$ is chosen
so that $\int_Y\kappa^2=1$. Put $s=\log|z|^2$, $\alpha=\ddc s$ and
$\beta=\mathrm ds\wedge\dc s$, so that $\kappa=c(\alpha+\beta)$.
The computation of $\ddc\beta$ in \eqref{eq:hopf-torsion} applies in
this dimension and gives $\ddc\beta=-\alpha^2$. Here $\alpha$ has rank
one, so $\ddc\kappa=-c\alpha^2=0$. Thus $\kappa$ is Gauduchon and
$\nu_\kappa=\kappa^2$. Polar coordinates identify $Y$ with
$S^3\times S^1$, hence $H^2(Y,\mathbb R)=0$. A closed positive form
would be exact and have positive square integral, contradicting Stokes'
theorem. Thus $Y$ is non-K\"ahler, and so is $\mathbb P^1\times Y$, by
restriction.

Let $\tau\in[-1,1]$ be the height on $\mathbb P^1$, set $b=2+\tau$,
and put
\[
 f(t)=\begin{cases}
       1/\log(\mathrm e/|t|),&0<|t|\leq1,\\
       0,&t=0,
      \end{cases}
\]
\[
 A=a+\varepsilon\left(f(\tau)-\int_{\mathbb P^1}f(\tau)\omega_1\right),
 \quad0<\varepsilon<a.
\]
Since $0\leq f\leq1$, this is a positive continuous function of mean
$a$. Along a normal segment to the equator, $|\tau|$ is comparable
to the distance $r$ from the equator. The oscillation of $A$ is therefore
bounded below by a positive multiple of $1/\log(\mathrm e/r)$. Its Dini
integral $\int_0^{r_0}\mathrm dr/(r\log(\mathrm e/r))$ diverges, and this
modulus is larger than every $r^\gamma$, $\gamma>0$, as $r\downarrow0$.
The logarithm of $A$ has the same lower bound, because $A$ is uniformly
positive and bounded. The smooth term $2\log b$ contributes only $O(r)$,
so $F=\log(3Ab^2)$ has neither regularity. The mixed component
$\mathrm db\wedge\kappa$ of $\mathrm d\omega$ shows that
$\omega=a\omega_1+b\kappa$ is nonclosed.
By \cref{prop:continuous-product-adjoint}, its supporting probability
is $(A/a)\,\omega_1\wedge\kappa^2$.
\end{proof}
