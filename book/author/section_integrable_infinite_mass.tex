\section{Integrable Hessians and infinite mass}
\label{sec:integrable-infinite-mass}

The Gauduchon identity fixes the integral of the trace of each positive summand in the preceding construction. Their weighted series therefore converges in the coefficient $L^1$ norm, although its determinant is not integrable. The trace of the tail gives a quantitative bound for this convergence.

\Needspace{11\baselineskip}
Fix a Gauduchon form $\gamma$ and set $M=\int_X\omega\wedge\gamma^{n-1}$.
For a real $(1,1)$-form $T$ with measurable coefficients, let $A_T$ be its Hermitian matrix relative to $\gamma$, and let $|A_T|=(A_T^2)^{1/2}$ be its positive matrix absolute value. Use the norm
\[
 \|T\|_{L^1(\gamma)}
 =\frac1n\int_X\operatorname{tr}|A_T|\,\gamma^n.
\]
For $T\geq0$ this equals $\int_XT\wedge\gamma^{n-1}$. Any smooth background norm gives the same space of integrable coefficients on compact $X$. We write $\det_\gamma T=\det A_T$, defined almost everywhere when the coefficients are integrable.

\Needspace{24\baselineskip}
\begin{proposition}\label{prop:integrable-infinite-mass}\index{Monge--Ampere mass@Monge--Amp\`ere mass!infinite with trivial ideals}
Suppose $v_+(\omega)=+\infty$. For the potential $U$ and smooth decreasing approximants $U_N$ constructed in \cref{thm:infinite-mass-witness}, the current $T=\omega+\ddc U$ has integrable coefficients and
\[
 \|\omega+\ddc U_N-T\|_{L^1(\gamma)}
 \leq2^{1-N}M.
\]
Its pointwise determinant satisfies
\[
 \mathrm{MA}^{\mathrm{ext}}_\omega(U)
 \geq (\det_\gamma T)\gamma^n,
 \qquad
 \int_X(\det_\gamma T)\gamma^n=+\infty.
\]
Moreover,
\[
 \int_X(\omega+\ddc U_N)^n\longrightarrow+\infty.
\]
\end{proposition}
\begin{proof}
Write $T_j=\omega+\ddc u_j$ for the smooth positive currents in that construction. Put
\[
 S_N=\sum_{j\leq N}a_jT_j,\qquad
 \delta_N=\sum_{j>N}a_j=2^{-N}.
\]
The smooth positive matrices $S_N$ increase pointwise. The Gauduchon identity gives
\[
 \int_XS_N\wedge\gamma^{n-1}=(1-\delta_N)M.
\]
Monotone convergence of their traces shows that the matrix series has a finite positive limit $S$ almost everywhere. Its coefficients are integrable: in a $\gamma$-unitary frame the absolute value of every coefficient is bounded by the trace. Positivity and the same trace identity give
\[
 \|S-S_N\|_{L^1(\gamma)}=\delta_N M.
\]
Thus $S_N\to S$ also in distributions. Since
\[
 \omega+\ddc U_N=S_N+\delta_N\omega
 \longrightarrow\omega+\ddc U
\]
in distributions by $L^1$ convergence of the potentials, we obtain $T=S$. The triangle inequality now gives
\[
 \|\omega+\ddc U_N-T\|_{L^1(\gamma)}
 \leq\|S_N-S\|_{L^1(\gamma)}
       +\delta_N\|\omega\|_{L^1(\gamma)}
 =2\delta_N M.
\]

Fix $N$ and set $V_N=U-U_N$. The remaining partial sums give
$\delta_N\omega+\ddc V_N\geq0$ and $V_N\leq0$. For $t>0$, put
$W_{N,t}=U_N+\max(V_N,-t)$.
It is bounded, and its current is the sum of two positive factors:
\[
 \omega+\ddc W_{N,t}
 =S_N+\bigl(\delta_N\omega+\ddc\max(V_N,-t)\bigr).
\]
On $D_t=\{U>-t\}$, the inequality $U_N\leq0$ gives $V_N\geq U>-t$. Consequently $W_{N,t}=U=\max(U,-t)$ there. Locality and positivity of bounded products imply
\[
 \mathbf1_{D_t}\mathrm{MA}^{\mathrm{ext}}_\omega(U)
 \geq\mathbf1_{D_t}S_N^n.
\]
The smooth measure $S_N^n$ ignores the pluripolar pole set of $U$. Letting $t\to\infty$ proves
$\mathrm{MA}^{\mathrm{ext}}_\omega(U)\geq S_N^n$.
The positive matrices $S_N$ increase to $T$ almost everywhere, so their determinants increase to $\det_\gamma T$. Monotone convergence on every Borel set gives the asserted determinant-measure inequality.

For every $N\geq j$, positivity gives $S_N\geq a_jT_j$. Hence both the measurable limit and the smooth approximating current satisfy
\[
 \int_X(\det_\gamma T)\gamma^n
 \geq a_j^n\int_XT_j^n\geq2^j,
 \qquad
 \int_X(\omega+\ddc U_N)^n
 \geq a_j^n\int_XT_j^n\geq2^j.
\]
The first bound holds for every $j$ and the second for every $N\geq j$. They give the two assertions about infinite mass.
\end{proof}

\Needspace{12\baselineskip}
\begin{corollary}\label{cor:strong-linear-infinite-mass}\index{Gauduchon trace!infinite nonlinear mass}
If bounded mass fails, there are nonpositive potentials $w_\varepsilon\in\PSH(X,\omega)$, $0<\varepsilon<1$, such that
\[
 \omega+\ddc w_\varepsilon\geq(1-\varepsilon)\omega,
 \qquad
 \mathcal I(pw_\varepsilon)=\mathcal O_X\quad(p>0),
\]
and both the pointwise determinant integral and the extended Monge--Amp\`ere mass are infinite. As $\varepsilon\downarrow0$, the potentials converge to zero in $L^1$ and their currents converge to $\omega$ in the coefficient $L^1$ norm. Their full Gauduchon trace measures converge to $\omega\wedge\gamma^{n-1}$ in total variation.
\end{corollary}
\begin{proof}
Take $w_\varepsilon=\varepsilon U$. Its current is
\[
 T_\varepsilon=(1-\varepsilon)\omega+\varepsilon T.
\]
The lower bound and the two $L^1$ convergence assertions follow immediately. The full trace measures have integrable densities, and their total-variation distance is at most
\[
 \varepsilon\|T-\omega\|_{L^1(\gamma)}.
\]
Every multiplier ideal is trivial by the exponential integrability already proved for $U$. Since $T_\varepsilon\geq\varepsilon T$ almost everywhere, its determinant integral is at least $\varepsilon^n$ times that of $T$, hence infinite. Its extended mass is infinite by \cref{cor:bmp-characterization}.
\end{proof}

\Needspace{18\baselineskip}
For smooth potentials, the total masses can be specified in advance while both the currents and the exponential weights converge.

\begin{corollary}\label{cor:smooth-prescribed-masses}
Suppose $v_+(\omega)=+\infty$. Let $(\varepsilon_j)_{j\geq1}$ and $(b_j)_{j\geq1}$ be real sequences with $0<\varepsilon_j<1$, $\varepsilon_j\to0$, and $b_j>\int_X\omega^n$. There are smooth potentials $w_j\in\PSH(X,\omega)$ with $\sup_Xw_j=0$ such that
\[
 \begin{gathered}
 \omega+\ddc w_j\geq(1-\varepsilon_j)\omega,
 \qquad
 \int_X(\omega+\ddc w_j)^n=b_j,
 \\
 \|\ddc w_j\|_{L^1(\gamma)}
 \leq2M\varepsilon_j.
 \end{gathered}
\]
Moreover, $w_j\to0$ in $L^1(\mathrm{d}V)$ and $\mathrm{e}^{-pw_j}\to1$ in $L^1(\mathrm{d}V)$ for every fixed $p>0$. The full Gauduchon trace measures converge to $\omega\wedge\gamma^{n-1}$ in total variation.
\end{corollary}
\Needspace{7\baselineskip}
\begin{proof}
By \cref{prop:bmp-metric}, choose a smooth $u_j\in\PSH(X,\omega)$ with $\sup_Xu_j=0$ and
\[
 \int_X(\omega+\ddc u_j)^n>\varepsilon_j^{-n}b_j.
\]
Put $T_j=\omega+\ddc u_j$. The polynomial
\[
 F_j(s)=\int_X\bigl((1-s)\omega+sT_j\bigr)^n,
 \qquad 0\leq s\leq\varepsilon_j,
\]
satisfies $F_j(0)<b_j$ and $F_j(\varepsilon_j)>b_j$, the latter by positivity of the mixed terms. The intermediate value theorem gives $s_j\in(0,\varepsilon_j)$ with $F_j(s_j)=b_j$. Set $w_j=s_ju_j$. This proves the normalization, strict positivity and prescribed mass.

The Gauduchon identity gives $\|T_j\|_{L^1(\gamma)}=\|\omega\|_{L^1(\gamma)}=M$. Hence
\[
 \|\ddc w_j\|_{L^1(\gamma)}
 =s_j\|T_j-\omega\|_{L^1(\gamma)}
 \leq2M\varepsilon_j.
\]
The absolute value of the trace is bounded by the trace of the matrix absolute value, so the same bound holds for the total variation of $\ddc w_j\wedge\gamma^{n-1}$. Also \eqref{eq:normalized-lone} gives $\|w_j\|_{L^1(\mathrm{d}V)}\leq K_\omega\varepsilon_j$.

Let $\alpha,C$ be the constants in \cref{lem:uniform-exponential} for $(X,\omega,\mathrm{d}V)$, and fix $p>0$. For all sufficiently large $j$, one has $ps_j\leq\alpha$. Convexity of the exponential function, applied with weight $ps_j/\alpha$, gives
\[
 0\leq\mathrm{e}^{-pw_j}-1
 \leq\frac{ps_j}{\alpha}\bigl(\mathrm{e}^{-\alpha u_j}-1\bigr).
\]
Integration bounds its $L^1(\mathrm{d}V)$ norm by $pC\varepsilon_j/\alpha$, proving the last convergence.
\end{proof}
