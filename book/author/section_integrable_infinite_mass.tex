\section{Integrable Hessians and infinite mass}
\label{sec:integrable-infinite-mass}

In the construction of \cref{thm:infinite-mass-witness}, the Gauduchon identity of \cref{prop:gauduchon-trace} fixes the trace integral of every positive summand $\omega+\ddc u_j$. The series defining $\omega+\ddc U$ therefore converges in the coefficient $L^1$ norm, although the determinant of its sum is not integrable. Thus the trace measures of the partial sums converge in total variation, yet the Monge--Amp\`ere mass of $U$ is infinite. In \cref{chap:damped-measures}, weights that decay on deep sublevel sets will control these masses.

\Needspace{11\baselineskip}
Fix a Gauduchon form $\gamma$ and set $M=\int_X\omega\wedge\gamma^{n-1}$.
For a real $(1,1)$-form $T$ with measurable coefficients, let $A_T$ be its Hermitian matrix relative to $\gamma$, and let $|A_T|=(A_T^2)^{1/2}$ be its positive matrix absolute value. We use the norm
\[
 \|T\|_{L^1(\gamma)}
 =\frac1n\int_X\operatorname{tr}|A_T|\,\gamma^n.
\]
For $T\geq0$ this equals $\int_XT\wedge\gamma^{n-1}$. On compact $X$, any other smooth Hermitian form gives an equivalent norm and hence the same space of integrable coefficients. When the coefficients of $T$ are integrable, we write $\det_\gamma T=\det A_T$; this function is defined almost everywhere, and $T^n=(\det_\gamma T)\gamma^n$ when $T$ is smooth.

\Needspace{24\baselineskip}
\begin{proposition}\label{prop:integrable-infinite-mass}\index{Monge--Ampere mass@Monge--Amp\`ere mass!infinite with trivial ideals}
\index{Complex Hessian!integrable coefficients}
Suppose $v_+(\omega)=+\infty$. For the potential $U$ and its smooth decreasing approximants $U_N$ constructed in the proof of \cref{thm:infinite-mass-witness}, the current $T=\omega+\ddc U$ has integrable coefficients and
\[
 \|\omega+\ddc U_N-T\|_{L^1(\gamma)}
 \leq2^{1-N}M.
\]
Its pointwise determinant satisfies
\[
 \mathrm{MA}^{\mathrm{ext}}_\omega(U)
 \geq (\det_\gamma T)\gamma^n,
 \qquad
 \int_X(\det_\gamma T)\gamma^n=+\infty.
\]
Moreover,
\[
 \int_X(\omega+\ddc U_N)^n\longrightarrow+\infty.
\]
\end{proposition}
\begin{proof}
Write $T_j=\omega+\ddc u_j$ for the smooth positive forms in that construction, and, as in its proof, put
\[
 S_N=\sum_{j\leq N}a_jT_j,\qquad
 \delta_N=\sum_{j>N}a_j=2^{-N}.
\]
The smooth positive forms $S_N$ increase pointwise, and the Gauduchon identity gives
\[
 \int_XS_N\wedge\gamma^{n-1}=(1-\delta_N)M.
\]
By monotone convergence of their traces, the series of matrices has a finite positive limit $S$ almost everywhere. The coefficients of $S$ are integrable, because in a $\gamma$-unitary frame the absolute value of every coefficient of a positive matrix is bounded by its trace. Positivity and the same trace identity give
\[
 \|S-S_N\|_{L^1(\gamma)}=\delta_N M.
\]
In particular, $S_N\to S$ in the sense of distributions. Since
\[
 \omega+\ddc U_N=S_N+\delta_N\omega
 \longrightarrow\omega+\ddc U
\]
in the sense of distributions, by $L^1$ convergence of the potentials, we obtain $T=S$. The triangle inequality now gives
\[
 \|\omega+\ddc U_N-T\|_{L^1(\gamma)}
 \leq\|S_N-S\|_{L^1(\gamma)}
       +\delta_N\|\omega\|_{L^1(\gamma)}
 =2\delta_N M.
\]

To compare $\mathrm{MA}^{\mathrm{ext}}_\omega(U)$ with the determinant, recall \eqref{eq:infinite-mass-partial-sums}: $\mathrm{MA}^{\mathrm{ext}}_\omega(U)\geq S_N^n$ for every $N$. The positive matrices $S_N$ increase to $T$ almost everywhere, so their determinants increase to $\det_\gamma T$. Monotone convergence on every Borel set gives the asserted inequality between $\mathrm{MA}^{\mathrm{ext}}_\omega(U)$ and the determinant measure.

Finally, for every $N\geq j$, positivity gives $T\geq S_N\geq a_jT_j$ and $\omega+\ddc U_N\geq S_N$. Hence both the measurable limit and the smooth approximating forms satisfy
\[
 \int_X(\det_\gamma T)\gamma^n
 \geq a_j^n\int_XT_j^n\geq2^j,
 \qquad
 \int_X(\omega+\ddc U_N)^n
 \geq a_j^n\int_XT_j^n\geq2^j.
\]
The first bound holds for every $j$, and the second for every $N\geq j$; together they prove the two assertions about infinite mass.
\end{proof}

\Needspace{12\baselineskip}
\begin{corollary}\label{cor:strong-linear-infinite-mass}\index{Gauduchon trace!infinite nonlinear mass}
If the bounded mass property fails, there are nonpositive potentials $w_\varepsilon\in\PSH(X,\omega)$, $0<\varepsilon<1$, such that
\[
 \omega+\ddc w_\varepsilon\geq(1-\varepsilon)\omega,
 \qquad
 \mathcal I(pw_\varepsilon)=\mathcal O_X\quad(p>0),
\]
and such that both the integral of the pointwise determinant $\det_\gamma(\omega+\ddc w_\varepsilon)$ against $\gamma^n$ and the extended Monge--Amp\`ere mass of $w_\varepsilon$ are infinite. As $\varepsilon\downarrow0$, the potentials converge to zero in $L^1$, their currents converge to $\omega$ in the coefficient $L^1$ norm, and their full trace measures $(\omega+\ddc w_\varepsilon)\wedge\gamma^{n-1}$ converge to $\omega\wedge\gamma^{n-1}$ in total variation.
\end{corollary}
\begin{proof}
Take $w_\varepsilon=\varepsilon U$, with $U$ and $T=\omega+\ddc U$ as in \cref{prop:integrable-infinite-mass}. Its current is
\[
 T_\varepsilon=(1-\varepsilon)\omega+\varepsilon T.
\]
The lower bound and the two $L^1$ convergence assertions follow immediately. The full trace measures have integrable densities, and their distance from $\omega\wedge\gamma^{n-1}$ in total variation is at most
\[
 \varepsilon\|T-\omega\|_{L^1(\gamma)}.
\]
Every multiplier ideal is trivial by the exponential integrability of $U$ proved in \cref{thm:infinite-mass-witness}. Since $T_\varepsilon\geq\varepsilon T$ almost everywhere, the determinant integral of $T_\varepsilon$ is at least $\varepsilon^n$ times that of $T$, hence infinite. The extended mass of $w_\varepsilon$ is infinite by the proof of \cref{cor:bmp-characterization}.
\end{proof}

\Needspace{18\baselineskip}
For smooth potentials, the total masses can be prescribed in advance while both the currents and the exponential weights converge.

\begin{corollary}\label{cor:smooth-prescribed-masses}
\index{Monge--Ampere mass@Monge--Amp\`ere mass!prescribed smooth masses}
Suppose $v_+(\omega)=+\infty$. Let $(\varepsilon_j)_{j\geq1}$ and $(b_j)_{j\geq1}$ be real sequences with $0<\varepsilon_j<1$, $\varepsilon_j\to0$, and $b_j>\int_X\omega^n$. There are smooth potentials $w_j\in\PSH(X,\omega)$ with $\sup_Xw_j=0$ such that
\[
 \begin{gathered}
 \omega+\ddc w_j\geq(1-\varepsilon_j)\omega,
 \qquad
 \int_X(\omega+\ddc w_j)^n=b_j,
 \\
 \|\ddc w_j\|_{L^1(\gamma)}
 \leq2M\varepsilon_j.
 \end{gathered}
\]
Moreover, $w_j\to0$ in $L^1(\mathrm{d}V)$ and $\mathrm{e}^{-pw_j}\to1$ in $L^1(\mathrm{d}V)$ for every fixed $p>0$. The full Gauduchon trace measures converge to $\omega\wedge\gamma^{n-1}$ in total variation.
\end{corollary}
\Needspace{7\baselineskip}
\begin{proof}
By \cref{prop:bmp-metric}, we can choose a smooth $u_j\in\PSH(X,\omega)$ with $\sup_Xu_j=0$ and
\[
 \int_X(\omega+\ddc u_j)^n>\varepsilon_j^{-n}b_j.
\]
Put $T_j=\omega+\ddc u_j$. The polynomial
\[
 F_j(s)=\int_X\bigl((1-s)\omega+sT_j\bigr)^n,
 \qquad 0\leq s\leq\varepsilon_j,
\]
satisfies $F_j(0)<b_j$, and also $F_j(\varepsilon_j)\geq\varepsilon_j^n\int_XT_j^n>b_j$ by positivity of the mixed terms. The intermediate value theorem gives $s_j\in(0,\varepsilon_j)$ with $F_j(s_j)=b_j$. Set $w_j=s_ju_j$. Then $\sup_Xw_j=0$, $\omega+\ddc w_j=(1-s_j)\omega+s_jT_j\geq(1-\varepsilon_j)\omega$, and $w_j$ has the prescribed mass $b_j$.

The Gauduchon identity gives $\|T_j\|_{L^1(\gamma)}=\|\omega\|_{L^1(\gamma)}=M$. Hence
\[
 \|\ddc w_j\|_{L^1(\gamma)}
 =s_j\|T_j-\omega\|_{L^1(\gamma)}
 \leq2M\varepsilon_j.
\]
Since the absolute value of the trace of a Hermitian matrix is bounded by the trace of its absolute value, the same bound holds for the total variation of $\ddc w_j\wedge\gamma^{n-1}$. With $K_\omega$ as in \eqref{eq:normalized-lone}, we also have $\|w_j\|_{L^1(\mathrm{d}V)}\leq K_\omega\varepsilon_j$.

Finally, let $\alpha,C$ be the constants of \cref{lem:uniform-exponential} for $(X,\omega,\mathrm{d}V)$, and fix $p>0$. For all sufficiently large $j$, one has $ps_j\leq\alpha$. Convexity of the exponential function, applied with weight $ps_j/\alpha$, gives
\[
 0\leq\mathrm{e}^{-pw_j}-1
 \leq\frac{ps_j}{\alpha}\bigl(\mathrm{e}^{-\alpha u_j}-1\bigr).
\]
Integrating, we bound the $L^1(\mathrm{d}V)$ norm of $\mathrm{e}^{-pw_j}-1$ by $pC\varepsilon_j/\alpha$, which proves the last convergence.
\end{proof}
