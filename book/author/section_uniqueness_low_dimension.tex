\section{Integral rigidity in dimensions two and three}
\label{uni-sec:lowdim}

In dimensions two and three, if two positive forms $A,B$ have equal determinants, an algebraic expression quadratic in $A-B$ has a strict sign unless the two forms agree. On a surface its integral vanishes because it is exact. On a threefold the integral is a term involving $\ddc\omega$, so a sign condition on $\ddc\omega$ leads to the same conclusion. Approximation in the Sobolev norm justifies the integral identities for $W^{2,n}$ potentials.

In this section $n\in\{2,3\}$. Norms of coefficients of forms are taken with respect to a fixed smooth metric and $\mathrm{d}V$.

\Needspace{16\baselineskip}
\begin{theorem}\label{uni-thm:sobolev}\index{Uniqueness!Sobolev potentials}
Let $u,v\in W^{2,n}(X,\mathbb R)$, and assume that
$A,B\geq0$ almost everywhere and
\[
  A^n=B^n=f\,\mathrm{d}V,\qquad f>0\quad\hbox{almost everywhere}.
\]
If $n=3$, assume in addition that
\begin{equation}
  \ddc\omega\wedge\mathrm{i}\eta\wedge\bar\eta\leq0
  \label{uni-eq:threefold-sign}
\end{equation}
for every $(1,0)$-covector $\eta$.  Then
$\ddc(u-v)=0$ in the sense of distributions, and $u-v$ is almost everywhere constant.
For the $\omega$-plurisubharmonic representatives of $u$ and $v$, the difference is constant everywhere.
\end{theorem}

\begin{proof}
Put
\[
 \begin{gathered}
 s=(u+v)/2,\qquad D=\ddc w=A-B,\\
 C=(A+B)/2=\omega+\ddc s.
 \end{gathered}
\]
The coefficients of $C$ and $D$ lie in $L^n$, so their products of top degree are
integrable by H\"older's inequality.  At almost every point the common
determinant is positive, so $A$ and $B$ are positive definite there, and we may diagonalize
them simultaneously by congruence: for a suitable coframe of $(1,0)$-covectors $\theta_1,\ldots,\theta_n$, write
$A=\sum e_j$, $B=\sum\lambda_j e_j$, where
$e_j=\mathrm{i}\theta_j\wedge\bar\theta_j$, $\lambda_j>0$, and
$\prod\lambda_j=1$.

In dimension two,
\begin{equation}\label{uni-eq:surface-sign}
 D^2=A^2+B^2-2A\wedge B
    =2(2-\lambda_1-\lambda_2)e_1\wedge e_2\leq0,
\end{equation}
with equality if and only if $A=B$.  In dimension three,
\begin{equation}\label{uni-eq:threefold-algebra}
 2C\wedge D^2=A^3+B^3-A^2\wedge B-A\wedge B^2\leq0.
\end{equation}
Indeed, with $E=e_1\wedge e_2\wedge e_3$, the four terms on the right
are
\[
 6E,\qquad 6E,\qquad 2\Bigl(\sum_j\lambda_j\Bigr)E,
 \qquad 2\Bigl(\sum_{i<j}\lambda_i\lambda_j\Bigr)E.
\]
By the arithmetic--geometric mean inequality and $\prod_j\lambda_j=1$, each sum is at least three, and equality forces all
$\lambda_j=1$, that is, $A=B$.

Choose smooth real functions $w_k\to w$ in $W^{2,n}$ and set
$D_k=\ddc w_k$; such approximations are obtained by a partition of unity and convolution, and they
need not preserve semipositivity.  For $n=2$, Stokes' theorem gives
$\int_XD_k^2=0$, since
$D_k^2=\mathrm{d}(\dc w_k\wedge D_k)$.  For a fixed norm-comparison constant $C_0>0$,
\[
 \|D_k^2-D^2\|_{L^1}
 \leq C_0\|D_k-D\|_{L^2}(\|D_k\|_{L^2}+\|D\|_{L^2})\longrightarrow0.
\]
Thus $\int_XD^2=0$, and the nonpositive integrand in
\eqref{uni-eq:surface-sign} vanishes almost everywhere.  By the equality
case, $D=0$ almost everywhere.

\Needspace{8\baselineskip}
For $n=3$, also choose smooth functions $s_k\to s$ in $W^{2,3}$, and put
$C_k=\omega+\ddc s_k$.  We need the identity
\begin{equation}\label{uni-eq:torsion-integral}
 \int_X C_k\wedge D_k^2
 =\int_X\omega\wedge D_k^2
 =-\int_X\mathrm{d}w_k\wedge \dc w_k\wedge\ddc\omega.
\end{equation}
The first equality holds because
$\ddc s_k\wedge D_k^2=\mathrm{d}(\dc s_k\wedge D_k^2)$.
For the second equality, including its sign, put
$J_k=\mathrm{d}w_k\wedge \dc w_k$.  Using
$\dc \mathrm{d}=-\ddc $ and $(\dc )^2=0$,
\[
 \dc J_k=-D_k\wedge \dc w_k,\qquad \ddc J_k=-D_k^2.
\]
Stokes' theorem, applied first to $\mathrm{d}(\omega\wedge \dc J_k)$ and then to
$\dc (\mathrm{d}\omega\wedge J_k)$, gives
\begin{align*}
 \int_X\omega\wedge\ddc J_k&=-\int_X\mathrm{d}\omega\wedge \dc J_k,\\
 0&=-\int_X\ddc\omega\wedge J_k-\int_X\mathrm{d}\omega\wedge \dc J_k.
\end{align*}
Combining them yields \eqref{uni-eq:torsion-integral}.

With a fixed norm-comparison constant $C_0>0$, strong $L^3$ convergence implies convergence in $L^1$ of all products of three
factors.  For example,
\[
 \|(C_k-C)\wedge D_k^2\|_1
 \leq C_0\|C_k-C\|_3\|D_k\|_3^2\longrightarrow0,
\]
and
\[
 \|C\wedge(D_k^2-D^2)\|_1
 \leq C_0\|C\|_3\|D_k-D\|_3(\|D_k\|_3+\|D\|_3)\longrightarrow0.
\]
Also $\mathrm{d}w_k\wedge \dc w_k\to \mathrm{d}w\wedge \dc w$ in $L^{3/2}$, so the
right side of \eqref{uni-eq:torsion-integral} converges.  Consequently
\begin{equation}\label{uni-eq:torsion-limit}
 \int_XC\wedge D^2=-\int_X\mathrm{d}w\wedge \dc w\wedge\ddc\omega\geq0.
\end{equation}
The sign follows from
$\mathrm{d}w\wedge \dc w=(\mathrm{i}/(2\pi))\partial w\wedge\bar\partial w$
and \eqref{uni-eq:threefold-sign}.  The integrand on the left is nonpositive by
\eqref{uni-eq:threefold-algebra}, so it vanishes almost everywhere, and the equality case,
which applies where the common determinant is positive, gives $D=0$ almost everywhere.

In both dimensions, therefore, $\ddc w=0$ in the sense of distributions. By \cref{prop:local-correction} with background zero, $w$ has a plurisubharmonic representative on $X$. It attains its maximum on the compact manifold $X$, so the maximum principle and connectedness make it constant.

For the $\omega$-plurisubharmonic representatives, \cref{lem:qpsh-almost-everywhere-order} upgrades the almost-everywhere equality $u=v+\text{constant}$ to equality everywhere. In particular, normalizing the means gives equality in the Sobolev class, and normalizing the suprema gives equality of the plurisubharmonic representatives.
\end{proof}

\begin{corollary}\label{uni-cor:dense-lowdim}\label{cor:gauduchon-surface-uniqueness}\index{Uniqueness!Gauduchon surface}
Let $n\in\{2,3\}$, and let $u,v\in C^2(X)$ satisfy $A,B\geq0$ and $A^n=B^n=f\,\mathrm{d}V$, where $f$ is positive on a dense set. If $n=3$, assume \eqref{uni-eq:threefold-sign}; for instance, $\ddc\omega=0$ suffices. Then $u-v$ is constant.
\end{corollary}

\begin{proof}
Put $D=A-B$ and $C=(A+B)/2$. At a point where the common top power vanishes, the algebraic expressions
in the proof of \cref{uni-thm:sobolev} are still nonpositive: in dimension two $D^2=-2A\wedge B$, and in
dimension three
$2C\wedge D^2=-A^2\wedge B-A\wedge B^2$.
The same integral identities force these continuous expressions to vanish
everywhere.  The equality cases give $D=0$ on $\{f>0\}$, and
density and continuity then give $D=0$ on $X$. As in the proof of the theorem, $u-v$ is then constant.
\end{proof}


For example, the Gauduchon metric $\omega$ of the Hopf threefold of \cref{chap:hopf} satisfies $\ddc\omega=-\alpha^2$ with $\alpha=\pi^*\omega_{\FS}$ semipositive, by \eqref{eq:hopf-torsion}, and hence \eqref{uni-eq:threefold-sign}. \Cref{uni-thm:sobolev} therefore applies to this metric, even though its upper mass $v_+(\omega)$ of \cref{def:bmp} is infinite by \cref{thm:hopf-unbounded}.
