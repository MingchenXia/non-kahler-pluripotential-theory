\subsection{Divisorial products and zero volume}
\label{subsec:zero-volume-product-obstacles}

For products over a curve, the fibration criterion becomes a classification by the polar curvature of one curve potential. A normalized divisor potential on the second factor makes the induced zero obstacle equal to zero on the curve. Projective products give examples with no analytic ordinary type, as well as distinct models with identical zero volumes. Polar curvature on the base nevertheless gives a complete metric on this product locus.

Let $C$ be a compact connected curve with a smooth positive form $\eta$, and let $Z$ be a connected compact K\"ahler manifold of dimension $k\geq1$. Suppose that $D\subset Z$ is a nonempty smooth reduced divisor, $\kappa$ is a K\"ahler form, and $\psi\in\PSH(Z,\kappa)$ is smooth off $D$ and satisfies
\[
 \kappa+\ddc\psi=[D],\qquad \sup_Z\psi=0.
\]
Thus $\psi$ is locally $\log|z_1|^2$ plus a smooth function at $D$. On $M=C\times Z$, write $p:M\to C$ and $r:M\to Z$ for the projections, and put
\[
 \Theta=p^*\eta+r^*\kappa,\qquad
 \Psi=\psi\circ r,\qquad H=C\times D.
\]
For $w\in\PSH(C,\eta)$ set $u=\Psi+w\circ p$.

\Needspace{14\baselineskip}
\begin{proposition}\label{prop:divisorial-product-obstacles}\index{Envelope!divisorial product}\index{Singularity volume!zero-volume product}
For $F\in C^\infty(M,\mathbb R)$, the function
\[
 \widehat F(y)=\min_{z\in Z\setminus D}\{F(y,z)-\psi(z)\}
 \qquad(y\in C)
\]
is finite and continuous. The two envelopes satisfy
\begin{equation}\label{eq:divisorial-product-obstacles}
 \begin{split}
 P_{\Theta,F}[u]&=\Psi+P_{\eta,\widehat F}[w]\circ p,\\
 P_{\Theta,F}[u]_{\mathcal I}
     &=\Psi+P_{\eta,\widehat F}[w]_{\mathcal I}\circ p.
 \end{split}
\end{equation}
\end{proposition}
\begin{proof}
Apply \cref{prop:fibration-obstacle-reduction} to $p:M\to C$, $H=C\times D$, $\beta=\eta$ and $\Psi=\psi\circ r$. The logarithmic local form of $\psi$ defines a smooth metric on $\mathcal O_M(H)$ with curvature $r^*\kappa$. Thus $\Theta+\ddc\Psi=p^*\eta+[H]$, and $H$ is smooth and transverse to the fibers. The induced fiber minimum is the displayed $\widehat F$. When $F=0$, the normalization $\sup_Z\psi=0$ gives $\widehat 0=0$.
\end{proof}

\Needspace{10\baselineskip}
\begin{example}\label{ex:zero-volume-nonanalytic-product}
Take $C=\mathbb P^1$ and $Z=\mathbb P^k$, $k\geq1$, with their standard Fubini--Study forms, and let $D$ be a hyperplane. There is an unbounded $u$ with no analytic ordinary type in this product setting and with
\[
 P_\Theta[u]=P_\Theta[u]_{\mathcal I}=\Psi,
 \qquad u\not\sim\Psi,
\]
for which the two envelopes agree below every smooth obstacle. Their common zero-obstacle model has zero singularity volume.
\end{example}
\begin{proof}
Use the unbounded curve potential $w$ constructed in the proof of \cref{prop:ordinary-topology-nonseparation}, whose two envelopes equal zero. A logarithmic hyperplane potential on $Z$ supplies $\psi$ with $\kappa+\ddc\psi=[D]$ and supremum zero. Set $u=\Psi+w\circ p$ and apply \cref{prop:divisorial-product-obstacles,cor:fibration-obstacle-equality}. Fixing $z\notin D$ shows that $u-\Psi=w\circ p$ is unbounded, proving $u\not\sim\Psi$. If an analytic potential $a$ were ordinarily equivalent to $u$, \cref{prop:analytic-envelope-type} would give $u\sim a\sim P_\Theta[a]=P_\Theta[u]=\Psi$, a contradiction.
\end{proof}

Zero volume alone does not force the two envelopes to agree.

\Needspace{18\baselineskip}
\begin{example}\label{ex:zero-volume-distinct-product-models}\index{Envelope!distinct zero-volume models}\index{Singularity volume!distinct zero-volume models}
In the product setting take $C=\mathbb P^1$ and $Z=\mathbb P^k$, $k\geq1$, with probability Fubini--Study forms and a hyperplane $D$. The positive closed background $\Theta$ admits normalized ordinary models $a\leq b$ with
\[
 \begin{gathered}
 P_\Theta[a]=a,\qquad
 P_\Theta[a]_{\mathcal I}=P_\Theta[b]=P_\Theta[b]_{\mathcal I}=b,\\
 a\sim_{\mathcal I}b,\qquad a\not\sim b,\\
 \underline{\operatorname{vol}}_\Theta(a)
 =\overline{\operatorname{vol}}_\Theta(a)
 =\underline{\operatorname{vol}}_\Theta(b)
 =\overline{\operatorname{vol}}_\Theta(b)=0.
 \end{gathered}
\]
Thus equal singularity volumes for a type and its ideal model do not imply envelope equality at zero volume.
\end{example}
\begin{proof}
Use the normalized diffuse polar curve model $w$ of \cref{ex:diffuse-polar-zero-model} on $\mathbb P^1$ with its probability Fubini--Study form. It satisfies $P_\eta[w]=w$, $P_\eta[w]_{\mathcal I}=0$, and is unbounded. Take the normalized hyperplane potential on $\mathbb P^k$ and put
\[
 a=\Psi+w\circ p,\qquad b=\Psi.
\]
The zero-obstacle identity in \cref{prop:divisorial-product-obstacles} gives all four envelope identities. Both functions have supremum zero, and $w\leq0$ gives $a\leq b$. Ideal projection preserves the ideal type by \cref{thm:hermitian-i-projection}, so $a\sim_{\mathcal I}b$. Restriction to $\mathbb P^1\times\{z\}$, for any $z\notin D$, gives the unbounded difference $w$, proving $a\not\sim b$. Finally, \cref{prop:fibration-obstacle-reduction} gives zero extended mass for every ordinary representative of either model. This proves all four volume identities.
\end{proof}

The same failure occurs in a genuinely nonclosed background after adding a second curve factor.

\Needspace{16\baselineskip}
\begin{example}\label{ex:nonclosed-zero-volume-distinct-models}\index{Envelope!nonclosed zero-volume obstruction}\index{Singularity volume!nonclosed zero-volume models}
In every complex dimension $n\geq3$, a projective manifold with bounded mass admits a smooth positive nonclosed background $\Theta$ and normalized ordinary models $a\leq b$ such that
\[
 \begin{gathered}
 P_\Theta[a]=a,\qquad
 P_\Theta[a]_{\mathcal I}=P_\Theta[b]=P_\Theta[b]_{\mathcal I}=b,\\
 a\sim_{\mathcal I}b,\qquad a\not\sim b,\\
 \underline{\operatorname{vol}}_\Theta(a)
 =\overline{\operatorname{vol}}_\Theta(a)
 =\underline{\operatorname{vol}}_\Theta(b)
 =\overline{\operatorname{vol}}_\Theta(b)=0.
 \end{gathered}
\]
In particular, equality of upper volumes with the ideal model does not force envelope equality when the ideal model has zero lower volume.
\end{example}
\begin{proof}
Let $Y=\mathbb P^1\times\mathbb P^1$, with projections $\pi_1,\pi_2$, and let $\sigma$ be the probability Fubini--Study form on $\mathbb P^1$. Choose a positive nonconstant smooth function $f$ on the first factor and put
\[
 \beta=\pi_1^*\sigma+(f\circ\pi_1)\pi_2^*\sigma.
\]
This form is positive and $\mathrm d\beta=\pi_1^*(\mathrm df)\wedge\pi_2^*\sigma\ne0$. Let $w_0$ be the normalized diffuse polar model of \cref{ex:diffuse-polar-zero-model}, and put $w=w_0\circ\pi_1$. Then $w\in\PSH(Y,\beta)$ and
\[
 P_\beta[w]=w,\qquad P_\beta[w]_{\mathcal I}=0.
\]
Indeed, for an ordinary competitor $v\leq0$, replace it by $\max(v,w)$. On every section $y\mapsto(y,z)$ this maximum is a genuine $\sigma$-plurisubharmonic function, since it dominates $w_0$, and it is more singular than $w_0$. Thus its restriction is at most $P_\sigma[w_0]=w_0$. This proves $v\leq w$ on $Y$; the reverse envelope inequality follows because $w$ is itself a competitor. All multiplier ideals of $w_0$ are trivial, so \cref{lem:submersion-multiplier-pullback} gives the same assertion for $w$. Since $\beta$ is positive, zero is a competitor for its ideal envelope, proving the second identity.

Set $k=n-2$ and $M=Y\times\mathbb P^k$, with projections $p,r$. On the second factor take the probability Fubini--Study form $\kappa$, a hyperplane $D$, and its normalized potential $\psi$ with $\kappa+\ddc\psi=[D]$. Put
\[
 \Theta=p^*\beta+r^*\kappa,\qquad
 \Psi=\psi\circ r,\qquad a=\Psi+w\circ p,\qquad b=\Psi.
\]
The form $\Theta$ is positive and nonclosed. A sufficiently large multiple of the closed positive product form dominates $\Theta$. The bounded mass property follows from Stokes' theorem for that closed form and metric independence in \cref{prop:bmp-metric}.

Apply \cref{prop:fibration-obstacle-reduction} to $p:M\to Y$ and the divisor $Y\times D$. The induced zero obstacle is zero because $\sup\psi=0$. The two base-envelope identities give all four upstairs identities, and both functions have supremum zero. Ideal projection preserves the ideal type, so $a\sim_{\mathcal I}b$. Moreover, $a\leq b$, while their difference restricts to the unbounded $w$ on any section outside the divisor, so $a\not\sim b$. The same fibration proposition gives zero extended mass for every ordinary representative of $a$ or $b$, since $\dim Y=2<n$. This proves the volume assertions.
\end{proof}

Return to the original product $M=C\times Z$ and its fixed divisor potential $\Psi$. Although its top-degree volumes vanish, the polar curvature on $C$ still distinguishes the induced ordinary models.

\begin{definition}\label{def:divisorial-product-model-locus}\index{Model potential!divisorial product locus}
For a normalized ordinary model $q$ in background $\eta$ on $C$, put
\[
 \Phi(q)=\Psi+q\circ p,\qquad
 \mathcal S_\Psi=\{\Phi(q):P_\eta[q]=q\}.
\]
Write $\mu_q=\mathbf1_{\{q=-\infty\}}(\eta+\ddc q)$ for its polar curvature. Define
\begin{equation}\label{eq:divisorial-product-polar-metric}
 \rho_\Psi(\Phi(q_1),\Phi(q_2))
       =\|\mu_{q_1}-\mu_{q_2}\|_{\mathrm{TV}}.
\end{equation}
\end{definition}

\Needspace{15\baselineskip}
\begin{proposition}\label{prop:divisorial-product-polar-metric}\index{Metric completeness!zero-volume product}\index{Model potential!polar product metric}
The set $\mathcal S_\Psi$ consists of normalized ordinary models in background $\Theta$, and $(\mathcal S_\Psi,\rho_\Psi)$ is a complete geodesic metric space. The volume comparison of \cref{def:model-volume-comparison} satisfies
\[
 d_\Theta(a,b)=0\qquad(a,b\in\mathcal S_\Psi).
\]
The ideal models in $\mathcal S_\Psi$ form a closed geodesically convex subspace, equal to the $\rho_\Psi$-closure of its analytic models. Ideal projection is a $1$-Lipschitz retraction onto this subspace and selects the unique nearest ideal model. For $a=\Phi(q)$ its distance is the mass of the nonatomic part of $\mu_q$. If $\rho_\Psi(a_j,a)\to0$, then $a_j\to a$ in $L^1(M)$ for every smooth positive volume form.
\end{proposition}
\begin{proof}
The zero-obstacle identities in \cref{prop:divisorial-product-obstacles} give
\[
 P_\Theta[\Phi(q)]=\Phi(q),\qquad
 P_\Theta[\Phi(q)]_{\mathcal I}
       =\Phi(P_\eta[q]_{\mathcal I}).
\]
Both summands have supremum zero on their respective factors, so $\Phi(q)$ is normalized. Restriction to any section outside $D$ shows that $\Phi$ is injective. The classification in \cref{lem:curve-polar-model} and the distance formula in \cref{thm:curve-model-volume-distance} therefore identify $\Phi$ with an isometry from the curve model space onto $\mathcal S_\Psi$. Completeness and geodesics follow from \cref{cor:curve-model-completeness}.

Every ordinary representative of $\Phi(q)$ has zero extended mass by \cref{prop:fibration-obstacle-reduction}. Moreover,
\[
 \max(\Phi(q_1),\Phi(q_2))
       =\Psi+\max(q_1,q_2)\circ p.
\]
The same proposition applies to this curve potential and gives zero upper volume for the maximum. Thus all three upper volumes in the definition of $d_\Theta$ vanish.

Ideal projection corresponds under $\Phi$ to taking the atomic part of $\mu_q$. The closedness, convexity and nearest-point assertions follow from \cref{cor:curve-model-completeness,prop:curve-analytic-distance-defect}. To check analytic closure upstairs, approximate an atomic measure by finite rational atomic measures of mass less than $\int_C\eta$. Their curve models have the corresponding logarithmic poles up to bounded remainder, by \cref{lem:curve-atomic-ideal-model}. Adding $\Psi$ preserves an analytic presentation: clear the denominators of the rational pole coefficients and the unit coefficient of $D$ in the local product monomial. The resulting models in $\mathcal S_\Psi$ converge in $\rho_\Psi$. Conversely, every analytic ordinary model in this locus is its ideal projection by \cref{prop:analytic-envelope-type}, so its closure lies in the closed ideal subspace.

Finally, \cref{thm:curve-model-capacity-convergence} gives $q_j\to q$ in $L^1(C)$ when $\rho_\Psi(\Phi(q_j),\Phi(q))\to0$. Off the pole sets, which have smooth volume zero, the difference of the two product potentials is $(q_j-q)\circ p$. For a smooth positive volume form $\mathrm dV$ on $M$, integration along the compact fibers makes $p_*\mathrm dV$ a smooth positive density on $C$. Fubini gives
\[
 \int_M|\Phi(q_j)-\Phi(q)|\,\mathrm dV
       \leq C_{\mathrm dV}\|q_j-q\|_{L^1(C,\eta)}\longrightarrow0.
\]
Here $C_{\mathrm dV}$ bounds the density of $p_*\mathrm dV$ relative to $\eta$.
\end{proof}

The complete metric also exists on a specified product locus in the nonclosed setting of \cref{ex:nonclosed-zero-volume-distinct-models}.

\Needspace{16\baselineskip}
\begin{corollary}\label{cor:nonclosed-product-polar-metric}\index{Metric completeness!nonclosed zero-volume product}\index{Model potential!nonclosed polar product metric}
Let $M=\mathbb P^1\times\mathbb P^1\times\mathbb P^k$, $k\geq1$, with projections $h,s,r$. Let $\sigma,\kappa$ be the probability Fubini--Study forms on $\mathbb P^1,\mathbb P^k$, respectively, and let $f$ be a positive nonconstant smooth real function on the first factor. Set
\[
 \Theta=h^*\sigma+(f\circ h)s^*\sigma+r^*\kappa,
 \qquad \Psi=\psi\circ r,
\]
where $\psi$ is the normalized hyperplane potential with $\kappa+\ddc\psi=[D]$. For every normalized ordinary model $q$ on $(\mathbb P^1,\sigma)$, put $\Phi_f(q)=\Psi+q\circ h$. Then
\[
 P_\Theta[\Phi_f(q)]=\Phi_f(q),\qquad
 P_\Theta[\Phi_f(q)]_{\mathcal I}
       =\Phi_f(P_\sigma[q]_{\mathcal I}).
\]
On $\mathcal T_\Psi=\{\Phi_f(q):P_\sigma[q]=q\}$, define
\[
 \begin{gathered}
 \rho_{\Psi,f}(\Phi_f(q_1),\Phi_f(q_2))
       =\|\mu_{q_1}-\mu_{q_2}\|_{\mathrm{TV}},\\
 \mu_q=\mathbf1_{\{q=-\infty\}}(\sigma+\ddc q).
 \end{gathered}
\]
All the conclusions of \cref{prop:divisorial-product-polar-metric} hold on this locus: completeness and geodesics, identically zero top-volume comparison, closed ideal subspace with analytic density, nearest-point ideal projection and $L^1$ convergence upstairs.
\end{corollary}
\begin{proof}
On $Y=\mathbb P^1\times\mathbb P^1$, write $\pi_1,\pi_2$ for the projections and put $\beta=\pi_1^*\sigma+(f\circ\pi_1)\pi_2^*\sigma$. The restriction argument in \cref{ex:nonclosed-zero-volume-distinct-models} applies to every normalized ordinary curve model, giving
\[
 P_\beta[q\circ\pi_1]=q\circ\pi_1.
\]
Indeed, the maximum of an ordinary competitor with $q\circ\pi_1$ restricts on every section to a genuine $\sigma$-plurisubharmonic competitor for $P_\sigma[q]=q$.

Put $q_{\mathcal I}=P_\sigma[q]_{\mathcal I}$ and let $E$ be the atomic part of $\mu_q$. The Green potential $w_E$ of \cref{lem:curve-atomic-ideal-model} satisfies $q_{\mathcal I}\sim w_E$ and $q\sim_{\mathcal I}w_E$. Its pullback has the decomposition
\[
 \ddc(w_E\circ\pi_1)
       =\sum_xE(\{x\})[\{x\}\times\mathbb P^1]
           -E(\mathbb P^1)\pi_1^*\sigma.
\]
The divisor current is closed and has finite mass, and $w_E\circ\pi_1$ is $\beta$-plurisubharmonic because $E(\mathbb P^1)\leq1$. \Cref{cor:divisorial-envelope-equality} therefore equates its ordinary and ideal envelopes. Multiplier ideals pull back by \cref{lem:submersion-multiplier-pullback}; ordinary equivalence and the preceding ordinary projection give
\[
 P_\beta[q\circ\pi_1]_{\mathcal I}
       =P_\beta[w_E\circ\pi_1]_{\mathcal I}
       =P_\beta[w_E\circ\pi_1]
       =q_{\mathcal I}\circ\pi_1.
\]
This includes $E=0$ and countably many atoms.

Apply \cref{prop:fibration-obstacle-reduction} to $M\to Y$ and $Y\times D$, whose induced zero obstacle is zero. The two base identities give the asserted upstairs identities. Thus $\Phi_f$ identifies the locus isometrically with the full curve model space, and ideal projection again takes the atomic part. The metric, nearest-point and analytic-closure arguments of \cref{prop:divisorial-product-polar-metric} apply unchanged: rational atomic coefficients and the unit hyperplane coefficient give local analytic monomials. Each pairwise maximum still has the form $\Psi+\max(q_1,q_2)\circ h$, so the fibration proposition makes all three upper volumes zero. Finally, integration of a smooth volume form along the compact fibers of $h$ transfers curve $L^1$ convergence to $M$, as in that proposition. The background is positive and genuinely nonclosed, and the dominating closed product metric gives bounded mass, as in \cref{ex:nonclosed-zero-volume-distinct-models}.
\end{proof}
