\subsection{Divisorial products and zero volume}
\label{subsec:zero-volume-product-obstacles}

For products over a curve, the fibration criterion becomes a classification by the polar curvature of one curve potential. A normalized divisor potential on the second factor makes the induced zero obstacle equal to zero on the curve. Projective products give examples with no analytic ordinary type, as well as distinct models with identical zero volumes.

Let $C$ be a compact connected curve with a smooth positive form $\eta$, and let $Z$ be a connected compact K\"ahler manifold of dimension $k\geq1$. Suppose that $D\subset Z$ is a nonempty smooth reduced divisor, $\kappa$ is a K\"ahler form, and $\psi\in\PSH(Z,\kappa)$ is smooth off $D$ and satisfies
\[
 \kappa+\ddc\psi=[D],\qquad \sup_Z\psi=0.
\]
Thus $\psi$ is locally $\log|z_1|^2$ plus a smooth function at $D$. On $M=C\times Z$, write $p:M\to C$ and $r:M\to Z$ for the projections, and put
\[
 \Theta=p^*\eta+r^*\kappa,\qquad
 \Psi=\psi\circ r,\qquad H=C\times D.
\]
For $w\in\PSH(C,\eta)$ set $u=\Psi+w\circ p$.

\Needspace{14\baselineskip}
\begin{proposition}\label{prop:divisorial-product-obstacles}\index{Envelope!divisorial product}\index{Singularity volume!zero-volume product}
For $F\in C^\infty(M,\mathbb R)$, the function
\[
 \widehat F(y)=\min_{z\in Z\setminus D}\{F(y,z)-\psi(z)\}
 \qquad(y\in C)
\]
is finite and continuous. The two envelopes satisfy
\begin{equation}\label{eq:divisorial-product-obstacles}
 \begin{split}
 P_{\Theta,F}[u]&=\Psi+P_{\eta,\widehat F}[w]\circ p,\\
 P_{\Theta,F}[u]_{\mathcal I}
     &=\Psi+P_{\eta,\widehat F}[w]_{\mathcal I}\circ p.
 \end{split}
\end{equation}
\end{proposition}
\begin{proof}
Apply \cref{prop:fibration-obstacle-reduction} to $p:M\to C$, $H=C\times D$, $\beta=\eta$ and $\Psi=\psi\circ r$. The logarithmic local form of $\psi$ defines a smooth metric on $\mathcal O_M(H)$ with curvature $r^*\kappa$. Thus $\Theta+\ddc\Psi=p^*\eta+[H]$, and $H$ is smooth and transverse to the fibers. The induced fiber minimum is the displayed $\widehat F$. When $F=0$, the normalization $\sup_Z\psi=0$ gives $\widehat 0=0$.
\end{proof}

\Needspace{10\baselineskip}
\begin{example}\label{ex:zero-volume-nonanalytic-product}
Take $C=\mathbb P^1$ and $Z=\mathbb P^k$, $k\geq1$, with their standard Fubini--Study forms, and let $D$ be a hyperplane. There is an unbounded $u$ with no analytic ordinary type in this product setting and with
\[
 P_\Theta[u]=P_\Theta[u]_{\mathcal I}=\Psi,
 \qquad u\not\sim\Psi,
\]
for which the two envelopes agree below every smooth obstacle. Their common zero-obstacle model has zero singularity volume.
\end{example}
\begin{proof}
Use the unbounded curve potential $w$ constructed in the proof of \cref{prop:ordinary-topology-nonseparation}, whose two envelopes equal zero. A logarithmic hyperplane potential on $Z$ supplies $\psi$ with $\kappa+\ddc\psi=[D]$ and supremum zero. Set $u=\Psi+w\circ p$ and apply \cref{prop:divisorial-product-obstacles,cor:fibration-obstacle-equality}. Fixing $z\notin D$ shows that $u-\Psi=w\circ p$ is unbounded, proving $u\not\sim\Psi$. If an analytic potential $a$ were ordinarily equivalent to $u$, \cref{prop:analytic-envelope-type} would give $u\sim a\sim P_\Theta[a]=P_\Theta[u]=\Psi$, a contradiction.
\end{proof}

Zero volume alone does not force the two envelopes to agree.

\Needspace{18\baselineskip}
\begin{example}\label{ex:zero-volume-distinct-product-models}\index{Envelope!distinct zero-volume models}\index{Singularity volume!distinct zero-volume models}
In the product setting take $C=\mathbb P^1$ and $Z=\mathbb P^k$, $k\geq1$, with probability Fubini--Study forms and a hyperplane $D$. The positive closed background $\Theta$ admits normalized ordinary models $a\leq b$ with
\[
 \begin{gathered}
 P_\Theta[a]=a,\qquad
 P_\Theta[a]_{\mathcal I}=P_\Theta[b]=P_\Theta[b]_{\mathcal I}=b,\\
 a\sim_{\mathcal I}b,\qquad a\not\sim b,\\
 \underline{\operatorname{vol}}_\Theta(a)
 =\overline{\operatorname{vol}}_\Theta(a)
 =\underline{\operatorname{vol}}_\Theta(b)
 =\overline{\operatorname{vol}}_\Theta(b)=0.
 \end{gathered}
\]
Thus equal singularity volumes for a type and its ideal model do not imply envelope equality at zero volume.
\end{example}
\begin{proof}
Use the normalized diffuse polar curve model $w$ of \cref{ex:diffuse-polar-zero-model} on $\mathbb P^1$ with its probability Fubini--Study form. It satisfies $P_\eta[w]=w$, $P_\eta[w]_{\mathcal I}=0$, and is unbounded. Take the normalized hyperplane potential on $\mathbb P^k$ and put
\[
 a=\Psi+w\circ p,\qquad b=\Psi.
\]
The zero-obstacle identity in \cref{prop:divisorial-product-obstacles} gives all four envelope identities. Both functions have supremum zero, and $w\leq0$ gives $a\leq b$. Ideal projection preserves the ideal type by \cref{thm:hermitian-i-projection}, so $a\sim_{\mathcal I}b$. Restriction to $\mathbb P^1\times\{z\}$, for any $z\notin D$, gives the unbounded difference $w$, proving $a\not\sim b$. Finally, \cref{prop:fibration-obstacle-reduction} gives zero extended mass for every ordinary representative of either model. This proves all four volume identities.
\end{proof}

The same failure occurs in a genuinely nonclosed background after adding a second curve factor.

\Needspace{16\baselineskip}
\begin{example}\label{ex:nonclosed-zero-volume-distinct-models}\index{Envelope!nonclosed zero-volume obstruction}\index{Singularity volume!nonclosed zero-volume models}
In every complex dimension $n\geq3$, a projective manifold with bounded mass admits a smooth positive nonclosed background $\Theta$ and normalized ordinary models $a\leq b$ such that
\[
 \begin{gathered}
 P_\Theta[a]=a,\qquad
 P_\Theta[a]_{\mathcal I}=P_\Theta[b]=P_\Theta[b]_{\mathcal I}=b,\\
 a\sim_{\mathcal I}b,\qquad a\not\sim b,\\
 \underline{\operatorname{vol}}_\Theta(a)
 =\overline{\operatorname{vol}}_\Theta(a)
 =\underline{\operatorname{vol}}_\Theta(b)
 =\overline{\operatorname{vol}}_\Theta(b)=0.
 \end{gathered}
\]
In particular, equality of upper volumes with the ideal model does not force envelope equality when the ideal model has zero lower volume.
\end{example}
\begin{proof}
Let $Y=\mathbb P^1\times\mathbb P^1$, with projections $\pi_1,\pi_2$, and let $\sigma$ be the probability Fubini--Study form on $\mathbb P^1$. Choose a positive nonconstant smooth function $f$ on the first factor and put
\[
 \beta=\pi_1^*\sigma+(f\circ\pi_1)\pi_2^*\sigma.
\]
This form is positive and $\mathrm d\beta=\pi_1^*(\mathrm df)\wedge\pi_2^*\sigma\ne0$. Let $w_0$ be the normalized diffuse polar model of \cref{ex:diffuse-polar-zero-model}, and put $w=w_0\circ\pi_1$. Then $w\in\PSH(Y,\beta)$ and
\[
 P_\beta[w]=w,\qquad P_\beta[w]_{\mathcal I}=0.
\]
Indeed, for an ordinary competitor $v\leq0$, replace it by $\max(v,w)$. On every section $y\mapsto(y,z)$ this maximum is a genuine $\sigma$-plurisubharmonic function, since it dominates $w_0$, and it is more singular than $w_0$. Thus its restriction is at most $P_\sigma[w_0]=w_0$. This proves $v\leq w$ on $Y$; the reverse envelope inequality follows because $w$ is itself a competitor. All multiplier ideals of $w_0$ are trivial, so \cref{lem:submersion-multiplier-pullback} gives the same assertion for $w$. Since $\beta$ is positive, zero is a competitor for its ideal envelope, proving the second identity.

Set $k=n-2$ and $M=Y\times\mathbb P^k$, with projections $p,r$. On the second factor take the probability Fubini--Study form $\kappa$, a hyperplane $D$, and its normalized potential $\psi$ with $\kappa+\ddc\psi=[D]$. Put
\[
 \Theta=p^*\beta+r^*\kappa,\qquad
 \Psi=\psi\circ r,\qquad a=\Psi+w\circ p,\qquad b=\Psi.
\]
The form $\Theta$ is positive and nonclosed. A sufficiently large multiple of the closed positive product form dominates $\Theta$. The bounded mass property follows from Stokes' theorem for that closed form and metric independence in \cref{prop:bmp-metric}.

Apply \cref{prop:fibration-obstacle-reduction} to $p:M\to Y$ and the divisor $Y\times D$. The induced zero obstacle is zero because $\sup\psi=0$. The two base-envelope identities give all four upstairs identities, and both functions have supremum zero. Ideal projection preserves the ideal type, so $a\sim_{\mathcal I}b$. Moreover, $a\leq b$, while their difference restricts to the unbounded $w$ on any section outside the divisor, so $a\not\sim b$. The same fibration proposition gives zero extended mass for every ordinary representative of $a$ or $b$, since $\dim Y=2<n$. This proves the volume assertions.
\end{proof}
