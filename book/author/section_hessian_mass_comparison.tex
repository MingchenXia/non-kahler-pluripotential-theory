\section{Hessian mass and global comparison}
\label{sec:hessian-mass-comparison}

The maximum inequality of \cref{hessian:bounded-maximum-measure}
compares measures on complementary Borel sets.
Integrating it leaves a difference of total masses. The recursive
currents express this difference through the complex Hessians of the
background powers, extending the mass calculation of
\cref{chap:mass} to bounded admissible potentials. Their vanishing
characterizes the unmodified comparison principle. When they do not
vanish, a small sublevel set limits the change made by the maximum;
the resulting torsion error can still be absorbed after rescaling
the upper potential.

For bounded admissible $t_1,\ldots,t_m$, put
\[
 \mathcal M_m(t_1,\ldots,t_m)=\int_XQ_m(t_1,\ldots,t_m),
\]
the total mass of a mixed positive measure by
\cref{hessian:bounded-shifted-positive}. Recall from
\cref{sec:hessian-bounded-locality} that $H_m(t)=Q_m(t,\ldots,t)$
is the finite positive Radon measure with all potentials equal
to~$t$. For a nonempty subset
$I=\{i_1<\cdots<i_q\}\subset\{1,\ldots,m\}$, put
$i(I)=i_q$ and $I^-=I\setminus\{i_q\}$. The currents $D_I$ and
$D_{I^-}$ are the recursive currents of the list $t_1,\ldots,t_m$
introduced before \eqref{eq:hessian-bounded-shifted-current};
in particular $D_\varnothing=1$.

\Needspace{17\baselineskip}
\begin{lemma}\label{hessian:bounded-mass-expansion}\index{Hessian measure!total mass}
For every bounded admissible list,
\begin{equation}\label{eq:hessian-bounded-mass-expansion}
 \mathcal M_m(t_1,\ldots,t_m)
 =\int_X\omega^n+
 \sum_{\varnothing\ne I\subset\{1,\ldots,m\}}
 \int_X t_{i(I)}D_{I^-}\wedge\ddc(\omega^{n-|I|}).
\end{equation}
Every integral on the right is an integral of bounded canonical
values against signed measure coefficients.
\end{lemma}
\begin{proof}
Expand $Q_m$ using \eqref{eq:hessian-bounded-shifted-current}.
The term for $I$ pairs $D_I$ with the smooth form
$\omega^{n-|I|}$. By \eqref{eq:hessian-bounded-recursion},
\[
 \langle D_I,\omega^{n-|I|}\rangle
 =\langle t_{i(I)}D_{I^-},\ddc(\omega^{n-|I|})\rangle.
\]
Since $|I^-|\leq m-1$, \cref{hessian:bounded-recursive-currents}
gives measure coefficients for $D_{I^-}$; their bounded product
with $t_{i(I)}$ is well defined. Thus the displayed pairing is
exactly the integral in the sum, and is finite on compact $X$.
The empty subset contributes $\int_X\omega^n$.
\end{proof}

Let $\mu_r$ be the repeated-slot mixed measure from
\cref{hessian:bounded-maximum-measure}, for fixed $r$ and
$z_1,\ldots,z_{m-r}$.
The total masses of $\mu_r(w)$ and $\mu_r(u)$ in the next proposition
are values of $\mathcal M_m$, so
\eqref{eq:hessian-bounded-mass-expansion} for the two corresponding
lists expresses their difference explicitly.

\Needspace{11\baselineskip}
\begin{proposition}\label{hessian:comparison-mass-defect}\index{Comparison principle!Hessian mass defect}
If $u,v$ are bounded admissible and $w=\max\{u,v\}$, then
\[
 \int_{\{u<v\}}\mathrm d\mu_r(v)
 \leq\int_{\{u<v\}}\mathrm d\mu_r(u)
                +\int_X\mathrm d\mu_r(w)-\int_X\mathrm d\mu_r(u).
\]
\end{proposition}
\begin{proof}
Set $E=\{u<v\}$. \Cref{hessian:bounded-maximum-measure}, whose
first term includes the contact set $\{u=v\}$, gives
\[
 \mu_r(w)\geq\mathbf1_{X\setminus E}\mu_r(u)
                     +\mathbf1_E\mu_r(v).
\]
All these measures are finite and positive. Integrating on $X$ and
using $\mu_r(u)(X\setminus E)=\mu_r(u)(X)-\mu_r(u)(E)$ gives
exactly the assertion. No boundary or closed-background hypothesis
enters this calculation.
\end{proof}

The unmodified global comparison principle holds precisely when the
powers occurring in this mass formula are $\ddc$-closed. Only the
powers $n-m,\ldots,n-1$ occur at Hessian degree $m$.

\Needspace{24\baselineskip}
\begin{theorem}\label{hessian:global-comparison-characterization}\index{Hessian measure!invariant mass}\index{Comparison principle!Hessian power identities}
The following conditions are equivalent:
\begin{enumerate}
\item $\ddc(\omega^{n-q})=0$ for every $1\leq q\leq m$;
\item $\int_XH_m(u)$ is independent of the smooth admissible
      potential $u$;
\item $\mathcal M_m(t_1,\ldots,t_m)=\int_X\omega^n$ for every
      bounded admissible list;
\item every pair of bounded admissible $u,v$ satisfies
\[
 \int_{\{u<v\}}H_m(v)\leq\int_{\{u<v\}}H_m(u).
\]
\end{enumerate}
Under these conditions, for every $1\leq r\leq m$, every fixed bounded
admissible list $z_1,\ldots,z_{m-r}$ and all bounded admissible $u,v$,
one also has
\[
 \int_{\{u<v\}}\mathrm d\mu_r(v)
           \leq\int_{\{u<v\}}\mathrm d\mu_r(u).
\]
\end{theorem}
\begin{proof}
The first condition implies the third by
\eqref{eq:hessian-bounded-mass-expansion}. The third implies the
second by taking all inputs equal to a smooth potential.

To prove that the second implies the first, fix any smooth real
$f$. For all sufficiently small real $t$ of either sign,
$\omega+t\ddc f$ is positive, so $tf$ is strictly admissible.
The polynomial
\[
 \int_X(\omega+t\ddc f)^m\wedge\omega^{n-m}
 =\sum_{q=0}^m\binom mq t^q
              \int_X(\ddc f)^q\wedge\omega^{n-q}
\]
is constant on an interval. Consequently every coefficient with
$q\geq1$ vanishes. Replacing $f$ by $\sum_{a=1}^q s_af_a$ in the
coefficient of $t^q$ and comparing the coefficients of
$s_1\cdots s_q$ gives
\[
 \int_X\ddc f_1\wedge\cdots\wedge\ddc f_q
                        \wedge\omega^{n-q}=0
\]
for all smooth real $f_a$. Integration by parts in the first
factor, using \cref{lem:ibp}, gives
\[
 \int_Xf_1\ddc f_2\wedge\cdots\wedge\ddc f_q
                         \wedge\ddc(\omega^{n-q})=0.
\]
Since $f_1$ is arbitrary, the smooth top-degree form paired with
it vanishes pointwise. At each point the other complex Hessians
can be prescribed by real quadratic functions times a cutoff.
Complexifying this multilinear identity gives all coordinate
$(q-1,q-1)$-forms, as in the proof of
\cref{thm:mass-invariance-characterization}. They detect every
coefficient of $\ddc(\omega^{n-q})$, so that form vanishes. When
$q=1$, there are no other Hessians and the same conclusion follows
directly from the arbitrariness of $f_1$.

Under the third condition, $\mu_r(w)$ and $\mu_r(u)$ both have
total mass $\int_X\omega^n$. \Cref{hessian:comparison-mass-defect} proves the mixed
comparison assertion, and $r=m$ gives the fourth condition.
Finally, suppose the fourth condition holds and fix smooth admissible
$u,v$. Choose $C$ so large that $u<v+C$ everywhere.
Constant invariance gives $H_m(v+C)=H_m(v)$, so comparison gives
$\int_XH_m(v)\leq\int_XH_m(u)$. Exchanging $u,v$ gives equality,
which is the second condition.
\end{proof}

The varying background of \cref{ex:torus-mass}, which obstructs
Monge--Amp\`ere comparison, also obstructs comparison at each
intermediate Hessian degree.

\Needspace{20\baselineskip}
\begin{example}\label{hessian:torus-comparison-obstruction}\index{Torus!Hessian comparison obstruction}
Let $X=(\mathbb C/(\mathbb Z+\mathrm i\mathbb Z))^n$, with
$z_j=x_j+\mathrm i y_j$, and put
\[
 \beta_j=\frac{\mathrm i}{2\pi}\,\mathrm dz_j\wedge\mathrm d\bar z_j,
 \qquad b=2+\cos(2\pi x_1),
 \qquad\omega=\beta_1+b\beta_2+\sum_{j=3}^n\beta_j.
\]
For $|t|<\pi^{-2}$ the potential $u_t=t\cos(2\pi x_1)$ is smooth
strictly admissible, and, with $B_n=\int_X\beta_1\wedge\cdots\wedge\beta_n$,
\[
 \int_X H_m(u_t)=n!\left(2-\frac{m\pi^2t}{2n}\right)B_n.
\]
Thus every $-\pi^{-2}<t<0$, together with a constant $C>|t|$,
gives a pair $u=0$, $v=u_t+C$ that violates the unmodified global
comparison inequality.
\end{example}
\begin{proof}
As in \cref{ex:torus-mass},
$\ddc u_t=-\pi^2t\cos(2\pi x_1)\beta_1$.
The coefficient of $\beta_1$ in $\omega+\ddc u_t$ is strictly
positive in the stated range, and all the other coefficients are
positive. Since $(\ddc u_t)^2=0$, expansion gives
\[
 H_m(u_t)=\omega^n+m\ddc u_t\wedge\omega^{n-1}
 =n!b\left(1-\frac mn\pi^2t\cos(2\pi x_1)\right)
                         \beta_1\wedge\cdots\wedge\beta_n.
\]
The period averages of $b$ and $b\cos(2\pi x_1)$ are respectively
$2$ and $1/2$, proving the mass formula. For $t<0$ this mass is
strictly larger than $\int_XH_m(0)=2n!B_n$. Meanwhile $C>|t|$
gives $\{0<u_t+C\}=X$, so the comparison inequality for this pair
would assert the opposite ordering of these masses.
\end{proof}

\subsection{Modified comparison}
\label{subsec:hessian-modified-comparison}

On a sublevel set just above the infimum of a difference, the maximum
changes the lower potential by at most a prescribed small number.
Integration by parts then expresses the mass defect through this
small difference and a torsion measure. Mixed lower-degree traces
control its variation. Rescaling the upper potential adds a positive
background term that relates these traces to the top Hessian measure.

For bounded admissible $u,V$, use the notation
\[
 \sigma_u=\omega+\ddc u,\qquad
 M_{p,q}=\sigma_u^p\sigma_V^q\omega^{n-p-q}
                    \quad(0\leq p+q\leq m).
\]
These expressions denote the positive mixed measure $Q_m$ with
$p$ slots equal to $u$, $q$ slots equal to $V$ and the remaining
slots equal to the constant zero. In particular $M_{0,0}=\omega^n$.
\Needspace{6\baselineskip}
For $p+q\leq m-1$, put
\[
 T_{p,q}=\sigma_u^p\sigma_V^q\omega^{n-p-q-1}.
\]
This denotes $R_{m-1}$ with the same nonconstant slots and zero
in its other slots, wedged with $\omega^{n-m}$.

\Needspace{15\baselineskip}
\begin{lemma}\label{hessian:mixed-torsion-comparison}\index{Comparison principle!mixed Hessian torsion}
Let $u,V$ be bounded admissible, $b$ a real constant and $s>0$.
Suppose $0\leq\max\{u,V+b\}-u\leq s$. Set
$E=\{u<V+b\}$ and $b_{p,q}=M_{p,q}(E)$.
There is a constant $C$, depending only on $X,\omega,m,n$, such that
\[
 b_{p,q+1}\leq b_{p+1,q}
              +Cs\sum_{a+d\leq p+q}b_{a,d}
                   \qquad(p+q\leq m-1).
\]
For $q=0$, the error can instead be bounded by
$Cs(b_{p,0}+b_{p-1,0}+b_{p-2,0})$, with negative indices zero.
\end{lemma}
\begin{proof}
First estimate the torsion of $T_{p,q}$, writing $r=p+q$, for
smooth admissible potentials. Since
$\mathrm d\sigma_u=\mathrm d\sigma_V=\mathrm d\omega$,
$\dc\sigma_u=\dc\sigma_V=\dc\omega$ and
$\ddc\sigma_u=\ddc\sigma_V=\ddc\omega$, the product rule
expresses $\ddc T_{p,q}$ as a sum of fixed smooth metric forms
wedged with products of $\sigma_u,\sigma_V$ of degrees $r,r-1,r-2$.
Every term contains either $\ddc\omega$ or
$\mathrm d\omega\wedge\dc\omega$, with its product-rule
multiplicity and the appropriate remaining background power.
The terms of degree $r$ come from differentiating the background
power; they cannot be discarded.
Terms with negative numbers of factors are absent.

All variable degrees are at most $r<m$.
Apply \cref{hessian:absolute-mixed-forms} to each term. Expanding
the resulting positive power of the sum of its variable factors
gives the pointwise bound
\[
 |\ddc T_{p,q}|\leq C\sum_{a+d\leq r}M_{a,d}.
\]
For $q=0$ every variable factor is $\sigma_u$, so the sharper
bound is $C(M_{p,0}+M_{p-1,0}+M_{p-2,0})$.
The constant bounds fixed metric coefficients on a finite coordinate
cover and the finitely many multiplicities; it is independent of
the potentials and their bounds.

Choose uniformly bounded smooth decreasing approximations to $u,V$.
By \cref{hessian:bounded-torsion-weight}, the torsion measures
converge weakly; by \cref{hessian:joint-decreasing-products},
all positive measures in these bounds also converge weakly.
Against nonnegative continuous tests, lower semicontinuity of
total variation preserves both bounds. They are consequently
inequalities of measures and may be restricted to the Borel set $E$.

Put $w=\max\{u,V+b\}$ and $h=w-u$.
Apply the mixed maximum theorem \cref{hessian:bounded-maximum-measure}
with $r=1$, with $V+b$ in place of $v$, and with the $m-1$ slots
defining $T_{p,q}$ as the fixed list. Since adding the constant $b$
does not change the measures, it gives
\[
 (\omega+\ddc w)T_{p,q}
 \geq\mathbf1_E\sigma_VT_{p,q}
                +\mathbf1_{X\setminus E}\sigma_uT_{p,q}.
\]
All products here are the corresponding positive mixed measures.
After integration, this implies
\[
 b_{p,q+1}\leq b_{p+1,q}
       +\int_X\bigl((\omega+\ddc w)-\sigma_u\bigr)T_{p,q}.
\]
Regularized maxima of the smooth decreasing approximations, from
\cref{hessian:regularized-maximum}, give a uniformly bounded
decreasing smooth list $w_j\downarrow w$.
Stokes' theorem on compact $X$ gives for these lists
\[
 \int_X\bigl((\omega+\ddc w_j)-\sigma_{u_j}\bigr)T_{p,q}^j
                      =\int_X(w_j-u_j)\ddc T_{p,q}^j.
\]
Joint decreasing convergence passes the positive mixed measures
on the left to their limits. The continuous two-variable weight
$F(x,y)=y-x$ in \cref{hessian:bounded-torsion-weight} passes the
right side to $\int_Xh\ddc T_{p,q}$. No monotonicity of $w_j-u_j$
is needed. Since $h$ is zero outside $E$ and $0\leq h\leq s$,
\[
 \int_Xh\ddc T_{p,q}\leq s|\ddc T_{p,q}|(E).
\]
Together with the torsion bounds restricted to $E$, the last three
displays prove both assertions. The argument never
passes a varying measure to a weak limit after restriction to $E$.
\end{proof}

\Needspace{17\baselineskip}
\begin{theorem}\label{hessian:weak-modified-comparison}\index{Comparison principle!modified Hessian comparison}
There is $C\geq1$, depending only on $X,\omega,m,n$, with the
following property. For bounded admissible $u,v$ and $0<\alpha<1$,
put $\eta=1-\alpha$, $V=\alpha v$ and $S=\inf_X(u-V)$.
For $0<s<\eta^3/C$, let $E_s=\{u<V+S+s\}$. Then
\[
 \int_{E_s}H_m(V)
       \leq\left(1+\frac{Cs}{\eta^m}\right)
                                  \int_{E_s}H_m(u).
\]
No differential identity for a power of $\omega$ is assumed.
\end{theorem}
\begin{proof}
Use the mixed measures above on $E_s$, writing
$b_{p,q}=M_{p,q}(E_s)$ and $a_k=b_{k,0}$.
The definition of $S$ gives
$0\leq\max\{u,V+S+s\}-u\leq s$, so \cref{hessian:mixed-torsion-comparison} applies.
Repeatedly moving a $V$ slot to a $u$ slot by this lemma gives,
for $p+q=r$,
\[
 b_{p,q}\leq a_r+C_1s\sum_{d=0}^{r-1}A_d,
 \qquad A_d=\max_{i+j=d}b_{i,j},\qquad 1\leq r\leq m.
\]
Here there are at most $m$ moves, and the multiplicity $d+1$
in the lower-degree sum is absorbed in $C_1$, which depends only
on $X,\omega,m,n$.
For $0<s\leq1$, induction on $r$ shows
\[
 A_r\leq a_r+K_rs\sum_{k<r}a_k.
\]
In fact $A_0=a_0$, and one may take
$K_r=C_1(1+\sum_{1\leq d<r}K_d)$: insert the earlier bounds
in the displayed recurrence and use $s^2\leq s$.
Thus, for a constant $C_2$ depending only on $X,\omega,m,n$,
\[
 b_{0,m}\leq a_m+C_2s\sum_{k<m}a_k.
\]

It remains to control the lower diagonal masses by $a_m$.
The identity $\sigma_V=\alpha\sigma_v+\eta\omega$ and mixed
positivity give $b_{k,1}\geq\eta a_k$ for $0\leq k<m$; this identity
and inequality pass from smooth approximations by joint convergence.
The assertion for $q=0$ in \cref{hessian:mixed-torsion-comparison},
with $C_3$ the constant of that lemma, gives
\[
 \eta a_k\leq a_{k+1}
          +C_3s(a_k+a_{k-1}+a_{k-2})\qquad(0\leq k<m),
\]
with $a_{-1}=a_{-2}=0$.
Set $B_k=\eta^{m-k}a_k$ for $0\leq k\leq m$, and set
$B_{-1}=B_{-2}=0$. Multiplying by $\eta^{m-k-1}$ and using
$0<\eta<1$ gives
\[
 B_k\leq B_{k+1}
          +\delta(B_k+B_{k-1}+B_{k-2}),
 \qquad\delta=C_3s\eta^{-3}.
\]
For $\bar B=\max_{0\leq k<m}B_k$, summing from any $j<m$ through
$m-1$ gives $B_j\leq B_m+3m\delta\bar B$.
If $s<\eta^3/(6mC_3)$, it follows that $\bar B\leq2B_m$.
Consequently
\[
 a_k\leq2\eta^{-(m-k)}a_m\quad(k<m),\qquad
 \sum_{k<m}a_k\leq2m\eta^{-m}a_m.
\]
Choose $C\geq\max\{1,6mC_3,2mC_2\}$. Then the asserted
range for $s$ supplies every preceding restriction, and substitution
in the bound for $b_{0,m}$ proves the theorem.
No division by $a_m$ occurs, so zero top mass on $E_s$ is included.
\end{proof}

\subsection{Domination and proper equations}
\label{subsec:hessian-bounded-domination}

The weak modified comparison estimate extends the strict domination
of \cref{hessian:strict-hessian-domination} to bounded potentials
on an arbitrary Hermitian background. A convex rescaling supplies a
positive volume term, while choosing a sufficiently small sublevel
set absorbs the torsion error. Canonical representatives show that
this sublevel set has positive volume even when it is not open.

\Needspace{14\baselineskip}
\begin{theorem}\label{hessian:strict-bounded-domination}\index{Domination principle!bounded Hessian potentials}
Let $u,v$ be bounded admissible. If, for some $0\leq c<1$,
\[
 \mathbf1_{\{u<v\}}H_m(u)
                  \leq c\mathbf1_{\{u<v\}}H_m(v),
\]
then $u\geq v$ everywhere, on an arbitrary Hermitian background.
In particular, $H_m(u)(\{u<v\})=0$ implies $u\geq v$.
\end{theorem}
\begin{proof}
Add the same constant to $u,v$ so that $v\geq1$.
This preserves the measure hypothesis and the desired conclusion.
Choose $0<\alpha<1$ with $\alpha^m>c$ and put $V=\alpha v$,
$\eta=1-\alpha$. Suppose $S=\inf_X(u-V)<0$.
For $0<s<-S$, the set $E_s=\{u<V+S+s\}$ is contained in
$\{u<V\}$ and hence in $\{u<v\}$.
It has positive volume. Indeed, if its volume were zero, then
$u\geq V+S+s$ almost everywhere. Both functions are bounded
admissible with their canonical representatives, so
\cref{hessian:canonical-order} would give this order everywhere,
contradicting the definition of $S$. The argument does not require
$E_s$ to be open or the infimum to be attained.

Choose also $s<\eta^3/C$, where $C$ is from \cref{hessian:weak-modified-comparison}.
By \cref{hessian:convex-background-term}, \cref{hessian:weak-modified-comparison}
and the measure hypothesis on $E_s\subset\{u<v\}$,
\[
\begin{aligned}
 \alpha^m H_m(v)(E_s)+\eta^m\int_{E_s}\omega^n
 &\leq H_m(V)(E_s)\\
 &\leq(1+Cs\eta^{-m})H_m(u)(E_s)\\
 &\leq c(1+Cs\eta^{-m})H_m(v)(E_s).
\end{aligned}
\]
If $c>0$, take $s$ still smaller so that
$c(1+Cs\eta^{-m})<\alpha^m$. If $c=0$, this strict inequality
already holds. All measures are finite and positive, so the
display forces the positive-volume term to be zero, a contradiction.
Hence $S\geq0$ and $u\geq\alpha v$. Let $\alpha$ increase
to one, with $\alpha^m>c$. Since $v$ is bounded, $\alpha v\to v$
uniformly, and therefore $u\geq v$. The final assertion is $c=0$.
\end{proof}

\Needspace{13\baselineskip}
\begin{corollary}\label{hessian:proper-bounded-uniqueness}\index{Hessian equation!bounded proper uniqueness}
Fix $a>0$.
If bounded admissible $u,v$ satisfy
\[
 e^{-av}H_m(v)\geq e^{-au}H_m(u),
\]
then $u\geq v$. Consequently, for any fixed finite positive Radon
measure $\nu$, the equation $H_m(u)=e^{au}\nu$ has at most one
bounded admissible solution.
\end{corollary}
\begin{proof}
For $\delta>0$, put $E_\delta=\{u<v-\delta\}$. Multiplying the
assumed measure inequality by the positive bounded Borel function
$e^{au}$ gives on $E_\delta$
\[
 H_m(u)\leq e^{a(u-v)}H_m(v)
                 \leq e^{-a\delta}H_m(v-\delta).
\]
The last step uses $H_m(v)=H_m(v-\delta)$, by invariance under
constants. Since
$e^{-a\delta}<1$, \cref{hessian:strict-bounded-domination} applied
to $u,v-\delta$ gives $u\geq v-\delta$ everywhere. Letting
$\delta\downarrow0$ gives $u\geq v$. Two solutions of the stated
equation have equal weighted measures $e^{-au}H_m(u)=\nu$.
Applying the order assertion in both directions proves uniqueness.
\end{proof}

A strictly increasing exponent also controls changes in logarithmic
data for a fixed Radon right-hand measure.

\Needspace{20\baselineskip}
\begin{proposition}\label{hessian:proper-bounded-stability}\index{Hessian equation!bounded data stability}
Let $A<B$ and $f$ be
smooth near $X\times[A,B]$, with $\partial_t f\geq\kappa>0$
on $X\times[A,B]$, and let $\nu$ be a finite positive Radon measure.
Suppose bounded admissible $u,v$ have values in $[A,B]$ and satisfy
\[
 H_m(u)=e^{f(x,u)+h_1}\nu,\qquad
 H_m(v)=e^{f(x,v)+h_2}\nu,
\]
where $h_1,h_2$ are bounded Borel functions. Then
\[
 \|u-v\|_\infty
            \leq\kappa^{-1}\|h_1-h_2\|_{L^\infty(\nu)}.
\]
\end{proposition}
\begin{proof}
Put $K=\|h_1-h_2\|_{L^\infty(\nu)}$ and fix $s>K/\kappa$.
On $E_s=\{v+s<u\}$, the pointwise values of $u,v$ lie in $[A,B]$,
so $f(x,u)-f(x,v)\geq\kappa s$ by the mean value theorem.
The equations and the essential bound on $h_2-h_1$ therefore give,
as restricted measures,
\[
 \mathbf1_{E_s}H_m(v+s)
 =\mathbf1_{E_s}H_m(v)
 \leq e^{K-\kappa s}\mathbf1_{E_s}H_m(u).
\]
The constant is smaller than one. Applying
\cref{hessian:strict-bounded-domination} to $v+s,u$ gives
$u\leq v+s$ pointwise. Let $s\downarrow K/\kappa$ and exchange
$u,v$ to obtain the norm bound. The exponent is evaluated only
at $u,v$; the shifted comparison potential need not take its
values in $[A,B]$.
\end{proof}

\Needspace{14\baselineskip}
\begin{corollary}\label{hessian:proper-bounded-smoothness}\index{Hessian equation!bounded proper smoothness}
Under the hypotheses of \cref{hessian:proper-variable-existence},
every bounded admissible solution with values in $[A,B]$ coincides
with the smooth solution given there. In particular, for every $a>0$ and smooth real
$H$, the equation $H_m(u)=e^{au+H}\omega^n$ has exactly one bounded
admissible solution, and that solution is smooth and strictly
admissible.
\end{corollary}
\begin{proof}
For a continuous admissible potential, smooth decreasing admissible
approximations converge uniformly on compact $X$ by Dini's theorem.
Thus \cref{hessian:joint-decreasing-products,hessian:continuous-products}
define the same Hessian measure on the continuous class.
The smooth solution of \cref{hessian:proper-variable-existence}
therefore solves the bounded equation as well. Apply
\cref{hessian:proper-bounded-stability} with $\nu=\omega^n$ and
$h_1=h_2=0$ to identify any other bounded solution in $[A,B]$
with it. For the final assertion, smooth existence is already
\cref{hessian:smooth-proper-existence}, and
\cref{hessian:proper-bounded-uniqueness} with
$\nu=e^H\omega^n$ gives uniqueness among all bounded solutions.
\end{proof}
