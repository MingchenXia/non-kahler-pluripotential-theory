\section{Relative projection and supersolutions}
\label{sec:big-relative-projection}

Relative projection turns the minimal-class solutions just constructed into supersolutions in $\mathcal E_\varphi$. The small stretches of \cref{lem:positive-volume-stretch} first produce a genuine rooftop with the full upper volume of $\varphi$. The same construction works between ordered model references and will allow those references to vary. Contact support then bounds the projected measure by the original measure on the coincidence set.

Let $\eta\in\PSH(X,\theta)$ satisfy $P_\theta[\eta]=\eta\geq\varphi$, and write $\mathcal E_\eta=\mathcal E_{\mathrm{env}}(X,\theta,\eta)$.

\begin{proposition}\label{prop:big-cross-class-roof}
For every $z\in\mathcal E_\eta$, the rooftop $P_\theta(\varphi,z)$ is genuine and belongs to $\mathcal E_\varphi$.
\end{proposition}
\Needspace{7\baselineskip}
\begin{proof}
First suppose $z\leq\eta$; also $\varphi\leq\eta\leq0$. For $b>1$ sufficiently close to one, \cref{lem:positive-volume-stretch} gives a genuine
\[
 h_b=Q_b(\varphi,\eta)\leq\varphi.
\]
Set $a=1-b^{-1}$. The same-class stretching lemma, now applied in $\mathcal E_\eta$, gives $z_{1/a}=Q_{1/a}(z,\eta)\in\mathcal E_\eta$. The potential
\[
 k_a=a z_{1/a}+(1-a)h_b
\]
is below both $\varphi$ and $z$. Indeed $z_{1/a}\leq\eta$ and $(1-a)h_b+a\eta\leq\varphi$ prove the first bound; $h_b\leq\eta$ and $a z_{1/a}+(1-a)\eta\leq z$ prove the second. Thus $p=P_\theta(\varphi,z)$ is genuine. Addition and monotonicity give
\[
 (1-a)^n\overline{\operatorname{vol}}_\theta(h_b)
 \leq\overline{\operatorname{vol}}_\theta(k_a)
 \leq\overline{\operatorname{vol}}_\theta(p)
 \leq\overline{\operatorname{vol}}_\theta(\varphi).
\]
As $b\downarrow1$, $a\downarrow0$, and the first term tends to the last by \cref{lem:positive-volume-stretch}. The upper volumes are therefore equal. Since $p\leq\varphi$, \cref{thm:relative-equivalence} gives $p\in\mathcal E_\varphi$.

\Needspace{6\baselineskip}
For general $z$, the envelope identity gives $z-\sup_Xz\leq\eta$. Set $S=\max(\sup_Xz,0)$. The preceding case gives $P_\theta(\varphi,z-S)\in\mathcal E_\varphi$, and
\[
 P_\theta(\varphi,z-S)\leq P_\theta(\varphi,z)\leq\varphi.
\]
Upward stability finishes the proof.
\end{proof}

\Needspace{12\baselineskip}
\begin{proposition}\label{prop:big-relative-projection}\index{Envelope!relative projection}
Let $z\in\mathcal E_\eta$ and set $Q=P_\theta[\varphi](z)$.
Then $Q\in\mathcal E_\varphi$, $Q\leq z$, and
\[
 \mu_Q\leq\mathbf1_{\{Q=z\}}\mu_z.
\]
\end{proposition}
\begin{proof}
Put $p_C=P_\theta(\varphi+C,z)$ for $C\geq0$. The preceding proposition gives $p_0\in\mathcal E_\varphi$. Since $p_0\leq p_C\preceq\varphi$, upward stability gives $p_C\in\mathcal E_\varphi$. Its envelope identity, together with $p_C\leq z$, yields the uniform bound
\[
 p_C\leq\varphi+\sup_Xp_C\leq\varphi+\sup_Xz.
\]
The increasing canonical limit is consequently genuine, belongs to $\mathcal E_\varphi$ and is at most $z$. The monotone convergence is in capacity by \cref{lem:monotone-psh-capacity}.

Applying \cref{lem:relative-obstacle-contact} with $u=\varphi$ and $w=z$ gives
\begin{equation}\label{eq:big-projection-contact-tail}
 \mu_Q\leq\mathbf1_{\{Q=z\}}\mu_z.
\end{equation}
\end{proof}

\Needspace{16\baselineskip}
\begin{corollary}\label{cor:big-relative-supersolution}
For every nonzero finite non-pluripolar measure $\nu$, there is $Q\in\mathcal E_\varphi$ such that
\[
 0\ne \mathrm{e}^{-Q}\mu_Q\leq\nu.
\]
\end{corollary}
\begin{proof}
The minimal-reference theorem supplies $z\in\mathcal E_V$ with $\mu_z=\mathrm{e}^z\nu$. Apply the proposition with $\eta=V$ to obtain $Q\in\mathcal E_\varphi$. On its contact set $Q=z$, so
\[
 \mu_Q\leq\mathbf1_{\{Q=z\}}\mathrm{e}^z\nu\leq \mathrm{e}^Q\nu.
\]
The weights are finite off the pole sets and are assigned zero measure there. Thus $\mathrm{e}^{-Q}\mu_Q\leq\nu$. It is nonzero because $\mu_Q(X)\geq\ell>0$.
\end{proof}

\subsection{Varying model references}
\label{subsec:big-varying-models}

The monotone model bounds of \cref{thm:model-volume-capacity-convergence} provide lower references whose upper volumes approach that of the limit. Projecting finite families to these references makes their rooftops comparable. To preserve the limiting relative class, domination is needed only where the potentials lie below one fixed member of that class. Domination on all of $X$ also gives capacity and measure convergence. Use the comparison $d_\theta$ and the classes $\mathcal M_\delta(\theta)$ of \cref{def:model-volume-comparison}.

\begin{theorem}\label{thm:big-varying-model-closure}\index{Relative full-mass class!varying models}
Let $\varphi_j,\varphi\in\mathcal M_\delta(\theta)$ for some $\delta>0$, with $d_\theta(\varphi_j,\varphi)\to0$. Suppose $u_j\in\mathcal E_{\varphi_j}$, $u_j\to u\in\PSH(X,\theta)$ in $L^1(X)$, and $\chi\in\mathcal E_\varphi$. If one finite non-pluripolar measure $\nu$ satisfies
\[
 \mathbf1_{\{u_j\leq\chi\}}\mu_{u_j}\leq\nu,
\]
then $u\in\mathcal E_\varphi$. After extraction and relabeling, the potentials
\[
 \psi_j=P_\theta\left(\inf_{l\geq j}\varphi_l\right),\qquad
 h_j=P_\theta\left(\inf_{l\geq j}u_l\right)
\]
satisfy $\psi_j\in\mathcal M_{\delta/2}(\theta)$ and $h_j\in\mathcal E_{\psi_j}$.
If $\mu_{u_j}\leq\nu$ on $X$, the original sequence converges to $u$ in capacity and $\mu_{u_j}\rightharpoonup\mu_u$.
\end{theorem}
\begin{proof}
Hartogs' lemma gives a common upper bound for $u_j,u,\chi$; translate them to be nonpositive. The envelope identity gives $u_j\leq\varphi_j$. Since $\varphi_j\to\varphi$ in $L^1$ by \cref{prop:model-volume-separation}, it follows that $u\leq\varphi$. Put $\Lambda=\nu+\mu_\chi$. By \cref{lem:fixed-npp-integrals}, extract so that $u_j\to u$ almost everywhere for smooth volume and
\[
 I_j:=\int_X|\mathrm{e}^{u_j}-\mathrm{e}^u|\,\mathrm{d}\Lambda\leq2^{-j}.
\]
Apply \cref{thm:model-volume-capacity-convergence} to extract further, retaining these estimates, and obtain increasing models $w_j\leq\varphi_j,\varphi$ with lower volume at least $\delta/2$ and $d_\theta(w_j,\varphi)\to0$. For $l\geq j$, $w_j\leq w_l\leq\varphi_l$. Hence $\psi_j$ is genuine and
\[
 w_j\leq\psi_j\leq\varphi.
\]
The upper bound follows by passing to the $L^1$ limit in $\psi_j\leq\varphi_l$. Envelope order gives $P_\theta[\psi_j]\leq\varphi_l$ for all $l\geq j$, so $\psi_j$ is a model. It increases with $j$, has lower volume at least $\delta/2$, and
\begin{equation}\label{eq:varying-reference-volume-limit}
 \overline{\operatorname{vol}}_\theta(\psi_j)
 \longrightarrow\overline{\operatorname{vol}}_\theta(\varphi).
\end{equation}

Set $g_j=P_\theta[\psi_j](\chi)$. The ordered-reference projection gives $g_j\in\mathcal E_{\psi_j}$, $g_j\leq\chi$ and $\mu_{g_j}\leq\mu_\chi$. Moreover $g_j$ increases with $j$. For $N\geq j$, consider
\[
 q_{j,N}=P_\theta(g_j,u_j,\ldots,u_N).
\]
For each $l\geq j$, $P_\theta[\psi_j](u_l)$ belongs to $\mathcal E_{\psi_j}$ by the same projection, now with upper reference $\varphi_l$. The finite rooftop of these projections and $g_j$ is a lower competitor for $q_{j,N}$. Since $q_{j,N}\leq g_j$, finite-rooftop and upward stability give $q_{j,N}\in\mathcal E_{\psi_j}$, and its mass is at least $\delta/2$.

Assign the contact set to disjoint pieces, one for each obstacle. On a piece where $q_{j,N}=u_l$, we have $u_l\leq g_j\leq\chi$, so the restricted hypothesis controls its measure by $\nu$. The remaining piece is controlled by $\mu_{g_j}\leq\mu_\chi$. Consequently $\mu_{q_{j,N}}\leq\Lambda$. Put $F_j=\min(\mathrm{e}^u,\mathrm{e}^{g_j})$. On the $u_l$ piece,
\[
 |\mathrm{e}^{q_{j,N}}-F_j|\leq|\mathrm{e}^{u_l}-\mathrm{e}^u|.
\]
On the $g_j$ piece, $g_j=q_{j,N}\leq u_N$, so the error is
$(\mathrm{e}^{g_j}-\mathrm{e}^u)_+\leq|\mathrm{e}^{u_N}-\mathrm{e}^u|$.
The two measures in $\Lambda$ account for both contributions at $N$. Thus
\begin{equation}\label{eq:varying-roof-exponential-error}
 \int_X|\mathrm{e}^{q_{j,N}}-F_j|\,\mu_{q_{j,N}}
 \leq\sum_{l=j}^N I_l\leq2^{1-j}.
\end{equation}

Since $F_j\geq F_1>0$ outside a pluripolar set, choose $\epsilon>0$ with $\Lambda(\{F_1\leq\epsilon\})<\delta/4$. The mass lower bound and domination by $\Lambda$ give
\[
 \epsilon\delta/4
 \leq\int_X F_j\,\mu_{q_{j,N}}
 \leq2^{1-j}+\Lambda(X)\mathrm{e}^{\sup_Xq_{j,N}}.
\]
For every sufficiently large $j$ this bounds $\sup_Xq_{j,N}$ below uniformly in $N$. The decreasing limit $q_j$ is genuine. Applying \cref{thm:big-relative-fixed-continuity} in the fixed class $\mathcal E_{\psi_j}$, with dominator $\Lambda$, gives $q_j\in\mathcal E_{\psi_j}$. Now
\[
 q_j\leq h_j\leq\psi_j,
\]
so $h_j\in\mathcal E_{\psi_j}$ by upward stability. Discard the finitely many earlier indices. Since $h_j\leq u\leq\varphi$, monotonicity and \eqref{eq:varying-reference-volume-limit} identify the upper volumes of $u$ and $\varphi$. Thus \cref{thm:relative-equivalence} gives $u\in\mathcal E_\varphi$.

Suppose now $\mu_{u_j}\leq\nu$ on $X$. The finite rooftops $p_{j,N}=P_\theta(u_j,\ldots,u_N)$ are genuine since they lie above $h_j$. Their disjoint contact pieces give
\[
 \mu_{p_{j,N}}\leq\nu,\qquad
 \int_X|\mathrm{e}^{p_{j,N}}-\mathrm{e}^u|\,\mu_{p_{j,N}}\leq2^{1-j}.
\]
For fixed $j$, they decrease to $h_j$. These lower envelopes increase canonically to $h\leq u\leq\varphi$, and their upper volumes tend to that of $\varphi$; hence $h\in\mathcal E_\varphi$. Both monotone passages converge in capacity by \cref{lem:monotone-psh-capacity}. Applying \cref{lem:weighted-extended-lsc} twice gives
\[
 \int_X|\mathrm{e}^h-\mathrm{e}^u|\,\mu_h=0.
\]
\Cref{thm:big-relative-domination} with coefficient zero gives $h=u$. The upper tail envelopes decrease to $u$ by Hartogs' lemma and the extracted almost-everywhere convergence. Squeezing proves convergence in capacity. Finally,
\[
 \mu_{u_j}(\{u_j\leq-T\})\leq\nu(\{h_1\leq-T\})\longrightarrow0
\]
uniformly in $j$. The general-background tail criterion of \cref{thm:general-background-damping} gives weak convergence of the measures. The extraction can start from any subsequence, so both conclusions hold for the original sequence.
\end{proof}
