\section{Relative projection and supersolutions}
\label{sec:big-relative-projection}

Relative projection turns the minimal-class solutions just constructed into supersolutions in $\mathcal E_\varphi$. The small stretches of \cref{lem:positive-volume-stretch} first produce a genuine rooftop between $\varphi$ and a member of $\mathcal E_V$, with the full upper volume of $\varphi$. Contact support then bounds the projected measure by the original measure on the coincidence set.

\begin{proposition}\label{prop:big-cross-class-roof}
For every $z\in\mathcal E_V$, the rooftop $P_\theta(\varphi,z)$ is genuine and belongs to $\mathcal E_\varphi$.
\end{proposition}
\Needspace{7\baselineskip}
\begin{proof}
First suppose $z\leq V$; also $\varphi\leq V\leq0$. For $b>1$ sufficiently close to one, \cref{lem:positive-volume-stretch} gives a genuine
\[
 h_b=Q_b(\varphi,V)\leq\varphi.
\]
Set $a=1-b^{-1}$. The same-class stretching lemma, now applied in $\mathcal E_V$, gives $z_{1/a}=Q_{1/a}(z,V)\in\mathcal E_V$. The potential
\[
 k_a=a z_{1/a}+(1-a)h_b
\]
is below both $\varphi$ and $z$. Indeed $z_{1/a}\leq V$ and $(1-a)h_b+aV\leq\varphi$ prove the first bound; $h_b\leq V$ and $a z_{1/a}+(1-a)V\leq z$ prove the second. Thus $p=P_\theta(\varphi,z)$ is genuine. Addition and monotonicity give
\[
 (1-a)^n\overline{\operatorname{vol}}_\theta(h_b)
 \leq\overline{\operatorname{vol}}_\theta(k_a)
 \leq\overline{\operatorname{vol}}_\theta(p)
 \leq\overline{\operatorname{vol}}_\theta(\varphi).
\]
As $b\downarrow1$, $a\downarrow0$, and the first term tends to the last by \cref{lem:positive-volume-stretch}. The upper volumes are therefore equal. Since $p\leq\varphi$, \cref{thm:relative-equivalence} gives $p\in\mathcal E_\varphi$.

\Needspace{6\baselineskip}
For general $z$, the envelope identity gives $z-\sup_Xz\leq V$. Set $S=\max(\sup_Xz,0)$. The preceding case gives $P_\theta(\varphi,z-S)\in\mathcal E_\varphi$, and
\[
 P_\theta(\varphi,z-S)\leq P_\theta(\varphi,z)\leq\varphi.
\]
Upward stability finishes the proof.
\end{proof}

\Needspace{12\baselineskip}
\begin{proposition}\label{prop:big-relative-projection}
Let $z\in\mathcal E_V$ and set $Q=P_\theta[\varphi](z)$.
Then $Q\in\mathcal E_\varphi$, $Q\leq z$, and
\[
 \mu_Q\leq\mathbf1_{\{Q=z\}}\mu_z.
\]
\end{proposition}
\begin{proof}
Put $p_C=P_\theta(\varphi+C,z)$ for $C\geq0$. The preceding proposition gives $p_0\in\mathcal E_\varphi$. Since $p_0\leq p_C\preceq\varphi$, upward stability gives $p_C\in\mathcal E_\varphi$. Its envelope identity, together with $p_C\leq z$, yields the uniform bound
\[
 p_C\leq\varphi+\sup_Xp_C\leq\varphi+\sup_Xz.
\]
The increasing canonical limit is consequently genuine, belongs to $\mathcal E_\varphi$ and is at most $z$. The monotone convergence is in capacity by \cref{lem:monotone-psh-capacity}.

Applying \cref{lem:relative-obstacle-contact} with $u=\varphi$ and $w=z$ gives
\begin{equation}\label{eq:big-projection-contact-tail}
 \mu_Q\leq\mathbf1_{\{Q=z\}}\mu_z.
\end{equation}
\end{proof}

\Needspace{16\baselineskip}
\begin{corollary}\label{cor:big-relative-supersolution}
For every nonzero finite non-pluripolar measure $\nu$, there is $Q\in\mathcal E_\varphi$ such that
\[
 0\ne \mathrm{e}^{-Q}\mu_Q\leq\nu.
\]
\end{corollary}
\begin{proof}
The minimal-reference theorem supplies $z\in\mathcal E_V$ with $\mu_z=\mathrm{e}^z\nu$. Apply the proposition to obtain $Q\in\mathcal E_\varphi$. On its contact set $Q=z$, so
\[
 \mu_Q\leq\mathbf1_{\{Q=z\}}\mathrm{e}^z\nu\leq \mathrm{e}^Q\nu.
\]
The weights are finite off the pole sets and are assigned zero measure there. Thus $\mathrm{e}^{-Q}\mu_Q\leq\nu$. It is nonzero because $\mu_Q(X)\geq\ell>0$.
\end{proof}
