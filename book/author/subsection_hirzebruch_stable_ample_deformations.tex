\subsection{Deforming to stable ample bundles}
\label{subsec:hirzebruch-stable-ample-deformations}
\index{Hirzebruch surface!stable ample approximation}

We keep the Hirzebruch surface $X=\mathbb F_e$ and the data $H,P,L_1,L_2,E_0$ of \cref{subsec:hirzebruch-ample-comparison,subsec:hirzebruch-stable-extensions}. The ample quotient $E_{\mathrm{amp}}$ and a nonsplit extension are two prioritary bundles with the same Chern classes as $E_0$; the former is ample and the latter is stable. Prioritary bundles with these invariants form an irreducible family in which ampleness and stability are open conditions, so bundles with both properties are dense in the family, and in particular they accumulate at $E_0$. We first prove the required openness assertions for algebraic families.

Throughout this subsection, $T$ denotes a smooth complex algebraic variety. A \emph{flat family} is a coherent sheaf $\mathcal E$ on $X\times T$ that is flat over $T$, and its fiber $\mathcal E_t$ at $t\in T$ is its restriction to $X\times\{t\}$.

\Needspace{6\baselineskip}
\begin{lemma}\label{lem:hirzebruch-family-local-freeness}
\index{Vector bundle!local freeness in families}
If a flat family $\mathcal E$ has a locally free fiber at $t_0\in T$, then $\mathcal E$ is a vector bundle on $X\times T'$ for some Zariski-open neighborhood $T'$ of $t_0$.
\end{lemma}
\begin{proof}
At a point of $X\times\{t_0\}$, lift a local basis of the fiber to $\mathcal E$. By Nakayama's lemma, the resulting map from a free module of finite rank is surjective near that point. Tensor its kernel sequence with the residue field of $t_0$. Flatness over $T$ makes the first Tor term zero, so the kernel restricts to the kernel of the chosen isomorphism on the fiber, namely zero, and Nakayama's lemma shows that the kernel vanishes near that point. Thus $\mathcal E$ is locally free near every point of the central fiber. The locus where $\mathcal E$ is not locally free is closed. Since the projection $X\times T\to T$ is proper, the image of that locus is closed and misses $t_0$, and we remove this image.
\end{proof}

\Needspace{6\baselineskip}
\begin{lemma}\label{lem:hirzebruch-family-ample-open}
\index{Vector bundle!openness of ampleness}
For a vector bundle $\mathcal E$ on $X\times T$, the parameters $t$ for which $\mathcal E_t$ is ample form a Zariski-open subset of $T$.
\end{lemma}
\begin{proof}
Put $Y=\mathbb P_{X\times T}(\mathcal E^*)$, with the line convention, and let $p\colon Y\to T$ be the projection, a projective flat morphism. Write $\mathcal L=\mathcal O_Y(1)$. If $\mathcal E_{t_0}$ is ample, then $\mathcal L_{t_0}$ is ample. Choose $k$ so that its $k$th power is very ample and has vanishing higher cohomology. By Serre vanishing, semicontinuity, and cohomology and base change, after shrinking $T$ around $t_0$ the sheaf $\mathcal V=p_*\mathcal L^k$ is locally free and commutes with restriction to fibers. After a further shrinking, the evaluation map $p^*\mathcal V\to\mathcal L^k$ is surjective: its cokernel vanishes on the central fiber, and the image of its support under the proper map $p$ is closed and misses $t_0$.

Evaluation gives a proper morphism
\[
 g\colon Y\longrightarrow\mathbb P_T(\mathcal V^*),
 \qquad g^*\mathcal O(1)=\mathcal L^k,
\]
whose restriction to the central fiber is a closed immersion. The locus where $g$ is quasi-finite is open; its complement has closed image in $T$ by properness and misses $t_0$, so after a further shrinking $g$ is proper and quasi-finite, hence finite. For a finite morphism, pushforward commutes with base change. The cokernel of
\[
 \mathcal O_{\mathbb P_T(\mathcal V^*)}\longrightarrow g_*\mathcal O_Y
\]
therefore vanishes on the central fiber, and its support projects to a closed subset of $T$ missing $t_0$. After removing that subset, the map is surjective, so $g$ is a closed immersion. Consequently $\mathcal L^k$ is very ample on every fiber, and $\mathcal E_t$ is ample there.
\end{proof}

\Needspace{6\baselineskip}
\begin{lemma}\label{lem:hirzebruch-family-stability-open}
\index{Vector bundle!openness of slope-stability}
For a family $\mathcal E$ of rank-two vector bundles on $X\times T$ whose fibers have a fixed first Chern class, the parameters $t$ for which $\mathcal E_t$ is $\mu_P$-stable form a Zariski-open subset of $T$.
\end{lemma}
\begin{proof}
Fix a stable fiber at $t_0$ and put $\nu=c_1(\mathcal E_t)\cdot P/2$, which is independent of $t$. Choose $k\geq1$ such that $\mathcal E_{t_0}^*(kP)$ is globally generated and has vanishing higher cohomology. The argument with Serre vanishing, base change and evaluation used in the previous proof gives, over a neighborhood of $t_0$, maps
\[
 \mathcal E_t\longrightarrow\mathcal O_X(kP)^{\oplus N}
\]
that are injective on every fiber. By the double-dual argument for rank-one subsheaves in the proof of \cref{prop:hirzebruch-stable-extensions}, a destabilizing subsheaf of $\mathcal E_t$ yields a nonzero map from a line bundle $M=\mathcal O_X(aS+bf)$ with $\deg_P M\geq\nu$. At least one component of its composite with the injection above is nonzero, and since effective classes have nonnegative coefficients,
\[
 a\leq4k,\qquad b\leq(12e+3)k.
\]
Since $\deg_P M=(8e+3)a+4b$, the inequality $\deg_PM\geq\nu$ also gives
\[
 a\geq\frac{\nu-4(12e+3)k}{8e+3},
 \qquad b\geq\frac{\nu-4k(8e+3)}4.
\]
Only finitely many integral pairs $(a,b)$ satisfy these four bounds. For each such pair with $\deg_P M\geq\nu$, semicontinuity makes the set
\[
 \{t:H^0(X,\mathcal E_t\otimes M^*)\neq0\}
\]
closed, and it does not contain $t_0$. Conversely, a nonzero map from such an $M$ is injective as a map of sheaves and violates stability. Thus the set of parameters with unstable fibers is, on this neighborhood, exactly the finite union of these closed subsets, and removing the union proves the claim.
\end{proof}

\Needspace{6\baselineskip}
\begin{lemma}\label{lem:dense-family-smooth-dolbeault-convergence}
\index{Dolbeault operator!algebraic family convergence}
\index{Vector bundle!parallel transport in families}
Let $T$ be irreducible, let $U\subset T$ be a nonempty Zariski-open subset, and let $\mathcal E$ be a vector bundle on $X\times T$. For every $t_0\in T$, there are points $t_j\in U$ tending to $t_0$ in the analytic topology, together with smooth complex-linear identifications of the bundles underlying $\mathcal E_{t_j}$ and $\mathcal E_{t_0}$, under which the Dolbeault operators of the fibers converge in the $C^m$ norm for every integer $m\geq0$. The dual operators converge in the same sense.
\end{lemma}
\begin{proof}
The proper algebraic closed subset $T\setminus U$ has empty interior in the analytic topology. In an analytic coordinate ball $B$ around $t_0$, choose $t_j\in U\cap B$ with $t_j\to t_0$. Choose a smooth complex connection on $\mathcal E|_{X\times B}$. Parallel transport along radial paths in $B$, keeping $x\in X$ fixed, gives identifications
\[
 \Phi_t\colon\mathcal E_{t_0}^{\mathrm{sm}}\longrightarrow
          \mathcal E_t^{\mathrm{sm}},\qquad \Phi_{t_0}=\operatorname{id},
\]
depending smoothly on $(x,t)$, where the superscript $\mathrm{sm}$ denotes the underlying smooth complex bundle. These identifications need not be holomorphic. Transport the Dolbeault operators by these maps and write
\[
 \bar\partial_t=\Phi_t^{-1}\bar\partial_{\mathcal E_t}\Phi_t,
 \qquad A_t=\bar\partial_t-\bar\partial_{\mathcal E_{t_0}}.
\]
Since both operators satisfy the same Leibniz rule, $A_t$ is an endomorphism-valued $(0,1)$-form. Its coefficients are smooth in $(x,t)$, and $A_{t_0}=0$. Fix background metrics and a smooth connection on the central bundle. Compactness of $X$ and continuity of every finite collection of derivatives give
\[
 \|A_{t_j}\|_{C^m(X)}\longrightarrow0
 \qquad\text{for every integer }m\geq0.
\]
The dual identifications are $(\Phi_t^{-1})^*$. In transported dual frames the perturbation is $-A_t^{\mathsf T}$, so the same convergence holds for the dual Dolbeault operators.
\end{proof}

\index{Prioritary bundle!stack irreducibility}
We use Walter's theorem on prioritary stacks: on a birationally ruled surface, a nonempty stack of prioritary torsion-free sheaves with fixed rank $r\geq2$ and fixed numerical Chern classes is smooth and irreducible. Here prioritary means that $\operatorname{Ext}^2(E,E(-f))=0$ for every ruling fiber $f$; on $X$ all ruling fibers are linearly equivalent. The theorem concerns the full prioritary stack and has no semistability hypothesis.

\Needspace{6\baselineskip}
\begin{theorem}\label{thm:hirzebruch-stable-ample-approximation}
\index{Vector bundle!stable ample approximation}
For every $e\geq0$, there are integrable Dolbeault operators on the smooth bundle underlying $E_0$ that converge to its split operator in every $C^m$ norm, $m\geq0$, and define ample $\mu_P$-stable bundles of rank two with $c_1=4H$ and $c_2=13H^2$. The induced dual operators converge to the split dual operator in every $C^m$ norm.
\end{theorem}
\begin{proof}
Let $\mathcal P$ be the prioritary stack with invariants $(2,4H,13H^2)$. It contains $E_0$ by \cref{prop:hirzebruch-extension-prioritary}, and Walter's theorem makes it smooth and irreducible. Let $\mathcal A$ and $\mathcal S$ be the loci of ample and of $\mu_P$-stable vector bundles in $\mathcal P$. They are open by \cref{lem:hirzebruch-family-ample-open,lem:hirzebruch-family-stability-open}, which apply on smooth local charts of $\mathcal P$; \cref{lem:hirzebruch-family-local-freeness} allows restriction to locally free families near a point given by a vector bundle. The ample bundle of \cref{prop:hirzebruch-ample-comparison} is prioritary by \cref{lem:hirzebruch-prioritary-comparison}, so $\mathcal A$ is nonempty. A nonsplit extension is stable and prioritary by \cref{prop:hirzebruch-stable-extensions,prop:hirzebruch-extension-prioritary}, so $\mathcal S$ is nonempty. By irreducibility, $\mathcal U=\mathcal A\cap\mathcal S$ is nonempty and dense.

Choose a smooth local chart $q\colon T\to\mathcal P$ of finite type over $\mathbb C$, with a point $t_0$ representing $E_0$, and let $\mathcal E$ be the corresponding flat family on $X\times T$. Fix an identification $\mathcal E_{t_0}\simeq E_0$. The variety $T$ is smooth because the stack and the chart are smooth. Shrink $T$ to an irreducible neighborhood of $t_0$ and use \cref{lem:hirzebruch-family-local-freeness} to make $\mathcal E$ a vector bundle on the whole product. The inverse image $q^{-1}(\mathcal U)$ is dense: every nonempty open $T'\subset T$ has nonempty open image $q(T')$ under the smooth map $q$, and this image meets $\mathcal U$. Apply \cref{lem:dense-family-smooth-dolbeault-convergence} to this dense open set. Each transported operator defines a bundle isomorphic to the corresponding ample stable fiber. The fixed invariants of the stack give its Chern classes, and the lemma also gives the convergence of the dual operators.
\end{proof}

\Needspace{6\baselineskip}
\begin{corollary}\label{cor:hirzebruch-stable-ample-finsler-obstruction}
\index{Kobayashi curvature!stable ample obstruction}
\index{Finsler metric!stable ample obstruction}
For every $e\geq0$, there is an ample $\mu_P$-stable bundle $E$ of rank two on $X$ with $c_1(E)=4H$ and $c_2(E)=13H^2$ that admits no smooth strongly pseudoconvex squared Finsler metric with semipositive Kobayashi curvature.
\end{corollary}
\begin{proof}
By \cref{prop:hirzebruch-positive-dual-tests}, the two line summands of $E_0^*$ have positive degrees with respect to suitable test forms. The preceding theorem gives an ample stable bundle $E$ whose dual Dolbeault operator is as close to the split dual operator in $C^4$ as required. Apply \cref{cor:gauduchon-semipositive-finsler-obstruction} to $F=E^*$.
\end{proof}
