\subsection{Higher moments in the minimal singularity type}
\label{sec:big-minimal-moments}

In this subsection the reference is the potential $V=V_\theta=P_\theta(0)$ with minimal singularities. It is an ordinary model, and we write $\mathcal E_V=\mathcal E_{\mathrm{env}}(X,\theta,V)$ and
\[
 \ell_V=\underline{\operatorname{vol}}_\theta(V)\geq\ell>0,
 \qquad M_V=\overline{\operatorname{vol}}_\theta(V)<\infty.
\]
The first inequality follows from volume monotonicity, since $\varphi\preceq V$, and the last from bounded mass. Thus $V$ satisfies the standing assumptions on $\varphi$, and the results above apply with $\varphi$ replaced by $V$. For a non-pluripolar probability measure $\nu$, recall from \cref{subsec:psh-measure-smoothing} that
\[
 A_\theta(\nu)=\sup\left\{\int_X(-w)\,\mathrm{d}\nu:
 w\in\PSH(X,\theta),\ \sup_Xw=0\right\}.
\]
We show that a first moment of $V-u$ against the measure $\mu_u$ of $u$ itself controls a higher moment of $V-u$ against every fixed $\nu$ with $A_\theta(\nu)<\infty$. An envelope with a concave obstacle converts the distribution of $V-u$ under $\mu_u$ into this stronger integrability estimate.

\begin{lemma}\label{lem:big-concave-envelope}
Let $u\in\PSH(X,\theta)$ satisfy $u\sim V$ and $u\leq V$. Let $\chi\colon\mathbb R_{\leq0}\to\mathbb R_{\leq0}$ be increasing and concave, with $\chi(0)=0$ and $\chi'\geq1$, and assume that $\chi$ is $C^{1,1}$ on compact intervals. Set
\[
 b=V+\chi(u-V),\qquad p=P_\theta(b),
\]
where the obstacle $b$ is defined off the pole set of $V$ and is set equal to $-\infty$ on that set. Then $p\sim V$ and
\[
 \mu_p\leq\mathbf1_{\{p=b\}}\chi'(u-V)^n\mu_u.
\]
The measures and contact sets in this inequality may be restricted to the complement of the pole set of any fixed strict analytic barrier for $\theta$, that is, of any $\rho\in\PSH(X,\theta)$ with analytic singularities such that $\theta+\ddc\rho\geq a\omega$ for some $a>0$.
\end{lemma}
\begin{proof}
The difference $u-V$ is bounded, so $V-C\leq b\leq u\leq V$ for some constant $C$. Hence $p\sim V$. The obstacle is quasi-continuous: away from sets of arbitrarily small capacity, $u$ and $V$ are finite and continuous, and their difference has a fixed bounded range. The general contact theorem \cref{thm:big-general-envelope-contact} therefore applies.

Choose a strict analytic barrier $\rho$ and translate it so that $\rho\leq V$; put $\Omega=X\setminus\{\rho=-\infty\}$. On this open set $u$, $V$ and $p$ are locally bounded. Let $\tau=\chi^{-1}$; then $\tau$ is convex and $0<\tau'\leq1$. By \cref{lem:bounded-convex-pair}, the function
\[
 w=V+\tau(p-V)
\]
is $\theta$-plurisubharmonic on $\Omega$ and satisfies
\[
 \mathrm{MA}_\theta(w)\geq\tau'(p-V)^n\mu_p.
\]
Since $p\leq w\leq V$ on $\Omega$, the removability lemma \cref{lem:qpsh-removable-analytic} extends $w$ across the analytic set $X\setminus\Omega$, preserving its smooth lower curvature bound.

\Needspace{6\baselineskip}
The inequality $p\leq b$ implies $w\leq u$, with equality on $D=\{p=b\}$. Ordered contact gives
\[
 \mathbf1_D\tau'(p-V)^n\mu_p
 \leq\mathbf1_D\mathrm{MA}_\theta(w)
 \leq\mathbf1_D\mu_u.
\]
On $D$ the derivative $\tau'(p-V)$ equals $1/\chi'(u-V)$. Together with contact support for $p$, this proves the claim on $\Omega$. The analytic set $X\setminus\Omega$ is not charged by the non-pluripolar measures involved.
\end{proof}

\Needspace{11\baselineskip}
\begin{proposition}\label{prop:big-minimal-higher-moment}
Set $N=2n+1$ and $r=1+(4N)^{-1}$. For $u\in\PSH(X,\theta)$ with $u\sim V$ and $\sup_Xu=0$, put $B=\int_X(V-u)\,\mu_u$. There is a constant $C$, depending only on $n,\ell_V,M_V$, such that, for every non-pluripolar probability measure $\nu$ with $A=A_\theta(\nu)<\infty$,
\[
 \int_X(V-u)^r\,\mathrm{d}\nu
 \leq C(1+A)(1+B)^{1+1/N}
 .
\]
In particular, normalized potentials $u\sim V$ with uniformly bounded $B$ are uniformly integrable with respect to $\nu$.
\end{proposition}
\begin{proof}
From the distribution of $V-u$ under $\mu_u$ we build a concave weight adapted to $u$, apply the preceding lemma, and use $A$ to bound the resulting envelope against $\nu$. Throughout, $c$ and $C$ denote positive constants depending only on $n,\ell_V,M_V$.

The normalization gives $u\leq V$. Put $x=V-u\geq0$ and
\[
 T=\sup\{t\geq0:\mu_u(\{x>t\})>0\},
\]
with $T=0$ if this set is empty. Since $u\sim V$, $T$ is finite. Zero domination, applied to $u$ and $V-t$ for $t>T$, shows that $u\geq V-T$; thus $x\leq T$ off the pole set. If $T\leq2$ the claim is immediate. We may therefore suppose $T>2$; in particular $B>0$.

Fix a real number $s$ with $0<s<T-2$. For $t\geq0$, define
\[
 g_s(t)=
 \begin{cases}
  \bigl((1+t)^{3/2}\mu_u(\{x>t\})\bigr)^{-1},&0\leq t\leq T-s,\\
  (1+t)^{-3/2},&t>T-s,
 \end{cases}
 \quad F_s(t)=1+\int_0^t g_s(a)\,\mathrm{d}a,
\]
and set
\[
 W_s(t)=\int_0^t F_s(a)^{1/N}\,\mathrm{d}a,
 \qquad\chi_s(-t)=-W_s(t).
\]
The tail $\mu_u(\{x>t\})$ in the first denominator has a positive lower bound on $[0,T-s]$. Hence $F_s$ is locally Lipschitz, and $\chi_s$ has the regularity and concavity required in the lemma.

Tonelli's theorem gives
\begin{equation}\label{eq:big-weight-derivative-moment}
 \int_X F_s(x)\,\mu_u
 =\mu_u(X)+\int_0^Tg_s(t)\mu_u(\{x>t\})\,\mathrm{d}t
 \leq3\max(1,M_V),
\end{equation}
since $\int_0^\infty(1+t)^{-3/2}\,\mathrm{d}t=2$ and the tail is at most $M_V$ on the second interval.
Chebyshev's inequality gives $\mu_u(\{x>t\})\leq B/t$. Integrating first for $F_s$ and then for $W_s$, we obtain
\begin{equation}\label{eq:big-weight-growth}
 W_s(t)\geq c(1+B)^{-1/N}t^{\beta},
 \qquad1\leq t\leq T-s,
 \qquad\beta=1+\frac1{2N}.
\end{equation}
Indeed, $g_s(t)\geq cB^{-1}t^{-1/2}$ for $t\geq1$ in this interval, so $F_s(t)\geq c(1+B)^{-1}t^{1/2}$. For $t\geq2$ integrate this estimate over $[t/2,t]$; for $1\leq t\leq2$ use $W_s(t)\geq t$.

Let $p_s=P_\theta(V-W_s(x))$, the envelope of the preceding lemma for $\chi=\chi_s$. The lemma, the bound $W_s(t)\leq tF_s(t)^{1/N}$ and the Cauchy--Schwarz inequality give, since $N=2n+1$,
\[
 \begin{split}
 \int_X(V-p_s)^{1/2}\,\mu_{p_s}
 &\leq\int_X x^{1/2}F_s(x)^{(n+1/2)/N}\,\mu_u\\
 &\leq B^{1/2}\left(\int_XF_s(x)\,\mu_u\right)^{1/2}
 \leq C B^{1/2}.
 \end{split}
\]
Write $a_s=\sup_Xp_s\leq0$. By the definition of $V$, $p_s-a_s\leq V$, so $V-p_s\geq-a_s$. Since $\mu_{p_s}(X)\geq\ell_V$, the last estimate yields $\ell_V\sqrt{-a_s}\leq C\sqrt B$, and hence $-a_s\leq CB$.

The definition of $A$ now gives
\[
 \int_XW_s(x)\,\mathrm{d}\nu
 \leq\int_X(V-p_s)\,\mathrm{d}\nu
 \leq A-a_s\leq A+CB.
\]
Combining this with \eqref{eq:big-weight-growth}, and then letting $s\downarrow0$ for each fixed $1\leq t<T$, we obtain
\[
 \nu(\{x>t\})\leq C(1+B)^{1/N}(A+B)t^{-\beta}.
\]
There is no tail for $t\geq T$. The layer-cake formula therefore gives
\[
 \int_Xx^r\,\mathrm{d}\nu
 =r\int_0^\infty t^{r-1}\nu(\{x>t\})\,\mathrm{d}t
 \leq1+C(1+B)^{1/N}(A+B),
\]
because $1<r<\beta$. This implies the stated bound.

Finally $-u=-V+(V-u)$, where $-V\in L^1(\nu)$ because $A<\infty$. The first summand is fixed and integrable, and the second has a uniform $L^r(\nu)$ bound. This proves uniform integrability; no $L^r$ bound for $V$ itself is claimed.
\end{proof}
