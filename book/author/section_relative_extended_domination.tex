\subsection{Domination for extended measures}

We now remove the finiteness assumption on $\mu_u$. The key case is an ordered pair $u\leq v$ in $\mathcal E_\varphi$ such that $\mu_u$ puts no mass on $\{u<v\}$. A difference envelope of $u$ and a smaller-background envelope of $v$ then has no mass where it is negative. Its measure is therefore finite, and \cref{thm:relative-finite-domination}, applied in a smaller background, proves that $u=v$. The general domination principle follows by the stretching argument used above.

\Needspace{10\baselineskip}
\begin{lemma}\label{lem:relative-zero-domination}
Let $u,v\in\mathcal E_\varphi$ satisfy $u\leq v$. If
\[
 \mu_u(\{u<v\})=0,
\]
then $u=v$.
\end{lemma}
\begin{proof}
Translate both potentials by a common constant so that $u\leq v\leq\varphi\leq0$. Fix $\delta$ with $\varepsilon<\delta<1$, put $\beta=1-\delta$, and set
\[
 q=P_{\delta\theta}(v).
\]
This envelope is not identically $-\infty$ by \cref{thm:relative-uniform-mass-lower}, applied to $v$. By \cref{lem:relative-difference-envelope},
\[
 w=P_\theta(u-\varphi)\in\mathcal E_0,
 \qquad w+\varphi\leq u,\qquad w\leq0.
\]
The stretching lemma \cref{lem:relative-stretch-closure}, applied with reference zero, shows that
$z=P_\theta(w/\beta)$ belongs to $\mathcal E_0$. Scaling the background and each defining competitor gives
\[
 \beta z\in\mathcal E_{\mathrm{env}}(X,\beta\theta,0).
\]
Indeed, if $a\in\PSH(X,\theta)$ lies below $t z$, then $\beta a\in\PSH(X,\beta\theta)$ lies below $t\beta z$, for every $t\geq1$.

Since $q\leq v\leq\varphi$, we have
\[
 \beta z\leq w\leq u-\varphi\leq u-q
\]
off the pole sets. Thus the envelope
\[
 k=P_{\beta\theta}(u-q)
\]
is not identically $-\infty$. Here the difference obstacle is taken off the pluripolar pole sets; it is quasi-continuous and finite almost everywhere. The general envelope theorem gives a finite upper bound for $k$ and the inequality $k+q\leq u$. Since $\beta z\leq k$, upward stability gives
$k\in\mathcal E_{\mathrm{env}}(X,\beta\theta,0)$.

Write $\nu_k=\mathrm{MA}^{\mathrm{ext}}_{\beta\theta}(k)$. By contact support, $\nu_k$ is carried by
\[
 D=\{k+q=u\}
\]
outside the pole sets. Positivity of mixed products and ordered contact give
\[
 \nu_k\leq\mathbf1_D\mu_u.
\]
These inequalities are first proved on common finite truncation levels and then pass to the limit by exhaustion. On $D\cap\{k<0\}$ we have $u=k+q<q\leq v$, so the hypothesis gives $\nu_k(\{k<0\})=0$.

This vanishing implies that $\nu_k$ is finite. Indeed, $\widehat k=\max(k,-1)$ is a bounded $\beta\theta$-plurisubharmonic function, and locality gives
\[
 \nu_k(X)=\nu_k(\{k\geq0\})
 =\int_{\{k\geq0\}}\mathrm{MA}_{\beta\theta}(\widehat k)
 \leq\int_X\mathrm{MA}_{\beta\theta}(\widehat k)<\infty.
\]

We now apply \cref{thm:relative-finite-domination} in the background $\beta\theta$, with reference zero, to the potentials $k$ and $0$ and the coefficient zero. Its hypotheses hold: $\beta\theta$ is semipositive and big, the reference zero satisfies the model condition, $\nu_k$ is finite and vanishes on $\{k<0\}$, and
\[
 v_-(\beta\theta)=\beta^n v_-(\theta)>0.
\]
The scaling identity follows from the bijection $a\mapsto\beta a$ between bounded potentials in the two backgrounds. The theorem yields $k\geq0$, and hence $u\geq q$.

It remains to let $\delta\uparrow1$. Fix $\delta_0\in(\varepsilon,1)$ and put $r=P_{\delta_0\theta}(v)$. For $\delta_0<\delta<1$, set $t=(\delta-\delta_0)/(1-\delta_0)$. The potential $t v+(1-t)r$ is $\delta\theta$-plurisubharmonic and lies below $v$. Therefore
\[
 u\geq P_{\delta\theta}(v)\geq t v+(1-t)r.
\]
Letting $t\uparrow1$ at points where $v$ and $r$ are finite gives $u\geq v$ almost everywhere, and then everywhere by \cref{lem:qpsh-almost-everywhere-order}. Together with $u\leq v$, this proves equality.
\end{proof}

\begin{theorem}\label{thm:relative-extended-domination}\index{Domination principle!extended measures}
Let $u\in\mathcal E_\varphi$ and $v\in\PSH(X,\theta)$ with $v\preceq\varphi$. If $0\leq c<1$ and
\[
 \mu_u\leq c\mu_v\quad\hbox{on }\{u<v\},
\]
then $u\geq v$.
\end{theorem}
\begin{proof}
Put $w=\max(u,v)\in\mathcal E_\varphi$ and choose $b>1$ with $(1-b^{-1})^n>c$. The stretching lemma gives $p=Q_b(u,w)\in\mathcal E_\varphi$ and $p\leq u$. By contact support, its measure is carried by
\[
 D=\{b^{-1}p+(1-b^{-1})w=u\}.
\]
On $D$, positivity and ordered contact imply
\[
 b^{-n}\mu_p+(1-b^{-1})^n\mu_w\leq\mu_u.
\]
On $\{u<w\}$, locality gives $\mu_u\leq c\mu_w$. We restrict both inequalities to $D\cap\{u<w\}$ and to common finite truncation levels, where all measures are finite and may be subtracted. Since $(1-b^{-1})^n>c$, it follows that $\mu_p$ vanishes there. Exhausting the finite levels, we conclude that $\mu_p$ is carried by $D\cap\{u=w\}\subseteq\{p=u\}$; that is, $\mu_p(\{p<u\})=0$.

Now \cref{lem:relative-zero-domination}, applied to $p\leq u$, both in $\mathcal E_\varphi$, gives $p=u$. Substituting this into the defining inequality $b^{-1}p+(1-b^{-1})w\leq u$ yields $w\leq u$ off the poles, hence everywhere by \cref{lem:qpsh-almost-everywhere-order}. Thus $v\leq w\leq u$.
\end{proof}

\begin{corollary}\label{cor:relative-extended-exponential-order}\label{cor:relative-finite-exponential-order}
Let $u,v$ be as in \cref{thm:relative-extended-domination}. Then, for every $\lambda>0$,
\[
 \mathrm{e}^{-\lambda u}\mu_u\leq \mathrm{e}^{-\lambda v}\mu_v
 \quad\Longrightarrow\quad u\geq v.
\]
Moreover, if $\mu_u\leq c\mu_v$ on $X$ for a finite constant $c\geq0$, then $c\geq1$.
\end{corollary}
\begin{proof}
Fix $a>0$. On $\{u<v-a\}$ the exponential inequality gives $\mu_u\leq \mathrm{e}^{-\lambda a}\mu_v$; here all weights are evaluated off the pluripolar pole sets. The theorem, applied to $v-a$, gives $u\geq v-a$, and we let $a\downarrow0$. For the second assertion, suppose $c<1$. Then the theorem applies to $v+A$ for every real $A$, since $\mu_{v+A}=\mu_v$, and gives $u\geq v+A$. This is impossible at a point where $u$ and $v$ are both finite.
\end{proof}

In particular, applying the exponential implication in both directions shows that a solution of $\mu_u=\mathrm{e}^{\lambda u}\nu$ in $\mathcal E_\varphi$, for a fixed $\lambda>0$, is unique. If $\mu_u=c_1\nu$ and $\mu_v=c_2\nu$ with $c_1,c_2>0$ and $\nu\ne0$, the second assertion, applied in both directions, gives $c_1=c_2$. \Cref{cor:analytic-nonuniqueness-envelope-class} at the end of the chapter shows that this uniqueness of the scalar does not extend to the normalized potentials.
