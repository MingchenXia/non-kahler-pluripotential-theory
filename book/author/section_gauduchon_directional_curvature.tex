\section{Conformal changes and directional curvature positivity}
\label{sec:gauduchon-directional-curvature}

Let $E\to X$ be a holomorphic vector bundle of positive rank on the connected compact complex manifold $X$, with $\dim_{\mathbb C}X=n\geq2$. The compatibility condition for a scalar trace equation can also improve a bundle metric. A conformal change adds the same complex Hessian to every fiber direction. Its trace can be positive away from a small region where the original curvature already supplies a positive direction.

For a smooth Hermitian metric $h$ on $E$, write its matrix in a holomorphic frame as $H$, using column vectors, an inner product linear in its first argument, and $h(v,v)=v^*Hv$. Its Chern curvature is
\[
 F_h=\bar\partial(H^{-1}\partial H)
     =\sum_{i,j}F_{i\bar j}\,\mathrm dz_i\wedge\mathrm d\bar z_j.
\]
Put $R_h(u,\bar u)=\sum_{i,j}F_{i\bar j}u_i\bar u_j$ and
\[
 \alpha_{h,v}=\mathrm i\,h(F_hv,v),\qquad
 K_{\omega,h}=\Lambda_\omega(\mathrm iF_h).
\]
Here $\alpha_{h,v}$ is a real $(1,1)$-form at the base point of $v$, and $K_{\omega,h}$ is an $h$-Hermitian endomorphism. The coefficients of $\alpha_{h,v}$ evaluate on $u$ as $h(R_h(u,\bar u)v,v)$. We call $K_{\omega,h}$ the \emph{mean curvature} relative to the Hermitian form $\omega$.

\Needspace{12\baselineskip}
\begin{definition}\label{def:directional-rc-positivity}\index{RC-positivity}\index{RC-positivity!uniform}\index{Mean curvature!of a bundle metric}
The metric $h$ is \emph{RC-positive} at $x$ if, for every $0\ne v\in E_x$, there is a tangent vector $u\in T_x^{1,0}X$ such that
\[
 h(R_h(u,\bar u)v,v)>0.
\]
It is \emph{uniformly RC-positive} at $x$ if one tangent vector $u$ works for every nonzero $v\in E_x$. A metric has either property on a set if it has that property at every point of the set.
\end{definition}

\begin{lemma}\label{lem:semipositive-mean-rc-criterion}
Suppose that $K_{\omega,h}(x)\geq0$. Then $h$ is RC-positive at $x$ if and only if $\alpha_{h,v}\ne0$ for every $0\ne v\in E_x$.
\end{lemma}
\begin{proof}
The trace of $\alpha_{h,v}$ relative to $\omega$ is $h(K_{\omega,h}v,v)\geq0$. A nonzero Hermitian form with nonnegative trace has a positive eigenvalue: if every eigenvalue were nonpositive, their nonnegative sum would force all of them to vanish. This gives the required tangent direction. The reverse implication follows from the positive evaluation.
\end{proof}

\Needspace{14\baselineskip}
\begin{theorem}\label{thm:conformal-rc-globalization}\index{RC-positivity!conformal construction}
Suppose that $K_{\omega,h}\geq0$ on $X$ and that $h$ is RC-positive on a nonempty open set $U$. There is a smooth real function $f$ such that $h_t=\mathrm e^{-tf}h$ is RC-positive on all of $X$ for every sufficiently small $t>0$.
\end{theorem}
\begin{proof}
Choose a conformal Gauduchon form $\gamma=\mathrm e^a\omega$ by \cref{thm:gauduchon}. Then $K_{\gamma,h}=\mathrm e^{-a}K_{\omega,h}\geq0$. Choose a closed coordinate ball $Q\subset U$ and a smooth function $\chi\geq0$ supported in its interior, with $\int_X\chi\gamma^n>0$. Set
\[
 \rho=1-\frac{\int_X\gamma^n}{\int_X\chi\gamma^n}\chi.
\]
Thus $\rho=1$ on $X\setminus Q$ and $\int_X\rho\gamma^n=0$. For the Gauduchon form $\gamma$, \cref{prop:adjoint-gauduchon} gives a smooth real solution of
\begin{equation}\label{eq:rc-gauduchon-poisson}
 P_\gamma f:=\operatorname{tr}_\gamma(\mathrm i\partial\bar\partial f)=\rho.
\end{equation}
Indeed, with $L_\gamma f=n\ddc f\wedge\gamma^{n-1}/\gamma^n$ as in \eqref{eq:linearized-ma}, we have $P_\gamma=2\pi L_\gamma$. Its adjoint probability is $\gamma^n/\int_X\gamma^n$.

The curvature transformation is
\begin{equation}\label{eq:rc-conformal-curvature}
 F_{h_t}=F_h+t\partial\bar\partial f\,\operatorname{Id}_E,
 \qquad
 \alpha_{h_t,v}=\mathrm e^{-tf}
       \bigl(\alpha_{h,v}+t|v|_h^2\mathrm i\partial\bar\partial f\bigr).
\end{equation}
On $X\setminus Q$, the trace of the expression in parentheses is at least $t|v|_h^2\rho=t|v|_h^2>0$ for $v\ne0$. Hence it has a positive eigenvalue there.

Over $Q$, the unit sphere bundle $|v|_h=1$ is compact. The largest $\gamma$-eigenvalue of $\alpha_{h,v}$ is continuous and strictly positive on it, so it has a positive minimum $\delta$. Let $C$ be the maximum on $X$ of the operator norm of $\mathrm i\partial\bar\partial f$ relative to $\gamma$. For every such unit vector over $Q$, the largest eigenvalue of the expression in parentheses in \eqref{eq:rc-conformal-curvature} is at least $\delta-tC$. Choose $t>0$ small enough that this is positive. Homogeneity in $v$ and the positive factor $\mathrm e^{-tf}$ prove the assertion on $Q$ as well.
\end{proof}

The conclusion concerns a direction that may depend on $v$. This distinction already occurs in a local metric with zero mean curvature.

\Needspace{14\baselineskip}
\begin{example}\label{ex:local-rc-nonuniform}\index{RC-positivity!dependence on the fiber vector}
On a sufficiently small ball in $\mathbb C^2$, equip the trivial rank-two bundle with the metric
\[
 H(z)=\operatorname{Id}-zz^*+\tfrac12|z|^2\operatorname{Id}.
\]
At the origin this metric has zero mean curvature for the Euclidean Hermitian form and is RC-positive, but is not uniformly RC-positive.
\end{example}
\begin{proof}
The two eigenvalues of $H(z)$ are $1-\tfrac12|z|^2$ and $1+\tfrac12|z|^2$, so the metric is positive on a small ball. Its first derivatives vanish at zero, where
\[
 F_{i\bar j}=e_ie_j^*-\tfrac12\delta_{ij}\operatorname{Id},\qquad
 h(R_h(u,\bar u)v,v)=|v^*u|^2-\tfrac12|u|^2|v|^2.
\]
The sum $F_{1\bar1}+F_{2\bar2}$ is zero. For each nonzero $v$, taking $u=v$ gives a positive evaluation. For each nonzero $u$, taking $v\perp u$ gives a negative evaluation, so no single $u$ works for every $v$.
\end{proof}

\Needspace{12\baselineskip}
\begin{proposition}\label{prop:trivial-determinant-rc-obstruction}\index{RC-positivity!determinant obstruction}
If $\det E$ is holomorphically trivial on $X$, then no smooth Hermitian metric on $E$ is uniformly RC-positive on $X$ or has positive definite mean curvature relative to any Hermitian form.
\end{proposition}
\begin{proof}
If $h$ were uniformly RC-positive, then at each point some $u$ would make the $h$-Hermitian endomorphism $R_h(u,\bar u)$ positive definite. Its trace would be strictly positive. The induced curvature on $\det E$ is $\operatorname{tr}F_h$, so the determinant metric would be RC-positive as a line-bundle metric. If instead $K_{\omega,h}>0$, the $\omega$-trace of the determinant curvature would be $\operatorname{tr}K_{\omega,h}>0$, which also gives a positive tangent evaluation.

Choose a nowhere-zero holomorphic section $s$ of $\det E$ and put $b=\log|s|_{\det h}^2$. Its curvature is $-\partial\bar\partial b$. At a global minimum of $b$, this curvature has no positive tangent evaluation. This contradicts RC-positivity of the determinant metric.
\end{proof}

Consequently, when $\det E$ is trivial and the hypotheses of \cref{thm:conformal-rc-globalization} hold, its conformal metric is RC-positive while the bundle admits neither a uniformly RC-positive metric nor a metric with positive definite mean curvature. Verifying those hypotheses is a separate geometric problem; the local example above supplies only the pointwise distinction.
