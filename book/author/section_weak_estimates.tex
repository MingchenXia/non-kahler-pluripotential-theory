\section{Capacity estimates and uniform compactness}

Let $\operatorname{Cap}=\operatorname{Cap}_\omega$ denote the capacity of \cref{sec:hermitian-capacity}. Modified comparison bounds the capacity of a thin sublevel set by the Monge--Amp\`ere mass of a slightly larger one. For an $L^p$ density, \cref{cor:lp-capacity-square} bounds this mass by the square of the capacity. Iterating the two inequalities gives a lower bound for the capacity of every sublevel set of fixed height above the minimum. Since the capacity of deep sublevel sets of normalized potentials is small by the tail estimate \eqref{eq:global-capacity-tail}, this bounds the depth of a normalized solution.

\Needspace{15\baselineskip}
\begin{lemma}\label{lem:thin-sublevel-capacity}
Let $u,v\in\PSH(X,\omega)\cap L^\infty(X)$, with $-H\leq v\leq0$ and $H\geq1$. Fix $0<\varepsilon<1$, and put
\[
 m=\inf_X(u-(1-\varepsilon)v),\qquad
 U_s=\{u<(1-\varepsilon)v+m+s\}.
\]
There are $\delta_\varepsilon>0$, depending only on $(X,\omega,\varepsilon)$, and $C_H>0$, depending only on $(n,H)$, such that
\begin{equation}\label{eq:thin-capacity}
 t^n\operatorname{Cap}(U_s)\leq C_H\int_{U_{s+t}}\omega_u^n
 \qquad(0<s,t,\ s+t<\delta_\varepsilon).
\end{equation}
Every $U_s$, $s>0$, has positive capacity.
\end{lemma}
\begin{proof}
Let $B_0$ be the constant of \cref{thm:modified-comparison}, and set
\[
 \delta_\varepsilon=\min\{1-\varepsilon,\varepsilon^3/(128B_0n^2)\}.
\]
For a capacity test $0\leq\rho\leq1$, put
\[
 \tau=\frac{t}{2(1-\varepsilon)(H+1)},\quad
 w=(1-\tau)v+\tau\rho,\quad
 m'=\inf_X(u-(1-\varepsilon)w).
\]
Then $0<\tau<1$ and
$0\leq(1-\varepsilon)(w-v)\leq t/2$, so $m-t/2\leq m'\leq m$. Consequently
\[
 U_s\subset V:=\{u<(1-\varepsilon)w+m'+s+t/2\}
 \subset U_{s+t}.
\]
We apply modified comparison to $u,w$ at height $s+t/2<\delta_\varepsilon$; by the choice of $\delta_\varepsilon$, its inverse factor is at most $(8/7)^n$. Moreover,
\[
 \omega_{(1-\varepsilon)w}
 \geq(1-\varepsilon)\tau\omega_\rho
 =\frac{t}{2(H+1)}\omega_\rho.
\]
Integrating over $U_s\subset V$, applying modified comparison on $V$, and taking the supremum over $\rho$, we obtain \eqref{eq:thin-capacity} with $C_H=[2(H+1)]^n(8/7)^n$. Finally, each $U_s$ is nonempty by the definition of the infimum, so it has positive smooth volume by \cref{lem:strict-set-positive-volume}, and hence positive capacity, as the constant test function shows.
\end{proof}

\Needspace{11\baselineskip}
\begin{lemma}\label{lem:backwards-capacity}
In the preceding setting, suppose
\[
 \int_E\omega_u^n\leq A\operatorname{Cap}(E)^2\qquad(E\subset X\ \hbox{Borel}),
 \qquad A>0.
\]
For every $0<D<\delta_\varepsilon$,
\begin{equation}\label{eq:sublevel-capacity-lower}
 \operatorname{Cap}(U_D)\geq\frac{D^n}{4^nC_HA}.
\end{equation}
\end{lemma}
\begin{proof}
Write $a(s)=\operatorname{Cap}(U_s)^{1/n}>0$ and $K=(C_HA)^{1/n}$. The preceding lemma and the hypothesis, applied to $E=U_{s+t}$, give
\[
 t\,a(s)\leq K a(s+t)^2\qquad(s+t<\delta_\varepsilon).
\]
Suppose, to the contrary, that $4Ka(D)<D$. Starting with $s_0=D$, define
\[
 t_j=2Ka(s_j),\qquad s_{j+1}=s_j-t_j.
\]
We check that $s_{j+1}>0$ before applying the sublevel inequality. If $a(s_i)\leq2^{-i}a(D)$ for $0\leq i\leq j$, then
\[
 \sum_{i=0}^j t_i\leq2Ka(D)\sum_{i=0}^j2^{-i}<4Ka(D)<D.
\]
Hence $s_{j+1}>0$, and the sublevel inequality with $s=s_{j+1}$ and $t=t_j$ gives $a(s_{j+1})\leq a(s_j)/2$. By induction, every step is justified, and $s_j\downarrow s_\infty\geq D-4Ka(D)>0$. Monotonicity gives
$0<a(s_\infty)\leq a(s_j)\leq2^{-j}a(D)$, a contradiction. Therefore $D\leq4Ka(D)$, which is \eqref{eq:sublevel-capacity-lower}.
\end{proof}

\begin{theorem}\label{thm:lp-potential-estimate}\index{A priori estimate!Lp densities@$L^p$ densities}
There are constants $C_p,c_p>0$, depending only on $(X,\omega,\mathrm{d}V,p)$, such that, whenever
$u\in\PSH(X,\omega)\cap L^\infty(X)$, $\sup_Xu=0$, and
\[
 \omega_u^n=F\,\mathrm{d}V,\qquad F\in L^p(X,\mathrm{d}V),\quad F\geq0,
\]
we have
\begin{equation}\label{eq:lp-potential-bound}
 \|u\|_\infty\leq C_p(1+\|F\|_p),
 \qquad \|F\|_p\geq c_p.
\end{equation}
\end{theorem}
\begin{proof}
By \cref{cor:bounded-mass-positive}, $F$ is not almost everywhere zero. Apply \cref{lem:backwards-capacity} with $v=0$, $H=1$, $\varepsilon=1/2$, and $A=C\|F\|_p>0$, where $C$ is the constant of \cref{cor:lp-capacity-square}. Fix $D=\delta_{1/2}/2$, and put $M=-\inf_Xu$. Since $U_D=\{u<-M+D\}$, the resulting inequality reads
\[
 \operatorname{Cap}(\{u<-M+D\})\geq c/\|F\|_p.
\]
If $M>D+1$, \eqref{eq:global-capacity-tail} bounds this capacity above by $C/(M-D)$, and the first assertion follows; the case $M\leq D+1$ is immediate. Bounding the same capacity above by $\operatorname{Cap}(X)$ gives the second assertion.
\end{proof}

Suppose that two uniformly bounded solutions are far apart at some point. The preceding estimates give a fixed positive lower bound for the capacity of a sublevel set of their difference. If their $L^1$ distance is small, this set has small volume, and the uniform $L^p$ bound for the densities then makes its Monge--Amp\`ere mass small. This contradicts the sublevel estimate \eqref{eq:thin-capacity}, and so upgrades $L^1$ compactness to uniform compactness. Put $p'=p/(p-1)$.

\begin{theorem}\label{thm:lp-uniform-compactness}\index{Compactness!uniform convergence of solutions}
Suppose $u_j\in\PSH(X,\omega)\cap C(X)$, $\sup_Xu_j=0$, and
\[
 \omega_{u_j}^n=F_j\,\mathrm{d}V,\qquad F_j\geq0,\qquad \sup_j\|F_j\|_p<\infty
\]
for the fixed $p>1$. If $(u_j)$ is Cauchy in $L^1(X,\mathrm{d}V)$, then it is Cauchy uniformly. In particular, every such family has a subsequence converging uniformly to a continuous $\omega$-plurisubharmonic function.
\end{theorem}
\begin{proof}
By \cref{thm:lp-potential-estimate}, there is a common $H\geq1$ with $-H\leq u_j\leq0$. Suppose that the sequence is not uniformly Cauchy. After exchanging indices if necessary, we find $a>0$ and arbitrarily large $j,k$ with
\[
 S:=\inf_X(u_k-u_j)\leq-a.
\]
\Needspace{10\baselineskip}
Choose $0<\varepsilon<1$ such that $\varepsilon H<a/4$, and choose a fixed $D>0$ with $2D<\delta_\varepsilon$ and $2D<a/4$. Apply the preceding sublevel lemmas with $u=u_k$ and $v=u_j$. Since $v\leq0$,
$m=\inf(u-(1-\varepsilon)v)\leq S$. Thus
\[
 U_{2D}\subset\{u-v<S+2D+\varepsilon H\}
 \subset\{|u-v|>a/2\}.
\]
The capacity lower bound \eqref{eq:sublevel-capacity-lower}, with a constant $A$ independent of $j,k$ given by \cref{cor:lp-capacity-square}, yields $\operatorname{Cap}(U_D)\geq c>0$. On the other hand, \eqref{eq:thin-capacity} and the inequalities of H\"older and Chebyshev give
\[
 \begin{split}
 D^n\operatorname{Cap}(U_D)
 &\leq C_H\int_{U_{2D}}F_k\,\mathrm{d}V\\
 &\leq C_H\|F_k\|_p
       \left(\frac{2\|u_k-u_j\|_1}{a}\right)^{1/p'}
 \longrightarrow0.
 \end{split}
\]
This contradiction proves that the sequence is uniformly Cauchy. For the last assertion, the normalized compactness theorem, \cref{thm:normalized-compactness}, gives an $L^1$-Cauchy subsequence of any such family. Its uniform limit is continuous, and distributional convergence preserves $\omega$-plurisubharmonicity.
\end{proof}

\section{Construction of continuous solutions}

Smooth approximation gives solutions with unknown normalization constants. A Gauduchon trace bounds these constants above, and the lower bound for the density norm in \cref{thm:lp-potential-estimate} bounds them away from zero. The resulting uniform $L^p$ bound puts the potentials in the compact family of \cref{thm:lp-uniform-compactness}, so their limit solves the prescribed equation.

\begin{proof}[of \cref{thm:kn-continuous}]
Choose smooth strictly positive functions $f_j\to f$ in $L^p(X,\mathrm{d}V)$. By \cref{thm:tw-smooth}, there are smooth $u_j$ with $\omega_{u_j}>0$ and constants $c_j$ such that
\[
 \omega_{u_j}^n=c_jf_j\,\mathrm{d}V,\qquad \sup_Xu_j=0,\qquad c_j>0.
\]
We first bound the constants. For $n\geq2$, fix a Gauduchon metric $\gamma$. By \cref{prop:gauduchon-amgm},
\[
 c_j^{1/n}A_j\leq\int_X\omega\wedge\gamma^{n-1},
 \qquad A_j=\int_X(f_j\mathrm{d}V/\gamma^n)^{1/n}\gamma^n.
\]
Since $|x^{1/n}-y^{1/n}|\leq|x-y|^{1/n}$ for $x,y\geq0$, H\"older's inequality gives $A_j\to A>0$, where $A=\int_X(f\mathrm{d}V/\gamma^n)^{1/n}\gamma^n$. Thus $c_j$ is bounded above. In dimension one, Stokes' theorem gives the same estimate, with $A_j=\int_Xf_j\,\mathrm{d}V$.

The lower bound for the density norm in \cref{thm:lp-potential-estimate} gives
$c_j\|f_j\|_p\geq c_p$, so the constants are bounded away from zero. The densities $c_jf_j$ are therefore uniformly bounded in $L^p$, and the same theorem gives a common bound for the $u_j$. By \cref{thm:normalized-compactness}, we may pass to a subsequence with $c_j\to c>0$ and $u_j$ Cauchy in $L^1$. By \cref{thm:lp-uniform-compactness}, $u_j\to u$ uniformly, with $u$ continuous and $\sup_Xu=0$.

Bounded Bedford--Taylor continuity gives $\omega_{u_j}^n\rightharpoonup\omega_u^n$, while $c_jf_j\to cf$ in $L^p$. Hence $\omega_u^n=cf\,\mathrm{d}V$, as required.
\end{proof}

For the exponential equation, the role of the normalization constant is played by $\mathrm{e}^{\lambda\sup_Xu}$. Its upper bound comes from Jensen's inequality: weighting the Gauduchon volume by the $n$th root of the density relates $\sup_Xu$ to the Gauduchon trace of $\omega_u$, which does not depend on $u$. The lower bound for the density norm again gives the reverse bound.

\begin{lemma}\label{lem:exponential-gauduchon-mean}
Fix $\lambda>0$. Suppose that smooth functions $u_j$ with $\omega_{u_j}>0$ solve
\[
 \omega_{u_j}^n=\mathrm{e}^{\lambda u_j}f_j\,\mathrm{d}V,
\]
where the smooth strictly positive functions $f_j$ converge in $L^p$ to a nonnegative nonzero $f$. Then $M_j=\sup_Xu_j$ is bounded above and below, uniformly in $j$.
\end{lemma}
\begin{proof}
Absorbing the fixed smooth positive ratio $\mathrm{d}V/\omega^n$ into $f_j$ and $f$ preserves their $L^p$ convergence, so we may assume $\mathrm{d}V=\omega^n$. Choose $\rho>0$ smooth with $\ddc(\rho\omega^{n-1})=0$, using \cref{thm:gauduchon}; for $n=1$, take $\rho=1$. Put
\[
 \alpha=\int_X\rho\omega^n,\quad
 A_j=\int_X f_j^{1/n}\rho\omega^n,\quad
 \psi_j=u_j-M_j\leq0.
\]
Set $A=\int_X f^{1/n}\rho\omega^n$. The inequality for $n$th roots used in the preceding existence proof gives $A_j\to A>0$. Since $\ddc(\rho\omega^{n-1})=0$, the trace $\int_X\omega_{u_j}\wedge\rho\omega^{n-1}$ equals $\alpha$; the arithmetic--geometric mean inequality therefore implies
\[
 \alpha=\int_X\omega_{u_j}\wedge\rho\omega^{n-1}
 \geq \mathrm{e}^{\lambda M_j/n}\int_X \mathrm{e}^{\lambda\psi_j/n}f_j^{1/n}\rho\omega^n.
\]
Applying Jensen's inequality to the probability measure
$A_j^{-1}f_j^{1/n}\rho\omega^n$ and taking logarithms, we obtain
\begin{equation}\label{eq:gauduchon-exponential-mean}
 \lambda M_j\leq n\log(\alpha/A_j)
 +\frac{\lambda}{A_j}\int_X(-\psi_j)f_j^{1/n}\rho\omega^n.
\end{equation}
By \cref{lem:uniform-exponential}, the normalized potentials $\psi_j$ have uniform $L^q$ bounds for every finite $q$. With $q=np/(np-1)$, H\"older's inequality gives a uniform bound for the last integral, since
$\|f_j^{1/n}\|_{np}=\|f_j\|_p^{1/n}$. This proves the upper bound for $M_j$.

Finally, the density $F_j=\mathrm{e}^{\lambda u_j}f_j$ satisfies
$\|F_j\|_p\leq \mathrm{e}^{\lambda M_j}\|f_j\|_p$. The lower-density bound of \cref{thm:lp-potential-estimate}, applied to $\psi_j$, gives $c_p\leq \mathrm{e}^{\lambda M_j}\|f_j\|_p$, which bounds $M_j$ below.
\end{proof}

\begin{proof}[of \cref{thm:nguyen-exp}]
Choose smooth strictly positive functions $f_j\to f$ in $L^p$. By \cref{thm:smooth-proper-existence}, there are smooth $u_j$ with $\omega_{u_j}>0$ and $\omega_{u_j}^n=\mathrm{e}^{\lambda u_j}f_j\,\mathrm{d}V$. The preceding lemma bounds $M_j=\sup_Xu_j$, and consequently bounds $F_j=\mathrm{e}^{\lambda u_j}f_j$ in $L^p$. The normalized potentials $\psi_j=u_j-M_j$ are uniformly bounded by \cref{thm:lp-potential-estimate}. We may pass to a subsequence with $M_j\to M$ and $\psi_j$ Cauchy in $L^1$. \Cref{thm:lp-uniform-compactness} gives uniform convergence $u_j\to u\in\PSH(X,\omega)\cap C(X)$.

Uniform convergence and $f_j\to f$ in $L^p$ give
$\mathrm{e}^{\lambda u_j}f_j\to \mathrm{e}^{\lambda u}f$ in $L^p$, and continuity of bounded products therefore gives $\omega_u^n=\mathrm{e}^{\lambda u}f\,\mathrm{d}V$. Uniqueness, even among all bounded solutions, follows from bounded exponential comparison, \cref{cor:bounded-exponential-comparison}.
\end{proof}
