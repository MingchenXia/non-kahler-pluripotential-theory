\section{Compactness with a common lower barrier}

We return to a semipositive big background $\theta$. Convergence of potentials in $L^1$ does not by itself imply convergence of their Monge--Amp\`ere measures. We give a criterion for convergence in capacity when the potentials stay within a bounded distance of a reference admitting a smaller-background envelope. This common lower barrier ensures that all finite and infinite rooftop envelopes of the sequence are $\theta$-plurisubharmonic. A small error integral of each potential against its own Monge--Amp\`ere measure bounds the corresponding error of the rooftops by a summable tail. Weighted lower semicontinuity passes this bound to the limit, and singular domination identifies the limit of the lower envelopes with the $L^1$ limit. Squeezing the sequence between lower and upper envelopes then gives convergence in capacity, and a domination of the measures also controls their mass on deep sublevels.

\Needspace{16\baselineskip}
\begin{theorem}\label{thm:common-barrier-criterion}\index{Barrier!common lower bound}
Let $\theta\geq0$ be big. Suppose $\varphi\in\PSH(X,\theta)$ satisfies $P_{\delta\theta}(\varphi)\not\equiv-\infty$ for some $0<\delta<1$. Let $u_j\in\PSH(X,\theta)$ satisfy, for a fixed $C$,
\[
 \varphi-C\leq u_j\leq\varphi+C,\qquad u_j\longrightarrow u\text{ in }L^1(X).
\]
If
\begin{equation}\label{eq:common-barrier-self-error}
 I_j:=\int_X|\mathrm{e}^{u_j}-\mathrm{e}^u|\,
               \mathrm{MA}^{\mathrm{ext}}_\theta(u_j)\longrightarrow0,
\end{equation}
then $u_j\to u$ in capacity.
\end{theorem}
\begin{proof}
The canonical representative of the limit also lies between $\varphi-C$ and $\varphi+C$. It suffices to show that every subsequence has a further subsequence converging in capacity. We may therefore assume that $I_j\leq2^{-j}$ and that $u_j\to u$ almost everywhere. For $N\geq j$ define
\[
 p_{j,N}=P_\theta(\min\{u_j,\ldots,u_N\}),\qquad
 p_j=P_\theta\left(\inf_{\ell\geq j}u_\ell\right).
\]
\Needspace{8\baselineskip}
All these envelopes lie between $\varphi-C$ and $\varphi+C$. The contact theorem and ordered contact give an inequality for the finite rooftops. Partition the contact set of $p_{j,N}$ according to the first index $\ell$ for which $p_{j,N}=u_\ell$. On the corresponding piece, the measure of $p_{j,N}$ is bounded by that of $u_\ell$, and $\mathrm{e}^{p_{j,N}}=\mathrm{e}^{u_\ell}$. Therefore
\[
 \int_X|\mathrm{e}^{p_{j,N}}-\mathrm{e}^u|\,
            \mathrm{MA}^{\mathrm{ext}}_\theta(p_{j,N})
 \leq\sum_{\ell=j}^N I_\ell\leq2^{1-j}.
\]
Only positive measures enter this argument, so it applies also when their total masses are infinite.

As $N\to\infty$, $p_{j,N}$ decreases to $p_j$; the limit is the envelope of the infimum because every competitor lies below every finite rooftop. The sequence $p_j$ increases, and its canonical limit $p$ is $\theta$-plurisubharmonic. By \cref{lem:monotone-psh-capacity}, both monotone limits hold in capacity. Applying \cref{lem:weighted-extended-lsc} to the bounded nonnegative weights $|\mathrm{e}^{p_{j,N}}-\mathrm{e}^u|$, first in $N$ and then in $j$, we obtain
\[
 \int_X|\mathrm{e}^p-\mathrm{e}^u|\,\mathrm{MA}^{\mathrm{ext}}_\theta(p)=0.
\]
The inequalities $p_j\leq u_\ell$ for all $\ell\geq j$ imply $p\leq u$, first almost everywhere and then for the canonical representatives. Also $p\sim u\sim\varphi$. Off the pole sets, $\mathrm{e}^p<\mathrm{e}^u$ exactly on $\{p<u\}$, so
$\mathrm{MA}^{\mathrm{ext}}_\theta(p)(\{p<u\})=0$.
Since $P_{\delta\theta}(\varphi)-C$ is a competitor for $P_{\delta\theta}(p)$, the latter is not identically $-\infty$. \Cref{thm:singular-strict-domination}, applied with coefficient zero, gives $p\geq u$, and hence $p=u$.

Finally, by Hartogs' lemma and the almost-everywhere convergence, $U_j=(\sup_{\ell\geq j}u_\ell)^*$ decreases to $u$. The inequalities
$p_j\leq u_j\leq U_j$, together with the convergence in capacity of both monotone sequences, prove the claim.
\end{proof}

\Needspace{10\baselineskip}
\begin{proposition}\label{prop:lower-barrier-criterion}
In \cref{thm:common-barrier-criterion}, the two bounds on $u_j$ may be replaced by
\[
 \varphi-C\leq u_j\leq C,
\]
provided that the $L^1$ limit satisfies $u\preceq\varphi$. In particular, the approximants may be bounded while $\varphi$ is unbounded.
\end{proposition}
\begin{proof}
The $L^1$ limit satisfies $u\geq\varphi-C$, so the additional hypothesis gives $u\sim\varphi$. We repeat the construction of the finite rooftops in the theorem. Every $p_{j,N}$ is still $\theta$-plurisubharmonic and lies between $\varphi-C$ and $C$, which is enough for contact support and for the boundedness of the exponential weights. The resulting inequality is again
\[
 \int_X|\mathrm{e}^{p_{j,N}}-\mathrm{e}^u|\,
             \mathrm{MA}^{\mathrm{ext}}_\theta(p_{j,N})\leq2^{1-j}.
\]
The limits $p_j$ exist because of the common lower bound, and $p_j\leq u$ follows from $p_j\leq u_\ell$ for every $\ell\geq j$ and almost-everywhere convergence. Thus
$\varphi-C\leq p_j\leq u$, so all $p_j$ and their increasing limit $p$ have the same ordinary singularity type as $\varphi$. Monotone convergence in capacity and weighted lower semicontinuity give
$\mathrm{MA}^{\mathrm{ext}}_\theta(p)(\{p<u\})=0$, and singular domination applies to $(p,u)$ exactly as before, so $p=u$. The upper envelopes of the tails decrease to $u$, and the same squeezing argument gives convergence in capacity. Neither the finite rooftops nor the individual approximants needed to have the singularity type of $\varphi$.
\end{proof}

\Needspace{12\baselineskip}
\begin{corollary}\label{cor:common-barrier-continuity}
In the setting of \cref{thm:common-barrier-criterion}, with the bounds $\varphi-C\leq u_j\leq\varphi+C$, suppose either
\begin{enumerate}
\item $\mathrm{MA}^{\mathrm{ext}}_\theta(u_j)\leq\nu$ for a fixed finite non-pluripolar measure $\nu$; or
\item $\mathrm{MA}^{\mathrm{ext}}_\theta(u_j)\leq A\omega_{w_j}^n$, where $A$ is fixed and $w_j\in\PSH(X,\omega)$ are uniformly bounded and converge in capacity to a bounded $w$.
\end{enumerate}
Then $u_j\to u$ in capacity, all the measures $\mathrm{MA}^{\mathrm{ext}}_\theta(u_j)$ and $\mathrm{MA}^{\mathrm{ext}}_\theta(u)$ have finite mass, and
\[
 \mathrm{MA}^{\mathrm{ext}}_\theta(u_j)
 \rightharpoonup\mathrm{MA}^{\mathrm{ext}}_\theta(u).
\]
\end{corollary}
\begin{proof}
In the first case, \cref{lem:fixed-npp-integrals}, applied to the exponentials $\mathrm{e}^{u_j}$, proves \eqref{eq:common-barrier-self-error}. In the second case we use \cref{lem:moving-bounded-integrals} instead, since these exponentials are bounded and have a common lower bound for their curvature. In both cases the criterion gives convergence in capacity.

The masses are uniformly bounded by those of the dominating measures. For the deep sublevels we use $u_j\geq\varphi-C$, which gives $\{u_j<-T\}\subset\{\varphi<C-T\}$. In the first case the masses of these sublevels are at most $\nu(\{\varphi<C-T\})$, which tends to zero as $T\to\infty$ because $\nu$ does not charge the pole set of $\varphi$. In the second case \cref{lem:bounded-mixed-capacity} bounds them by
$C'\operatorname{Cap}_\omega(\{\varphi<C-T\})$, which also tends to zero. The tail criterion \cref{thm:undamped-tail-continuity}, in the general-background form of \cref{thm:general-background-damping}, now gives finiteness of the limiting measure and weak convergence.
\end{proof}

The reference $\varphi$ in these results need not be an envelope fixed point; the smaller-background condition suffices. Envelope fixed points enter when the two barriers must be derived from an a priori estimate, as in \cref{thm:relative-linfty}. When no common lower barrier is available, one has to be constructed in a relative envelope class; this problem is treated in \cref{sec:general-measure-relative}.
